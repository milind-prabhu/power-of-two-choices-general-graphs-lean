  12 theorem sufficientThreshold_upper (mean f z A B : ℝ)
  13     (hfm : 0 ≤ f*mean) (hA : 0 ≤ A) (hB : 0 ≤ B)
  14     (hbase : f*mean ≤ A) (hlog : z ≤ B) :
  15     (sufficientThreshold mean f z : ℝ) ≤ A+B+2 := by
  16   have hnonneg : 0 ≤ max (f*mean) z := hfm.trans (le_max_left _ _)
  17   have hc := Nat.ceil_lt_add_one hnonneg
  18   have hm : max (f*mean) z ≤ A+B := max_le
  19     (hbase.trans (le_add_of_nonneg_right hB))
  20     (hlog.trans (le_add_of_nonneg_left hA))
  21   simp only [sufficientThreshold, Nat.cast_add, Nat.cast_one]
  22   linarith

  24 theorem log_branching_bound (x f L σ K : ℝ)
  25     (hx : 0 ≤ x) (hf : 1 < f) (hL : 1 ≤ L) (hσ : 0 ≤ σ) (hK : 1 ≤ K)
  26     (hbranch : x ≤ K*(f*L)^(2*σ)) :
  27     Real.log x ≤ Real.log K + 2*σ*(Real.log f + Real.log L) := by
  28   have hfpos : 0 < f := by linarith
  29   have hLpos : 0 < L := by linarith
  30   have hKpos : 0 < K := by linarith
  31   by_cases hxpos : 0 < x
  32   · have h := Real.log_le_log hxpos hbranch
  33     rwa [Real.log_mul hKpos.ne' (Real.rpow_pos_of_pos (mul_pos hfpos hLpos) _).ne',
  34       Real.log_rpow (mul_pos hfpos hLpos), Real.log_mul hfpos.ne' hLpos.ne'] at h
  35   · have hxzero : x = 0 := by linarith
  36     rw [hxzero, Real.log_zero]
  37     exact add_nonneg (Real.log_nonneg hK)
  38       (mul_nonneg (by positivity) (add_nonneg (Real.log_pos hf).le (Real.log_nonneg hL)))
  39 
  40 theorem log_branching_requirement (x f L σ K : ℝ)
  41     (hx : 0 ≤ x) (hf : 1 < f) (hL : 1 ≤ L) (hσ : 0 ≤ σ) (hK : 1 ≤ K)
  42     (hbranch : x ≤ K*(f*L)^(2*σ)) :
  43     Real.log x / Real.log f ≤ 2*σ + (Real.log K + 2*σ*Real.log L)/Real.log f := by
  44   have hlog := Real.log_pos hf
  45   apply (div_le_iff₀ hlog).mpr
  46   have h := log_branching_bound x f L σ K hx hf hL hσ hK hbranch
  47   have heq : (2*σ + (Real.log K + 2*σ*Real.log L)/Real.log f) * Real.log f =
  48       Real.log K + 2*σ*(Real.log f + Real.log L) := by
  49     field_simp
  50     ring
  51   rwa [heq]

  57 theorem inverse_shifted_powers_sum_le_one (h : ℕ) :
  58     ∑ i ∈ range h, (1:ℝ)/2^(i+1) ≤ 1 := by
  59   have heq : (∑ i ∈ range h, (1:ℝ)/2^(i+1)) =
  60       (1/2:ℝ)*(∑ i ∈ range h, (1:ℝ)/2^i) := by
  61     rw [mul_sum]
  62     apply sum_congr rfl
  63     intro i _
  64     rw [pow_succ]
  65     ring
  66   rw [heq]
  67   have hsum := ThresholdGreedy.inverse_powers_sum_le_two h

  70 theorem inverse_scale_logs_sum_le (h : ℕ) :
  71     ∑ i ∈ range h, (1:ℝ)/Real.log (Peeling.scaleNat i : ℝ) ≤ 1/Real.log 2 := by
  72   have hp : (0:ℝ) < Real.log 2 := Real.log_pos (by norm_num)
  73   have heq : (∑ i ∈ range h, (1:ℝ)/Real.log (Peeling.scaleNat i : ℝ)) =
  74       (1/Real.log 2)*(∑ i ∈ range h, (1:ℝ)/2^(i+1)) := by
  75     rw [mul_sum]
  76     apply sum_congr rfl
  77     intro i _
  78     rw [scale_log]
  79     ring
  80   rw [heq]
  81   exact (mul_le_mul_of_nonneg_left (inverse_shifted_powers_sum_le_one h) (by positivity)).trans_eq (by ring)

  83 /-- At each scale, mu is the binomial-tail base mean including the factor e,
  84 and x is the right branching factor divided by the desired decay target.
  85 The total budget is explicit, including both ceiling and increment errors. -/
  86 theorem sufficientThreshold_scale_budget (h : ℕ) (mean x : ℕ → ℝ) (A L σ K : ℝ)
  87     (hmean : ∀ i < h, 0 ≤ mean i) (hx : ∀ i < h, 0 ≤ x i)
  88     (hA : 0 ≤ A) (hL : 1 ≤ L) (hσ : 0 ≤ σ) (hK : 1 ≤ K)
  89     (hbase : ∀ i < h, (Peeling.scaleNat i : ℝ)*mean i ≤ A)
  90     (hbranch : ∀ i < h, x i ≤ K*((Peeling.scaleNat i : ℝ)*L)^(2*σ)) :
  91     (∑ i ∈ range h, (sufficientThreshold (mean i) (Peeling.scaleNat i)
  92       (Real.log (x i)/Real.log (Peeling.scaleNat i)) : ℝ)) ≤
  93       h*(A+2*σ+2) + (Real.log K + 2*σ*Real.log L)/Real.log 2 := by
  94   have hq : 0 ≤ Real.log K + 2*σ*Real.log L :=
  95     add_nonneg (Real.log_nonneg hK) (mul_nonneg (by positivity) (Real.log_nonneg hL))
  96   have hlocal (i : ℕ) (hi : i < h) :
  97       (sufficientThreshold (mean i) (Peeling.scaleNat i)
  98         (Real.log (x i)/Real.log (Peeling.scaleNat i)) : ℝ) ≤
  99       A+2*σ+2 + (Real.log K + 2*σ*Real.log L)/Real.log (Peeling.scaleNat i : ℝ) := by
 100     have hf : (1:ℝ) < Peeling.scaleNat i := by
 101       have hh := Peeling.scaleNat_ge_two i
 102       exact_mod_cast (by omega : 1 < Peeling.scaleNat i)
 103     have hlog := log_branching_requirement (x i) _ L σ K (hx i hi) hf hL hσ hK (hbranch i hi)
 104     have hB : 0 ≤ 2*σ + (Real.log K + 2*σ*Real.log L)/Real.log (Peeling.scaleNat i : ℝ) := by
 105       exact add_nonneg (by positivity) (div_nonneg hq (Real.log_pos hf).le)
 106     have ht := sufficientThreshold_upper (mean i) (Peeling.scaleNat i)
 107       (Real.log (x i)/Real.log (Peeling.scaleNat i)) A _
 108       (mul_nonneg (by positivity) (hmean i hi)) hA hB (hbase i hi) hlog
 109     linarith
 110   calc
 111     _ ≤ ∑ i ∈ range h, (A+2*σ+2 + (Real.log K + 2*σ*Real.log L)/Real.log (Peeling.scaleNat i : ℝ)) :=
 112       sum_le_sum fun i hi => hlocal i (mem_range.mp hi)
 113     _ = h*(A+2*σ+2) + (Real.log K + 2*σ*Real.log L)*
 114         (∑ i ∈ range h, (1:ℝ)/Real.log (Peeling.scaleNat i : ℝ)) := by
 115       rw [sum_add_distrib, mul_sum]
 116       simp only [sum_const, card_range, nsmul_eq_mul, mul_one_div]
 117     _ ≤ h*(A+2*σ+2) + (Real.log K + 2*σ*Real.log L)*(1/Real.log 2) :=
 118       add_le_add_left (mul_le_mul_of_nonneg_left (inverse_scale_logs_sum_le h) hq) _
 119     _ = _ := by ring

 121 /-- Specialization to the real degree bounds provided by the structural
 122 decomposition. z is the expected number of occurrences per base-edge label.
 123 The result includes all ceiling errors and does not assume a log bound. -/
 124 theorem decomposition_threshold_budget (h : ℕ) (DL DR : ℕ → ℕ) (B z b L σ : ℝ)
 125     (hB : 0 < B) (hz : 0 ≤ z) (hb : 0 < b) (hL : 1 ≤ L) (hσ : 0 ≤ σ)
 126     (hK : 1 ≤ 16*B*z/b)
 127     (hDL : ∀ i < h, (DL i : ℝ) ≤ 16*B/(Peeling.scaleNat i : ℝ))
 128     (hDR : ∀ i < h, (DR i : ℝ) ≤ 16*B*((Peeling.scaleNat i : ℝ)*L)^(2*σ)) :
 129     (∑ i ∈ range h, (sufficientThreshold (Real.exp 1*z*(DL i : ℝ)) (Peeling.scaleNat i)
 130       (Real.log ((DR i : ℝ)*z/b)/Real.log (Peeling.scaleNat i)) : ℝ)) ≤
 131       h*(16*Real.exp 1*B*z+2*σ+2) +
 132         (Real.log (16*B*z/b) + 2*σ*Real.log L)/Real.log 2 := by
 133   apply sufficientThreshold_scale_budget h (fun i => Real.exp 1*z*(DL i : ℝ))
 134     (fun i => (DR i : ℝ)*z/b) (16*Real.exp 1*B*z) L σ (16*B*z/b)
 135     (fun _ _ => by positivity) (fun _ _ => by positivity) (by positivity) hL hσ hK
 136   · intro i hi
 137     have hpos : (0:ℝ) < Peeling.scaleNat i := by exact_mod_cast Peeling.scaleNat_pos i
 138     have hd := (le_div_iff₀ hpos).mp (hDL i hi)
 139     calc
 140       _ = (Real.exp 1*z)*((DL i : ℝ)*(Peeling.scaleNat i : ℝ)) := by ring
 141       _ ≤ (Real.exp 1*z)*(16*B) := mul_le_mul_of_nonneg_left hd (by positivity)
 142       _ = _ := by ring
 143   · intro i hi
 144     calc
 145       _ = (DR i : ℝ)*(z/b) := by ring
 146       _ ≤ (16*B*((Peeling.scaleNat i : ℝ)*L)^(2*σ))*(z/b) :=
 147         mul_le_mul_of_nonneg_right (hDR i hi) (div_nonneg hz hb.le)
 148       _ = _ := by ring
