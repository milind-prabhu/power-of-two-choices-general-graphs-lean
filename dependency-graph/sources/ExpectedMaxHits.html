  10 theorem exp_average_le {Ω : Type*} [Fintype Ω] [Nonempty Ω] (f : Ω → ℝ) :
  11     Real.exp (average f) ≤ average (fun ω => Real.exp (f ω)) := by
  12   have hm : (Fintype.card Ω : ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  13   have h := convexOn_exp.map_sum_le (t := (univ : Finset Ω))
  14     (w := fun _ => (Fintype.card Ω : ℝ)⁻¹) (p := f)
  15     (fun _ _ => by positivity) (by simp [hm]) (fun _ _ => Set.mem_univ _)
  16   have havg (g : Ω → ℝ) : (∑ ω, (Fintype.card Ω:ℝ)⁻¹*g ω) = average g := by
  17     simp only [average, ← mul_sum, div_eq_mul_inv]
  18     ring
  19   simp only [smul_eq_mul] at h
  20   rwa [havg, havg] at h

  24 theorem average_two_pow_hits (A : Finset E) (T : ℕ) :
  25     average (fun ω : Fin T → E => (2:ℝ)^(hitCount A ω)) =
  26       (1+(A.card:ℝ)/Fintype.card E)^T := by
  27   have hp (ω : Fin T → E) : (2:ℝ)^(hitCount A ω) =
  28       ∏ t, if ω t ∈ A then (2:ℝ) else 1 := by
  29     simp only [prod_ite, prod_const_one, mul_one, prod_const, hitCount]
  30   have hs : (∑ e : E, if e ∈ A then (2:ℝ) else 1) = Fintype.card E+A.card := by
  31     have hh (e : E) : (if e ∈ A then (2:ℝ) else 1) = 1+(if e ∈ A then 1 else 0) := by
  32       split <;> norm_num
  33     calc
  34       _ = ∑ e : E, (1+(if e ∈ A then (1:ℝ) else 0)) := sum_congr rfl (fun e _ => hh e)
  35       _ = _ := by
  36         simp only [sum_add_distrib, sum_const, card_univ, nsmul_eq_mul, mul_one, sum_boole]
  37         congr 3
  38         ext e
  39         simp
  40   simp only [average, hp]
  41   rw [← Fintype.prod_sum (fun (_ : Fin T) (e : E) => if e ∈ A then (2:ℝ) else 1)]
  42   simp only [hs, prod_const, card_univ,
  43     Fintype.card_fin, Fintype.card_fun, Nat.cast_pow, ← div_pow]
  44   have hm : (Fintype.card E:ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  45   congr 1
  46   field_simp

  48 theorem average_two_pow_hits_le (A : Finset E) (T : ℕ) (μ : ℝ)
  49     (hmean : (T:ℝ)*((A.card:ℝ)/Fintype.card E) ≤ μ) :
  50     average (fun ω : Fin T → E => (2:ℝ)^(hitCount A ω)) ≤ Real.exp μ := by
  51   rw [average_two_pow_hits]
  52   calc
  53     _ ≤ (Real.exp ((A.card:ℝ)/Fintype.card E))^T :=
  54       pow_le_pow_left₀ (by positivity) (by simpa only [add_comm] using Real.add_one_le_exp ((A.card:ℝ)/Fintype.card E)) T
  55     _ = Real.exp ((T:ℝ)*((A.card:ℝ)/Fintype.card E)) := (Real.exp_nat_mul _ T).symm
  56     _ ≤ _ := Real.exp_le_exp.mpr hmean

  58 /-- Every bucket may overlap other buckets; only its marginal mean matters. -/
  59 theorem expected_max_hits {I : Type*} [Fintype I] [Nonempty I]
  60     (bucket : I → Finset E) (T : ℕ) (μ : ℝ)
  61     (hmean : ∀ i, (T:ℝ)*(((bucket i).card:ℝ)/Fintype.card E) ≤ μ) :
  62     average (fun ω : Fin T → E => ((univ.sup fun i => hitCount (bucket i) ω : ℕ):ℝ)) ≤
  63       2*(Real.log (Fintype.card I:ℝ)+μ) := by
  64   classical
  65   let X := fun ω : Fin T → E => ((univ.sup fun i => hitCount (bucket i) ω : ℕ):ℝ)
  66   have hpoint (ω : Fin T → E) : Real.exp (Real.log 2*X ω) ≤
  67       ∑ i, (2:ℝ)^(hitCount (bucket i) ω) := by
  68     obtain ⟨i,_,hi⟩ := exists_mem_eq_sup (s := (univ : Finset I)) (by simp) (fun i => hitCount (bucket i) ω)
  69     have he : Real.exp (Real.log 2*X ω) = (2:ℝ)^(hitCount (bucket i) ω) := by
  70       dsimp only [X]
  71       rw [hi, mul_comm, Real.exp_nat_mul, Real.exp_log (by norm_num)]
  72     rw [he]
  73     exact single_le_sum (f := fun i => (2:ℝ)^(hitCount (bucket i) ω)) (fun _ _ => by positivity) (mem_univ i)
  74   have hupper : average (fun ω => Real.exp (Real.log 2*X ω)) ≤
  75       (Fintype.card I:ℝ)*Real.exp μ := by
  76     apply (average_mono hpoint).trans
  77     rw [average_sum]
  78     simpa using sum_le_sum (fun i (_ : i ∈ (univ : Finset I)) => average_two_pow_hits_le (bucket i) T μ (hmean i))
  79   have hj := exp_average_le (fun ω => Real.log 2*X ω)
  80   rw [average_mul] at hj
  81   have hexp := hj.trans hupper
  82   have hcard : (0:ℝ) < Fintype.card I := by exact_mod_cast Fintype.card_pos
  83   rw [← Real.exp_log hcard, ← Real.exp_add, Real.exp_le_exp] at hexp
  84   have hX' : 0 ≤ average X := by unfold average X; positivity
  85   have hl2 : (1:ℝ)/2 ≤ Real.log 2 := by linarith [Real.log_two_gt_d9]
  86   change average X ≤ _
  87   nlinarith [mul_le_mul_of_nonneg_right hl2 hX']
