   4 /-! Turn a tail for actual residual component sizes into an expected load
   5 bound. Exceptional samples cost at most T times their probability. This
   6 module does not assume or claim a proved component-shattering estimate. -/
   7 namespace GraphChoices
   8 open Finset

  10 theorem average_add {Ω : Type*} [Fintype Ω] (f g : Ω → ℝ) :
  11     average (fun ω => f ω + g ω) = average f + average g := by
  12   simp only [average, sum_add_distrib, add_div]
  13 
  14 theorem average_indicator {Ω : Type*} [Fintype Ω] [DecidableEq Ω]
  15     (A : Finset Ω) (M : ℝ) :
  16     average (fun ω => if ω ∈ A then M else 0) = M * eventProbability A := by
  17   simp [average, eventProbability, mul_div_assoc, mul_comm]
  18 
  19 /-- A finite expectation bound with an explicit additive exceptional-sample cost. -/
  20 theorem average_le_good_add_bad {Ω : Type*} [Fintype Ω] [DecidableEq Ω]
  21     (X B : Ω → ℝ) (A : Finset Ω) (M : ℝ)
  22     (hgood : ∀ ω, ω ∉ A → X ω ≤ B ω)
  23     (hbad : ∀ ω, ω ∈ A → X ω ≤ B ω + M) :
  24     average X ≤ average B + M * eventProbability A := by
  25   calc
  26     _ ≤ average (fun ω => B ω + if ω ∈ A then M else 0) := by
  27       apply average_mono
  28       intro ω
  29       split_ifs with h
  30       · exact hbad ω h
  31       · simpa using hgood ω h
  32     _ = _ := by rw [average_add, average_indicator]
  33 
  34 end GraphChoices

  37 open Finset
  38 variable {E L R C : Type*} [Fintype E] [DecidableEq E] [Nonempty E]
  39   [Fintype L] [Fintype R] [Fintype C]
  40   [DecidableEq L] [DecidableEq R] [DecidableEq C]

  42 /-- The actual large-component event, rather than a canonical pattern event. -/
  43 noncomputable def largeComponentEvent (encode : E → Arrival L R C) (α : C → ℕ)
  44     (T s : ℕ) : Finset (Fin T → E) := by
  45   classical
  46   exact univ.filter fun ω => ∃ v,
  47     s < (componentVertices (residualInstance α (sampleArrivals encode ω)) v).card
  48 
  49 omit [DecidableEq E] [Nonempty E] [Fintype C] in
  50 theorem not_mem_largeComponentEvent (encode : E → Arrival L R C) (α : C → ℕ)
  51     (T s : ℕ) (ω : Fin T → E) :
  52     ω ∉ largeComponentEvent encode α T s ↔
  53       ∀ v, (componentVertices (residualInstance α (sampleArrivals encode ω)) v).card ≤ s := by
  54   classical
  55   simp [largeComponentEvent]
  56 
  57 /-- Expected offline optimum of the full encoded arrival sequence. -/
  58 noncomputable def expectedReplayOpt (encode : E → Arrival L R C) (T : ℕ) : ℝ :=
  59   average (fun ω : Fin T → E => (opt (replayInstance (sampleArrivals encode ω)) : ℝ))
  60 
  61 noncomputable def expectedTotalMaxLoad (encode : E → Arrival L R C) (α : C → ℕ) (T : ℕ) : ℝ :=
  62   average (fun ω : Fin T → E => (totalMaxLoad α (sampleArrivals encode ω) : ℝ))
  63 
  64 /-- The expectation step of Theorem 4.1. Its error is T times the probability
  65 of an actual large residual component; that probability is not assumed small. -/
  66 theorem expected_load_le_component_tail (encode : E → Arrival L R C) (α : C → ℕ)
  67     (T s : ℕ) :
  68     expectedTotalMaxLoad encode α T ≤ (∑ c, (α c : ℝ)) +
  69       (2 * (Nat.log 2 s + 1) : ℕ) * expectedReplayOpt encode T +
  70       T * eventProbability (largeComponentEvent encode α T s) := by
  71   let B : (Fin T → E) → ℝ := fun ω => (∑ c, (α c : ℝ)) +
  72     (2 * (Nat.log 2 s + 1) : ℕ) * (opt (replayInstance (sampleArrivals encode ω)) : ℝ)
  73   have hb : ∀ ω : Fin T → E, (∑ c, (α c : ℝ)) ≤ B ω := by
  74     intro ω
  75     dsimp [B]
  76     exact le_add_of_nonneg_right (by positivity)
  77   have h := average_le_good_add_bad
  78     (fun ω : Fin T → E => (totalMaxLoad α (sampleArrivals encode ω) : ℝ))
  79     B (largeComponentEvent encode α T s) (T : ℝ) ?_ ?_
  80   · change expectedTotalMaxLoad encode α T ≤ _ at h
  81     have heq : average B = (∑ c, (α c : ℝ)) +
  82         (2 * (Nat.log 2 s + 1) : ℕ) * expectedReplayOpt encode T := by
  83       dsimp only [B]
  84       rw [average_add, average_const, average_mul]
  85       rfl
  86     rwa [heq] at h
  87   · intro ω hω
  88     have hsmall := (not_mem_largeComponentEvent encode α T s ω).mp hω
  89     have hh := totalMaxLoad_of_small_components α (sampleArrivals encode ω) s hsmall
  90     have hh' : (totalMaxLoad α (sampleArrivals encode ω) : ℝ) ≤
  91         (∑ c, (α c : ℝ)) +
  92           (2 * opt (replayInstance (sampleArrivals encode ω)) * (Nat.log 2 s + 1) : ℕ) := by
  93       exact_mod_cast hh
  94     dsimp [B]
  95     convert hh' using 1 <;> push_cast <;> ring
  96   · intro ω _
  97     have hh := totalMaxLoad_le_budget_add_length α (sampleArrivals encode ω)
  98     rw [sampleArrivals_length] at hh
  99     have hh' : (totalMaxLoad α (sampleArrivals encode ω) : ℝ) ≤ (∑ c, (α c : ℝ)) + T := by
 100       exact_mod_cast hh
 101     exact hh'.trans (add_le_add_right (hb ω) _)

 103 theorem expected_load_of_component_tail (encode : E → Arrival L R C) (α : C → ℕ)
 104     (T s : ℕ) (ε : ℝ)
 105     (htail : eventProbability (largeComponentEvent encode α T s) ≤ ε) :
 106     expectedTotalMaxLoad encode α T ≤ (∑ c, (α c : ℝ)) +
 107       (2 * (Nat.log 2 s + 1) : ℕ) * expectedReplayOpt encode T + T * ε := by
 108   apply (expected_load_le_component_tail encode α T s).trans
 109   exact add_le_add_left (mul_le_mul_of_nonneg_left htail (Nat.cast_nonneg T)) _
 110 
 111 end GraphChoices.ThresholdGreedy
