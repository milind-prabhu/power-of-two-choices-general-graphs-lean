   6 /-! Rooted class-labelled patterns in parent-before-child order, and a bound on
   7 their graph assignments. Inconsistent bipartite side annotations simply have
   8 no assignments. Injective embeddings form a subset of these assignments. -/
   9 namespace GraphChoices.Patterns
  10 open Finset
  11 variable {L R C : Type*} [Fintype L] [Fintype R]
  12   [DecidableEq L] [DecidableEq R] [DecidableEq C]
  13 
  14 structure ParentPattern (C : Type*) (m : ℕ) where
  15   parent : (i : Fin m) → Fin (i.val+1)
  16   cls : Fin m → C
  17   childLeft : Fin m → Bool
  18 
  19 noncomputable def patternMenu {m : ℕ} (p : ParentPattern C m)
  20     (H : C → Peeling.Graph L R) : (k : ℕ) → (Fin k → L ⊕ R) → Finset (L ⊕ R)
  21   | 0, _ => univ.image Sum.inr
  22   | k+1, f => if hk : k < m then
  23       let i : Fin m := ⟨k,hk⟩
  24       if p.childLeft i then
  25         match f (p.parent i) with
  26         | Sum.inl _ => ∅
  27         | Sum.inr v => (Peeling.rightNeighbors (H (p.cls i)) v).image Sum.inl
  28       else
  29         match f (p.parent i) with
  30         | Sum.inl u => (Peeling.leftNeighbors (H (p.cls i)) u).image Sum.inr
  31         | Sum.inr _ => ∅
  32     else ∅
  33 
  34 def branchBound {m : ℕ} (p : ParentPattern C m) (D : ℕ) (DR : C → ℕ) : ℕ → ℕ
  35   | 0 => Fintype.card R
  36   | k+1 => if hk : k < m then if p.childLeft ⟨k,hk⟩ then DR (p.cls ⟨k,hk⟩) else D else 0
  37 
  38  theorem patternMenu_card_le {m : ℕ} (p : ParentPattern C m) (H : C → Peeling.Graph L R)
  39     (D : ℕ) (DR : C → ℕ)
  40     (hL : ∀ c u, Peeling.leftDegree (H c) u ≤ D)
  41     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c)
  42     (k : ℕ) (f : Fin k → L ⊕ R) :
  43     (patternMenu p H k f).card ≤ branchBound (R := R) p D DR k := by
  44   cases k with
  45   | zero =>
  46     simpa only [patternMenu, branchBound, card_univ] using
  47       (card_image_le (s := (univ : Finset R)) (f := (Sum.inr : R → L ⊕ R)))
  48   | succ k =>
  49     by_cases hk : k < m
  50     · cases hs : p.childLeft ⟨k,hk⟩
  51       · cases hv : f (p.parent ⟨k,hk⟩) with
  52         | inl u =>
  53           simp only [patternMenu, branchBound, dif_pos hk, hs, Bool.false_eq_true, if_false, hv]
  54           exact card_image_le.trans (hL _ _)
  55         | inr v => simp [patternMenu, branchBound, hk, hs, hv]
  56       · cases hv : f (p.parent ⟨k,hk⟩) with
  57         | inl u => simp [patternMenu, branchBound, hk, hs, hv]
  58         | inr v =>
  59           simp only [patternMenu, branchBound, dif_pos hk, hs, if_true, hv]
  60           exact card_image_le.trans (hR _ _)
  61     · simp [patternMenu, branchBound, hk]
  62 
  63 noncomputable def patternAssignments {m : ℕ} (p : ParentPattern C m) (H : C → Peeling.Graph L R) :
  64     Finset (Fin (m+1) → L ⊕ R) := GraphChoices.menuAssignments (patternMenu p H) (m+1)
  65 
  66 /-- The fixed-pattern embedding count underlying Equation (8), before grouping
  67 left vertices with equal edge classes. The right root has at most |R| choices. -/
  68  theorem patternAssignments_card_le {m : ℕ} (p : ParentPattern C m) (H : C → Peeling.Graph L R)
  69     (D : ℕ) (DR : C → ℕ)
  70     (hL : ∀ c u, Peeling.leftDegree (H c) u ≤ D)
  71     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c) :
  72     (patternAssignments p H).card ≤ Fintype.card R *
  73       ∏ i : Fin m, if p.childLeft i then DR (p.cls i) else D := by
  74   have hh := GraphChoices.menuAssignments_card_le (patternMenu p H)
  75     (branchBound (R := R) p D DR) (m+1) (fun k _ f => patternMenu_card_le p H D DR hL hR k f)
  76   unfold patternAssignments
  77   apply hh.trans_eq
  78   rw [prod_range_succ']
  79   rw [← Fin.prod_univ_eq_prod_range (fun k => branchBound (R := R) p D DR (k+1)) m]
  80   simp only [branchBound, Fin.isLt, dif_pos]
  81   rw [mul_comm]
  82 
  83 end GraphChoices.Patterns
