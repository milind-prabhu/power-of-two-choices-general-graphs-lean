  11 noncomputable def localLogSkewness (a b d D n : ℝ) : ℝ :=
  12   Real.log (a/b) / Real.log (D * (Real.log n)^2 / d)

  14  theorem log_ratio_bound_iff {a b q σ : ℝ} (ha : 0 < a) (hb : 0 < b) (hq : 1 < q) :
  15     Real.log (a/b) / Real.log q ≤ σ ↔ a ≤ b * q^σ := by
  16   rw [div_le_iff₀ (Real.log_pos hq)]
  17   rw [mul_comm σ, Real.log_le_iff_le_exp (div_pos ha hb)]
  18   rw [← Real.rpow_def_of_pos (lt_trans zero_lt_one hq)]
  19   rw [div_le_iff₀ hb, mul_comm]

  21 /-- A local skewness bound implies the neighborhood expansion inequality.
  22 The parameter q is the denominator's argument in Definition 3.2. -/
  23  theorem expansion_of_log_skew {a b q Q σ : ℝ}
  24     (ha : 0 < a) (hb : 0 < b) (hq : 1 < q) (hQ : q ≤ Q) (hσ : 0 ≤ σ)
  25     (hskew : Real.log (a/b) / Real.log q ≤ σ) :
  26     a ≤ b * Q^σ := by
  27   exact ((log_ratio_bound_iff ha hb hq).mp hskew).trans
  28     (mul_le_mul_of_nonneg_left (Real.rpow_le_rpow (by linarith) hQ hσ) (le_of_lt hb))

  30 /-- The exact scale substitution used for b_i in Lemma 3.7. -/
  31  theorem expansion_at_scale {a b d D n f σ : ℝ}
  32     (ha : 0 < a) (hb : 0 < b) (hd : 0 < d)
  33     (hf : 0 < f) (hn : 0 < Real.log n) (hσ : 0 ≤ σ)
  34     (hdegree : D / f^2 ≤ d)
  35     (hdenom : 1 < D * (Real.log n)^2 / d)
  36     (hskew : localLogSkewness a b d D n ≤ σ) :
  37     a ≤ b * (f * Real.log n)^(2*σ) := by
  38   have hf2 : 0 < f^2 := sq_pos_of_pos hf
  39   have hDle : D ≤ d * f^2 := (div_le_iff₀ hf2).mp hdegree
  40   have hq : D * (Real.log n)^2 / d ≤ (f * Real.log n)^2 := by
  41     apply (div_le_iff₀ hd).mpr
  42     nlinarith [mul_le_mul_of_nonneg_right hDle (sq_nonneg (Real.log n))]
  43   have hh := expansion_of_log_skew ha hb hdenom hq hσ hskew
  44   have hp : 0 ≤ f * Real.log n := le_of_lt (mul_pos hf hn)
  45   have heq : ((f * Real.log n)^2 : ℝ)^σ = (f * Real.log n)^(2*σ) := by
  46     rw [← Real.rpow_natCast, ← Real.rpow_mul hp]
  47     norm_num
  48   simpa [heq] using hh
