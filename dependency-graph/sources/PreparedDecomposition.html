   5 /-! The structural partition as an operational input to Threshold-Greedy.
   6 The stored bounds are derived from the proved decomposition, and the source
   7 edge labels, classes and degree certificates all use that exact partition. -/

  14 structure PreparedDecomposition (G : Graph L R) (B N : ℕ) (d : ℝ) where
  15   pieces : List (Graph L R)
  16   count : pieces.length = numberRounds B
  17   disjoint : pieces.Pairwise Disjoint
  18   cover : unionPieces pieces = G
  19   positive_average : 0 < d
  20   large_log : 1 < Real.log (N:ℝ)
  21   average_cap : d ≤ 16*B
  22   global_cap : ∀ u, leftDegree G u ≤ 4*B
  23   preprocessing : ∀ u, (leftDegree G u : ℝ) ≤ d*Real.log (N:ℝ)
  24   left_bound : ∀ i : Fin pieces.length,
  25     (maxLeftDegree (pieces.get i) : ℝ) ≤ 16*B/(scaleNat i.val : ℝ)
  26   right_bound : ∀ i : Fin pieces.length,
  27     (maxRightDegree (pieces.get i) : ℝ) ≤
  28       16*B*((scaleNat i.val : ℝ)*Real.log (N:ℝ))^(2*graphLogSkewness G d (N:ℝ))
  29 
  30 theorem exists_preparedDecomposition (G : Graph L R) (B N : ℕ) (d : ℝ)
  31     (hd : 0 < d) (hN : 1 < Real.log (N:ℝ))
  32     (hleft : ∀ u, (leftDegree G u : ℝ) ≤ d*Real.log (N:ℝ))
  33     (hB : d ≤ 16*B) (hD : ∀ u, leftDegree G u ≤ 4*B) :
  34     Nonempty (PreparedDecomposition G B N d) := by
  35   obtain ⟨Ps, hlen, hdisj, hcover, hbounds⟩ := exists_skew_biregular_partition G B d (N:ℝ) hd hN hleft hB hD
  36   refine ⟨⟨Ps, hlen, hdisj, hcover, hd, hN, hB, hD, hleft, ?_, ?_⟩⟩
  37   · intro i
  38     exact maxLeftDegree_real_le _ _ (by positivity) (hbounds i).1
  39   · intro i
  40     exact maxRightDegree_real_le _ _ (by positivity) (hbounds i).2

  44 noncomputable def PreparedDecomposition.encode (P : PreparedDecomposition G B N d) :
  45     G → Arrival L R (Fin P.pieces.length) := pieceEncode G P.pieces P.cover
  46 
  47 def PreparedDecomposition.DL (P : PreparedDecomposition G B N d) (i : Fin P.pieces.length) : ℕ :=
  48   maxLeftDegree (P.pieces.get i)
  49 
  50 def PreparedDecomposition.DR (P : PreparedDecomposition G B N d) (i : Fin P.pieces.length) : ℕ :=
  51   maxRightDegree (P.pieces.get i)
  52 
  53 noncomputable def PreparedDecomposition.thresholds (P : PreparedDecomposition G B N d) (T : ℕ) :
  54     Fin P.pieces.length → ℕ :=
  55   automaticThresholds (E := G) P.DL P.DR (4*B) T (fun i => scaleNat i.val)
  56 
  57 def graphBase (G : Graph L R) : BaseGraph (L ⊕ R) G where
  58   left e := Sum.inl e.val.1
  59   right e := Sum.inr e.val.2
  60 
  61 noncomputable def PreparedDecomposition.policy (P : PreparedDecomposition G B N d) (T : ℕ) : OnlinePolicy G :=
  62   thresholdPolicy P.encode (P.thresholds T)
  63 
  64 noncomputable def PreparedDecomposition.budget (P : PreparedDecomposition G B N d) (T : ℕ) : ℝ :=
  65   P.pieces.length*(16*Real.exp 1*B*((T:ℝ)/Fintype.card G)+2*graphLogSkewness G d (N:ℝ)+2) +
  66     (Real.log (16*B*((T:ℝ)/Fintype.card G)/branchTarget (4*B) P.pieces.length ((T:ℝ)/Fintype.card G)) +
  67       2*graphLogSkewness G d (N:ℝ)*Real.log (Real.log (N:ℝ)))/Real.log 2

  69 theorem PreparedDecomposition.source (P : PreparedDecomposition G B N d) :
  70     ∀ e, ((P.encode e).left, (P.encode e).right) ∈ P.pieces.get (P.encode e).cls :=

  73 theorem PreparedDecomposition.simple (P : PreparedDecomposition G B N d) :
  74     Function.Injective fun e => ((P.encode e).left, (P.encode e).right) :=

  77 theorem PreparedDecomposition.bucket_bound (P : PreparedDecomposition G B N d) (u : L) (i : Fin P.pieces.length) :
  78     (sourceBucket P.encode u i).card ≤ P.DL i :=
  79   (piece_sourceBucket_card_le G P.pieces P.cover u i).trans (leftDegree_le_max _ _)
  80 
  81 theorem PreparedDecomposition.global_bucket_bound (P : PreparedDecomposition G B N d) (u : L) :
  82     (leftSourceBucket P.encode u).card ≤ maxLeftDegree G :=
  83   (piece_leftSourceBucket_card_le G P.pieces P.cover u).trans (leftDegree_le_max _ _)
  84 
  85 theorem PreparedDecomposition.piece_left_cap (P : PreparedDecomposition G B N d) (i : Fin P.pieces.length) (u : L) :
  86     leftDegree (P.pieces.get i) u ≤ 4*B :=
  87   (leftDegree_le_max _ _).trans ((piece_maxLeftDegree_le G P.pieces P.cover i).trans (maxLeftDegree_le G _ P.global_cap))
  88 
  89 theorem PreparedDecomposition.piece_right_cap (P : PreparedDecomposition G B N d) (i : Fin P.pieces.length) (v : R) :
  90     rightDegree (P.pieces.get i) v ≤ P.DR i := rightDegree_le_max _ _
  91 
  92 theorem PreparedDecomposition.max_degree_preprocessing (P : PreparedDecomposition G B N d) :
  93     (maxLeftDegree G : ℝ) ≤ d*Real.log (N:ℝ) :=
  94   maxLeftDegree_real_le G _ (mul_nonneg P.positive_average.le (by linarith [P.large_log])) P.preprocessing
  95 
  96 theorem PreparedDecomposition.vertex_count_pos (P : PreparedDecomposition G B N d) : 0 < N := by
  97   have h := P.large_log
  98   by_contra hn
  99   have hz : N = 0 := by omega
 100   norm_num [hz] at h

 102 theorem PreparedDecomposition.scale_gt_one (P : PreparedDecomposition G B N d) (i : Fin P.pieces.length) :
 103     (1:ℝ) < scaleNat i.val := by
 104   have hh := scaleNat_ge_two i.val
 105   exact_mod_cast (by omega : 1 < scaleNat i.val)

 109 theorem PreparedDecomposition.class_count_pos (P : PreparedDecomposition G B N d) : 0 < P.pieces.length := by

 112 theorem PreparedDecomposition.reference_cap_pos (P : PreparedDecomposition G B N d) : 0 < B := by
 113   classical
 114   let e : G := Classical.choice (inferInstance : Nonempty G)
 115   have hmem : e.val.2 ∈ leftNeighbors G e.val.1 := by simpa [leftNeighbors] using e.property
 116   have hp : 0 < leftDegree G e.val.1 := card_pos.mpr ⟨e.val.2, hmem⟩
 117   have hc := P.global_cap e.val.1
 118   omega

 120 theorem PreparedDecomposition.threshold_budget (P : PreparedDecomposition G B N d) (T : ℕ)
 121     (hmass : 1 ≤ 16*(B:ℝ)*((T:ℝ)/Fintype.card G)) :
 122     (∑ i, (P.thresholds T i : ℝ)) ≤ P.budget T := by
 123   exact automaticThresholds_budget P.pieces.length (4*B) T P.DL P.DR (B:ℝ)
 124     (Real.log (N:ℝ)) (graphLogSkewness G d (N:ℝ)) P.class_count_pos
 125     (by exact_mod_cast P.reference_cap_pos) P.large_log.le (graphLogSkewness_nonneg G d (N:ℝ))
 126     hmass P.left_bound P.right_bound

 128 theorem PreparedDecomposition.mass_at_vertex_horizon (P : PreparedDecomposition G B N d)
 129     (haverage : 2*(G.card : ℝ) = (N:ℝ)*d) :
 130     1 ≤ 16*(B:ℝ)*((N:ℝ)/Fintype.card G) := by
 131   have hE : (0:ℝ) < Fintype.card G := by exact_mod_cast Fintype.card_pos
 132   rw [← mul_div_assoc]
 133   apply (le_div_iff₀ hE).mpr
 134   have hp := mul_le_mul_of_nonneg_left P.average_cap (Nat.cast_nonneg N : (0:ℝ) ≤ N)
 135   have heq : (Fintype.card G : ℝ) = G.card := by rw [Fintype.card_coe]
 136   rw [heq]
 137   nlinarith
