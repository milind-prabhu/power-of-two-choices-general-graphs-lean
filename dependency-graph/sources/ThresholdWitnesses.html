   3 /-! Distinct, earlier threshold occurrences are necessary witnesses for every
   4 residual arrival. Their time labels are disjoint from the entire residual log,
   5 so they cannot be reused as selected residual-tree occurrences. -/
   6 namespace GraphChoices.ThresholdGreedy
   7 variable {L R C : Type*} [DecidableEq L] [DecidableEq R] [DecidableEq C]
   8 
   9 def thresholdBucket (log : List (Event L R C)) (u : L) (i : C) : List (Event L R C) :=
  10   log.filter fun ev => ev.threshold && decide (ev.arrival.left = u ∧ ev.arrival.cls = i)
  11 
  12 def thresholdWitnessTimes (α : C → ℕ) (es : List (Arrival L R C)) (u : L) (i : C) : List ℕ :=
  13   (thresholdBucket (trace α es) u i).map Event.time
  14 
  15  theorem thresholdBucket_length (log : List (Event L R C)) (u : L) (i : C) :
  16     (thresholdBucket log u i).length = incidentCount (thresholdArrivals log) u i := by
  17   simp [thresholdBucket, incidentCount, thresholdArrivals, List.filter_map, List.filter_filter, Bool.and_comm]
  18 
  19  theorem thresholdWitnessTimes_length (α : C → ℕ) (es : List (Arrival L R C)) (u : L) (i : C) :
  20     (thresholdWitnessTimes α es u i).length = (run α es).threshold u i := by
  21   simp only [thresholdWitnessTimes, List.length_map, thresholdBucket_length, threshold_counter_exact]
  22 
  23  theorem thresholdWitnessTimes_nodup (α : C → ℕ) (es : List (Arrival L R C)) (u : L) (i : C) :
  24     (thresholdWitnessTimes α es u i).Nodup := by
  25   apply List.Nodup.sublist (List.Sublist.map Event.time (List.filter_sublist))
  26   exact traceFrom_unique_times α initial 0 es
  27 
  28  theorem trace_prefix_mem (α : C → ℕ) (xs ys : List (Arrival L R C))
  29     (ev : Event L R C) (hev : ev ∈ trace α xs) : ev ∈ trace α (xs ++ ys) := by
  30   unfold trace
  31   rw [traceFrom_append]
  32   exact List.mem_append_left _ hev
  33 
  34  theorem thresholdWitnessTimes_spec (α : C → ℕ) (es : List (Arrival L R C)) (u : L) (i : C)
  35     (t : ℕ) (ht : t ∈ thresholdWitnessTimes α es u i) :
  36     ∃ ev ∈ trace α es, ev.time = t ∧ ev.threshold = true ∧ ev.arrival.left = u ∧ ev.arrival.cls = i := by
  37   obtain ⟨ev,hev,hetime⟩ := List.mem_map.mp ht
  38   have h := List.mem_filter.mp hev
  39   have hh : ev.threshold = true ∧ (ev.arrival.left = u ∧ ev.arrival.cls = i) := by
  40     simpa using h.2
  41   exact ⟨ev,h.1,hetime,hh.1,hh.2⟩
  42 
  43  theorem traceFrom_time_lt (α : C → ℕ) (s : State L R C) (k : ℕ)
  44     (es : List (Arrival L R C)) (ev : Event L R C) (hev : ev ∈ traceFrom α s k es) :
  45     ev.time < k + es.length := by
  46   induction es generalizing s k with
  47   | nil => simp [traceFrom] at hev
  48   | cons e es ih =>
  49     rcases List.mem_cons.mp hev with rfl | hev
  50     · simp
  51     · have ht := ih (step α s e) (k+1) hev
  52       simp only [List.length_cons]
  53       omega
  54 
  55  theorem thresholdWitnessTimes_earlier (α : C → ℕ) (es : List (Arrival L R C)) (u : L) (i : C)
  56     (t : ℕ) (ht : t ∈ thresholdWitnessTimes α es u i) : t < es.length := by
  57   obtain ⟨ev,hev,hetime,_⟩ := thresholdWitnessTimes_spec α es u i t ht
  58   have h := traceFrom_time_lt α initial 0 es ev hev
  59   simpa [hetime] using h
  60 
  61  theorem thresholdWitnessTimes_disjoint_residual (α : C → ℕ) (xs ys : List (Arrival L R C))
  62     (u : L) (i : C) :
  63     List.Disjoint (thresholdWitnessTimes α xs u i)
  64       (((trace α (xs ++ ys)).filter fun ev => !ev.threshold).map Event.time) := by
  65   apply List.disjoint_left.mpr
  66   intro t ht hr
  67   obtain ⟨ev,hev,hetime,heth,_,_⟩ := thresholdWitnessTimes_spec α xs u i t ht
  68   have hfull := trace_prefix_mem α xs ys ev hev
  69   have htfull : t ∈ (((trace α (xs ++ ys)).filter fun ev => ev.threshold).map Event.time) :=
  70     List.mem_map.mpr ⟨ev,List.mem_filter.mpr ⟨hfull,heth⟩,hetime⟩
  71   exact List.disjoint_left.mp (trace_threshold_residual_disjoint α (xs ++ ys)) htfull hr
  72 
  73 /-- A residual class-i arrival forces alpha_i distinct earlier threshold
  74 occurrences in its bucket, disjoint from every residual occurrence in the full
  75 execution. The suffix is arbitrary and can include the current arrival. -/
  76  theorem residual_distinct_threshold_witnesses (α : C → ℕ) (xs ys : List (Arrival L R C))
  77     (e : Arrival L R C) (he : isThreshold α (run α xs) e = false) :
  78     ∃ W : List ℕ, W.Nodup ∧ W.length = α e.cls ∧
  79       (∀ t ∈ W, t < xs.length) ∧
  80       List.Disjoint W (((trace α (xs ++ ys)).filter fun ev => !ev.threshold).map Event.time) ∧
  81       ∀ t ∈ W, ∃ ev ∈ trace α xs,
  82         ev.time = t ∧ ev.threshold = true ∧ ev.arrival.left = e.left ∧ ev.arrival.cls = e.cls := by
  83   refine ⟨thresholdWitnessTimes α xs e.left e.cls,
  84     thresholdWitnessTimes_nodup α xs _ _, ?_, ?_, ?_, ?_⟩
  85   · rw [thresholdWitnessTimes_length]
  86     exact residual_counter_saturated α xs e he
  87   · exact thresholdWitnessTimes_earlier α xs _ _
  88   · exact thresholdWitnessTimes_disjoint_residual α xs ys _ _
  89   · exact thresholdWitnessTimes_spec α xs _ _
  90 
  91 end GraphChoices.ThresholdGreedy
