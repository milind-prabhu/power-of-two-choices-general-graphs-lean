   5 /-- The count of vertex subsets cancels against the binomial density tail.
   6 The remaining small-side factor makes the bound polynomially small in m. -/
   7 theorem block_size_pair_bound (m t a b k : ℕ) (hm : 0 < m) (ht : 1 ≤ t)
   8     (ha : 0 < a) (hb : 0 < b) (ham : a ≤ m) (hbtm : b ≤ t*m) (hk : 16 ≤ k)
   9     (hscale : Real.exp 1*(t:ℝ)*(2*Real.exp 1/(k:ℝ))^k ≤ 1/256) :
  10     (m.choose a:ℝ)*((t*m).choose b:ℝ)*
  11       (2*Real.exp 1*(a:ℝ)*b/((m:ℝ)*k*(a+b)))^(k*(a+b)) ≤ 1/(m:ℝ)^8 := by
  12   let z := min a b
  13   let n := a+b
  14   have hz : 0 < z := lt_min ha hb
  15   have hzm : z ≤ m := (min_le_left _ _).trans ham
  16   have hzn : z ≤ n := (min_le_left a b).trans (by dsimp [n]; omega)
  17   have hmr : (0:ℝ) < m := by exact_mod_cast hm
  18   have hzr : (0:ℝ) < z := by exact_mod_cast hz
  19   have hkr : (0:ℝ) < k := by exact_mod_cast (show 0 < k by omega)
  20   have har : (0:ℝ) < a := by exact_mod_cast ha
  21   have hbr : (0:ℝ) < b := by exact_mod_cast hb
  22   have htr : (1:ℝ) ≤ t := by exact_mod_cast ht
  23   have hnr : (0:ℝ) < n := by dsimp [n]; positivity
  24   let C := Real.exp 1*t*m/z
  25   let D := 2*Real.exp 1*z/((m:ℝ)*k)
  26   have hC : 0 ≤ C := by dsimp [C]; positivity
  27   have hD : 0 ≤ D := by dsimp [D]; positivity
  28   have hca : (m.choose a:ℝ) ≤ C^a := by
  29     apply (choose_le_exp_div_pow m a).trans
  30     apply pow_le_pow_left₀ (by positivity)
  31     dsimp [C]
  32     have hza : (z:ℝ) ≤ a := by dsimp only [z]; exact_mod_cast min_le_left a b
  33     calc
  34       _ ≤ Real.exp 1*m/z := div_le_div_of_nonneg_left (by positivity) hzr hza
  35       _ ≤ _ := by
  36         have hh := mul_le_mul_of_nonneg_right htr (show 0 ≤ Real.exp 1*m/z by positivity)
  37         convert hh using 1 <;> ring
  38   have hcb : ((t*m).choose b:ℝ) ≤ C^b := by
  39     apply (choose_le_exp_div_pow (t*m) b).trans
  40     apply pow_le_pow_left₀ (by positivity)
  41     dsimp [C]
  42     push_cast
  43     have hzb : (z:ℝ) ≤ b := by dsimp only [z]; exact_mod_cast min_le_right a b
  44     convert div_le_div_of_nonneg_left (show 0 ≤ Real.exp 1*(t:ℝ)*m by positivity) hzr hzb using 1 <;> ring
  45   have htail : 2*Real.exp 1*(a:ℝ)*b/((m:ℝ)*k*(a+b)) ≤ D := by
  46     dsimp [D]
  47     apply (div_le_div_iff₀ (by positivity : (0:ℝ)<(m:ℝ)*k*(a+b)) (by positivity : (0:ℝ)<(m:ℝ)*k)).mpr
  48     have hab : (a:ℝ)*b ≤ (z:ℝ)*(a+b) := by
  49       by_cases hh : a ≤ b
  50       · have he : z = a := min_eq_left hh
  51         rw [he]
  52         nlinarith
  53       · have he : z = b := min_eq_right (le_of_not_ge hh)
  54         rw [he]
  55         nlinarith
  56     nlinarith [mul_nonneg (by positivity : 0 ≤ 2*Real.exp 1*m*k) (sub_nonneg.mpr hab)]
  57   have hfactor : C*D^k = (Real.exp 1*(t:ℝ)*(2*Real.exp 1/(k:ℝ))^k)*((z:ℝ)/m)^(k-1) := by
  58     have hk' : k = (k-1)+1 := by omega
  59     have heq : D = (2*Real.exp 1/(k:ℝ))*((z:ℝ)/m) := by dsimp [D]; ring
  60     have hzpow : ((z:ℝ)/m)^k = ((z:ℝ)/m)^(k-1)*((z:ℝ)/m) := by
  61       conv_lhs => rw [hk']
  62       rw [pow_succ]
  63     rw [heq, mul_pow, hzpow]
  64     dsimp [C]
  65     field_simp <;> ring
  66   have hpow : ((z:ℝ)/m)^((k-1)*n) ≤ ((z:ℝ)/m)^8 := by
  67     apply pow_le_pow_of_le_one (by positivity) ((div_le_one hmr).mpr (by exact_mod_cast hzm))
  68     have hn : 2 ≤ n := by dsimp [n]; omega
  69     have hkm : 15 ≤ k-1 := by omega
  70     nlinarith
  71   have hz8 : (z:ℝ)^8 ≤ (256:ℝ)^n := by
  72     have hz2 : z ≤ 2^z := (Nat.lt_two_pow_self : z < 2^z).le
  73     have hh : z^8 ≤ 256^n := by
  74       calc
  75         _ ≤ (2^z)^8 := Nat.pow_le_pow_left hz2 _
  76         _ = 256^z := by rw [← pow_mul, Nat.mul_comm z 8, pow_mul]; norm_num
  77         _ ≤ _ := Nat.pow_le_pow_right (by norm_num) hzn
  78     exact_mod_cast hh
  79   calc
  80     _ ≤ C^a*C^b*D^(k*n) := by
  81       apply mul_le_mul (mul_le_mul hca hcb (by positivity) (pow_nonneg hC _)) _ (by positivity) (by positivity)
  82       apply pow_le_pow_left₀ (by positivity) htail
  83     _ = (C*D^k)^n := by rw [mul_pow, ← pow_mul, ← pow_add]
  84     _ ≤ ((1/256:ℝ)*((z:ℝ)/m)^(k-1))^n := by
  85       apply pow_le_pow_left₀ (by positivity)
  86       rw [hfactor]
  87       exact mul_le_mul_of_nonneg_right hscale (by positivity)
  88     _ = (1/256:ℝ)^n*((z:ℝ)/m)^((k-1)*n) := by rw [mul_pow, ← pow_mul]
  89     _ ≤ (1/256:ℝ)^n*((z:ℝ)/m)^8 := mul_le_mul_of_nonneg_left hpow (by positivity)
  90     _ ≤ 1/(m:ℝ)^8 := by
  91       rw [div_pow, div_pow, one_pow]
  92       have hh : (z:ℝ)^8/(256:ℝ)^n ≤ 1 := (div_le_one (by positivity)).mpr hz8
  93       calc
  94         _ = ((z:ℝ)^8/(256:ℝ)^n)/(m:ℝ)^8 := by ring
  95         _ ≤ _ := div_le_div_of_nonneg_right hh (by positivity)
  96 
  97 end GraphChoices.LowerBound
