   5 /-! The ordinary residual replay is an actual indexed Greedy orientation.
   6 Its counted loads equal the implemented state, so the checked adversarial
   7 Greedy theorem applies directly to Threshold-Greedy's residual execution. -/

   9 variable {L R C : Type*} [Fintype L] [Fintype R]
  10   [DecidableEq L] [DecidableEq R] [DecidableEq C]

  12 def zeroGreedy : GreedyLoads L R := ⟨fun _ => 0,fun _ => 0⟩
  13 
  14 def greedyVertexLoad (s : GreedyLoads L R) : L ⊕ R → ℕ := Sum.elim s.left s.right
  15 
  16 def greedyTarget (s : GreedyLoads L R) (e : Arrival L R C) : L ⊕ R :=
  17   if s.left e.left < s.right e.right then Sum.inl e.left else Sum.inr e.right
  18 
  19  theorem greedyVertexLoad_step (s : GreedyLoads L R) (e : Arrival L R C) (v : L ⊕ R) :
  20     greedyVertexLoad (greedyStep s e) v = greedyVertexLoad s v + if greedyTarget s e = v then 1 else 0 := by
  21   by_cases hs : s.left e.left < s.right e.right <;> cases v <;>
  22     simp [greedyVertexLoad, greedyStep, greedyTarget, hs, eq_comm]
  23 
  24  theorem greedyTarget_min (s : GreedyLoads L R) (e : Arrival L R C) :
  25     greedyVertexLoad s (greedyTarget s e) ≤ s.left e.left ∧
  26     greedyVertexLoad s (greedyTarget s e) ≤ s.right e.right := by
  27   by_cases hs : s.left e.left < s.right e.right
  28   · simp [greedyVertexLoad, greedyTarget, hs, Nat.le_of_lt hs]
  29   · simp [greedyVertexLoad, greedyTarget, hs, Nat.le_of_not_gt hs]
  30 
  31  theorem greedyRun_append (s : GreedyLoads L R) (xs ys : List (Arrival L R C)) :
  32     greedyRun s (xs ++ ys) = greedyRun (greedyRun s xs) ys := by
  33   induction xs generalizing s with
  34   | nil => rfl
  35   | cons e es ih => exact ih (greedyStep s e)
  36 
  37  theorem greedyRun_take_succ (es : List (Arrival L R C)) (t : Fin es.length) :
  38     greedyRun (zeroGreedy : GreedyLoads L R) (es.take (t.val+1)) =
  39       greedyStep (greedyRun zeroGreedy (es.take t.val)) (es.get t) := by
  40   rw [List.take_succ_eq_append_getElem t.isLt, greedyRun_append]
  41   rfl

  43 def replayInstance (es : List (Arrival L R C)) : Instance (L ⊕ R) es.length where
  44   left t := Sum.inl (es.get t).left
  45   right t := Sum.inr (es.get t).right
  46 
  47 def replayOrientation (es : List (Arrival L R C)) : Orientation es.length := fun t =>
  48   let s := greedyRun (zeroGreedy : GreedyLoads L R) (es.take t.val)
  49   decide (s.left (es.get t).left < s.right (es.get t).right)
  50 
  51 def replayState (es : List (Arrival L R C)) (k : ℕ) : L ⊕ R → ℕ :=
  52   greedyVertexLoad (greedyRun (zeroGreedy : GreedyLoads L R) (es.take k))
  53 
  54  theorem replay_destination (es : List (Arrival L R C)) (t : Fin es.length) :
  55     destination (replayInstance es) (replayOrientation es) t =
  56       greedyTarget (greedyRun zeroGreedy (es.take t.val)) (es.get t) := by
  57   simp [destination, replayInstance, replayOrientation, greedyTarget]
  58 
  59  theorem replay_zero (es : List (Arrival L R C)) (v : L ⊕ R) : replayState es 0 v = 0 := by
  60   cases v <;> rfl
  61 
  62  theorem replay_increment (es : List (Arrival L R C)) (t : Fin es.length) (v : L ⊕ R) :
  63     replayState es (t.val+1) v = replayState es t.val v +
  64       if destination (replayInstance es) (replayOrientation es) t = v then 1 else 0 := by
  65   unfold replayState
  66   rw [greedyRun_take_succ, greedyVertexLoad_step, replay_destination]

  68  theorem replay_isGreedy (es : List (Arrival L R C)) :
  69     IsGreedy (replayInstance es) (replayOrientation es) := by
  70   apply isGreedy_of_execution (replayInstance es) (replayOrientation es) (replayState es)
  71     (replay_zero es) (replay_increment es)
  72   intro t v hv
  73   have hmin := greedyTarget_min (greedyRun zeroGreedy (es.take t.val)) (es.get t)
  74   rw [replay_destination]
  75   rcases hv with rfl | rfl
  76   · exact hmin.1
  77   · exact hmin.2
  78 
  79  theorem replay_final_load (es : List (Arrival L R C)) (v : L ⊕ R) :
  80     greedyVertexLoad (greedyRun (zeroGreedy : GreedyLoads L R) es) v =
  81       load (replayInstance es) (replayOrientation es) v := by
  82   simpa [replayState] using execution_final_load (replayInstance es) (replayOrientation es)
  83     (replayState es) (replay_zero es) (replay_increment es) v

  85 /-- The actual residual left load obeys the proved deterministic Greedy bound.
  86 The optimum here is the samplewise optimum of the residual arrival instance. -/
  87  theorem residual_left_competitive (α : C → ℕ) (es : List (Arrival L R C))
  88     (u : L) (hres : 0 < (residualArrivals (trace α es)).length) :
  89     (run α es).greedyLeft u <
  90       2 * opt (replayInstance (residualArrivals (trace α es))) *
  91         (Nat.log 2 (Fintype.card (L ⊕ R)) + 1) := by
  92   let rs := residualArrivals (trace α es)
  93   have hr := residual_replay α es
  94   have hl : (run α es).greedyLeft u =
  95       greedyVertexLoad (greedyRun (zeroGreedy : GreedyLoads L R) rs) (Sum.inl u) := by
  96     exact congrArg (fun s : GreedyLoads L R => s.left u) hr
  97   rw [hl, replay_final_load]
  98   exact (load_le_maxLoad (replayInstance rs) (replayOrientation rs) (Sum.inl u)).trans_lt
  99     (greedy_competitive (replayInstance rs) (replayOrientation rs) (replay_isGreedy rs) hres)
