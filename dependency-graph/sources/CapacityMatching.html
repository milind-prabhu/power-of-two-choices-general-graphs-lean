  16 /-- A capacity-b matching covering every left vertex. Capacities are integers. -/
  17  theorem exists_capacity_matching (r : L → R → Prop) [DecidableRel r] (b : ℕ)
  18     (hexpand : ∀ A : Finset L, A.card ≤ b * (neighbors r A).card) :
  19     ∃ f : L → R, (∀ u, r u (f u)) ∧
  20       ∀ v, (univ.filter fun u => f u = v).card ≤ b := by
  21   classical
  22   have hhall : ∀ A : Finset L,
  23       A.card ≤ (univ.filter fun p : R × Fin b => ∃ u ∈ A, r u p.1).card := by
  24     intro A
  25     have heq : (univ.filter fun p : R × Fin b => ∃ u ∈ A, r u p.1) =
  26         (neighbors r A) ×ˢ (univ : Finset (Fin b)) := by
  27       ext p
  28       simp [neighbors]
  29     rw [heq, card_product]
  30     simpa [mul_comm] using hexpand A
  31   let family (u : L) : Finset (R × Fin b) := univ.filter fun p => r u p.1
  32   have hfamily (A : Finset L) : A.biUnion family =
  33       (univ.filter fun p : R × Fin b => ∃ u ∈ A, r u p.1) := by
  34     ext p
  35     simp [family]
  36   obtain ⟨g, hginj, hg⟩ :=
  37     (Finset.all_card_le_biUnion_card_iff_existsInjective' family).mp
  38       (fun A => by rw [hfamily]; exact hhall A)
  39   simp only [family, mem_filter, mem_univ, true_and] at hg
  40   refine ⟨fun u => (g u).1, hg, ?_⟩
  41   intro v
  42   have hh := card_le_card_of_injOn (fun u : L => (g u).2)
  43     (s := univ.filter fun u => (g u).1 = v) (t := univ)
  44     (fun _ _ => mem_univ _)
  45     (by
  46       intro u hu w hw hslots
  47       apply hginj
  48       apply Prod.ext
  49       · exact (mem_filter.mp hu).2.trans (mem_filter.mp hw).2.symm
  50       · exact hslots)
  51   simpa using hh
