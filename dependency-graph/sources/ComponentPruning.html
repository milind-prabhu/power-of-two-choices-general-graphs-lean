   3 /-! The pruning theorem applied to an actual replay component. Both graph
   4 degrees and bipartite sides are proved to agree with the implementation's
   5 list neighborhoods; no component-to-tree hypothesis is assumed. -/
   6 namespace GraphChoices
   7 open Finset
   8 variable {V : Type*} [Fintype V] [DecidableEq V]
   9 
  10 theorem connected_both_colors (G : SimpleGraph V) (left : V → Bool)
  11     (hc : G.Connected) (hcolor : ∀ x y, G.Adj x y → left x ≠ left y)
  12     (hcard : 1 < Fintype.card V) :
  13     (∃ v, left v = true) ∧ (∃ v, left v = false) := by
  14   haveI : Nontrivial V := Fintype.one_lt_card_iff_nontrivial.mp hcard
  15   obtain ⟨u, v, hne⟩ := exists_pair_ne V
  16   obtain ⟨p⟩ := hc u v
  17   cases p with
  18   | nil => exact (hne rfl).elim
  19   | @cons u w v h p =>
  20     have hdiff := hcolor u w h
  21     cases hu : left u <;> cases hw : left w
  22     · exact (hdiff (hu.trans hw.symm)).elim
  23     · exact ⟨⟨w, hw⟩, ⟨u, hu⟩⟩
  24     · exact ⟨⟨u, hu⟩, ⟨w, hw⟩⟩
  25     · exact (hdiff (hu.trans hw.symm)).elim
  26 
  27 theorem component_induce_degree {T : ℕ} (I : Instance V T) (v : V)
  28     (x : {w : V // w ∈ componentVertices I v}) :
  29     let G := arrivalGraph I
  30     letI : DecidableRel G.Adj := Classical.decRel _
  31     (G.induce (componentVertices I v : Set V)).degree x = G.degree x.val := by
  32   classical
  33   dsimp only
  34   simp only [← SimpleGraph.card_neighborSet_eq_degree]
  35   apply Fintype.card_congr
  36   refine {
  37     toFun := fun w => ⟨w.val.val, w.property⟩
  38     invFun := fun w => ⟨⟨w.val, ?_⟩, w.property⟩
  39     left_inv := fun w => rfl
  40     right_inv := fun w => rfl }
  41   exact (mem_componentVertices I v w.val).mpr
  42     (((mem_componentVertices I v x.val).mp x.property).trans w.property.reachable)
  43 
  44 end GraphChoices

  51 /-- Every nontrivial replay component has a large left-heavy tree with a
  52 right root and edges in the actual residual graph. The subtree's vertex type
  53 is a subset of the component, so its ambient vertex map is injective. -/
  54 theorem replay_component_left_heavy_tree (es : List (Arrival L R C)) (v : L ⊕ R)
  55     (D : ℕ) (hD : ∀ u, (listLeftNeighbors es u).card ≤ D)
  56     (hlarge : 1 < (componentVertices (replayInstance es) v).card) :
  57     let U := {w : L ⊕ R // w ∈ componentVertices (replayInstance es) v}
  58     ∃ s : Set U, ∃ H : SimpleGraph s,
  59       H.IsTree ∧
  60       (∀ x y, H.Adj x y → (arrivalGraph (replayInstance es)).Adj x.val.val y.val.val) ∧
  61       (componentVertices (replayInstance es) v).card ≤ (D+1) * Nat.card s ∧
  62       Nat.card {x : s // isLeft x.val.val = false} + 1 ≤
  63         Nat.card {x : s // isLeft x.val.val = true} ∧
  64       (1 < Nat.card s → ∃ root : s, isLeft root.val.val = false) := by
  65   classical
  66   let U := {w : L ⊕ R // w ∈ componentVertices (replayInstance es) v}
  67   let G : SimpleGraph U := (arrivalGraph (replayInstance es)).induce
  68     (componentVertices (replayInstance es) v : Set (L ⊕ R))
  69   have hconn : G.Connected := component_induce_connected _ _
  70   have hcolor : ∀ x y, G.Adj x y → isLeft x.val ≠ isLeft y.val :=
  71     fun x y hxy => replay_adj_sides es x.val y.val hxy
  72   have hcard : 1 < Fintype.card U := by simpa only [U, Fintype.card_coe] using hlarge
  73   obtain ⟨root, hroot⟩ := (connected_both_colors G (fun x => isLeft x.val) hconn hcolor hcard).1
  74   have hdeg : ∀ x : U, isLeft x.val = true → G.degree x ≤ D := by
  75     intro x hx
  76     rw [component_induce_degree]
  77     cases heq : x.val with
  78     | inl u => rw [replay_left_degree]; exact hD u
  79     | inr w => simp [heq, isLeft] at hx
  80   obtain ⟨s, H, htree, hsub, _, hsize, hheavy⟩ :=
  81     exists_large_left_heavy_tree G (fun x => isLeft x.val) hconn hcolor root hroot D hdeg
  82   refine ⟨s, H, htree, ?_, ?_, hheavy, ?_⟩
  83   · intro x y hxy
  84     exact hsub hxy
  85   · simpa only [U, Fintype.card_coe] using hsize
  86   · intro hs
  87     have hcol : ∀ x y, H.Adj x y → isLeft x.val.val ≠ isLeft y.val.val :=
  88       fun x y hxy => hcolor x.val y.val (hsub hxy)
  89     have hcard' : 1 < Fintype.card s := by simpa only [Nat.card_eq_fintype_card] using hs
  90     exact (connected_both_colors H (fun x => isLeft x.val.val) htree.isConnected hcol hcard').2
  91 
  92 end GraphChoices.ThresholdGreedy
