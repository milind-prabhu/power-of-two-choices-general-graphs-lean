   3 /-! A fully probabilistic lower bound for every concrete skewness witness.
   4 Its hypotheses concern only the base subgraph's degrees and cardinalities. -/
   5 namespace GraphChoices.Peeling
   6 open Finset
   7 variable {L R : Type*} [Fintype L] [Fintype R] [DecidableEq L] [DecidableEq R]
   8 
   9 def subgraphBucket (G F : Graph L R) (u : L) : Finset G :=
  10   univ.filter fun e => e.val ∈ F ∧ e.val.1 = u
  11 
  12 theorem subgraphBucket_degree_lower {G F : Graph L R} (hF : F ⊆ G) (u : L) :
  13     leftDegree F u ≤ (subgraphBucket G F u).card := by
  14   classical
  15   have hedge (v : leftNeighbors F u) : (u,v.val) ∈ F := by
  16     simpa only [leftNeighbors, mem_filter, mem_univ, true_and] using v.property
  17   let f : leftNeighbors F u → subgraphBucket G F u := fun v =>
  18     ⟨⟨(u,v.val), hF (hedge v)⟩, mem_filter.mpr ⟨mem_univ _, hedge v, rfl⟩⟩
  19   have hf : Function.Injective f := by
  20     intro v w h
  21     exact Subtype.ext (congrArg (fun x => x.val.val.2) h)
  22   have hh := Fintype.card_le_of_injective f hf
  23   simpa only [Fintype.card_coe, leftDegree] using hh
  24 
  25 theorem subgraphBucket_source (G F : Graph L R) (A : Finset L) :
  26     ∀ u ∈ A, ∀ e ∈ subgraphBucket G F u,
  27       e.val.1 = u ∧ e.val.2 ∈ neighborSet F A := by
  28   intro u hu e he
  29   have hs := (mem_filter.mp he).2
  30   refine ⟨hs.2, ?_⟩
  31   simp only [neighborSet, GraphChoices.neighbors, mem_filter, mem_univ, true_and]
  32   exact ⟨u, hu, by simpa only [← hs.2] using hs.1⟩
  33 
  34 /-- If the block-hit lower bound predicts at least |B| successful left
  35 vertices, the actual sampled optimum is at least k/3 in expectation. -/
  36 theorem expectedOpt_subgraph_blocks (G F : Graph L R) [Nonempty G]
  37     (hF : F ⊆ G) (A : Finset L) (hB : (neighborSet F A).Nonempty)
  38     (k s : ℕ) (x : ℝ) (hx : 0 ≤ x) (hx1 : x ≤ 1)
  39     (hmean : x ≤ (s:ℝ)*((minLeftDegree F A : ℝ)/G.card))
  40     (hpop : ((neighborSet F A).card : ℝ) ≤ (A.card:ℝ)*(x/2)^k) :
  41     (k:ℝ)/3 ≤ expectedOpt (graphBase G) (k*s) := by
  42   have hprob (u : A) : (x/2)^k ≤ eventProbability (allBlocksEvent (subgraphBucket G F u.val) k s) := by
  43     rw [allBlocks_probability]
  44     apply pow_le_pow_left₀ (by positivity)
  45     apply blockHit_probability_lower_of_mean _ _ x hx hx1
  46     apply hmean.trans
  47     apply mul_le_mul_of_nonneg_left _ (Nat.cast_nonneg s)
  48     rw [Fintype.card_coe]
  49     apply div_le_div_of_nonneg_right _ (Nat.cast_nonneg G.card)
  50     exact_mod_cast (minLeftDegree_le F u.property).trans (subgraphBucket_degree_lower hF u.val)
  51   have hh := expectedOpt_block_lower (fun e : G => e.val.1) (fun e : G => e.val.2)
  52     A (neighborSet F A) (subgraphBucket G F) (subgraphBucket_source G F A) k s hB
  53     (hpop.trans (by simpa only [sum_const, card_univ, Fintype.card_coe, nsmul_eq_mul] using
  54       (sum_le_sum fun u (_ : u ∈ (univ : Finset A)) => hprob u)))
  55   exact hh
  56 
  57 end GraphChoices.Peeling
