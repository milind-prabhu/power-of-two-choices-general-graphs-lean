  12 theorem path_internal_degree {u v : V} (p : G.Walk u v) (hp : p.IsPath)
  13     (w : V) (hw : w ∈ p.support) (hwu : w ≠ u) (hwv : w ≠ v) :
  14     2 ≤ G.degree w := by
  15   induction p generalizing w with
  16   | nil => simp only [SimpleGraph.Walk.support_nil, List.mem_singleton] at hw; exact (hwu hw).elim
  17   | @cons u x v h p ih =>
  18     have hps := (SimpleGraph.Walk.cons_isPath_iff h p).mp hp
  19     simp only [SimpleGraph.Walk.support_cons, List.mem_cons] at hw
  20     rcases hw with hw | hw
  21     · exact (hwu hw).elim
  22     by_cases hx : w = x
  23     · subst w
  24       cases p with
  25       | nil => exact (hwv rfl).elim
  26       | @cons x y v hxy q =>
  27         have huy : u ≠ y := by
  28           intro heq
  29           apply hps.2
  30           simp [heq]
  31         have hpair : ({u,y} : Finset V) ⊆ G.neighborFinset x := by
  32           intro z hz
  33           rcases mem_insert.mp hz with rfl | hz
  34           · exact (G.mem_neighborFinset _ _).mpr h.symm
  35           · have hz' := mem_singleton.mp hz
  36             subst z
  37             exact (G.mem_neighborFinset _ _).mpr hxy
  38         have hc := card_le_card hpair
  39         simpa only [card_pair huy, SimpleGraph.card_neighborFinset_eq_degree] using hc
  40     · exact ih hps.1 w hw hx hwv

  43 theorem reachable_induce_of_walk {s : Set V} {u v : V} (p : G.Walk u v)
  44     (hu : u ∈ s) (hv : v ∈ s) (hs : ∀ w ∈ p.support, w ∈ s) :
  45     (G.induce s).Reachable ⟨u,hu⟩ ⟨v,hv⟩ := by
  46   induction p with
  47   | nil => exact SimpleGraph.Reachable.rfl
  48   | @cons u w v h p ih =>
  49     have hw : w ∈ s := hs w (by simp)
  50     have hp : ∀ z ∈ p.support, z ∈ s := fun z hz => hs z (by simp [hz])
  51     exact (SimpleGraph.Adj.reachable (show (G.induce s).Adj ⟨u,hu⟩ ⟨w,hw⟩ from h)).trans
  52       (ih hw hv hp)

  54 def prunedVertices (left : V → Bool) : Set V :=
  55   {v | left v = true ∨ 2 ≤ G.degree v}
  56 
  57 instance decidablePrunedVertices (left : V → Bool) :
  58     DecidablePred (fun v => v ∈ prunedVertices G left) :=
  59   fun v => inferInstanceAs (Decidable (left v = true ∨ 2 ≤ G.degree v))

  61 theorem pruned_induce_connected (left : V → Bool) (hc : G.Connected)
  62     (root : V) (hroot : left root = true) :
  63     (G.induce (prunedVertices G left)).Connected := by
  64   apply (SimpleGraph.connected_iff_exists_forall_reachable _).mpr
  65   refine ⟨⟨root, Or.inl hroot⟩, ?_⟩
  66   intro v
  67   obtain ⟨p, hp⟩ := (hc root v.val).exists_isPath
  68   apply reachable_induce_of_walk G p (Or.inl hroot) v.property
  69   intro w hw
  70   by_cases hwu : w = root
  71   · subst w; exact Or.inl hroot
  72   by_cases hwv : w = v.val
  73   · subst w; exact v.property
  74   exact Or.inr (path_internal_degree G p hp w hw hwu hwv)

  76 theorem pruned_induce_isTree (left : V → Bool) (ht : G.IsTree)
  77     (root : V) (hroot : left root = true) :
  78     (G.induce (prunedVertices G left)).IsTree :=
  79   ⟨pruned_induce_connected G left ht.isConnected root hroot,
  80     ht.IsAcyclic.induce _⟩

  83 theorem bool_bipartite (left : V → Bool)
  84     (hcolor : ∀ x y, G.Adj x y → left x ≠ left y) :
  85     G.IsBipartiteWith {v | left v = true} {v | left v = false} := by
  86   constructor
  87   · simp [Set.disjoint_left]
  88   · intro x y hxy
  89     have h := hcolor x y hxy
  90     cases hx : left x <;> cases hy : left y <;> simp_all

  99 theorem right_degree_preserved_by_pruning (left : V → Bool)
 100     (hcolor : ∀ x y, G.Adj x y → left x ≠ left y)
 101     (v : prunedVertices G left) (hv : left v.val = false) :

 110 /-- A tree without right leaves has strictly more left vertices than right
 111 vertices. This is the degree-sum argument with the exact tree edge count. -/
 112 theorem tree_left_heavy_of_right_min_degree (left : V → Bool) (ht : G.IsTree)
 113     (hcolor : ∀ x y, G.Adj x y → left x ≠ left y)
 114     (hdeg : ∀ v, left v = false → 2 ≤ G.degree v) :
 115     (univ.filter fun v => left v = false).card + 1 ≤
 116       (univ.filter fun v => left v = true).card := by
 117   classical
 118   have hbi : G.IsBipartiteWith
 119       (↑(univ.filter fun v => left v = true)) (↑(univ.filter fun v => left v = false)) := by
 120     simpa using bool_bipartite G left hcolor
 121   have hsum := SimpleGraph.isBipartiteWith_sum_degrees_eq_card_edges' hbi
 122   have hbound : 2 * (univ.filter fun v => left v = false).card ≤ G.edgeFinset.card := by
 123     calc
 124       _ = ∑ v ∈ univ.filter (fun v => left v = false), 2 := by simp [mul_comm]
 125       _ ≤ ∑ v ∈ univ.filter (fun v => left v = false), G.degree v := by
 126         apply sum_le_sum
 127         intro v hv
 128         exact hdeg v (mem_filter.mp hv).2
 129       _ = _ := hsum
 130   have hc := ht.card_edgeFinset
 131   have hp := bool_partition_card left
 132   omega

 134 theorem pruned_tree_left_heavy (left : V → Bool) (ht : G.IsTree)
 135     (hcolor : ∀ x y, G.Adj x y → left x ≠ left y)
 136     (root : V) (hroot : left root = true) :
 137     let U := prunedVertices G left
 138     (univ.filter fun v : U => left v.val = false).card + 1 ≤
 139       (univ.filter fun v : U => left v.val = true).card := by
 140   apply tree_left_heavy_of_right_min_degree (G.induce (prunedVertices G left))
 141     (fun v => left v.val) (pruned_induce_isTree G left ht root hroot)
 142   · intro x y hxy
 143     exact hcolor x.val y.val hxy
 144   · intro v hv
 145     rw [right_degree_preserved_by_pruning G left hcolor v hv]
 146     rcases v.property with hleft | hdeg
 147     · simp [hv] at hleft
 148     · exact hdeg

 158 /-- Deleting right leaves loses at most a factor D+1 in tree size, when all
 159 left degrees are at most D. This also covers a singleton left tree. -/
 160 theorem pruning_size_bound (left : V → Bool) (hc : G.Connected)
 161     (hcolor : ∀ x y, G.Adj x y → left x ≠ left y)
 162     (root : V) (hroot : left root = true) (D : ℕ)
 163     (hD : ∀ v, left v = true → G.degree v ≤ D) :
 164     Fintype.card V ≤ (D+1) * Fintype.card (prunedVertices G left) := by
 165   classical
 166   have hbi : G.IsBipartiteWith
 167       (↑(univ.filter fun v => left v = true)) (↑(univ.filter fun v => left v = false)) := by
 168     simpa using bool_bipartite G left hcolor
 169   have hright : (univ.filter fun v => left v = false).card ≤ G.edgeFinset.card := by
 170     calc
 171       _ = ∑ v ∈ univ.filter (fun v => left v = false), 1 := by simp
 172       _ ≤ ∑ v ∈ univ.filter (fun v => left v = false), G.degree v := by
 173         apply sum_le_sum
 174         intro v hv
 175         have hvfalse := (mem_filter.mp hv).2
 176         have hne : v ≠ root := by intro heq; subst v; simp [hroot] at hvfalse
 177         letI : Nontrivial V := ⟨⟨v, root, hne⟩⟩
 178         exact hc.preconnected.degree_pos_of_nontrivial v
 179       _ = _ := SimpleGraph.isBipartiteWith_sum_degrees_eq_card_edges' hbi
 180   have hleft : G.edgeFinset.card ≤ D * (univ.filter fun v => left v = true).card := by
 181     calc
 182       _ = ∑ v ∈ univ.filter (fun v => left v = true), G.degree v :=
 183         (SimpleGraph.isBipartiteWith_sum_degrees_eq_card_edges hbi).symm
 184       _ ≤ ∑ v ∈ univ.filter (fun v => left v = true), D := by
 185         apply sum_le_sum
 186         intro v hv
 187         exact hD v (mem_filter.mp hv).2
 188       _ = _ := by simp [mul_comm]
 189   have hpart := bool_partition_card left
 190   calc
 191     Fintype.card V ≤ (D+1)*(univ.filter fun v => left v = true).card := by nlinarith
 192     _ ≤ _ := Nat.mul_le_mul_left _ (left_card_le_pruned_card G left)

 194 /-- Deterministic large-component reduction: any connected bipartite graph
 195 with left degrees at most D contains a left-heavy tree with at least a
 196 1/(D+1) fraction of its vertices. All original left vertices are retained. -/
 197 theorem exists_large_left_heavy_tree (left : V → Bool) (hc : G.Connected)
 198     (hcolor : ∀ x y, G.Adj x y → left x ≠ left y)
 199     (root : V) (hroot : left root = true) (D : ℕ)
 200     (hD : ∀ v, left v = true → G.degree v ≤ D) :
 201     ∃ s : Set V, ∃ H : SimpleGraph s,
 202       H.IsTree ∧ H ≤ G.induce s ∧
 203       (∀ v, left v = true → v ∈ s) ∧
 204       Fintype.card V ≤ (D+1) * Nat.card s ∧
 205       Nat.card {v : s // left v.val = false} + 1 ≤ Nat.card {v : s // left v.val = true} := by
 206   classical
 207   obtain ⟨F, hFG, htree⟩ := hc.exists_isTree_le
 208   let s := prunedVertices F left
 209   refine ⟨s, F.induce s, pruned_induce_isTree F left htree root hroot, ?_, ?_, ?_, ?_⟩
 210   · intro x y hxy
 211     exact hFG hxy
 212   · intro v hv
 213     exact Or.inl hv
 214   · have hFD : ∀ v, left v = true → F.degree v ≤ D := by
 215       intro v hv
 216       exact (F.degree_le_of_le hFG).trans (hD v hv)
 217     simpa only [Nat.card_eq_fintype_card] using
 218       pruning_size_bound F left htree.isConnected (fun x y h => hcolor x y (hFG h)) root hroot D hFD
 219   · simpa only [Nat.card_eq_fintype_card, Fintype.card_subtype] using
 220       pruned_tree_left_heavy F left htree (fun x y h => hcolor x y (hFG h)) root hroot
