   9 theorem nat_log_le_twice_real_log (a : ℕ) (ha : 0 < a) :
  10     (Nat.log 2 a : ℝ) ≤ 2*Real.log (a:ℝ) := by
  11   have hp : (2:ℝ)^(Nat.log 2 a) ≤ (a:ℝ) := by
  12     exact_mod_cast Nat.pow_log_le_self 2 ha.ne'
  13   have hlog := Real.log_le_log (by positivity : (0:ℝ) < (2:ℝ)^(Nat.log 2 a)) hp
  14   rw [Real.log_pow] at hlog
  15   have htwo : (1:ℝ)/2 ≤ Real.log 2 := by linarith [Real.log_two_gt_d9]
  16   have hn : (0:ℝ) ≤ Nat.log 2 a := Nat.cast_nonneg _
  17   nlinarith

  19 theorem iteratedLog_le_real_factor (N : ℕ) (hlog : 1 < Real.log (N:ℝ)) :
  20     (iteratedLog N : ℝ) ≤ 3*loglogFactor N := by
  21   have hN : 2 ≤ N := by
  22     by_contra hn
  23     have hc : N = 0 ∨ N = 1 := by omega
  24     rcases hc with rfl | rfl <;> norm_num at hlog
  25   have hNr : (2:ℝ) ≤ N := by exact_mod_cast hN
  26   let a := Nat.log 2 N+1
  27   have ha : 0 < a := by dsimp [a]; omega
  28   have hinner := nat_log_le_twice_real_log N (by omega)
  29   have houter := nat_log_le_twice_real_log a ha
  30   have hap : (0:ℝ) < a := by exact_mod_cast ha
  31   have haBound : (a:ℝ) ≤ 2*(1+Real.log (N:ℝ)) := by
  32     dsimp [a]
  33     push_cast
  34     linarith
  35   have hla := Real.log_le_log hap haBound
  36   rw [Real.log_mul (by norm_num) (ne_of_gt (by linarith : 0 < 1+Real.log (N:ℝ)))] at hla
  37   have hln : 0 ≤ Real.log (1+Real.log (N:ℝ)) := Real.log_nonneg (by linarith)
  38   have htwo : Real.log 2 ≤ 1 := by linarith [Real.log_two_lt_d9]
  39   unfold loglogFactor
  40   rw [max_eq_right hNr]
  41   have heq : (iteratedLog N : ℝ) = (Nat.log 2 a : ℝ)+1 := by
  42     simp only [iteratedLog, Nat.cast_add, Nat.cast_one, a]
  43   rw [heq]
  44   linarith
