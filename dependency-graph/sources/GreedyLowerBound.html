  75 theorem family_competitive_gap (b : ℕ) (hb : 16 ≤ b) :
  76     letI := family_edges_nonempty b hb
  77     (1/30000:ℝ)*Real.log (size b:ℝ)/(Real.log (Real.log (size b:ℝ)))^2*
  78       expectedOpt (graph b (branching b) (width b)) (size b) ≤
  79         RandomGreedy.expectedLoad (graph b (branching b) (width b)) (size b) := by
  80   letI := family_edges_nonempty b hb
  81   let A := Real.log (size b:ℝ)
  82   let D := Real.log A
  83   let l := Real.log ((b:ℝ)+1)
  84   let O := expectedOpt (graph b (branching b) (width b)) (size b)
  85   have hl : 1 ≤ l := (family_elementary_logs b hb).1
  86   have hD : l ≤ D := Real.log_le_log (by positivity) (family_log_bounds b hb).1
  87   have hA : A ≤ 26*(b:ℝ)*l := (family_log_bounds b hb).2
  88   have hO : O ≤ 500*l := family_optimum_bound b hb
  89   have hO0 : 0 ≤ O := expectedOpt_nonneg _ _
  90   have hprod : A*O ≤ 13000*(b:ℝ)*l^2 := by
  91     have h1 := mul_le_mul_of_nonneg_right hA hO0
  92     have h2 := mul_le_mul_of_nonneg_left hO (show 0 ≤ 26*(b:ℝ)*l by positivity)
  93     nlinarith
  94   have hsq : l^2 ≤ D^2 := by nlinarith
  95   have hprod' : A*O ≤ 13000*(b:ℝ)*D^2 := hprod.trans
  96     (mul_le_mul_of_nonneg_left hsq (by positivity))
  97   apply le_trans _ (family_greedy_lower b hb)
  98   change (1/30000:ℝ)*A/D^2*O ≤ (b:ℝ)/2
  99   have hDpos : 0 < D := by linarith
 100   apply (le_of_mul_le_mul_right (a := D^2) ?_ (sq_pos_of_pos hDpos))
 101   have heq : ((1/30000:ℝ)*A/D^2*O)*D^2 = A*O/30000 := by field_simp
 102   rw [heq]
 103   nlinarith [mul_nonneg (Nat.cast_nonneg b : (0:ℝ) ≤ b) (sq_nonneg D)]
