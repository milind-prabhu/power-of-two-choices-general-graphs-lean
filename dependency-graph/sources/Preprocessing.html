  13 def BaseGraph.split (G : BaseGraph V E) (b : E → Bool) : BaseGraph (V ⊕ V) E where
  14   left e := Sum.inl (if b e then G.left e else G.right e)
  15   right e := Sum.inr (if b e then G.right e else G.left e)

  17 omit [Fintype V] [DecidableEq V] [Fintype E] [DecidableEq E] [Nonempty E] in
  18  theorem split_sample (G : BaseGraph V E) (b : E → Bool) (ω : Fin T → E) :
  19     (G.split b).sample ω = splitInstance (G.sample ω) (fun t => b (ω t)) := by
  20   apply Instance.ext <;> funext t
  21   · rfl
  22   · cases h : b (ω t) <;>
  23       simp [BaseGraph.split, BaseGraph.sample, splitInstance, destination, flip, h]

  26  theorem split_expectedOpt_le (G : BaseGraph V E) (b : E → Bool) (T : ℕ) :
  27     expectedOpt (G.split b) T ≤ expectedOpt G T := by
  28   apply average_mono
  29   intro ω
  30   rw [split_sample]
  31   exact_mod_cast split_opt_le (G.sample ω) (fun t => b (ω t))

  43 /-- Expected maximum load of a policy. This definition also permits policies
  44 that see the future; applications must separately establish nonanticipation. -/
  45 noncomputable def expectedCost (G : BaseGraph V E) (a : (Fin T → E) → Orientation T) : ℝ :=
  46   average (fun ω => (maxLoad (G.sample ω) (a ω) : ℝ))
  47 
  48  theorem expected_competitive_transfer (G : BaseGraph V E) (b : E → Bool)
  49     (a : (Fin T → E) → Orientation T) (c : ℝ) (hc : 0 ≤ c)
  50     (hcost : expectedCost (G.split b) a ≤ c * expectedOpt (G.split b) T) :
  51     expectedCost G (fun ω => transport (fun t => b (ω t)) (a ω)) ≤
  52       (2*c) * expectedOpt G T := by
  53   have hcollapse : expectedCost G (fun ω => transport (fun t => b (ω t)) (a ω)) ≤
  54       2 * expectedCost (G.split b) a := by
  55     unfold expectedCost
  56     rw [← average_mul]
  57     apply average_mono
  58     intro ω
  59     rw [split_sample]
  60     exact_mod_cast collapse_maxLoad_le (G.sample ω) (fun t => b (ω t)) (a ω)
  61   calc
  62     _ ≤ 2 * expectedCost (G.split b) a := hcollapse
  63     _ ≤ 2 * (c * expectedOpt (G.split b) T) := mul_le_mul_of_nonneg_left hcost (by norm_num)
  64     _ ≤ 2 * (c * expectedOpt G T) :=
  65       mul_le_mul_of_nonneg_left (mul_le_mul_of_nonneg_left (split_expectedOpt_le G b T) hc) (by norm_num)
  66     _ = _ := by ring

  68 /-- An actual reference orientation with maximum left degree ceil(rho*).
  69 The index set here is an arbitrary finite sequence of base-edge occurrences. -/
  70  theorem exists_left_degree_bounded_split (I : Instance V T) :
  71     ∃ b : Orientation T, ∀ u : V,
  72       (univ.filter fun t => (splitInstance I b).left t = Sum.inl u).card ≤ ⌈maxDensity I⌉₊ := by
  73   obtain ⟨b, hb⟩ := exists_exact_opt I
  74   refine ⟨b, ?_⟩
  75   intro u
  76   have h := load_le_maxLoad I b u
  77   rw [hb, opt_eq_ceil_maxDensity] at h
  78   simpa [splitInstance, load] using h
