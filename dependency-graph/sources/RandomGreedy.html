   3 /-! Ordinary Greedy with a fresh independent fair coin for ties. The source
   4 alphabet E × Bool records both the edge and the otherwise unused coin. -/
   5 namespace GraphChoices.RandomGreedy
   6 open Finset
   7 variable {V E : Type*} [Fintype V] [DecidableEq V]
   8 
   9 def choice (G : BaseGraph V E) (s : V → ℕ) (x : E × Bool) : Bool :=
  10   if s (G.left x.1) < s (G.right x.1) then true
  11   else if s (G.right x.1) < s (G.left x.1) then false else x.2
  12 
  13 def target (G : BaseGraph V E) (s : V → ℕ) (x : E × Bool) : V :=
  14   if choice G s x then G.left x.1 else G.right x.1
  15 
  16 def step (G : BaseGraph V E) (s : V → ℕ) (x : E × Bool) : V → ℕ :=
  17   fun v => s v + if target G s x = v then 1 else 0
  18 
  19 def run (G : BaseGraph V E) (xs : List (E × Bool)) : V → ℕ :=
  20   xs.foldl (step G) (fun _ => 0)
  21 
  22 def policy (G : BaseGraph V E) : OnlinePolicy (E × Bool) := fun history x => choice G (run G history) x
  23 
  24 def coinGraph (G : BaseGraph V E) : BaseGraph V (E × Bool) where
  25   left x := G.left x.1
  26   right x := G.right x.1
  27 
  28 def state (G : BaseGraph V E) {T : ℕ} (ω : Fin T → E × Bool) (k : ℕ) : V → ℕ :=
  29   run G ((List.ofFn ω).take k)
  30 
  31 theorem run_append (G : BaseGraph V E) (xs : List (E × Bool)) (x : E × Bool) :
  32     run G (xs ++ [x]) = step G (run G xs) x := by simp only [run, List.foldl_append, List.foldl_cons, List.foldl_nil]
  33 
  34 theorem state_zero (G : BaseGraph V E) {T : ℕ} (ω : Fin T → E × Bool) (v : V) : state G ω 0 v = 0 := by
  35   simp only [state, List.take_zero, run, List.foldl_nil]
  36 
  37 theorem state_step (G : BaseGraph V E) {T : ℕ} (ω : Fin T → E × Bool) (t : Fin T) (v : V) :
  38     state G ω (t.val+1) v = state G ω t.val v +
  39       if destination ((coinGraph G).sample ω) (onlineOrientation (policy G) ω) t = v then 1 else 0 := by
  40   have ht : t.val < (List.ofFn ω).length := by simpa using t.isLt
  41   simp only [state, List.take_succ_eq_append_getElem ht, run_append, step,
  42     List.getElem_ofFn, destination, BaseGraph.sample, coinGraph, onlineOrientation, policy, target]
  43 
  44 theorem state_eq_prefixLoad (G : BaseGraph V E) {T : ℕ} (ω : Fin T → E × Bool)
  45     (k : ℕ) (hk : k ≤ T) (v : V) :
  46     state G ω k v = prefixLoad ((coinGraph G).sample ω) (onlineOrientation (policy G) ω) k v :=
  47   execution_counts_prefix _ _ _ (state_zero G ω) (state_step G ω) k hk v
  48 
  49 theorem state_mono (G : BaseGraph V E) {T : ℕ} (ω : Fin T → E × Bool)
  50     {a b : ℕ} (hab : a ≤ b) (hb : b ≤ T) (v : V) : state G ω a v ≤ state G ω b v := by
  51   rw [state_eq_prefixLoad G ω a (hab.trans hb), state_eq_prefixLoad G ω b hb]
  52   apply card_le_card
  53   intro t ht
  54   simp only [mem_filter, mem_univ, true_and] at ht ⊢
  55   exact ⟨by omega,ht.2⟩
  56 
  57 theorem choice_favours_left (G : BaseGraph V E) (s : V → ℕ) (x : E × Bool)
  58     (hcoin : x.2 = true) (hload : s (G.left x.1) ≤ s (G.right x.1)) : choice G s x = true := by
  59   unfold choice
  60   split_ifs with h1 h2
  61   · rfl
  62   · omega
  63   · exact hcoin
  64 
  65 theorem policy_isGreedy (G : BaseGraph V E) {T : ℕ} (ω : Fin T → E × Bool) :
  66     IsGreedy ((coinGraph G).sample ω) (onlineOrientation (policy G) ω) := by
  67   apply isGreedy_of_execution _ _ _ (state_zero G ω) (state_step G ω)
  68   intro t v hv
  69   change v = G.left (ω t).1 ∨ v = G.right (ω t).1 at hv
  70   change state G ω t.val (target G (state G ω t.val) (ω t)) ≤ state G ω t.val v
  71   by_cases h1 : state G ω t.val (G.left (ω t).1) < state G ω t.val (G.right (ω t).1)
  72   · simp only [target, choice, if_pos h1, ↓reduceIte]
  73     rcases hv with rfl | rfl <;> omega
  74   · by_cases h2 : state G ω t.val (G.right (ω t).1) < state G ω t.val (G.left (ω t).1)
  75     · simp only [target, choice, if_neg h1, if_pos h2, Bool.false_eq_true, ↓reduceIte]
  76       rcases hv with rfl | rfl <;> omega
  77     · simp only [target, choice, if_neg h1, if_neg h2]
  78       split <;> rcases hv with rfl | rfl <;> omega
  79 
  80 variable [Fintype E] [DecidableEq E] [Nonempty E]
  81 noncomputable def expectedLoad (G : BaseGraph V E) (T : ℕ) : ℝ :=
  82   expectedCost (coinGraph G) (onlineOrientation (T := T) (policy G))
  83 
  84 end GraphChoices.RandomGreedy
