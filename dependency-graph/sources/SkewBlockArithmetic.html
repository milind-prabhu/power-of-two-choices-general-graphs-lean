   8 theorem skew_denominator_ge_log {δ d n : ℝ} (hδ : 0 < δ) (hd : 0 < d)
   9     (hn : 2 ≤ Real.log n) (hdeg : δ ≤ d*Real.log n) :
  10     Real.log n ≤ d*(Real.log n)^2/δ := by
  11   apply (le_div_iff₀ hδ).mpr
  12   nlinarith [mul_le_mul_of_nonneg_right hdeg (by linarith : 0 ≤ Real.log n)]

  14 theorem skew_threshold_le_log {a b δ d n : ℝ} (ha : 0 < a) (hb : 1 ≤ b)
  15     (han : a ≤ n) (hδ : 0 < δ) (hd : 0 < d) (hn : 2 ≤ Real.log n)
  16     (hdeg : δ ≤ d*Real.log n) (k : ℕ)
  17     (hk : 2*(k:ℝ) ≤ localLogSkewness a b δ d n) : (k:ℝ) ≤ Real.log n := by
  18   let q := d*(Real.log n)^2/δ
  19   have hq : Real.log n ≤ q := skew_denominator_ge_log hδ hd hn hdeg
  20   have hq2 : 2 ≤ q := hn.trans hq
  21   have hlogq : 1/2 ≤ Real.log q := by
  22     have hh := Real.log_le_log (by norm_num : (0:ℝ)<2) hq2
  23     linarith [Real.log_two_gt_d9]
  24   have hratio : a/b ≤ n := by
  25     apply (div_le_iff₀ (by linarith : 0 < b)).mpr
  26     nlinarith
  27   have hlogratio := Real.log_le_log (div_pos ha (by linarith : 0 < b)) hratio
  28   have hk' : 2*(k:ℝ)*Real.log q ≤ Real.log (a/b) :=
  29     (le_div_iff₀ (by linarith : 0 < Real.log q)).mp hk
  30   nlinarith [mul_le_mul_of_nonneg_left hlogq (Nat.cast_nonneg k)]

  32 theorem skew_block_population {a b δ d n : ℝ}
  33     (ha : 0 < a) (hb : 1 ≤ b) (han : a ≤ n) (hδ : 0 < δ) (hd : 0 < d)
  34     (hn : 2 ≤ Real.log n) (hdeg : δ ≤ d*Real.log n)
  35     (k : ℕ) (hk : 0 < k) (hks : 2*(k:ℝ) ≤ localLogSkewness a b δ d n) :
  36     b ≤ a*((min 1 (δ/d)/(k:ℝ))/2)^k := by
  37   let q := d*(Real.log n)^2/δ
  38   let rate := min 1 (δ/d)
  39   have hkpos : (0:ℝ) < k := by exact_mod_cast hk
  40   have hrate : 0 < rate := lt_min (by norm_num) (div_pos hδ hd)
  41   have hqlog : Real.log n ≤ q := skew_denominator_ge_log hδ hd hn hdeg
  42   have hq2 : 2 ≤ q := hn.trans hqlog
  43   have hqpos : 0 < q := by linarith
  44   have hklog : (k:ℝ) ≤ Real.log n := skew_threshold_le_log ha hb han hδ hd hn hdeg k hks
  45   have hscale : 2*(k:ℝ)/rate ≤ q^2 := by
  46     apply (div_le_iff₀ hrate).mpr
  47     by_cases hδd : 1 ≤ δ/d
  48     · have hl : rate = 1 := min_eq_left hδd
  49       rw [hl]
  50       nlinarith
  51     · have hl : rate = δ/d := min_eq_right (le_of_not_ge hδd)
  52       have heq : q*rate = (Real.log n)^2 := by dsimp [q]; rw [hl]; field_simp
  53       have hh : q*rate ≤ q^2*rate := by nlinarith [mul_nonneg hrate.le (by nlinarith : 0 ≤ q^2-q)]
  54       nlinarith
  55   have hpow : (2*(k:ℝ)/rate)^k ≤ a/b := by
  56     apply (Real.log_le_log_iff (pow_pos (div_pos (by positivity) hrate) _) (div_pos ha (by linarith))).mp
  57     rw [Real.log_pow]
  58     have hlogscale := Real.log_le_log (div_pos (by positivity : 0 < 2*(k:ℝ)) hrate) hscale
  59     rw [Real.log_pow] at hlogscale
  60     have hk' : 2*(k:ℝ)*Real.log q ≤ Real.log (a/b) :=
  61       (le_div_iff₀ (Real.log_pos (by linarith : 1 < q))).mp hks
  62     nlinarith [mul_le_mul_of_nonneg_left hlogscale hkpos.le]
  63   have hpowerpos : 0 < (2*(k:ℝ)/rate)^k := pow_pos (div_pos (by positivity) hrate) _
  64   have hprod : ((rate/(k:ℝ))/2)^k * (2*(k:ℝ)/rate)^k = 1 := by
  65     rw [← mul_pow]
  66     have heq : ((rate/(k:ℝ))/2) * (2*(k:ℝ)/rate) = 1 := by field_simp
  67     rw [heq, one_pow]
  68   change b ≤ a*((rate/(k:ℝ))/2)^k
  69   apply (mul_le_mul_right hpowerpos).mp
  70   rw [mul_assoc, hprod, mul_one]
  71   have hh := (le_div_iff₀ (by linarith : 0 < b)).mp hpow
  72   nlinarith

  74 theorem block_length_mean (N k : ℕ) (hk : 0 < k) (hkN : k ≤ N)
  75     (δ d m : ℝ) (hδ : 0 ≤ δ) (hd : 0 < d) (hm : 0 < m)
  76     (haverage : 2*m = (N:ℝ)*d) :
  77     min 1 (δ/d)/(k:ℝ) ≤ ((N/k : ℕ):ℝ)*(δ/m) := by
  78   have hs : 1 ≤ N/k := Nat.div_pos hkN hk
  79   have hN : N < (N/k)*k+k := Nat.lt_div_mul_add hk
  80   have hnk : k ≤ (N/k)*k := by nlinarith
  81   have hfloor : (N:ℝ) ≤ 2*((N/k : ℕ):ℝ)*(k:ℝ) := by exact_mod_cast (by nlinarith : N ≤ 2*(N/k)*k)
  82   have hkpos : (0:ℝ) < k := by exact_mod_cast hk
  83   apply le_trans (div_le_div_of_nonneg_right (min_le_right _ _) hkpos.le)
  84   apply (div_le_iff₀ hkpos).mpr
  85   apply (div_le_iff₀ hd).mpr
  86   have hh : (N:ℝ)*δ ≤ 2*((N/k : ℕ):ℝ)*(k:ℝ)*δ :=
  87     mul_le_mul_of_nonneg_right hfloor hδ
  88   apply (le_of_mul_le_mul_right (a := m) ?_ hm)
  89   field_simp
  90   nlinarith [mul_nonneg hδ (sub_nonneg.mpr hfloor)]
