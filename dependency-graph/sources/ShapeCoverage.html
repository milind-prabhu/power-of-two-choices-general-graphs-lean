   3 /-! A geometric embedding of an unlabelled canonical shape in the actual
   4 residual graph forces the class-labelled shape event. Edge classes and
   5 distinct occurrence witnesses are constructed from the actual execution.
   6 -/
   7 namespace GraphChoices.Patterns
   8 open Finset
   9 open ThresholdGreedy
  10 
  11 abbrev UnlabelledShape (m : ℕ) := {t : Tree Unit // t ∈ Tree.treesOfNumNodesEq m}
  12 
  13 def skeletonPattern {m : ℕ} (s : UnlabelledShape m) : ParentPattern Unit m :=
  14   shapePattern (s, fun _ => ())
  15 
  16 variable {E L R C : Type*} [Fintype E] [DecidableEq E] [Nonempty E]
  17   [Fintype L] [Fintype R] [Fintype C]
  18   [DecidableEq L] [DecidableEq R] [DecidableEq C] {T m : ℕ}
  19 
  20 /-- The only input is an unlabelled geometric embedding. Classes, edge labels
  21 and globally distinct occurrence witnesses follow from actual adjacency. -/
  22 theorem mem_heavyShapeEvent_of_geometric_embedding (H : C → Peeling.Graph L R)
  23     (encode : E → Arrival L R C) (α : C → ℕ)
  24     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
  25     (ω : Fin T → E) (s : UnlabelledShape m) (f : Fin (m+1) → L ⊕ R)
  26     (hf : Function.Injective f) (hroot : ∃ v, f ⟨0, by omega⟩ = Sum.inr v)
  27     (hside : ∀ i, isLeft (f (childIndex i)) = (skeletonPattern s).childLeft i)
  28     (hadj : ∀ i, (arrivalGraph (residualInstance α (sampleArrivals encode ω))).Adj
  29       (f (childIndex i)) (f (parentIndex (skeletonPattern s) i)))
  30     (hheavy : m ≤ 2*(univ.filter fun i : Fin m => isLeft (f (childIndex i)) = true).card) :
  31     ω ∈ heavyShapeEvent H encode α T m := by
  32   classical
  33   have ht : ∀ i : Fin m, ∃ t : Fin T, residualTime α encode ω t ∧
  34       if (skeletonPattern s).childLeft i then
  35         f (childIndex i) = Sum.inl (encode (ω t)).left ∧
  36         f (parentIndex (skeletonPattern s) i) = Sum.inr (encode (ω t)).right
  37       else
  38         f (parentIndex (skeletonPattern s) i) = Sum.inl (encode (ω t)).left ∧
  39         f (childIndex i) = Sum.inr (encode (ω t)).right := by
  40     intro i
  41     obtain ⟨_, t, hres, hxy⟩ := (residual_adj_iff_time encode α ω _ _).mp (hadj i)
  42     refine ⟨t, hres, ?_⟩
  43     have hsi := hside i
  44     cases hs : (skeletonPattern s).childLeft i <;> simp only [hs, Bool.false_eq_true, if_false, if_true]
  45     · rcases hxy with hxy | hxy
  46       · rw [← hxy.1, hs] at hsi
  47         simp [isLeft] at hsi
  48       · exact ⟨hxy.1.symm, hxy.2.symm⟩
  49     · rcases hxy with hxy | hxy
  50       · exact ⟨hxy.1.symm, hxy.2.symm⟩
  51       · rw [← hxy.2, hs] at hsi
  52         simp [isLeft] at hsi
  53   choose τ hτ using ht
  54   let g : Fin m → E := fun i => ω (τ i)
  55   let labelled : ShapeCode C m := (s, fun i => (encode (g i)).cls)
  56   have he : EdgeData (shapePattern labelled) encode f g := by
  57     intro i
  58     exact ⟨rfl, (hτ i).2⟩
  59   have hp := mem_residualPatternEvent_of_data (shapePattern labelled) H encode α hsource
  60     ω f g hf hroot he (fun i => ⟨τ i, rfl, (hτ i).1⟩)
  61   have hheavy' : m ≤ 2*(univ.filter fun i : Fin m => (shapePattern labelled).childLeft i = true).card := by
  62     simpa only [hside] using hheavy
  63   apply mem_biUnion.mpr
  64   refine ⟨labelled, mem_univ _, ?_⟩
  65   simpa only [if_pos hheavy'] using hp
  66 
  67 end GraphChoices.Patterns
