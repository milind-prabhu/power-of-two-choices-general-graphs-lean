  11 theorem paired_power_bound (a b r : ℝ) (q l : ℕ)
  12     (ha : 0 ≤ a) (hb : 0 ≤ b) (hr : 0 ≤ r) (hql : q ≤ l)
  13     (hab : a*b ≤ r^2) (hbr : b ≤ r) : a^q*b^l ≤ r^(q+l) := by
  14   have hsplit : l = q + (l-q) := by omega
  15   calc
  16     _ = (a*b)^q * b^(l-q) := by nth_rw 1 [hsplit]; rw [pow_add, mul_pow]; ring
  17     _ ≤ (r^2)^q * r^(l-q) :=
  18       mul_le_mul (pow_le_pow_left₀ (mul_nonneg ha hb) hab q) (pow_le_pow_left₀ hb hbr (l-q))
  19         (pow_nonneg hb _) (by positivity)
  20     _ = _ := by rw [← pow_mul, ← pow_add]; congr 1; omega

  22  theorem two_class_product_bound {I : Type*} [Fintype I] (P : I → Prop) [DecidablePred P]
  23     (a b r : ℝ) (ha : 0 ≤ a) (hb : 0 ≤ b) (hr : 0 ≤ r)
  24     (hab : a*b ≤ r^2) (hbr : b ≤ r)
  25     (hheavy : Fintype.card I ≤ 2 * (univ.filter P).card) :
  26     (∏ i : I, if P i then b else a) ≤ r^(Fintype.card I) := by
  27   classical
  28   let l := (univ.filter P).card
  29   let q := (univ.filter fun i => ¬P i).card
  30   have hsum : l+q = Fintype.card I := by
  31     simpa [l,q] using filter_card_add_filter_neg_card_eq_card (s := (univ : Finset I)) P
  32   have hql : q ≤ l := by dsimp [l,q] at *; omega
  33   have heq : (∏ i : I, if P i then b else a) = b^l*a^q := by
  34     rw [prod_ite]
  35     simp [l,q]
  36   rw [heq, mul_comm, ← hsum]
  37   have hp := paired_power_bound a b r q l ha hb hr hql hab hbr
  38   simpa [Nat.add_comm] using hp
