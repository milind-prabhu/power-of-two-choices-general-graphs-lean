   4 /-! Theorem 1.1: a deterministic nonanticipating O(log log n)-competitive
   5 policy for every nonempty simple base graph and every arrival horizon. -/
   6 namespace GraphChoices
   7 open Finset
   8 variable {V E : Type*} [Fintype V] [DecidableEq V]
   9   [Fintype E] [DecidableEq E] [Nonempty E]
  10 
  11 theorem exists_competitive_policy (G : BaseGraph V E) (hsimple : G.IsSimple) (T : ℕ) :
  12     ∃ p : OnlinePolicy E, expectedCost G (onlineOrientation (T := T) p) ≤
  13       120000*loglogFactor (Fintype.card V)*expectedOpt G T := by
  14   classical
  15   let x : E := Classical.choice (inferInstance : Nonempty E)
  16   have hV : 2 ≤ Fintype.card V := by
  17     have hh := card_le_univ ({G.left x, G.right x}:Finset V)
  18     simpa [hsimple.1 x] using hh
  19   obtain ⟨b,hD⟩ := exists_preprocessed_split G hsimple
  20   let H := splitGraph G b
  21   let e := splitEdgeEquiv G hsimple b
  22   letI : Nonempty H := ⟨e x⟩
  23   have hvertices : Fintype.card (V ⊕ V) ≤ 2*Fintype.card V := by simp only [Fintype.card_sum]; omega
  24   obtain ⟨p,hp⟩ := Peeling.exists_bipartite_policy_all_horizons H (2*Fintype.card V) T hvertices hD
  25   let q := pullbackPolicy e p
  26   have hq : expectedCost (G.split b) (onlineOrientation (T := T) q) ≤
  27       (30000*loglogFactor (2*Fintype.card V))*expectedOpt (G.split b) T := by
  28     rw [relabel_expectedCost (G.split b) (Peeling.graphBase H) e
  29       (splitEdgeEquiv_left G hsimple b) (splitEdgeEquiv_right G hsimple b) T p,
  30       relabel_expectedOpt (G.split b) (Peeling.graphBase H) e
  31         (splitEdgeEquiv_left G hsimple b) (splitEdgeEquiv_right G hsimple b) T]
  32     exact hp
  33   have hfactor := loglogFactor_ge_one (2*Fintype.card V)
  34   have hc : 0 ≤ 30000*loglogFactor (2*Fintype.card V) := by positivity
  35   refine ⟨collapsePolicy b q, (online_expected_transfer G b q _ hc hq).trans ?_⟩
  36   have hlog := loglogFactor_square_bound (Fintype.card V) (2*Fintype.card V) hV (by nlinarith)
  37   have ho := expectedOpt_nonneg G T
  38   have hm := mul_le_mul_of_nonneg_right hlog (show 0 ≤ 60000*expectedOpt G T by positivity)
  39   nlinarith
  40 
  41 /-- The exact proposition stated in Targets.lean, now proved. -/
  42 theorem mainTheorem : MainTheorem := by
  43   refine ⟨120000, by norm_num, ?_⟩
  44   intro n m G hsimple T
  45   simpa only [Fintype.card_fin] using exists_competitive_policy G hsimple T
