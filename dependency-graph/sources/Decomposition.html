   3 /-! Iteration of the proved peeling round: every finite valid schedule yields
   4 edge-disjoint pieces with their per-round degree bounds. A final cutoff of zero
   5 gives an exact edge partition, not just a collection of partial subgraphs. -/
   6 namespace GraphChoices.Peeling
   7 open Finset
   8 variable {L R : Type*} [Fintype L] [Fintype R] [DecidableEq L] [DecidableEq R]
   9 
  10 def unionPieces : List (Graph L R) → Graph L R
  11   | [] => ∅
  12   | P::Ps => P ∪ unionPieces Ps
  13 
  14 def finalBudget : ℕ → List (ℕ × ℕ) → ℕ
  15   | D, [] => D
  16   | _, (k,_)::ss => finalBudget k ss
  17 
  18 /-- The left cap and right cap promised for each successive piece. -/
  19 def pieceBudgets : ℕ → List (ℕ × ℕ) → List (ℕ × ℕ)
  20   | _, [] => []
  21   | D, (k,b)::ss => (D,D*b) :: pieceBudgets k ss
  22 
  23 inductive PeelingRun : Graph L R → ℕ → List (ℕ × ℕ) → List (Graph L R) → Graph L R → Prop
  24   | nil (H : Graph L R) (D : ℕ) (hD : ∀ u, leftDegree H u ≤ D) : PeelingRun H D [] [] H
  25   | cons {H P : Graph L R} {D k b : ℕ} {ss : List (ℕ × ℕ)} {Ps : List (Graph L R)} {F : Graph L R}
  26       (hP : P ⊆ H) (hPL : ∀ u, leftDegree P u ≤ D)
  27       (hPR : ∀ v, rightDegree P v ≤ D*b)
  28       (tail : PeelingRun (H \ P) k ss Ps F) :
  29       PeelingRun H D ((k,b)::ss) (P::Ps) F
  30 
  31  theorem mem_subset_unionPieces {P : Graph L R} {Ps : List (Graph L R)} (hP : P ∈ Ps) :
  32     P ⊆ unionPieces Ps := by
  33   induction Ps with
  34   | nil => simp at hP
  35   | cons Q Qs ih =>
  36     rcases List.mem_cons.mp hP with rfl | hp
  37     · exact subset_union_left
  38     · exact (ih hp).trans subset_union_right
  39 
  40 /-- Finite construction of a complete peeling execution. -/
  41  theorem exists_peeling_run (G H : Graph L R) (hHG : H ⊆ G) (D : ℕ)
  42     (hD : ∀ u, leftDegree H u ≤ D) (ss : List (ℕ × ℕ))
  43     (hexpand : ∀ kb ∈ ss, PeelingExpansion G kb.1 kb.2) :
  44     ∃ Ps F, PeelingRun H D ss Ps F := by
  45   induction ss generalizing H D with
  46   | nil => exact ⟨[],H,PeelingRun.nil H D hD⟩
  47   | cons kb ss ih =>
  48     obtain ⟨k,b⟩ := kb
  49     obtain ⟨P,hPH,hnext,hPR⟩ := exists_peeling_piece H k b D hD
  50       ((hexpand (k,b) (by simp)).mono hHG)
  51     obtain ⟨Ps,F,htail⟩ := ih (H \ P) (sdiff_subset.trans hHG) k hnext
  52       (fun kb hkb => hexpand kb (by simp [hkb]))
  53     refine ⟨P::Ps,F,PeelingRun.cons hPH ?_ hPR htail⟩
  54     intro u
  55     exact (leftDegree_mono hPH u).trans (hD u)
  56 
  57  theorem PeelingRun.length_eq {H F : Graph L R} {D : ℕ}
  58     {ss : List (ℕ × ℕ)} {Ps : List (Graph L R)} (h : PeelingRun H D ss Ps F) :
  59     Ps.length = ss.length := by
  60   induction h with
  61   | nil => rfl
  62   | cons _ _ _ _ ih => simpa using ih
  63 
  64  theorem PeelingRun.degree_bounds {H F : Graph L R} {D : ℕ}
  65     {ss : List (ℕ × ℕ)} {Ps : List (Graph L R)} (h : PeelingRun H D ss Ps F) :
  66     List.Forall₂ (fun P kb => (∀ u, leftDegree P u ≤ kb.1) ∧
  67       (∀ v, rightDegree P v ≤ kb.2)) Ps (pieceBudgets D ss) := by
  68   induction h with
  69   | nil => exact List.Forall₂.nil
  70   | cons _ hPL hPR _ ih => exact List.Forall₂.cons ⟨hPL,hPR⟩ ih
  71 
  72  theorem PeelingRun.final_degree_bound {H F : Graph L R} {D : ℕ}
  73     {ss : List (ℕ × ℕ)} {Ps : List (Graph L R)} (h : PeelingRun H D ss Ps F) :
  74     ∀ u, leftDegree F u ≤ finalBudget D ss := by
  75   induction h with
  76   | nil _ _ hD => exact hD
  77   | cons _ _ _ _ ih => exact ih
  78 
  79 /-- Every recorded execution really is an edge-disjoint extraction; the final
  80 residual is precisely the complement of all pieces in the original graph. -/
  81  theorem PeelingRun.partition_properties {H F : Graph L R} {D : ℕ}
  82     {ss : List (ℕ × ℕ)} {Ps : List (Graph L R)} (h : PeelingRun H D ss Ps F) :
  83     unionPieces Ps ⊆ H ∧ F = H \ unionPieces Ps ∧ Ps.Pairwise Disjoint := by
  84   induction h with
  85   | nil => simp [unionPieces]
  86   | @cons H P D k b ss Ps F hPH hPL hPR htail ih =>
  87     obtain ⟨hPs,hF,hdisj⟩ := ih
  88     refine ⟨union_subset hPH (hPs.trans sdiff_subset), ?_, ?_⟩
  89     · rw [hF]
  90       ext p
  91       simp [unionPieces]
  92       tauto
  93     · apply List.pairwise_cons.mpr
  94       refine ⟨?_,hdisj⟩
  95       intro Q hQ
  96       apply Finset.disjoint_left.mpr
  97       intro e heP heQ
  98       have he : e ∈ H \ P := hPs (mem_subset_unionPieces hQ heQ)
  99       exact (mem_sdiff.mp he).2 heP

 101  theorem graph_empty_of_left_degrees_zero (H : Graph L R)
 102     (h : ∀ u, leftDegree H u ≤ 0) : H = ∅ := by
 103   apply eq_empty_iff_forall_notMem.mpr
 104   intro p hp
 105   have hn : p.2 ∈ leftNeighbors H p.1 := by simpa [leftNeighbors] using hp
 106   have hc : 0 < leftDegree H p.1 := card_pos.mpr ⟨p.2,hn⟩
 107   have := h p.1
 108   omega

 110 /-- An exact edge partition with one piece per scheduled round and all degree
 111 bounds, provided the schedule ends at zero. -/
 112  theorem exists_edge_partition (G : Graph L R) (D : ℕ)
 113     (hD : ∀ u, leftDegree G u ≤ D) (ss : List (ℕ × ℕ))
 114     (hexpand : ∀ kb ∈ ss, PeelingExpansion G kb.1 kb.2)
 115     (hfinal : finalBudget D ss = 0) :
 116     ∃ Ps : List (Graph L R), Ps.length = ss.length ∧ Ps.Pairwise Disjoint ∧
 117       unionPieces Ps = G ∧
 118       List.Forall₂ (fun P kb => (∀ u, leftDegree P u ≤ kb.1) ∧
 119         (∀ v, rightDegree P v ≤ kb.2)) Ps (pieceBudgets D ss) := by
 120   obtain ⟨Ps,F,h⟩ := exists_peeling_run G G (subset_refl _) D hD ss hexpand
 121   obtain ⟨hsub,hF,hpair⟩ := h.partition_properties
 122   have hz : F = ∅ := graph_empty_of_left_degrees_zero F (by simpa [hfinal] using h.final_degree_bound)
 123   have hcover : unionPieces Ps = G := by
 124     apply subset_antisymm hsub
 125     intro e he
 126     by_contra hn
 127     have : e ∈ F := by rw [hF]; exact mem_sdiff.mpr ⟨he,hn⟩
 128     simpa [hz] using this
 129   exact ⟨Ps,h.length_eq,hpair,hcover,h.degree_bounds⟩
