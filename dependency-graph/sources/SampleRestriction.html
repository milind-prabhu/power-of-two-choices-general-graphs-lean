   8 def Instance.restrict {V : Type*} {M N : ℕ} (I : Instance V N) (f : Fin M ↪ Fin N) :
   9     Instance V M where
  10   left t := I.left (f t)
  11   right t := I.right (f t)
  12 
  13 theorem opt_restrict_le {V : Type*} [Fintype V] [DecidableEq V] {M N : ℕ}
  14     (I : Instance V N) (f : Fin M ↪ Fin N) : opt (I.restrict f) ≤ opt I := by
  15   obtain ⟨a,ha⟩ := exists_exact_opt I
  16   apply (opt_le_maxLoad (I.restrict f) (fun t => a (f t))).trans
  17   rw [← ha]
  18   apply (maxLoad_le_iff _ _ _).mpr
  19   intro v
  20   apply le_trans _ (load_le_maxLoad I a v)
  21   apply card_le_card_of_injOn f
  22   · intro t ht
  23     change t ∈ univ.filter (fun t => destination (I.restrict f) (fun t => a (f t)) t = v) at ht
  24     change f t ∈ univ.filter (fun t => destination I a t = v)
  25     simpa only [mem_filter, mem_univ, true_and, destination, Instance.restrict] using ht
  26   · exact fun _ _ _ _ h => f.injective h

  28 theorem average_product_fst {Ω Ω' : Type*} [Fintype Ω] [Fintype Ω'] [Nonempty Ω']
  29     (f : Ω → ℝ) : average (fun p : Ω × Ω' => f p.1) = average f := by
  30   simp only [average, Fintype.sum_prod_type, sum_const, card_univ, nsmul_eq_mul,
  31     ← mul_sum, Fintype.card_prod, Nat.cast_mul]
  32   have h : (Fintype.card Ω' : ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  33   by_cases hΩ : Fintype.card Ω = 0
  34   · simp [hΩ]
  35   · have hΩ' : (Fintype.card Ω : ℝ) ≠ 0 := by exact_mod_cast hΩ
  36     field_simp

  38 def appendSamples {E : Type*} (M Q : ℕ) :
  39     (Fin M → E) × (Fin Q → E) ≃ (Fin (M+Q) → E) :=
  40   (Equiv.sumArrowEquivProdArrow (Fin M) (Fin Q) E).symm.trans
  41     (Equiv.arrowCongr finSumFinEquiv (Equiv.refl E))
  42 
  43 theorem appendSamples_left {E : Type*} (M Q : ℕ)
  44     (p : (Fin M → E) × (Fin Q → E)) (t : Fin M) :
  45     appendSamples M Q p (Fin.castAdd Q t) = p.1 t := by
  46   change appendSamples M Q p (finSumFinEquiv (Sum.inl t)) = p.1 t
  47   simp only [appendSamples, Equiv.trans_apply, Equiv.arrowCongr_apply, Function.comp_apply, Equiv.refl_apply, Equiv.symm_apply_apply]
  48   rcases p with ⟨p,q⟩
  49   rfl
  50 
  51 theorem average_prefix {E : Type*} [Fintype E] [Nonempty E] (M Q : ℕ)
  52     (f : (Fin M → E) → ℝ) :
  53     average (fun ω : Fin (M+Q) → E => f (fun t => ω (Fin.castAdd Q t))) = average f := by
  54   rw [← average_equiv (appendSamples M Q)]
  55   simp_rw [appendSamples_left]
  56   exact average_product_fst f

  58 theorem expectedOpt_mono_horizon {V E : Type*} [Fintype V] [DecidableEq V]
  59     [Fintype E] [Nonempty E] (G : BaseGraph V E) {M N : ℕ} (hMN : M ≤ N) :
  60     expectedOpt G M ≤ expectedOpt G N := by
  61   obtain ⟨Q,rfl⟩ := Nat.exists_eq_add_of_le hMN
  62   unfold expectedOpt
  63   rw [← average_prefix M Q (fun ω => (opt (G.sample ω):ℝ))]
  64   apply average_mono
  65   intro ω
  66   have hh := opt_restrict_le (G.sample ω) ⟨Fin.castAdd Q, Fin.castAdd_injective M Q⟩
  67   exact_mod_cast hh
