  12 theorem eventProbability_union_le {Ω : Type*} [Fintype Ω] [DecidableEq Ω]
  13     (A B : Finset Ω) : eventProbability (A ∪ B) ≤ eventProbability A + eventProbability B := by
  14   unfold eventProbability
  15   rw [← add_div]
  16   apply div_le_div_of_nonneg_right _ (Nat.cast_nonneg _)
  17   exact_mod_cast card_union_le A B

  19 end GraphChoices

  22 open Finset
  23 variable {E L R C : Type*} [Fintype E] [DecidableEq E] [Nonempty E]

  26 /-- A genuine geometric tree in an actual replay component. Its vertices are
  27 a subset of that component, and every edge is an actual replay adjacency. -/
  28 def HasLargeHeavyTree (es : List (Arrival L R C)) (k : ℕ) : Prop :=
  29   ∃ v : L ⊕ R,
  30   ∃ s : Set {w : L ⊕ R // w ∈ componentVertices (replayInstance es) v},
  31   ∃ H : SimpleGraph s,
  32     H.IsTree ∧
  33     (∀ x y, H.Adj x y → (arrivalGraph (replayInstance es)).Adj x.val.val y.val.val) ∧
  34     k+1 < Nat.card s ∧
  35     Nat.card {x : s // isLeft x.val.val = false} + 1 ≤
  36       Nat.card {x : s // isLeft x.val.val = true} ∧
  37     ∃ root : s, isLeft root.val.val = false
  38 
  39 noncomputable def largeHeavyTreeEvent (encode : E → Arrival L R C) (α : C → ℕ)
  40     (T k : ℕ) : Finset (Fin T → E) := by
  41   classical
  42   exact univ.filter fun ω => HasLargeHeavyTree (residualArrivals (trace α (sampleArrivals encode ω))) k
  43 
  44 noncomputable def largeLeftDegreeEvent (encode : E → Arrival L R C) (α : C → ℕ)
  45     (T D : ℕ) : Finset (Fin T → E) := by
  46   classical
  47   exact univ.filter fun ω => ∃ u,
  48     D < (listLeftNeighbors (residualArrivals (trace α (sampleArrivals encode ω))) u).card
  49 
  50 /-- The deterministic reduction applied to actual random executions. There is
  51 no assumed tree-extraction or side/degree agreement in this containment. -/
  52 theorem largeComponent_subset_degree_or_tree (encode : E → Arrival L R C) (α : C → ℕ)
  53     (T D k : ℕ) :
  54     largeComponentEvent encode α T ((D+1)*(k+1)) ⊆
  55       largeLeftDegreeEvent encode α T D ∪ largeHeavyTreeEvent encode α T k := by
  56   classical
  57   intro ω hω
  58   by_cases hbad : ω ∈ largeLeftDegreeEvent encode α T D
  59   · exact mem_union_left _ hbad
  60   apply mem_union_right
  61   have hD : ∀ u, (listLeftNeighbors (residualArrivals (trace α (sampleArrivals encode ω))) u).card ≤ D := by
  62     simpa only [largeLeftDegreeEvent, mem_filter, mem_univ, true_and, not_exists, not_lt] using hbad
  63   obtain ⟨v, hv⟩ := (mem_filter.mp hω).2
  64   have hlarge : 1 < (componentVertices (residualInstance α (sampleArrivals encode ω)) v).card := by
  65     have hpos : 1 ≤ (D+1)*(k+1) := Nat.mul_pos (by omega) (by omega)
  66     omega
  67   obtain ⟨s, H, ht, hadj, hsize, hheavy, hroot⟩ :=
  68     replay_component_left_heavy_tree (residualArrivals (trace α (sampleArrivals encode ω))) v D hD hlarge
  69   have hs : k+1 < Nat.card s := by
  70     have hmult : (D+1)*(k+1) < (D+1)*Nat.card s := hv.trans_le hsize
  71     exact (Nat.mul_lt_mul_left (by omega : 0 < D+1)).mp hmult
  72   exact mem_filter.mpr ⟨mem_univ _, v, s, H, ht, hadj, hs, hheavy, hroot (by omega)⟩
  73 
  74 theorem largeLeftDegree_probability_le (encode : E → Arrival L R C) (α : C → ℕ)
  75     (T D : ℕ) (r : ℝ)
  76     (hmean : ∀ u, Real.exp 1*T*(((leftSourceBucket encode u).card : ℝ)/Fintype.card E)/(D+1) ≤ r) :
  77     eventProbability (largeLeftDegreeEvent encode α T D) ≤ Fintype.card L * r^(D+1) := by
  78   have h := residual_left_degree_tail encode α T (D+1) r
  79     (by simpa only [Nat.cast_add, Nat.cast_one] using hmean)
  80   simpa only [largeLeftDegreeEvent, Nat.add_one_le_iff] using h
  81 
  82 theorem largeComponent_probability_le_tree_tail (encode : E → Arrival L R C) (α : C → ℕ)
  83     (T D k : ℕ) (r ε : ℝ)
  84     (hmean : ∀ u, Real.exp 1*T*(((leftSourceBucket encode u).card : ℝ)/Fintype.card E)/(D+1) ≤ r)
  85     (htree : eventProbability (largeHeavyTreeEvent encode α T k) ≤ ε) :
  86     eventProbability (largeComponentEvent encode α T ((D+1)*(k+1))) ≤
  87       Fintype.card L * r^(D+1) + ε := by
  88   calc
  89     _ ≤ eventProbability (largeLeftDegreeEvent encode α T D ∪ largeHeavyTreeEvent encode α T k) :=
  90       eventProbability_mono (largeComponent_subset_degree_or_tree encode α T D k)
  91     _ ≤ eventProbability (largeLeftDegreeEvent encode α T D) +
  92         eventProbability (largeHeavyTreeEvent encode α T k) := eventProbability_union_le _ _
  93     _ ≤ _ := add_le_add (largeLeftDegree_probability_le encode α T D r hmean) htree
  94 
  95 end GraphChoices.ThresholdGreedy

 103 /-- Conditional shattering assembly. `hcoverage` is the remaining deterministic
 104 canonical-enumeration containment; it is not established by this theorem. -/
 105 theorem largeComponent_tail_of_canonical_coverage (H : C → Peeling.Graph L R)
 106     (encode : E → Arrival L R C) (α : C → ℕ) (T D k B : ℕ) (r : ℝ)
 107     (hmean : ∀ u, Real.exp 1*T*(((leftSourceBucket encode u).card : ℝ)/Fintype.card E)/(D+1) ≤ r)
 108     (hcoverage : largeHeavyTreeEvent encode α T k ⊆ largeShapeEvent H encode α T k B)
 109     (hshape : ∀ j ≤ B, eventProbability (heavyShapeEvent H encode α T (k+j)) ≤
 110       (Fintype.card R : ℝ)/2^(k+j)) :
 111     eventProbability (largeComponentEvent encode α T ((D+1)*(k+1))) ≤
 112       Fintype.card L * r^(D+1) + 2*(Fintype.card R : ℝ)/2^k := by
 113   apply largeComponent_probability_le_tree_tail encode α T D k r _ hmean
 114   exact (eventProbability_mono hcoverage).trans (largeShape_probability_le H encode α T k B hshape)
