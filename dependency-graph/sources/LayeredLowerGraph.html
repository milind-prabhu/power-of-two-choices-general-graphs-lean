   6 abbrev Vertices (b t m : ℕ) := (i : Fin (b+1)) × (Fin (t^i.val) × Fin m)
   7 abbrev Blocks (b t : ℕ) := (i : Fin b) × Fin (t^i.val)
   8 abbrev Edges (b t m : ℕ) := Blocks b t × (Fin m × Fin (t*m))
   9 
  10 def leftVertex {b t m : ℕ} (p : Blocks b t × Fin m) : Vertices b t m :=
  11   ⟨p.1.1.castSucc,(p.1.2,p.2)⟩
  12 
  13 def childGroup {t m i : ℕ} (g : Fin (t^i)) (r : Fin (t*m)) : Fin (t^(i+1)) :=
  14   Fin.cast (pow_succ t i).symm (finProdFinEquiv (g,(finProdFinEquiv.symm r).1))
  15 
  16 def rightVertex {b t m : ℕ} (p : Blocks b t × Fin (t*m)) : Vertices b t m :=
  17   ⟨p.1.1.succ,(childGroup p.1.2 p.2,(finProdFinEquiv.symm p.2).2)⟩
  18 
  19 def graph (b t m : ℕ) : BaseGraph (Vertices b t m) (Edges b t m) where
  20   left e := leftVertex (e.1,e.2.1)
  21   right e := rightVertex (e.1,e.2.2)
  22 
  23 theorem leftVertex_injective (b t m : ℕ) : Function.Injective (@leftVertex b t m) := by
  24   rintro ⟨⟨i,g⟩,a⟩ ⟨⟨j,h⟩,a'⟩ he
  25   have hij : i = j := Fin.ext (congrArg (fun v => v.1.val) he)
  26   subst j
  27   have hp : (g,a) = (h,a') := eq_of_heq (Sigma.mk.inj_iff.mp he).2
  28   obtain ⟨rfl,rfl⟩ := Prod.mk.inj hp
  29   rfl
  30 
  31 theorem rightVertex_injective (b t m : ℕ) : Function.Injective (@rightVertex b t m) := by
  32   rintro ⟨⟨i,g⟩,r⟩ ⟨⟨j,h⟩,s⟩ he
  33   have hij : i = j := by
  34     have hh := congrArg (fun v => v.1.val) he
  35     exact Fin.ext (by simpa only [rightVertex, Fin.val_succ] using Nat.succ.inj hh)
  36   subst j
  37   have hp : (childGroup g r,(finProdFinEquiv.symm r).2) =
  38       (childGroup h s,(finProdFinEquiv.symm s).2) := eq_of_heq (Sigma.mk.inj_iff.mp he).2
  39   have hc := Fin.cast_injective (pow_succ t i.val).symm (congrArg Prod.fst hp)
  40   have hpair := finProdFinEquiv.injective hc
  41   have hg : g = h := congrArg Prod.fst hpair
  42   have hr1 : (finProdFinEquiv.symm r).1 = (finProdFinEquiv.symm s).1 := congrArg (fun p : Fin (t^i.val) × Fin t => p.2) hpair
  43   have hr2 : (finProdFinEquiv.symm r).2 = (finProdFinEquiv.symm s).2 := congrArg (fun p : Fin (t^(i.val+1)) × Fin m => p.2) hp
  44   have hr : r = s := finProdFinEquiv.symm.injective (Prod.ext hr1 hr2)
  45   subst h
  46   subst s
  47   rfl

  49 theorem graph_isSimple (b t m : ℕ) : (graph b t m).IsSimple := by
  50   constructor
  51   · intro e he
  52     have hh := congrArg (fun v => v.1.val) he
  53     change e.1.1.val = e.1.1.val+1 at hh
  54     omega
  55   · intro e f hef
  56     rcases hef with ⟨hl,hr⟩ | ⟨hl,hr⟩
  57     · have hleft := leftVertex_injective b t m hl
  58       have hright := rightVertex_injective b t m hr
  59       have hc : e.1 = f.1 := congrArg (fun p : Blocks b t × Fin m => p.1) hleft
  60       have ha : e.2.1 = f.2.1 := congrArg (fun p : Blocks b t × Fin m => p.2) hleft
  61       have hr : e.2.2 = f.2.2 := congrArg (fun p : Blocks b t × Fin (t*m) => p.2) hright
  62       exact Prod.ext hc (Prod.ext ha hr)
  63     · have h1 := congrArg (fun v => v.1.val) hl
  64       have h2 := congrArg (fun v => v.1.val) hr
  65       change e.1.1.val = f.1.1.val+1 at h1
  66       change e.1.1.val+1 = f.1.1.val at h2
  67       omega

  69 def layerSum (t b : ℕ) : ℕ := ∑ i ∈ range b, t^i
  70 
  71 theorem card_blocks (b t : ℕ) : Fintype.card (Blocks b t) = layerSum t b := by
  72   simp only [Blocks, Fintype.card_sigma, Fintype.card_fin, layerSum, Fin.sum_univ_eq_sum_range]
  73 
  74 theorem card_vertices (b t m : ℕ) : Fintype.card (Vertices b t m) = m*layerSum t (b+1) := by
  75   simp only [Vertices, Fintype.card_sigma, Fintype.card_prod, Fintype.card_fin,
  76     ← sum_mul, Fin.sum_univ_eq_sum_range, layerSum]
  77   ring
  78 
  79 theorem card_edges (b t m : ℕ) : Fintype.card (Edges b t m) = m^2*t*layerSum t b := by
  80   simp only [Edges, Fintype.card_prod, Fintype.card_fin, card_blocks]
  81   ring
  82 
  83 theorem layerSum_succ (t b : ℕ) : layerSum t (b+1) = layerSum t b+t^b := sum_range_succ _ _
  84 
  85 theorem layerSum_shift (t b : ℕ) : layerSum t (b+1) = 1+t*layerSum t b := by
  86   simp only [layerSum, sum_range_succ', pow_succ, ← sum_mul, pow_zero]
  87   ring
  88 
  89 theorem layerSum_pos (t b : ℕ) (hb : 0 < b) : 0 < layerSum t b := by
  90   have hh : (1:ℕ) ≤ layerSum t b := by
  91     simpa only [pow_zero] using (single_le_sum (f := fun i => t^i) (fun _ _ => Nat.zero_le _) (mem_range.mpr hb))
  92   omega

  94 theorem layerSum_upper (t b : ℕ) (ht : 1 ≤ t) : layerSum t (b+1) ≤ (b+1)*t^b := by
  95   apply le_trans (sum_le_card_nsmul _ _ (t^b) ?_) (by simp)
  96   intro i hi
  97   exact Nat.pow_le_pow_right (by omega) (by have := mem_range.mp hi; omega)
