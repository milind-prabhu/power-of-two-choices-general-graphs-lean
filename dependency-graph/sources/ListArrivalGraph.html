   4 /-! Agreement between the graph of a replay, distinct list neighborhoods, and
   5 the bipartite side annotation needed by component pruning. Multiplicities are
   6 forgotten only in graph adjacency; the original list keeps every occurrence. -/
   7 namespace GraphChoices.ThresholdGreedy
   8 open Finset
   9 variable {L R C : Type*} [Fintype L] [Fintype R]
  10   [DecidableEq L] [DecidableEq R] [DecidableEq C]
  11 
  12 def isLeft : L ⊕ R → Bool := Sum.elim (fun _ => true) (fun _ => false)
  13 
  14 theorem mem_listLeftNeighbors (es : List (Arrival L R C)) (u : L) (v : R) :
  15     v ∈ listLeftNeighbors es u ↔ ∃ e ∈ es, e.left = u ∧ e.right = v := by
  16   simp only [listLeftNeighbors, List.mem_toFinset, List.mem_map, List.mem_filter,
  17     decide_eq_true_eq]
  18   constructor
  19   · rintro ⟨e, ⟨he, hl⟩, hr⟩
  20     exact ⟨e, he, hl, hr⟩
  21   · rintro ⟨e, he, hl, hr⟩
  22     exact ⟨e, ⟨he, hl⟩, hr⟩
  23 
  24 theorem replay_adj_iff (es : List (Arrival L R C)) (x y : L ⊕ R) :
  25     (arrivalGraph (replayInstance es)).Adj x y ↔
  26       x ≠ y ∧ ∃ e ∈ es,
  27         (Sum.inl e.left = x ∧ Sum.inr e.right = y) ∨
  28         (Sum.inl e.left = y ∧ Sum.inr e.right = x) := by
  29   constructor
  30   · rintro ⟨hne, t, ht⟩
  31     exact ⟨hne, es.get t, List.get_mem _ _, ht⟩
  32   · rintro ⟨hne, e, he, hxy⟩
  33     obtain ⟨t, ht⟩ := List.mem_iff_get.mp he
  34     refine ⟨hne, t, ?_⟩
  35     simpa only [replayInstance, ht] using hxy
  36 
  37 theorem replay_adj_sides (es : List (Arrival L R C)) (x y : L ⊕ R)
  38     (hxy : (arrivalGraph (replayInstance es)).Adj x y) : isLeft x ≠ isLeft y := by
  39   obtain ⟨_, e, _, h | h⟩ := (replay_adj_iff es x y).mp hxy
  40   · rcases h with ⟨rfl, rfl⟩
  41     simp [isLeft]
  42   · rcases h with ⟨rfl, rfl⟩
  43     simp [isLeft]
  44 
  45 theorem replay_adj_inl_inr (es : List (Arrival L R C)) (u : L) (v : R) :
  46     (arrivalGraph (replayInstance es)).Adj (Sum.inl u) (Sum.inr v) ↔
  47       v ∈ listLeftNeighbors es u := by
  48   rw [replay_adj_iff, mem_listLeftNeighbors]
  49   simp
  50 
  51 theorem replay_left_degree (es : List (Arrival L R C)) (u : L) :
  52     let G := arrivalGraph (replayInstance es)
  53     letI : DecidableRel G.Adj := Classical.decRel _
  54     G.degree (Sum.inl u) = (listLeftNeighbors es u).card := by
  55   classical
  56   dsimp only
  57   have heq : (arrivalGraph (replayInstance es)).neighborFinset (Sum.inl u) =
  58       (listLeftNeighbors es u).image Sum.inr := by
  59     ext w
  60     cases w with
  61     | inl a =>
  62       simp only [SimpleGraph.mem_neighborFinset, mem_image, Sum.inr_ne_inl, and_false, exists_false]
  63       constructor
  64       · intro h
  65         exact (replay_adj_sides es _ _ h) rfl
  66       · exact False.elim
  67     | inr v => simp [replay_adj_inl_inr]
  68   rw [← SimpleGraph.card_neighborFinset_eq_degree, heq, card_image_of_injective _ Sum.inr_injective]
  69 
  70 end GraphChoices.ThresholdGreedy
