   4 /-! Combines the fixed-pattern assignment count with the actual residual-tree
   5 probability bound. Embeddings carry real base-edge labels; uniqueness follows
   6 from the base graph having at most one label per bipartite endpoint pair. -/

  14 def childIndex (i : Fin m) : Fin (m+1) := ⟨i.val+1,by omega⟩
  15 
  16 def parentIndex (p : ParentPattern C m) (i : Fin m) : Fin (m+1) :=
  17   ⟨(p.parent i).val,by have := (p.parent i).isLt; have := i.isLt; omega⟩
  18 
  19 def EdgeData (p : ParentPattern C m) (encode : E → Arrival L R C)
  20     (f : Fin (m+1) → L ⊕ R) (g : Fin m → E) : Prop :=
  21   ∀ i, (encode (g i)).cls = p.cls i ∧
  22     if p.childLeft i then
  23       f (childIndex i) = Sum.inl (encode (g i)).left ∧
  24       f (parentIndex p i) = Sum.inr (encode (g i)).right
  25     else
  26       f (parentIndex p i) = Sum.inl (encode (g i)).left ∧
  27       f (childIndex i) = Sum.inr (encode (g i)).right
  28 
  29 noncomputable def embeddingSet (p : ParentPattern C m) (H : C → Peeling.Graph L R)
  30     (encode : E → Arrival L R C) :
  31     Finset ((Fin (m+1) → L ⊕ R) × (Fin m → E)) := by
  32   classical
  33   exact univ.filter fun fg => fg.1 ∈ patternAssignments p H ∧ Function.Injective fg.1 ∧ EdgeData p encode fg.1 fg.2
  34 
  35  theorem embedding_projection_inj (p : ParentPattern C m) (H : C → Peeling.Graph L R)
  36     (encode : E → Arrival L R C)
  37     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right)) :
  38     Set.InjOn (fun fg : ((Fin (m+1) → L ⊕ R) × (Fin m → E)) => fg.1)
  39       (↑(embeddingSet p H encode) : Set _) := by
  40   classical
  41   intro a ha b hb hab
  42   change a.1 = b.1 at hab
  43   have hA := (mem_filter.mp ha).2.2.2
  44   have hB := (mem_filter.mp hb).2.2.2
  45   apply Prod.ext hab
  46   funext i
  47   apply hsimple
  48   have hx := (hA i).2
  49   have hy := (hB i).2
  50   rw [hab] at hx
  51   cases hs : p.childLeft i <;> simp only [hs, Bool.false_eq_true, if_false, if_true] at hx hy
  52   · exact Prod.ext (Sum.inl.inj (hx.1.symm.trans hy.1)) (Sum.inr.inj (hx.2.symm.trans hy.2))
  53   · exact Prod.ext (Sum.inl.inj (hx.1.symm.trans hy.1)) (Sum.inr.inj (hx.2.symm.trans hy.2))
  54 
  55  theorem embeddingSet_card_le (p : ParentPattern C m) (H : C → Peeling.Graph L R)
  56     (encode : E → Arrival L R C)
  57     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right)) :
  58     (embeddingSet p H encode).card ≤ (patternAssignments p H).card := by
  59   classical
  60   exact card_le_card_of_injOn Prod.fst
  61     (fun fg hfg => (mem_filter.mp hfg).2.1) (embedding_projection_inj p H encode hsimple)
  62 
  63 abbrev LeftChildren (p : ParentPattern C m) := {i : Fin m // p.childLeft i = true}
  64 
  65  theorem embedding_left_buckets_distinct (p : ParentPattern C m) (encode : E → Arrival L R C)
  66     (f : Fin (m+1) → L ⊕ R) (g : Fin m → E) (hf : Function.Injective f) (he : EdgeData p encode f g) :
  67     Function.Injective (fun j : LeftChildren p => ((encode (g j.val)).left,p.cls j.val)) := by
  68   intro j k hjk
  69   have hj := (he j.val).2
  70   have hk := (he k.val).2
  71   simp only [j.property, k.property, if_true] at hj hk
  72   have hl := congrArg Prod.fst hjk
  73   have hc : childIndex j.val = childIndex k.val := hf (hj.1.trans ((congrArg Sum.inl hl).trans hk.1.symm))
  74   apply Subtype.ext
  75   apply Fin.ext
  76   have hv := congrArg Fin.val hc
  77   simp only [childIndex] at hv
  78   omega

  80 noncomputable def patternProbabilityFactor (p : ParentPattern C m) (α : C → ℕ)
  81     (DL : C → ℕ) (T : ℕ) : ℝ :=
  82   ((T:ℝ) / Fintype.card E)^m * ∏ j : LeftChildren p,
  83     (Real.exp 1 * T * (DL (p.cls j.val) : ℝ) / Fintype.card E / α (p.cls j.val))^(α (p.cls j.val))
  84 
  85  theorem embedding_probability_le (p : ParentPattern C m) (H : C → Peeling.Graph L R)
  86     (encode : E → Arrival L R C) (α : C → ℕ) (DL : C → ℕ)
  87     (hDL : ∀ u c, (sourceBucket encode u c).card ≤ DL c)
  88     (fg : (Fin (m+1) → L ⊕ R) × (Fin m → E)) (hfg : fg ∈ embeddingSet p H encode) :
  89     eventProbability (fixedResidualEvent α encode T m fg.2) ≤ patternProbabilityFactor (E := E) p α DL T := by
  90   classical
  91   have hf := (mem_filter.mp hfg).2.2.1
  92   have he := (mem_filter.mp hfg).2.2.2
  93   have hp := fixedResidual_probability_le (T := T) (J := LeftChildren p) α encode fg.2
  94     (fun j => (encode (fg.2 j.val)).left) (fun j => p.cls j.val) (fun j => j.val)
  95     (fun j => ⟨rfl,(he j.val).1⟩) (embedding_left_buckets_distinct p encode fg.1 fg.2 hf he)
  96   apply hp.trans
  97   apply mul_le_mul_of_nonneg_left _ (by positivity)
  98   apply prod_le_prod
  99   · intro j _; positivity
 100   · intro j _
 101     apply pow_le_pow_left₀ (by positivity) _
 102     apply div_le_div_of_nonneg_right _ (Nat.cast_nonneg _)
 103     apply div_le_div_of_nonneg_right _ (Nat.cast_nonneg _)
 104     apply mul_le_mul_of_nonneg_left _ (by positivity)
 105     exact_mod_cast hDL (encode (fg.2 j.val)).left (p.cls j.val)

 107 noncomputable def residualPatternEvent (p : ParentPattern C m) (H : C → Peeling.Graph L R)
 108     (encode : E → Arrival L R C) (α : C → ℕ) (T : ℕ) : Finset (Fin T → E) :=
 109   (embeddingSet p H encode).biUnion fun fg => fixedResidualEvent α encode T m fg.2

 111 /-- Fixed-pattern union bound combining the concrete embedding count and raw
 112 residual-tree probability. Threshold parameter simplification is still separate. -/
 113  theorem residualPattern_probability_le (p : ParentPattern C m) (H : C → Peeling.Graph L R)
 114     (encode : E → Arrival L R C) (α : C → ℕ) (DL DR : C → ℕ) (D : ℕ)
 115     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
 116     (hDL : ∀ u c, (sourceBucket encode u c).card ≤ DL c)
 117     (hL : ∀ c u, Peeling.leftDegree (H c) u ≤ D)
 118     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c) :
 119     eventProbability (residualPatternEvent p H encode α T) ≤
 120       (Fintype.card R * (∏ i : Fin m, if p.childLeft i then DR (p.cls i) else D) : ℕ) *
 121         patternProbabilityFactor (E := E) p α DL T := by
 122   classical
 123   have hP : 0 ≤ patternProbabilityFactor (E := E) p α DL T := by unfold patternProbabilityFactor; positivity
 124   calc
 125     _ ≤ ∑ fg ∈ embeddingSet p H encode, eventProbability (fixedResidualEvent α encode T m fg.2) := witness_union_bound _ _
 126     _ ≤ ∑ _fg ∈ embeddingSet p H encode, patternProbabilityFactor (E := E) p α DL T :=
 127       sum_le_sum fun fg hfg => embedding_probability_le p H encode α DL hDL fg hfg
 128     _ = (embeddingSet p H encode).card * patternProbabilityFactor (E := E) p α DL T := by simp
 129     _ ≤ (Fintype.card R * (∏ i : Fin m, if p.childLeft i then DR (p.cls i) else D) : ℕ) *
 130         patternProbabilityFactor (E := E) p α DL T := by
 131       apply mul_le_mul_of_nonneg_right _ hP
 132       exact_mod_cast (embeddingSet_card_le p H encode hsimple).trans (patternAssignments_card_le p H D DR hL hR)
