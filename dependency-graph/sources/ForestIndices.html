   8 theorem map_numNodes {W : Type*} (t : Tree V) (f : V → W) : (t.map f).numNodes = t.numNodes := by
   9   induction t with
  10   | nil => rfl
  11   | node v l r hl hr => simp [Tree.map, Tree.numNodes, hl, hr]
  12 
  13 def eraseForest (t : Tree V) : Tree Unit := t.map (fun _ => ())
  14 
  15 theorem eraseForest_numNodes (t : Tree V) : (eraseForest t).numNodes = t.numNodes := by
  16   induction t with
  17   | nil => rfl
  18   | node v l r hl hr => simp [eraseForest, Tree.map, Tree.numNodes] at *; omega
  19 
  20 def forestValue : Tree V → V → ℕ → V
  21   | .nil, fallback, _ => fallback
  22   | .node v l r, fallback, i =>
  23     if i = 0 then v else if i ≤ l.numNodes then forestValue l fallback (i-1)
  24     else forestValue r fallback (i-1-l.numNodes)
  25 
  26 def parentPos : Tree V → ℕ → ℕ → ℕ → ℕ
  27   | .nil, p, _, _ => p
  28   | .node _ l r, p, start, i =>
  29     if i = 0 then p else if i ≤ l.numNodes then parentPos l (start+1) (start+1) (i-1)
  30     else parentPos r p (start+1+l.numNodes) (i-1-l.numNodes)
  31 
  32 def sidePos : Tree V → Bool → ℕ → Bool
  33   | .nil, side, _ => side
  34   | .node _ l r, side, i =>
  35     if i = 0 then side else if i ≤ l.numNodes then sidePos l (!side) (i-1)
  36     else sidePos r side (i-1-l.numNodes)
  37 
  38 theorem forestValue_eq_get (t : Tree V) (fallback : V) (i : ℕ) (hi : i < t.numNodes) :
  39     forestValue t fallback i = (forestLabels t)[i]'(by rw [forestLabels_length]; exact hi) := by
  40   induction t generalizing i with
  41   | nil => simp at hi
  42   | node v l r hl hr =>
  43     cases i with
  44     | zero => simp [forestValue, forestLabels]
  45     | succ i =>
  46       simp only [forestLabels, List.getElem_cons_succ]
  47       by_cases hil : i+1 ≤ l.numNodes
  48       · rw [List.getElem_append_left (by rw [forestLabels_length]; omega)]
  49         simpa [forestValue, hil] using hl i (by omega)
  50       · rw [List.getElem_append_right (by rw [forestLabels_length]; omega)]
  51         have hir : i-l.numNodes < r.numNodes := by simp only [Tree.numNodes] at hi; omega
  52         simpa [forestValue, hil, forestLabels_length] using hr (i-l.numNodes) hir

  54 theorem forestParent_eq_parentPos (t : Tree V) (p start : ℕ) (hp : p ≤ start)
  55     (i : Fin (eraseForest t).numNodes) :
  56     (forestParent (eraseForest t) p start hp i).val = parentPos t p start i.val := by
  57   induction t generalizing p start with
  58   | nil => exact Fin.elim0 i
  59   | node v l r hl hr =>
  60     by_cases hz : i.val = 0
  61     · simp [eraseForest, Tree.map, forestParent, parentPos, hz]
  62     by_cases hil : i.val ≤ l.numNodes
  63     · simp only [eraseForest, Tree.map, forestParent, parentPos, hz, if_false,
  64         map_numNodes, hil, dite_true, dite_false, if_true]
  65       exact hl (start+1) (start+1) le_rfl _
  66     · simp only [eraseForest, Tree.map, forestParent, parentPos, hz, if_false,
  67         map_numNodes, hil, dite_true, dite_false]
  68       have hir : i.val-1-(eraseForest l).numNodes < (eraseForest r).numNodes := by
  69         have hbound := i.isLt
  70         simp only [eraseForest_numNodes, Tree.numNodes] at hbound ⊢
  71         omega
  72       have h := hr p (start+1+(eraseForest l).numNodes) (by omega)
  73         ⟨i.val-1-(eraseForest l).numNodes, hir⟩
  74       convert h using 1 <;> simp only [eraseForest_numNodes]
  75 
  76 theorem forestSide_eq_sidePos (t : Tree V) (side : Bool) (i : Fin (eraseForest t).numNodes) :
  77     forestSide (eraseForest t) side i = sidePos t side i.val := by
  78   induction t generalizing side with
  79   | nil => exact Fin.elim0 i
  80   | node v l r hl hr =>
  81     by_cases hz : i.val = 0
  82     · simp [eraseForest, Tree.map, forestSide, sidePos, hz]
  83     by_cases hil : i.val ≤ l.numNodes
  84     · simp only [eraseForest, Tree.map, forestSide, sidePos, hz, if_false,
  85         map_numNodes, hil, dite_true, dite_false, if_true]
  86       exact hl (!side) _
  87     · simp only [eraseForest, Tree.map, forestSide, sidePos, hz, if_false,
  88         map_numNodes, hil, dite_true, dite_false]
  89       exact hr side _

  91 /-- Indexed edges and side labels are inherited from the labelled forest's
  92 recursive graph edges. The lookup may be embedded in any larger preorder. -/
  93 theorem forest_index_spec (G : SimpleGraph V) (color : V → Bool)
  94     (hcolor : ∀ u v, G.Adj u v → color u ≠ color v)
  95     (t : Tree V) (F : ℕ → V) (fallback : V) (p start : ℕ) (side : Bool)
  96     (ht : ForestValid G (F p) t) (hparent : color (F p) = !side)
  97     (hvalues : ∀ i < t.numNodes, F (start+i+1) = forestValue t fallback i) :
  98     ∀ i < t.numNodes,
  99       G.Adj (F (start+i+1)) (F (parentPos t p start i)) ∧
 100       color (F (start+i+1)) = sidePos t side i := by
 101   induction t generalizing p start side with
 102   | nil => intro i hi; simp at hi
 103   | node v l r hl hr =>
 104     obtain ⟨hpv, hvl, hpr⟩ := ht
 105     have hv : F (start+1) = v := by simpa [forestValue] using hvalues 0 (by simp)
 106     have hcv : color v = side := by
 107       have hne := hcolor (F p) v hpv
 108       cases side <;> cases hc : color v <;> simp_all
 109     intro i hi
 110     by_cases hz : i = 0
 111     · subst i
 112       simpa [parentPos, sidePos, hv] using And.intro hpv.symm hcv
 113     by_cases hil : i ≤ l.numNodes
 114     · have hvals : ∀ j < l.numNodes, F ((start+1)+j+1) = forestValue l fallback j := by
 115         intro j hj
 116         have h := hvalues (j+1) (by simp only [Tree.numNodes]; omega)
 117         have hsplit : forestValue (.node v l r) fallback (j+1) = forestValue l fallback j := by
 118           simp [forestValue, show j+1 ≤ l.numNodes by omega]
 119         rw [hsplit] at h
 120         convert h using 1 <;> congr 1 <;> omega
 121       have hvalid : ForestValid G (F (start+1)) l := by simpa only [hv] using hvl
 122       have hcparent : color (F (start+1)) = !(!side) := by simpa [hv] using hcv
 123       have h := hl (start+1) (start+1) (!side) hvalid hcparent hvals (i-1) (by omega)
 124       have heq : start+1+(i-1)+1 = start+i+1 := by omega
 125       simpa only [parentPos, sidePos, hz, if_false, hil, if_true, heq] using h
 126     · have hvals : ∀ j < r.numNodes, F ((start+1+l.numNodes)+j+1) = forestValue r fallback j := by
 127         intro j hj
 128         have h := hvalues (j+1+l.numNodes) (by simp only [Tree.numNodes]; omega)
 129         have hnot : ¬j+1+l.numNodes ≤ l.numNodes := by omega
 130         have hsplit : forestValue (.node v l r) fallback (j+1+l.numNodes) = forestValue r fallback j := by
 131           rw [forestValue, if_neg (by omega), if_neg hnot]
 132           congr 1
 133           omega
 134         rw [hsplit] at h
 135         convert h using 1 <;> congr 1 <;> omega
 136       have h := hr p (start+1+l.numNodes) side hpr hparent hvals (i-1-l.numNodes)
 137         (by simp only [Tree.numNodes] at hi; omega)
 138       have heq : start+1+l.numNodes+(i-1-l.numNodes)+1 = start+i+1 := by omega
 139       simpa only [parentPos, sidePos, hz, if_false, hil, heq] using h
