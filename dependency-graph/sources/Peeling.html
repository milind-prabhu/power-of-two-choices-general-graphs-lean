   4 /-! The finite graph-theoretic peeling step behind Lemma 3.7. This module
   5 proves extraction and termination from a hereditary neighborhood-expansion
   6 condition. The analytic-to-graph certificate is kept as an explicit premise. -/
   7 namespace GraphChoices.Peeling
   8 open Finset
   9 variable {L R : Type*} [Fintype L] [Fintype R] [DecidableEq L] [DecidableEq R]
  10 
  11 abbrev Graph (L R : Type*) := Finset (L × R)
  12 
  13 def leftNeighbors (H : Graph L R) (u : L) : Finset R :=
  14   univ.filter fun v => (u,v) ∈ H
  15 
  16 def rightNeighbors (H : Graph L R) (v : R) : Finset L :=
  17   univ.filter fun u => (u,v) ∈ H
  18 
  19 def leftDegree (H : Graph L R) (u : L) : ℕ := (leftNeighbors H u).card
  20 
  21 def rightDegree (H : Graph L R) (v : R) : ℕ := (rightNeighbors H v).card
  22 
  23  theorem leftNeighbors_mono {F H : Graph L R} (h : F ⊆ H) (u : L) :
  24     leftNeighbors F u ⊆ leftNeighbors H u := by
  25   intro v hv
  26   simp only [leftNeighbors, mem_filter, mem_univ, true_and] at hv ⊢
  27   exact h hv
  28 
  29  theorem leftDegree_mono {F H : Graph L R} (h : F ⊆ H) (u : L) :
  30     leftDegree F u ≤ leftDegree H u := card_le_card (leftNeighbors_mono h u)
  31 
  32  theorem leftDegree_sdiff {F H : Graph L R} (h : F ⊆ H) (u : L) :
  33     leftDegree (H \ F) u = leftDegree H u - leftDegree F u := by
  34   have heq : leftNeighbors (H \ F) u = leftNeighbors H u \ leftNeighbors F u := by
  35     ext v; simp [leftNeighbors]
  36   rw [leftDegree, heq, card_sdiff_of_subset (leftNeighbors_mono h u)]
  37   rfl
  38 
  39  theorem rightDegree_union_le (F H : Graph L R) (v : R) :
  40     rightDegree (F ∪ H) v ≤ rightDegree F v + rightDegree H v := by
  41   have heq : rightNeighbors (F ∪ H) v = rightNeighbors F v ∪ rightNeighbors H v := by
  42     ext u; simp [rightNeighbors]
  43   exact (congrArg card heq).le.trans (card_union_le _ _)
  44 
  45 /-- Every sufficiently high-degree left set expands, in every residual graph. -/
  46 def PeelingExpansion (H : Graph L R) (k b : ℕ) : Prop :=
  47   ∀ F : Graph L R, F ⊆ H → ∀ A : Finset L,
  48     (∀ u ∈ A, k < leftDegree F u) →
  49       A.card ≤ b * (GraphChoices.neighbors (fun u v => (u,v) ∈ F) A).card
  50 
  51  theorem PeelingExpansion.mono {F H : Graph L R} {k b : ℕ}
  52     (h : PeelingExpansion H k b) (hF : F ⊆ H) : PeelingExpansion F k b := by
  53   intro J hJ A hA
  54   exact h J (hJ.trans hF) A hA
  55 
  56 /-- A matching covers exactly a specified active set, with right capacity b. -/
  57  theorem exists_active_matching (H : Graph L R) (A : Finset L) (b : ℕ)
  58     (hexpand : ∀ B : Finset L, B ⊆ A →
  59       B.card ≤ b * (GraphChoices.neighbors (fun u v => (u,v) ∈ H) B).card) :
  60     ∃ M : Graph L R, M ⊆ H ∧
  61       (∀ u, leftDegree M u = if u ∈ A then 1 else 0) ∧
  62       (∀ v, rightDegree M v ≤ b) := by
  63   classical
  64   let U := {u : L // u ∈ A}
  65   let r (u : U) (v : R) := (u.val,v) ∈ H
  66   have hcap (B : Finset U) : B.card ≤ b * (GraphChoices.neighbors r B).card := by
  67     have hb := hexpand (B.image Subtype.val) (by
  68       intro u hu
  69       obtain ⟨w, hw, rfl⟩ := mem_image.mp hu
  70       exact w.property)
  71     have hi : (B.image Subtype.val).card = B.card := card_image_of_injective _ Subtype.val_injective
  72     have heq : GraphChoices.neighbors (fun u v => (u,v) ∈ H) (B.image Subtype.val) =
  73         GraphChoices.neighbors r B := by
  74       ext v
  75       simp only [GraphChoices.neighbors, mem_filter, mem_univ, true_and, mem_image]
  76       constructor
  77       · rintro ⟨u, ⟨w, hw, rfl⟩, h⟩; exact ⟨w, hw, h⟩
  78       · rintro ⟨w, hw, h⟩; exact ⟨w.val, ⟨w, hw, rfl⟩, h⟩
  79     simpa [hi, heq] using hb
  80   obtain ⟨f, hf, hfb⟩ := GraphChoices.exists_capacity_matching r b hcap
  81   let M : Graph L R := univ.image fun u : U => (u.val, f u)
  82   refine ⟨M, ?_, ?_, ?_⟩
  83   · intro p hp
  84     obtain ⟨u, hu, rfl⟩ := mem_image.mp hp
  85     exact hf u
  86   · intro u
  87     by_cases hu : u ∈ A
  88     · have heq : leftNeighbors M u = {f ⟨u,hu⟩} := by
  89         ext v
  90         simp only [leftNeighbors, mem_filter, mem_univ, true_and, M, mem_image, mem_singleton]
  91         constructor
  92         · rintro ⟨w, hw⟩
  93           have hwval : w.val = u := congrArg Prod.fst hw
  94           have hwsub : w = ⟨u,hu⟩ := Subtype.ext hwval
  95           have := congrArg Prod.snd hw
  96           simpa [hwsub] using this.symm
  97         · intro hv
  98           exact ⟨⟨u,hu⟩, by simp [hv]⟩
  99       simp [leftDegree, heq, hu]
 100     · have heq : leftNeighbors M u = ∅ := by
 101         ext v
 102         simp only [leftNeighbors, mem_filter, mem_univ, true_and, M, mem_image, notMem_empty, iff_false]
 103         rintro ⟨w, hw⟩
 104         have hwval : w.val = u := congrArg Prod.fst hw
 105         exact hu (hwval ▸ w.property)
 106       simp [leftDegree, heq, hu]
 107   · intro v
 108     have heq : rightNeighbors M v = (univ.filter fun u : U => f u = v).image Subtype.val := by
 109       ext u
 110       simp only [rightNeighbors, mem_filter, mem_univ, true_and, M, mem_image]
 111       constructor
 112       · rintro ⟨w, hw⟩
 113         exact ⟨w, congrArg Prod.snd hw, congrArg Prod.fst hw⟩
 114       · rintro ⟨w, hw, rfl⟩
 115         exact ⟨w, by simp [hw]⟩
 116     calc
 117       rightDegree M v = ((univ.filter fun u : U => f u = v).image Subtype.val).card := congrArg card heq
 118       _ ≤ (univ.filter fun u : U => f u = v).card := card_image_le
 119       _ ≤ b := hfb v

 121 /-- Extraction terminates after at most D matching rounds. The removed piece
 122 has right degree at most D b, and the residual has left degree at most k. -/
 123  theorem exists_peeling_piece (H : Graph L R) (k b D : ℕ)
 124     (hD : ∀ u, leftDegree H u ≤ D) (hexpand : PeelingExpansion H k b) :
 125     ∃ P : Graph L R, P ⊆ H ∧
 126       (∀ u, leftDegree (H \ P) u ≤ k) ∧
 127       (∀ v, rightDegree P v ≤ D*b) := by
 128   induction D generalizing H with
 129   | zero =>
 130     refine ⟨∅, empty_subset _, ?_, ?_⟩
 131     · intro u; simpa using (hD u).trans (Nat.zero_le k)
 132     · intro v; simp [rightDegree, rightNeighbors]
 133   | succ D ih =>
 134     by_cases hk : D+1 ≤ k
 135     · refine ⟨∅, empty_subset _, ?_, ?_⟩
 136       · intro u; simpa using (hD u).trans hk
 137       · intro v; simp [rightDegree, rightNeighbors]
 138     · let A : Finset L := univ.filter fun u => k < leftDegree H u
 139       obtain ⟨M, hMH, hML, hMR⟩ := exists_active_matching H A b (by
 140         intro B hBA
 141         apply hexpand H (subset_refl _) B
 142         intro u hu
 143         exact (mem_filter.mp (hBA hu)).2)
 144       have hnext (u : L) : leftDegree (H \ M) u ≤ D := by
 145         rw [leftDegree_sdiff hMH, hML]
 146         by_cases hu : u ∈ A
 147         · rw [if_pos hu]
 148           have := hD u
 149           omega
 150         · rw [if_neg hu]
 151           have hlow : leftDegree H u ≤ k := by
 152             simpa only [A, mem_filter, mem_univ, true_and, not_lt] using hu
 153           omega
 154       obtain ⟨Q, hQ, hQleft, hQright⟩ := ih (H \ M) hnext
 155         (hexpand.mono sdiff_subset)
 156       have hQH : Q ⊆ H := hQ.trans sdiff_subset
 157       refine ⟨M ∪ Q, union_subset hMH hQH, ?_, ?_⟩
 158       · intro u
 159         have heq : H \ (M ∪ Q) = (H \ M) \ Q := by ext p; simp; tauto
 160         rw [heq]
 161         exact hQleft u
 162       · intro v
 163         have hdeg := rightDegree_union_le M Q v
 164         have hm := hMR v
 165         have hq := hQright v
 166         nlinarith
 167 
 168 end GraphChoices.Peeling
