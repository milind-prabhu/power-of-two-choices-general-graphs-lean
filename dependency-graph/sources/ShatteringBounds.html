  15  theorem heavyShape_decay_of_thresholds (H : C → Peeling.Graph L R)
  16     (encode : E → Arrival L R C) (α : C → ℕ) (DL DR : C → ℕ) (D : ℕ) (f : C → ℝ)
  17     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
  18     (hDL : ∀ u c, (sourceBucket encode u c).card ≤ DL c)
  19     (hL : ∀ c u, Peeling.leftDegree (H c) u ≤ D)
  20     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c)
  21     (b r : ℝ) (hT : 0 < T) (hb : 0 < b) (hr : 0 ≤ r) (hbr : b ≤ r)
  22     (hpair : (D : ℝ)*((T:ℝ)/Fintype.card E)*b ≤ r^2)
  23     (hf : ∀ c, 1 < f c)
  24     (hbase : ∀ c, Real.exp 1*T*(DL c : ℝ)/Fintype.card E/α c ≤ 1/f c)
  25     (hthreshold : ∀ c, 0 < DR c →
  26       Real.log (((DR c : ℝ)*((T:ℝ)/Fintype.card E))/b) / Real.log (f c) ≤ (α c : ℝ)) :
  27     eventProbability (heavyShapeEvent H encode α T m) ≤
  28       (Fintype.card R : ℝ)*(4*(Fintype.card C : ℝ)*r)^m := by
  29   apply heavyShape_probability_le H encode α T m _ r (Nat.cast_nonneg _) hr
  30   intro s hs
  31   exact residualPattern_decay_of_thresholds (shapePattern s) H encode α DL DR D f
  32     hsimple hDL hL hR b r hT hb hr hbr hpair hf hbase hthreshold hs

  34  theorem heavyShape_geometric_tail (H : C → Peeling.Graph L R)
  35     (encode : E → Arrival L R C) (α : C → ℕ) (T m : ℕ) (r : ℝ) (hr : 0 ≤ r)
  36     (hdecay : eventProbability (heavyShapeEvent H encode α T m) ≤
  37       (Fintype.card R : ℝ)*(4*(Fintype.card C : ℝ)*r)^m)
  38     (hsmall : 4*(Fintype.card C : ℝ)*r ≤ 1/2) :
  39     eventProbability (heavyShapeEvent H encode α T m) ≤ (Fintype.card R : ℝ)/2^m := by
  40   calc
  41     _ ≤ (Fintype.card R : ℝ)*(4*(Fintype.card C : ℝ)*r)^m := hdecay
  42     _ ≤ (Fintype.card R : ℝ)*((1:ℝ)/2)^m :=
  43       mul_le_mul_of_nonneg_left (pow_le_pow_left₀ (by positivity) hsmall m) (Nat.cast_nonneg _)
  44     _ = _ := by rw [div_pow]; ring

  46 /-- An explicit inverse-polynomial failure probability for the canonical
  47 heavy-tree event. The exponent condition is exact and uses no asymptotic O. -/
  48  theorem heavyShape_polynomial_tail (H : C → Peeling.Graph L R)
  49     (encode : E → Arrival L R C) (α : C → ℕ) (T m N c : ℕ) (hN : 0 < N)
  50     (hR : Fintype.card R ≤ N)
  51     (hprob : eventProbability (heavyShapeEvent H encode α T m) ≤ (Fintype.card R : ℝ)/2^m)
  52     (hexponent : N^(c+1) ≤ 2^m) :
  53     eventProbability (heavyShapeEvent H encode α T m) ≤ 1/(N:ℝ)^c := by
  54   have hNr : (0:ℝ) < N := by exact_mod_cast hN
  55   have hexp : (N:ℝ)^(c+1) ≤ (2:ℝ)^m := by exact_mod_cast hexponent
  56   calc
  57     _ ≤ (Fintype.card R : ℝ)/2^m := hprob
  58     _ ≤ (N:ℝ)/2^m := div_le_div_of_nonneg_right (by exact_mod_cast hR) (by positivity)
  59     _ ≤ (N:ℝ)/(N:ℝ)^(c+1) := div_le_div_of_nonneg_left hNr.le (by positivity) hexp
  60     _ = _ := by rw [pow_succ]; field_simp
