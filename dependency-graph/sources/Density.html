   6 /-! Exact integer density characterization of the offline optimum, Section 2.2.
   7 The upper bound is proved by capacitated Hall, rather than assumed. -/
   8 namespace GraphChoices
   9 open Finset
  10 variable {V : Type*} [Fintype V] [DecidableEq V] {T : ℕ}

  12 /-- An orientation of capacity c exists iff all induced subgraphs have at most
  13 c times as many edge occurrences as vertices. This includes multiplicities. -/
  14  theorem exists_orientation_iff_density (I : Instance V T) (c : ℕ) :
  15     (∃ a : Orientation T, maxLoad I a ≤ c) ↔
  16       ∀ S : Finset V, internalCount I S ≤ S.card * c := by
  17   constructor
  18   · rintro ⟨a, ha⟩ S
  19     exact (density_obstruction I a S).trans (Nat.mul_le_mul_left _ ha)
  20   · intro hdensity
  21     let r (t : Fin T) (v : V) := v = I.left t ∨ v = I.right t
  22     have hexpand (A : Finset (Fin T)) : A.card ≤ c * (neighbors r A).card := by
  23       have hsub : A ⊆ univ.filter (fun t => I.left t ∈ neighbors r A ∧ I.right t ∈ neighbors r A) := by
  24         intro t ht
  25         simp only [mem_filter, mem_univ, true_and, neighbors]
  26         exact ⟨⟨t, ht, Or.inl rfl⟩, ⟨t, ht, Or.inr rfl⟩⟩
  27       exact (card_le_card hsub).trans (by simpa [mul_comm] using hdensity (neighbors r A))
  28     obtain ⟨f, hf, hcap⟩ := exists_capacity_matching r c hexpand
  29     let a : Orientation T := fun t => decide (f t = I.left t)
  30     have hdest (t : Fin T) : destination I a t = f t := by
  31       by_cases h : f t = I.left t
  32       · simp [destination, a, h]
  33       · have hr : f t = I.right t := (hf t).resolve_left h
  34         simp only [destination, a, decide_eq_true_eq, if_neg h]
  35         exact hr.symm
  36     refine ⟨a, (maxLoad_le_iff I a c).2 ?_⟩
  37     intro v
  38     simpa only [load, hdest] using hcap v
  39 
  40  theorem opt_le_iff_density (I : Instance V T) (c : ℕ) :
  41     opt I ≤ c ↔ ∀ S : Finset V, internalCount I S ≤ S.card * c := by
  42   constructor
  43   · intro hc S
  44     exact (density_le_opt I S).trans (Nat.mul_le_mul_left _ hc)
  45   · intro h
  46     obtain ⟨a, ha⟩ := (exists_orientation_iff_density I c).2 h
  47     exact (opt_le_maxLoad I a).trans ha
  48 
  49 noncomputable def density (I : Instance V T) (S : Finset V) : ℝ :=
  50   internalCount I S / (S.card : ℝ)
  51 
  52 noncomputable def maxDensity (I : Instance V T) : ℝ :=
  53   (univ.powerset).sup' (by simp) (density I)
  54

  56  theorem density_nonneg (I : Instance V T) (S : Finset V) : 0 ≤ density I S := by
  57   exact div_nonneg (Nat.cast_nonneg _) (Nat.cast_nonneg _)

  59 omit [Fintype V] in
  60  theorem density_le_nat_iff (I : Instance V T) (S : Finset V) (c : ℕ) :
  61     density I S ≤ c ↔ internalCount I S ≤ S.card * c := by
  62   by_cases hS : S.card = 0
  63   · have hs : S = ∅ := card_eq_zero.mp hS
  64     simp [hs, density, internalCount]
  65   · have hp : (0:ℝ) < S.card := by exact_mod_cast Nat.pos_of_ne_zero hS
  66     rw [density, div_le_iff₀ hp]
  67     norm_cast
  68     rw [mul_comm]
  69 
  70  theorem maxDensity_le_nat_iff (I : Instance V T) (c : ℕ) :
  71     maxDensity I ≤ c ↔ ∀ S : Finset V, internalCount I S ≤ S.card * c := by
  72   simp only [maxDensity, sup'_le_iff, mem_powerset, subset_univ, forall_true_left,
  73     density_le_nat_iff]
  74 
  75  theorem maxDensity_nonneg (I : Instance V T) : 0 ≤ maxDensity I := by
  76   exact (density_nonneg I ∅).trans (le_sup' (density I) (by simp))
  77 
  78 /-- The ceiling in the paper's exact density identity is retained. -/
  79  theorem opt_eq_ceil_maxDensity (I : Instance V T) : opt I = ⌈maxDensity I⌉₊ := by
  80   apply Nat.le_antisymm
  81   · apply (opt_le_iff_density I _).2
  82     apply (maxDensity_le_nat_iff I _).1
  83     exact Nat.le_ceil _
  84   · apply Nat.ceil_le.mpr
  85     exact (maxDensity_le_nat_iff I _).2 (density_le_opt I)
