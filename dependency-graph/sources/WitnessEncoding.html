   3 /-! Deterministic encoding of disjoint tree and threshold occurrence times into
   4 the finite witness family used by the proved joint probability bound. -/
   5 namespace GraphChoices
   6 open Finset
   7 variable {E J : Type*} [Fintype E] [DecidableEq E] [Nonempty E]
   8   [Fintype J] [DecidableEq J] {T m : ℕ} {α : J → ℕ}
   9 
  10  theorem witnessTime_injective (w : TimeWitness J T m α)
  11     (htree : Function.Injective w.1)
  12     (hsep : ∀ i j, w.1 i ∉ (w.2 j).val)
  13     (hdisj : ∀ j j', j ≠ j' → Disjoint (w.2 j).val (w.2 j').val) :
  14     Function.Injective (witnessTime w) := by
  15   intro a b hab
  16   cases a with
  17   | inl i =>
  18     cases b with
  19     | inl i' => exact congrArg Sum.inl (htree hab)
  20     | inr p =>
  21       obtain ⟨j,k⟩ := p
  22       have hk := (w.2 j).val.orderEmbOfFin_mem ((mem_powersetCard.mp (w.2 j).property).2) k
  23       simp only [witnessTime] at hab
  24       exact False.elim (hsep i j (hab.symm ▸ hk))
  25   | inr p =>
  26     obtain ⟨j,k⟩ := p
  27     cases b with
  28     | inl i =>
  29       have hk := (w.2 j).val.orderEmbOfFin_mem ((mem_powersetCard.mp (w.2 j).property).2) k
  30       simp only [witnessTime] at hab
  31       exact False.elim (hsep i j (hab ▸ hk))
  32     | inr p' =>
  33       obtain ⟨j',k'⟩ := p'
  34       by_cases hj : j = j'
  35       · subst j'
  36         have hk : k = k' := ((w.2 j).val.orderEmbOfFin ((mem_powersetCard.mp (w.2 j).property).2)).injective hab
  37         subst k'
  38         rfl
  39       · have hk := (w.2 j).val.orderEmbOfFin_mem ((mem_powersetCard.mp (w.2 j).property).2) k
  40         have hk' := (w.2 j').val.orderEmbOfFin_mem ((mem_powersetCard.mp (w.2 j').property).2) k'
  41         simp only [witnessTime] at hab
  42         exact False.elim (Finset.disjoint_left.mp (hdisj j j' hj) hk (hab.symm ▸ hk'))

  44 /-- Exhibiting actual, globally disjoint occurrence times is sufficient for
  45 membership in the abstract pattern witness event. -/
  46  theorem mem_patternWitnessEvent_of_times (ω : Fin T → E) (tree : Fin m → E)
  47     (bucket : J → Finset E) (τ : Fin m → Fin T) (K : J → Finset (Fin T))
  48     (hcard : ∀ j, (K j).card = α j)
  49     (htree : Function.Injective τ) (hlabel : ∀ i, ω (τ i) = tree i)
  50     (hsep : ∀ i j, τ i ∉ K j)
  51     (hdisj : ∀ j j', j ≠ j' → Disjoint (K j) (K j'))
  52     (hbucket : ∀ j t, t ∈ K j → ω t ∈ bucket j) :
  53     ω ∈ patternWitnessEvent T m α tree bucket := by
  54   classical
  55   let w : TimeWitness J T m α := (τ, fun j => ⟨K j,mem_powersetCard.mpr ⟨subset_univ _,hcard j⟩⟩)
  56   have hinj : Function.Injective (witnessTime w) := witnessTime_injective w htree hsep hdisj
  57   unfold patternWitnessEvent
  58   apply mem_biUnion.mpr
  59   refine ⟨w,mem_univ _,?_⟩
  60   rw [if_pos hinj]
  61   simp only [requirementsEvent, mem_filter, mem_univ, true_and]
  62   intro s
  63   cases s with
  64   | inl i => simpa [slotAllowed, witnessTime, w] using hlabel i
  65   | inr p =>
  66     obtain ⟨j,k⟩ := p
  67     apply hbucket j
  68     exact (K j).orderEmbOfFin_mem (hcard j) k
  69 
  70 end GraphChoices
