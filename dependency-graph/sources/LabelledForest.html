   4 /-! A spanning first-child/next-sibling forest in every finite connected
   5 graph. Maximality is over the bounded node count, not an assumed enumeration.
   6 The distinguished root is external to the binary forest. -/
   7 namespace GraphChoices.Patterns
   8 open Finset
   9 variable {V : Type*}
  10 
  11 def forestLabels : Tree V → List V
  12   | .nil => []
  13   | .node v l r => v :: (forestLabels l ++ forestLabels r)
  14 
  15 def ForestValid (G : SimpleGraph V) : V → Tree V → Prop
  16   | _, .nil => True
  17   | p, .node v l r => G.Adj p v ∧ ForestValid G v l ∧ ForestValid G p r
  18 
  19 theorem forestLabels_length (t : Tree V) : (forestLabels t).length = t.numNodes := by
  20   induction t with
  21   | nil => rfl
  22   | node v l r hl hr => simp [forestLabels, Tree.numNodes, hl, hr]
  23 
  24 /-- Insert a new child at any named vertex in the forest, without imposing
  25 an ordering on the old vertices. The labels change by exactly one insertion. -/
  26 theorem forest_grow {G : SimpleGraph V} (t : Tree V) (p u x : V)
  27     (ht : ForestValid G p t) (hu : u = p ∨ u ∈ forestLabels t) (hux : G.Adj u x) :
  28     ∃ t' : Tree V, ForestValid G p t' ∧ (forestLabels t').Perm (x :: forestLabels t) := by
  29   induction t generalizing p with
  30   | nil =>
  31     have hup : u = p := by simpa [forestLabels] using hu
  32     subst u
  33     exact ⟨.node x .nil .nil, ⟨hux, trivial, trivial⟩, by simp [forestLabels]⟩
  34   | node v l r hl hr =>
  35     by_cases hup : u = p
  36     · subst u
  37       exact ⟨.node x .nil (.node v l r), ⟨hux, trivial, ht⟩, by simp [forestLabels]⟩
  38     have hu' : u ∈ forestLabels (.node v l r) := hu.resolve_left hup
  39     obtain ⟨hpv, hvl, hpr⟩ := ht
  40     simp only [forestLabels, List.mem_cons, List.mem_append] at hu'
  41     rcases hu' with rfl | hu | hu
  42     · refine ⟨.node u (.node x .nil l) r, ⟨hpv, ⟨hux, trivial, hvl⟩, hpr⟩, ?_⟩
  43       simp only [forestLabels, List.nil_append, List.cons_append]
  44       exact List.Perm.swap _ _ _
  45     · obtain ⟨l', hl', hperm⟩ := hl v hvl (Or.inr hu)
  46       refine ⟨.node v l' r, ⟨hpv, hl', hpr⟩, ?_⟩
  47       have h := (hperm.append_right (forestLabels r)).cons v
  48       exact h.trans (List.Perm.swap _ _ _)
  49     · obtain ⟨r', hr', hperm⟩ := hr p hpr (Or.inr hu)
  50       refine ⟨.node v l r', ⟨hpv, hvl, hr'⟩, ?_⟩
  51       have h := (hperm.append_left (forestLabels l)).cons v
  52       apply h.trans
  53       simpa only [forestLabels, List.cons_append] using
  54         (List.perm_middle (a := x) (l₁ := v :: forestLabels l) (l₂ := forestLabels r))
  55 
  56 theorem walk_mem_of_closed {G : SimpleGraph V} {u v : V} (p : G.Walk u v)
  57     (S : Set V) (hu : u ∈ S) (hclosed : ∀ a ∈ S, ∀ b, G.Adj a b → b ∈ S) : v ∈ S := by
  58   induction p with
  59   | nil => exact hu
  60   | @cons a b c hab p ih => exact ih (hclosed a hu b hab)

  62 /-- Every finite connected graph has a nonrepeating spanning labelled binary
  63 forest with the designated root external to that forest. -/
  64 theorem exists_spanning_labelled_forest [Fintype V] (G : SimpleGraph V)
  65     (hc : G.Connected) (root : V) :
  66     ∃ t : Tree V, ForestValid G root t ∧ (root :: forestLabels t).Nodup ∧
  67       (∀ v, v ∈ root :: forestLabels t) ∧ t.numNodes+1 = Fintype.card V := by
  68   classical
  69   let P (n : ℕ) : Prop := ∃ t : Tree V,
  70     t.numNodes = n ∧ ForestValid G root t ∧ (root :: forestLabels t).Nodup
  71   have hzero : P 0 := ⟨.nil, rfl, trivial, by simp [forestLabels]⟩
  72   let n := Nat.findGreatest P (Fintype.card V)
  73   obtain ⟨t, hcount, ht, hn⟩ := Nat.findGreatest_spec (Nat.zero_le (Fintype.card V)) hzero
  74   have hclosed : ∀ u ∈ root :: forestLabels t, ∀ x, G.Adj u x → x ∈ root :: forestLabels t := by
  75     intro u hu x hux
  76     by_contra hx
  77     have hu' : u = root ∨ u ∈ forestLabels t := by simpa using hu
  78     obtain ⟨t', ht', hperm⟩ := forest_grow t root u x ht hu' hux
  79     have hn0 : (x :: root :: forestLabels t).Nodup := List.nodup_cons.mpr ⟨hx, hn⟩
  80     have hn1 : (root :: x :: forestLabels t).Nodup := (List.Perm.swap _ _ _).nodup_iff.mpr hn0
  81     have hn' : (root :: forestLabels t').Nodup := (hperm.cons root).nodup_iff.mpr hn1
  82     have hlen := hperm.length_eq
  83     simp only [forestLabels_length, List.length_cons] at hlen
  84     have hsize : t'.numNodes = n+1 := by omega
  85     have hbound := hn'.length_le_card
  86     simp only [List.length_cons, forestLabels_length] at hbound
  87     have hnew : P (n+1) := ⟨t', hsize, ht', hn'⟩
  88     exact Nat.findGreatest_is_greatest (show n < n+1 by omega) (by omega) hnew
  89   have hcover : ∀ v, v ∈ root :: forestLabels t := by
  90     intro v
  91     obtain ⟨p⟩ := hc root v
  92     exact walk_mem_of_closed p {w | w ∈ root :: forestLabels t} (by simp) hclosed
  93   refine ⟨t, ht, hn, hcover, ?_⟩
  94   have hfin : (root :: forestLabels t).toFinset = univ := by
  95     apply Finset.eq_univ_of_forall
  96     intro v
  97     exact List.mem_toFinset.mpr (hcover v)
  98   have hcard := congrArg Finset.card hfin
  99   simpa only [List.toFinset_card_of_nodup hn, card_univ, List.length_cons, forestLabels_length] using hcard
