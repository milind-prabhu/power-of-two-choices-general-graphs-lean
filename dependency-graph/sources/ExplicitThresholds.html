   8 noncomputable def sufficientThreshold (mean f logRequirement : ℝ) : ℕ :=
   9   ⌈max (f * mean) logRequirement⌉₊ + 1
  10 
  11 theorem sufficientThreshold_pos (mean f z : ℝ) : 0 < sufficientThreshold mean f z := by
  12   unfold sufficientThreshold
  13   omega
  14 
  15 theorem sufficientThreshold_dominates (mean f z : ℝ) :
  16     max (f*mean) z ≤ (sufficientThreshold mean f z : ℝ) := by
  17   have h := Nat.le_ceil (max (f*mean) z)
  18   simp only [sufficientThreshold, Nat.cast_add, Nat.cast_one]
  19   linarith
  20 
  21 theorem sufficientThreshold_base (mean f z : ℝ) (hf : 0 < f) :
  22     mean / sufficientThreshold mean f z ≤ 1/f := by
  23   have hpos : (0:ℝ) < sufficientThreshold mean f z := by
  24     exact_mod_cast sufficientThreshold_pos mean f z
  25   apply (div_le_div_iff₀ hpos hf).mpr
  26   have h := (le_max_left (f*mean) z).trans (sufficientThreshold_dominates mean f z)
  27   nlinarith
  28 
  29 theorem sufficientThreshold_log (mean f z : ℝ) :
  30     z ≤ (sufficientThreshold mean f z : ℝ) :=
  31   (le_max_right (f*mean) z).trans (sufficientThreshold_dominates mean f z)
  32 
  33 end GraphChoices
