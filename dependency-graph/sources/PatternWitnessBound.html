   6 /-! Joint witness bound of the form used in Equations (10)--(12). Tree slots
   7 require fixed base edges. Each threshold bucket chooses an unordered set of
   8 alpha_j time steps. Only globally injective assignments are retained, so all
   9 required occurrences are distinct and the product probability is justified.
  10 The graph-to-witness encoding and pattern-embedding count are separate results.
  11 -/
  12 namespace GraphChoices
  13 open Finset
  14 variable {E J : Type*} [Fintype E] [DecidableEq E] [Nonempty E]
  15   [Fintype J] [DecidableEq J]

  17 abbrev TimeChoice (T k : ℕ) :=
  18   {K : Finset (Fin T) // K ∈ (univ : Finset (Fin T)).powersetCard k}
  19 
  20 abbrev TimeWitness (J : Type*) (T m : ℕ) (α : J → ℕ) :=
  21   (Fin m → Fin T) × ((j : J) → TimeChoice T (α j))
  22 
  23 abbrev WitnessSlot (J : Type*) (m : ℕ) (α : J → ℕ) :=
  24   Fin m ⊕ ((j : J) × Fin (α j))
  25 
  26 noncomputable def witnessTime {T m : ℕ} {α : J → ℕ} (w : TimeWitness J T m α) :
  27     WitnessSlot J m α → Fin T
  28   | Sum.inl i => w.1 i
  29   | Sum.inr ⟨j,i⟩ => (w.2 j).val.orderEmbOfFin ((mem_powersetCard.mp (w.2 j).property).2) i
  30 
  31 def slotAllowed {m : ℕ} {α : J → ℕ} (tree : Fin m → E) (bucket : J → Finset E) :
  32     WitnessSlot J m α → Finset E
  33   | Sum.inl i => {tree i}
  34   | Sum.inr ⟨j,_⟩ => bucket j

  36  theorem timeChoice_card (T k : ℕ) : Fintype.card (TimeChoice T k) = T.choose k := by
  37   simp [TimeChoice, Fintype.card_coe, card_powersetCard]

  39  theorem timeWitness_card (T m : ℕ) (α : J → ℕ) :
  40     Fintype.card (TimeWitness J T m α) = T^m * ∏ j, T.choose (α j) := by
  41   simp [TimeWitness, Fintype.card_prod, Fintype.card_fun, Fintype.card_pi, timeChoice_card]

  43  theorem slot_probability_product {m : ℕ} (α : J → ℕ) (tree : Fin m → E) (bucket : J → Finset E) :
  44     (∏ s : WitnessSlot J m α, ((slotAllowed tree bucket s).card : ℝ) / Fintype.card E) =
  45       ((1:ℝ) / Fintype.card E)^m * ∏ j, (((bucket j).card : ℝ) / Fintype.card E)^(α j) := by
  46   rw [Fintype.prod_sum_type, Fintype.prod_sigma]
  47   simp [slotAllowed, div_pow, prod_div_distrib]

  49 noncomputable def patternWitnessEvent (T m : ℕ) (α : J → ℕ)
  50     (tree : Fin m → E) (bucket : J → Finset E) : Finset (Fin T → E) := by
  51   classical
  52   exact univ.biUnion fun w : TimeWitness J T m α =>
  53     if Function.Injective (witnessTime w) then requirementsEvent (witnessTime w) (slotAllowed tree bucket) else ∅

  55 /-- Joint counting and probability bound with exact binomial coefficients. -/
  56  theorem patternWitness_probability_le (T m : ℕ) (α : J → ℕ)
  57     (tree : Fin m → E) (bucket : J → Finset E) :
  58     eventProbability (patternWitnessEvent T m α tree bucket) ≤
  59       ((T:ℝ) / Fintype.card E)^m *
  60         ∏ j, (T.choose (α j) : ℝ) * (((bucket j).card : ℝ) / Fintype.card E)^(α j) := by
  61   classical
  62   have hh := injected_witness_union_bound
  63     (W := TimeWitness J T m α) (J := WitnessSlot J m α) witnessTime (slotAllowed tree bucket)
  64   change eventProbability (patternWitnessEvent T m α tree bucket) ≤ _ at hh
  65   calc
  66     _ ≤ (Fintype.card (TimeWitness J T m α) : ℝ) *
  67         (∏ s : WitnessSlot J m α, ((slotAllowed tree bucket s).card : ℝ) / Fintype.card E) := hh
  68     _ = _ := by
  69       rw [timeWitness_card, slot_probability_product]
  70       simp only [Nat.cast_mul, Nat.cast_pow, Nat.cast_prod, prod_mul_distrib, div_pow]
  71       ring

  73 /-- The exponential version corresponding to the raw pattern bound in (12).
  74 All time-step collisions have been discarded before applying independence. -/
  75  theorem patternWitness_probability_le_exp (T m : ℕ) (α : J → ℕ)
  76     (tree : Fin m → E) (bucket : J → Finset E) :
  77     eventProbability (patternWitnessEvent T m α tree bucket) ≤
  78       ((T:ℝ) / Fintype.card E)^m *
  79         ∏ j, (Real.exp 1 * T * (((bucket j).card : ℝ) / Fintype.card E) / α j)^(α j) := by
  80   apply (patternWitness_probability_le T m α tree bucket).trans
  81   apply mul_le_mul_of_nonneg_left _ (by positivity)
  82   apply prod_le_prod
  83   · intro j _; positivity
  84   · intro j _
  85     calc
  86       (T.choose (α j) : ℝ) * (((bucket j).card : ℝ) / Fintype.card E)^(α j) ≤
  87           (Real.exp 1 * T / α j)^(α j) * (((bucket j).card : ℝ) / Fintype.card E)^(α j) :=
  88         mul_le_mul_of_nonneg_right (choose_le_exp_div_pow T (α j)) (by positivity)
  89       _ = _ := by rw [← mul_pow]; congr 1; ring
