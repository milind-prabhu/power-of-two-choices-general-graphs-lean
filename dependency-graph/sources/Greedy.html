   4 /-! A quantitative version of Fact 3.6, for every arrival sequence and either
   5 legal tie choice. The proof uses a doubling inequality for high-load layers. -/
   6 namespace GraphChoices
   7 open Finset
   8 variable {V : Type*} [Fintype V] [DecidableEq V] {T : ℕ}

  10 def prefixLoad (I : Instance V T) (a : Orientation T) (k : ℕ) (v : V) : ℕ :=
  11   (univ.filter fun t => t.val < k ∧ destination I a t = v).card
  12 
  13 /-- A relational specification: both ways of breaking ties are included. -/
  14 def IsGreedy (I : Instance V T) (a : Orientation T) : Prop :=
  15   ∀ t, ∀ v, (v = I.left t ∨ v = I.right t) →
  16     prefixLoad I a t.val (destination I a t) ≤ prefixLoad I a t.val v

  19  theorem prefixLoad_le_load (I : Instance V T) (a : Orientation T) (k : ℕ) (v : V) :
  20     prefixLoad I a k v ≤ load I a v := by
  21   apply card_le_card
  22   intro t ht
  23   simp only [mem_filter, mem_univ, true_and] at ht ⊢
  24   exact ht.2

  27  theorem prefixLoad_strict (I : Instance V T) (a : Orientation T) (v : V)
  28     (t u : Fin T) (htu : t < u) (ht : destination I a t = v) :
  29     prefixLoad I a t.val v < prefixLoad I a u.val v := by
  30   have hnot : t ∉ univ.filter (fun s : Fin T => s.val < t.val ∧ destination I a s = v) := by simp
  31   have hsub : insert t (univ.filter fun s : Fin T => s.val < t.val ∧ destination I a s = v) ⊆
  32       univ.filter (fun s : Fin T => s.val < u.val ∧ destination I a s = v) := by
  33     intro s hs
  34     rcases mem_insert.mp hs with rfl | hs
  35     · simp [htu, ht]
  36     · simp only [mem_filter, mem_univ, true_and] at hs ⊢
  37       exact ⟨lt_trans hs.1 htu, hs.2⟩
  38   have hc := card_le_card hsub
  39   rw [card_insert_of_notMem hnot] at hc
  40   exact hc

  43 /-- A vertex has at most k arrivals assigned before its load reaches k. -/
  44  theorem low_arrivals_le (I : Instance V T) (a : Orientation T) (v : V) (k : ℕ) :
  45     (univ.filter fun t => destination I a t = v ∧ prefixLoad I a t.val v < k).card ≤ k := by
  46   have h := card_le_card_of_injOn (fun t : Fin T => prefixLoad I a t.val v)
  47     (s := univ.filter fun t => destination I a t = v ∧ prefixLoad I a t.val v < k)
  48     (t := range k)
  49     (by intro t ht; simpa using (mem_filter.mp ht).2.2)
  50     (by
  51       intro t ht u hu heq
  52       dsimp at heq
  53       by_contra hne
  54       rcases lt_or_gt_of_ne hne with h | h
  55       · have := prefixLoad_strict I a v t u h (mem_filter.mp ht).2.1
  56         omega
  57       · have := prefixLoad_strict I a v u t h (mem_filter.mp hu).2.1
  58         omega)
  59   simpa using h

  61 def highVertices (I : Instance V T) (a : Orientation T) (k : ℕ) : Finset V :=
  62   univ.filter fun v => k ≤ load I a v
  63 
  64 def highArrivals (I : Instance V T) (a : Orientation T) (k : ℕ) : Finset (Fin T) :=
  65   univ.filter fun t => k ≤ prefixLoad I a t.val (destination I a t)

  67  theorem highArrivals_internal (I : Instance V T) (a : Orientation T) (hg : IsGreedy I a)
  68     (k : ℕ) :
  69     highArrivals I a k ⊆ univ.filter
  70       (fun t => I.left t ∈ highVertices I a k ∧ I.right t ∈ highVertices I a k) := by
  71   intro t ht
  72   have hk := (mem_filter.mp ht).2
  73   simp only [mem_filter, mem_univ, true_and, highVertices]
  74   exact ⟨le_trans hk ((hg t _ (Or.inl rfl)).trans (prefixLoad_le_load I a _ _)),
  75     le_trans hk ((hg t _ (Or.inr rfl)).trans (prefixLoad_le_load I a _ _))⟩

  78  theorem load_le_low_add_high (I : Instance V T) (a : Orientation T) (v : V) (k : ℕ) :
  79     load I a v ≤ k + ((highArrivals I a k).filter fun t => destination I a t = v).card := by
  80   have heq : load I a v =
  81       (univ.filter fun t => destination I a t = v ∧ prefixLoad I a t.val v < k).card +
  82       ((highArrivals I a k).filter fun t => destination I a t = v).card := by
  83     rw [load, ← filter_card_add_filter_neg_card_eq_card (s := univ.filter fun t => destination I a t = v)
  84       (p := fun t => prefixLoad I a t.val v < k)]
  85     congr 1
  86     · congr 1
  87       ext t
  88       simp
  89     · congr 1
  90       ext t
  91       simp only [highArrivals, mem_filter, mem_univ, true_and, not_lt]
  92       constructor
  93       · rintro ⟨hd, hk⟩; exact ⟨by simpa [hd] using hk, hd⟩
  94       · rintro ⟨hk, hd⟩; exact ⟨hd, by simpa [hd] using hk⟩
  95   rw [heq]
  96   exact Nat.add_le_add_right (low_arrivals_le I a v k) _

  98 /-- The key deterministic layer inequality, with the true offline optimum. -/
  99  theorem greedy_layer_bound (I : Instance V T) (a : Orientation T) (hg : IsGreedy I a)
 100     (k d : ℕ) :
 101     d * (highVertices I a (k + d)).card ≤ opt I * (highVertices I a k).card := by
 102   calc
 103     d * (highVertices I a (k + d)).card = ∑ v ∈ highVertices I a (k+d), d := by simp [mul_comm]
 104     _ ≤ ∑ v ∈ highVertices I a (k+d),
 105         ((highArrivals I a k).filter fun t => destination I a t = v).card := by
 106       apply sum_le_sum
 107       intro v hv
 108       have hl := (mem_filter.mp hv).2
 109       have hb := load_le_low_add_high I a v k
 110       omega
 111     _ ≤ ∑ v, ((highArrivals I a k).filter fun t => destination I a t = v).card :=
 112       sum_le_sum_of_subset_of_nonneg (subset_univ _) (fun _ _ _ => Nat.zero_le _)
 113     _ = (highArrivals I a k).card := by
 114       symm
 115       exact card_eq_sum_card_fiberwise (fun _ _ => mem_univ _)
 116     _ ≤ internalCount I (highVertices I a k) := card_le_card (highArrivals_internal I a hg k)
 117     _ ≤ opt I * (highVertices I a k).card := by
 118       simpa [mul_comm] using density_le_opt I (highVertices I a k)

 120  theorem greedy_layers_decay (I : Instance V T) (a : Orientation T) (hg : IsGreedy I a)
 121     (hc : 0 < opt I) (j : ℕ) :
 122     2^j * (highVertices I a (2 * opt I * j)).card ≤ Fintype.card V := by
 123   induction j with
 124   | zero => simp [highVertices]
 125   | succ j ih =>
 126     have h := greedy_layer_bound I a hg (2 * opt I * j) (2 * opt I)
 127     have hstep : 2 * (highVertices I a (2 * opt I * (j+1))).card ≤
 128         (highVertices I a (2 * opt I * j)).card := by
 129       have heq : 2 * opt I * j + 2 * opt I = 2 * opt I * (j+1) := by ring
 130       rw [heq] at h
 131       nlinarith
 132     calc
 133       _ = 2^j * (2 * (highVertices I a (2 * opt I * (j+1))).card) := by rw [pow_succ]; ring
 134       _ ≤ 2^j * (highVertices I a (2 * opt I * j)).card := Nat.mul_le_mul_left _ hstep
 135       _ ≤ _ := ih

 137 /-- Explicit adversarial bound. Any j with 2^j > |V| is sufficient. -/
 138  theorem greedy_maxLoad_lt (I : Instance V T) (a : Orientation T) (hg : IsGreedy I a)
 139     (hT : 0 < T) (j : ℕ) (hj : Fintype.card V < 2^j) :
 140     maxLoad I a < 2 * opt I * j := by
 141   have hd := greedy_layers_decay I a hg (opt_pos I hT) j
 142   have hz : (highVertices I a (2 * opt I * j)).card = 0 := by
 143     by_contra hn
 144     have : 1 ≤ (highVertices I a (2 * opt I * j)).card := by omega
 145     have := Nat.mul_le_mul_left (2^j) this
 146     omega
 147   have hn : Nonempty V := ⟨I.left ⟨0,hT⟩⟩
 148   have hp : 0 < 2 * opt I * j := by
 149     have : 0 < j := by
 150       by_contra h
 151       have hj0 : j = 0 := by omega
 152       have := Fintype.card_pos_iff.mpr hn
 153       simp [hj0] at hj
 154     exact Nat.mul_pos (Nat.mul_pos (by omega) (opt_pos I hT)) this
 155   have hbound : maxLoad I a ≤ 2 * opt I * j - 1 := by
 156     apply (maxLoad_le_iff I a _).2
 157     intro v
 158     have hnot : v ∉ highVertices I a (2 * opt I * j) := by
 159       rw [card_eq_zero.mp hz]; simp
 160     simp only [highVertices, mem_filter, mem_univ, true_and, not_le] at hnot
 161     omega
 162   omega

 164 /-- Fact 3.6, with an explicit constant and base-two logarithm. -/
 165  theorem greedy_competitive (I : Instance V T) (a : Orientation T) (hg : IsGreedy I a)
 166     (hT : 0 < T) :
 167     maxLoad I a < 2 * opt I * (Nat.log 2 (Fintype.card V) + 1) := by
 168   exact greedy_maxLoad_lt I a hg hT _ (Nat.lt_pow_succ_log_self (by decide) _)
