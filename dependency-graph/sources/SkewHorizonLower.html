   9 theorem skewValue_le_eight_expectedOpt_horizon (G F : Graph L R) [Nonempty G]
  10     (hF : F ⊆ G) (A : Finset L) (N T : ℕ) (d : ℝ)
  11     (hd : 0 < d) (hlog : 2 ≤ Real.log (N:ℝ)) (hlogT : Real.log (N:ℝ) ≤ T)
  12     (hleft : ∀ u, (leftDegree G u:ℝ) ≤ d*Real.log (N:ℝ))
  13     (hvertices : Fintype.card (L ⊕ R) ≤ N)
  14     (haverage : 2*(G.card:ℝ) = (T:ℝ)*d)
  15     (hadm : SkewAdmissible F A d (N:ℝ)) :
  16     skewValue F A d (N:ℝ) ≤ 8*expectedOpt (graphBase G) T := by
  17   let a : ℝ := A.card
  18   let b : ℝ := (neighborSet F A).card
  19   let δ : ℝ := minLeftDegree F A
  20   let σ := skewValue F A d (N:ℝ)
  21   let k : ℕ := ⌊σ/2⌋₊
  22   have ha : 0 < a := by dsimp only [a]; exact_mod_cast hadm.1.card_pos
  23   have hb : 1 ≤ b := by
  24     dsimp only [b]
  25     exact_mod_cast (show 1 ≤ (neighborSet F A).card from hadm.2.1.card_pos)
  26   have hδ : 0 < δ := by dsimp only [δ]; exact_mod_cast hadm.2.2.2.1
  27   have han : a ≤ N := by
  28     have hh : A.card ≤ N := (card_le_univ A).trans
  29       ((by simp only [Fintype.card_sum]; omega : Fintype.card L ≤ Fintype.card (L ⊕ R)).trans hvertices)
  30     dsimp only [a]
  31     exact_mod_cast hh
  32   obtain ⟨u,hu⟩ := hadm.1
  33   have hdeg : δ ≤ d*Real.log (N:ℝ) := by
  34     apply le_trans _ (hleft u)
  35     dsimp only [δ]
  36     exact_mod_cast (minLeftDegree_le F hu).trans (leftDegree_mono hF u)
  37   have hσ : 0 ≤ σ := by
  38     apply div_nonneg
  39     · apply Real.log_nonneg
  40       apply (one_le_div (by linarith : 0 < b)).mpr
  41       dsimp only [b]
  42       exact_mod_cast hadm.2.2.1
  43     · exact (Real.log_pos hadm.2.2.2.2).le
  44   have hkfloor : 2*(k:ℝ) ≤ σ := by
  45     have hh := Nat.floor_le (by linarith : 0 ≤ σ/2)
  46     change (⌊σ/2⌋₊:ℝ) ≤ σ/2 at hh
  47     dsimp only [k]
  48     linarith
  49   have hσk : σ < 2*((k:ℝ)+1) := by
  50     have hh := Nat.lt_floor_add_one (σ/2)
  51     change σ/2 < (k:ℝ)+1 at hh
  52     linarith
  53   have hN : 0 < N := by
  54     by_contra h
  55     have hh : N = 0 := by omega
  56     norm_num [hh] at hlog
  57   have hT : 0 < T := by
  58     have hh : (0:ℝ) < T := by linarith
  59     exact_mod_cast hh
  60   have hopt := expectedOpt_ge_one (graphBase G) T hT
  61   by_cases hk : k = 0
  62   · have hh : (k:ℝ) = 0 := by simp [hk]
  63     change σ ≤ _
  64     linarith
  65   have hkpos : 0 < k := Nat.pos_of_ne_zero hk
  66   have hkr : (0:ℝ) < k := by exact_mod_cast hkpos
  67   have hklog : (k:ℝ) ≤ Real.log (N:ℝ) :=
  68     skew_threshold_le_log ha hb han hδ hd hlog hdeg k hkfloor
  69   have hkT : k ≤ T := by exact_mod_cast hklog.trans hlogT
  70   let x := min 1 (δ/d)/(k:ℝ)
  71   have hx : 0 ≤ x := div_nonneg (le_min (by norm_num) (div_nonneg hδ.le hd.le)) hkr.le
  72   have hx1 : x ≤ 1 := by
  73     apply (div_le_one hkr).mpr
  74     have hh : (1:ℝ) ≤ k := by exact_mod_cast hkpos
  75     exact (min_le_left _ _).trans hh
  76   have hmean : x ≤ ((T/k : ℕ):ℝ)*(δ/G.card) :=
  77     block_length_mean T k hkpos hkT δ d G.card hδ.le hd
  78       (by exact_mod_cast (show 0 < G.card by simpa only [← Fintype.card_coe] using Fintype.card_pos (α := G))) haverage
  79   have hpop : b ≤ a*(x/2)^k :=
  80     skew_block_population ha hb han hδ hd hlog hdeg k hkpos hkfloor
  81   have hlower := expectedOpt_subgraph_blocks G F hF A hadm.2.1 k (T/k) x hx hx1 hmean hpop
  82   have hmon := expectedOpt_mono_horizon (graphBase G) (by simpa only [Nat.mul_comm] using Nat.div_mul_le_self T k)
  83   have hlow : (k:ℝ)/3 ≤ expectedOpt (graphBase G) T := hlower.trans hmon
  84   change σ ≤ _
  85   linarith

  87 /-- Concrete graph log-skewness is absorbed into the actual expected
  88 offline optimum; no skewness-lower-bound premise remains. -/
  89 theorem graphLogSkewness_le_eight_expectedOpt_horizon (G : Graph L R) [Nonempty G]
  90     (N T : ℕ) (d : ℝ) (hd : 0 < d) (hlog : 2 ≤ Real.log (N:ℝ)) (hlogT : Real.log (N:ℝ) ≤ T)
  91     (hleft : ∀ u, (leftDegree G u:ℝ) ≤ d*Real.log (N:ℝ))
  92     (hvertices : Fintype.card (L ⊕ R) ≤ N)
  93     (haverage : 2*(G.card:ℝ) = (T:ℝ)*d) :
  94     graphLogSkewness G d (N:ℝ) ≤ 8*expectedOpt (graphBase G) T := by
  95   classical
  96   unfold graphLogSkewness
  97   apply (sup'_le_iff _ _).mpr
  98   intro y hy
  99   rcases mem_insert.mp hy with rfl | hy
 100   · have hh := expectedOpt_nonneg (graphBase G) T
 101     dsimp only [id]
 102     linarith
 103   · obtain ⟨⟨F,A⟩,hFA,rfl⟩ := mem_image.mp hy
 104     have hh := mem_filter.mp hFA
 105     have hF : F ⊆ G := mem_powerset.mp (mem_product.mp hh.1).1
 106     exact skewValue_le_eight_expectedOpt_horizon G F hF A N T d hd hlog hlogT hleft hvertices haverage hh.2
