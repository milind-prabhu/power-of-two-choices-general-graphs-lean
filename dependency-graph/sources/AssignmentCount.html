   4 /-! Counting sequentially generated assignments by their available choices.
   5 Menus may depend on every preceding choice. No independence is assumed. -/
   6 namespace GraphChoices
   7 open Finset
   8 variable {V : Type*} [Fintype V] [DecidableEq V]
   9 
  10 def appendChoice {n : ℕ} (f : Fin n → V) (v : V) : Fin (n+1) → V := Fin.snoc f v
  11 
  12 noncomputable def menuAssignments (menu : (k : ℕ) → (Fin k → V) → Finset V) :
  13     (n : ℕ) → Finset (Fin n → V)
  14   | 0 => univ
  15   | n+1 => (menuAssignments menu n).biUnion fun f =>
  16       (menu n f).image fun v => appendChoice f v
  17 
  18 def MenuValid (menu : (k : ℕ) → (Fin k → V) → Finset V) : (n : ℕ) → (Fin n → V) → Prop
  19   | 0, _ => True
  20   | n+1, f => MenuValid menu n (Fin.init f) ∧ f (Fin.last n) ∈ menu n (Fin.init f)
  21 
  22  theorem mem_menuAssignments_iff (menu : (k : ℕ) → (Fin k → V) → Finset V)
  23     (n : ℕ) (f : Fin n → V) : f ∈ menuAssignments menu n ↔ MenuValid menu n f := by
  24   classical
  25   induction n with
  26   | zero =>
  27     change f ∈ (univ : Finset (Fin 0 → V)) ↔ True
  28     simp only [mem_univ]
  29   | succ n ih =>
  30     simp only [menuAssignments, mem_biUnion, mem_image]
  31     constructor
  32     · rintro ⟨g,hg,v,hv,rfl⟩
  33       simp only [MenuValid, appendChoice, Fin.init_snoc, Fin.snoc_last]
  34       exact ⟨(ih g).mp hg,hv⟩
  35     · rintro ⟨hg,hv⟩
  36       exact ⟨Fin.init f,(ih _).mpr hg,f (Fin.last n),hv,Fin.snoc_init_self f⟩
  37 
  38 /-- The number of complete assignments is at most the product of the menu
  39 bounds, even though each menu can depend on the prefix. -/
  40  theorem menuAssignments_card_le (menu : (k : ℕ) → (Fin k → V) → Finset V)
  41     (d : ℕ → ℕ) (n : ℕ) (hmenu : ∀ k, k < n → ∀ f, (menu k f).card ≤ d k) :
  42     (menuAssignments menu n).card ≤ ∏ k ∈ range n, d k := by
  43   classical
  44   induction n with
  45   | zero => simp [menuAssignments]
  46   | succ n ih =>
  47     have hi := ih (fun k hk f => hmenu k (Nat.lt_succ_of_lt hk) f)
  48     rw [menuAssignments]
  49     calc
  50       _ ≤ ∑ f ∈ menuAssignments menu n, ((menu n f).image fun v => appendChoice f v).card := card_biUnion_le (s := menuAssignments menu n)
  51         (t := fun f : Fin n → V => (menu n f).image (appendChoice f))
  52       _ ≤ ∑ f ∈ menuAssignments menu n, d n := by
  53         apply sum_le_sum
  54         intro f _
  55         exact card_image_le.trans (hmenu n (by omega) f)
  56       _ = (menuAssignments menu n).card * d n := by simp
  57       _ ≤ (∏ k ∈ range n, d k) * d n := Nat.mul_le_mul_right _ hi
  58       _ = _ := (prod_range_succ _ _).symm
  59 
  60 end GraphChoices
