  10 theorem rateReference_bounds (G : Graph L R) [Nonempty G] (N T : ℕ) (d : ℝ)
  11     (hT : 0 < T) (hd : 0 < d) (hvertices : Fintype.card (L ⊕ R) ≤ N)
  12     (haverage : 2*(G.card:ℝ) = (T:ℝ)*d) :
  13     d ≤ 16*(rateReference G d:ℝ) ∧ rateReference G d ≤ N^2 ∧
  14       (rateReference G d:ℝ)*((T:ℝ)/G.card) ≤ 6*expectedOpt (graphBase G) T := by
  15   have hTr : (1:ℝ) ≤ T := by exact_mod_cast hT
  16   have hM : (0:ℝ) < G.card := by
  17     exact_mod_cast (show 0 < G.card by simpa only [← Fintype.card_coe] using Fintype.card_pos (α := G))
  18   have hdM : d ≤ 2*(G.card:ℝ) := by nlinarith
  19   have hceilM : ⌈d/16⌉₊ ≤ G.card := Nat.ceil_le.mpr (by linarith)
  20   have hbaseM : (graphBase G).referenceCap ≤ G.card := by simpa only [Fintype.card_coe] using (graphBase G).referenceCap_le_edges
  21   have hBM : rateReference G d ≤ 2*G.card := by unfold rateReference; omega
  22   have hMprod : G.card ≤ Fintype.card L*Fintype.card R := by simpa only [Fintype.card_prod] using card_le_univ G
  23   have hsum : Fintype.card L+Fintype.card R ≤ N := by simpa only [Fintype.card_sum] using hvertices
  24   have hM2 : 2*G.card ≤ N^2 := by nlinarith [Nat.mul_self_le_mul_self hsum]
  25   have hc := Nat.le_ceil (d/16)
  26   have hcap : d ≤ 16*(rateReference G d:ℝ) := by
  27     unfold rateReference
  28     push_cast
  29     have hbase : (0:ℝ) ≤ (graphBase G).referenceCap := Nat.cast_nonneg _
  30     linarith
  31   refine ⟨hcap, hBM.trans hM2, ?_⟩
  32   have hceil : (⌈d/16⌉₊:ℝ) ≤ d/16+1 := (Nat.ceil_lt_add_one (by linarith : 0 ≤ d/16)).le
  33   have hz : 0 ≤ (T:ℝ)/G.card := by positivity
  34   have hh := mul_le_mul_of_nonneg_right hceil hz
  35   have heq : (d/16)*((T:ℝ)/G.card) = 1/8 := by
  36     apply (eq_div_iff (by norm_num : (8:ℝ) ≠ 0)).mpr
  37     field_simp
  38     nlinarith
  39   have hb := expected_reference_cap_bound (graphBase G) T
  40   have he := expected_edge_rate_le_two_opt (graphBase G) T
  41   rw [Fintype.card_coe] at hb he
  42   have ho := expectedOpt_ge_one (graphBase G) T hT
  43   unfold rateReference
  44   push_cast
  45   rw [add_mul]
  46   nlinarith

  48 theorem exists_rate_competitive_policy (G : Graph L R) [Nonempty G] (N T : ℕ) (d : ℝ)

  50     (hlogT : Real.log (N:ℝ) ≤ T) (hTN : T ≤ N)
  51     (hleft : ∀ u, (leftDegree G u:ℝ) ≤ d*Real.log (N:ℝ))
  52     (hD : ∀ u, leftDegree G u ≤ 4*(graphBase G).referenceCap)
  53     (hvertices : Fintype.card (L ⊕ R) ≤ N)
  54     (haverage : 2*(G.card:ℝ) = (T:ℝ)*d) :
  55     ∃ p : OnlinePolicy G, expectedCost (graphBase G) (onlineOrientation (T := T) p) ≤
  56       15000*loglogFactor N*expectedOpt (graphBase G) T := by

  60     (fun u => (hD u).trans (Nat.mul_le_mul_left 4 hBbase))
  61   refine ⟨P.policy T, ?_⟩
  62   have hh := P.expected_cost_horizon T hT hTN hvertices haverage hBsquare
  63   have hs := graphLogSkewness_le_eight_expectedOpt_horizon G N T d hd hlog hlogT hleft hvertices haverage
  64   have ho := expectedOpt_ge_one (graphBase G) T hT

  66   have hsum : (rateReference G d:ℝ)*((T:ℝ)/G.card)+graphLogSkewness G d (N:ℝ)+1 ≤
  67       15*expectedOpt (graphBase G) T := by linarith

  69       5000*(iteratedLog N:ℝ)*expectedOpt (graphBase G) T := by

  72     nlinarith
