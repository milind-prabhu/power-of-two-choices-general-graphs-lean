  14 theorem largeHeavyTree_subset_largeShape (H : C → Peeling.Graph L R)
  15     (encode : E → Arrival L R C) (α : C → ℕ)
  16     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
  17     (T k : ℕ) :
  18     largeHeavyTreeEvent encode α T k ⊆
  19       largeShapeEvent H encode α T k (Fintype.card (L ⊕ R)) := by
  20   classical
  21   intro ω hω
  22   obtain ⟨v, s, G, htree, hadj, hlarge, hheavy, root, hroot⟩ := (mem_filter.mp hω).2
  23   letI : Fintype s := Fintype.ofFinite s
  24   let color : s → Bool := fun x => isLeft x.val.val
  25   have hcolor : ∀ x y, G.Adj x y → color x ≠ color y :=
  26     fun x y hxy => replay_adj_sides _ _ _ (hadj x y hxy)
  27   obtain ⟨m, shape, e, hsize, heroot, hspec⟩ :=
  28     exists_canonical_embedding G htree.isConnected color hcolor root hroot
  29   let f : Fin (m+1) → L ⊕ R := fun i => (e i).val.val
  30   have hf : Function.Injective f :=
  31     Subtype.val_injective.comp (Subtype.val_injective.comp e.injective)
  32   have hroot' : ∃ w, f ⟨0, by omega⟩ = Sum.inr w := by
  33     change ∃ w, (e ⟨0, by omega⟩).val.val = Sum.inr w
  34     rw [heroot]
  35     cases hr : root.val.val with
  36     | inl u => simp [hr, isLeft] at hroot
  37     | inr w => exact ⟨w, rfl⟩
  38   have hside : ∀ i, isLeft (f (childIndex i)) = (skeletonPattern shape).childLeft i :=
  39     fun i => (hspec i).2
  40   have hedge : ∀ i, (arrivalGraph (residualInstance α (sampleArrivals encode ω))).Adj
  41       (f (childIndex i)) (f (parentIndex (skeletonPattern shape) i)) :=
  42     fun i => hadj _ _ (hspec i).1
  43   have hsize' : m+1 = Nat.card s := by simpa only [Nat.card_eq_fintype_card] using hsize
  44   have hpart : Nat.card {x : s // color x = true} + Nat.card {x : s // color x = false} = Nat.card s := by
  45     simpa only [Nat.card_eq_fintype_card, Fintype.card_subtype] using bool_partition_card color
  46   have hcount := left_index_card m e color (by rw [heroot]; exact hroot)
  47   have hleft : m ≤ 2*(univ.filter fun i : Fin m => isLeft (f (childIndex i)) = true).card := by
  48     change m ≤ 2*(univ.filter fun i : Fin m => color (e (childIndex i)) = true).card
  49     rw [hcount]
  50     change Nat.card {x : s // color x = false}+1 ≤ Nat.card {x : s // color x = true} at hheavy
  51     omega
  52   have hshape := mem_heavyShapeEvent_of_geometric_embedding H encode α hsource
  53     ω shape f hf hroot' hside hedge hleft
  54   have hk : k ≤ m := by omega
  55   have hbound : m+1 ≤ Fintype.card (L ⊕ R) := by
  56     simpa only [Fintype.card_fin] using Fintype.card_le_of_injective f hf
  57   change ω ∈ (range (Fintype.card (L ⊕ R)+1)).biUnion (fun j => heavyShapeEvent H encode α T (k+j))
  58   apply mem_biUnion.mpr
  59   refine ⟨m-k, mem_range.mpr (by omega), ?_⟩
  60   have heq : k+(m-k) = m := by omega
  61   simpa only [heq] using hshape

  63 /-- Actual component shattering for concrete automatic thresholds. There is
  64 no canonical-coverage or per-shape probability premise in this theorem. -/
  65 theorem largeComponent_polynomial_automatic (H : C → Peeling.Graph L R)
  66     (encode : E → Arrival L R C) (DL DR : C → ℕ) (D T d k N c : ℕ) (f : C → ℝ)
  67     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
  68     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
  69     (hDL : ∀ u c, (sourceBucket encode u c).card ≤ DL c)
  70     (hL : ∀ c u, Peeling.leftDegree (H c) u ≤ D)
  71     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c)
  72     (hT : 0 < T) (hf : ∀ c, 1 < f c)
  73     (hN : 0 < N) (hLN : Fintype.card L ≤ N) (hRN : Fintype.card R ≤ N)
  74     (hd : polynomialCutoff N c ≤ d+1) (hk : polynomialCutoff N c ≤ k)
  75     (hmean : ∀ u, Real.exp 1*T*(((leftSourceBucket encode u).card : ℝ)/Fintype.card E)/(d+1) ≤ 1/2) :
  76     eventProbability (largeComponentEvent encode (automaticThresholds (E := E) DL DR D T f)
  77       T ((d+1)*(k+1))) ≤ 3/(N:ℝ)^c := by
  78   apply largeComponent_polynomial_automatic_of_coverage H encode DL DR D T d k
  79     (Fintype.card (L ⊕ R)) N c f hsimple hDL hL hR hT hf hN hLN hRN hd hk hmean
  80   exact largeHeavyTree_subset_largeShape H encode _ hsource T k

  82 theorem expected_load_polynomial_automatic (H : C → Peeling.Graph L R)
  83     (encode : E → Arrival L R C) (DL DR : C → ℕ) (D T d k N c : ℕ) (f : C → ℝ)
  84     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
  85     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
  86     (hDL : ∀ u c, (sourceBucket encode u c).card ≤ DL c)
  87     (hL : ∀ c u, Peeling.leftDegree (H c) u ≤ D)
  88     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c)
  89     (hT : 0 < T) (hf : ∀ c, 1 < f c)
  90     (hN : 0 < N) (hLN : Fintype.card L ≤ N) (hRN : Fintype.card R ≤ N)
  91     (hd : polynomialCutoff N c ≤ d+1) (hk : polynomialCutoff N c ≤ k)
  92     (hmean : ∀ u, Real.exp 1*T*(((leftSourceBucket encode u).card : ℝ)/Fintype.card E)/(d+1) ≤ 1/2) :
  93     expectedTotalMaxLoad encode (automaticThresholds (E := E) DL DR D T f) T ≤
  94       (∑ a, (automaticThresholds (E := E) DL DR D T f a : ℝ)) +
  95       (2*(Nat.log 2 ((d+1)*(k+1))+1) : ℕ)*expectedReplayOpt encode T + 3*T/(N:ℝ)^c := by
  96   apply expected_load_polynomial_automatic_of_coverage H encode DL DR D T d k
  97     (Fintype.card (L ⊕ R)) N c f hsimple hDL hL hR hT hf hN hLN hRN hd hk hmean
