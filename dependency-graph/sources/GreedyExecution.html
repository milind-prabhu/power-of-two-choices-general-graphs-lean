  10  theorem prefixLoad_succ (I : Instance V T) (a : Orientation T) (t : Fin T) (v : V) :
  11     prefixLoad I a (t.val+1) v = prefixLoad I a t.val v + if destination I a t = v then 1 else 0 := by
  12   classical
  13   have htime (s : Fin T) : s.val < t.val+1 ↔ s.val < t.val ∨ s = t := by
  14     constructor
  15     · intro hs
  16       by_cases heq : s.val = t.val
  17       · exact Or.inr (Fin.ext heq)
  18       · exact Or.inl (by omega)
  19     · rintro (h | rfl) <;> omega
  20   have hnot : t ∉ univ.filter (fun s : Fin T => s.val < t.val ∧ destination I a s = v) := by simp
  21   by_cases hd : destination I a t = v
  22   · have heq : (univ.filter fun s : Fin T => s.val < t.val+1 ∧ destination I a s = v) =
  23         insert t (univ.filter fun s : Fin T => s.val < t.val ∧ destination I a s = v) := by
  24       ext s
  25       simp only [mem_filter, mem_univ, true_and, mem_insert, htime]
  26       constructor
  27       · rintro ⟨h,hs⟩
  28         rcases h with h | rfl
  29         · exact Or.inr ⟨h,hs⟩
  30         · exact Or.inl rfl
  31       · rintro (rfl | h)
  32         · exact ⟨Or.inr rfl,hd⟩
  33         · exact ⟨Or.inl h.1,h.2⟩
  34     simp only [prefixLoad, heq, card_insert_of_notMem hnot, if_pos hd]
  35   · have heq : (univ.filter fun s : Fin T => s.val < t.val+1 ∧ destination I a s = v) =
  36         univ.filter (fun s : Fin T => s.val < t.val ∧ destination I a s = v) := by
  37       ext s
  38       simp only [mem_filter, mem_univ, true_and, htime]
  39       constructor
  40       · rintro ⟨h,hs⟩
  41         rcases h with h | rfl
  42         · exact ⟨h,hs⟩
  43         · exact False.elim (hd hs)
  44       · intro h; exact ⟨Or.inl h.1,h.2⟩
  45     simp only [prefixLoad, heq, if_neg hd, Nat.add_zero]

  47  theorem execution_counts_prefix (I : Instance V T) (a : Orientation T) (x : ℕ → V → ℕ)
  48     (hzero : ∀ v, x 0 v = 0)
  49     (hstep : ∀ t : Fin T, ∀ v, x (t.val+1) v = x t.val v + if destination I a t = v then 1 else 0)
  50     (k : ℕ) (hk : k ≤ T) (v : V) : x k v = prefixLoad I a k v := by
  51   induction k with
  52   | zero => simp [hzero, prefixLoad]
  53   | succ k ih =>
  54     let t : Fin T := ⟨k,by omega⟩
  55     have hs := hstep t v
  56     have hp := prefixLoad_succ I a t v
  57     have hi := ih (by omega)
  58     simpa only [t, hi, hp] using hs
  59 
  60  theorem isGreedy_of_execution (I : Instance V T) (a : Orientation T) (x : ℕ → V → ℕ)
  61     (hzero : ∀ v, x 0 v = 0)
  62     (hstep : ∀ t : Fin T, ∀ v, x (t.val+1) v = x t.val v + if destination I a t = v then 1 else 0)
  63     (hchoice : ∀ t : Fin T, ∀ v, (v = I.left t ∨ v = I.right t) →
  64       x t.val (destination I a t) ≤ x t.val v) : IsGreedy I a := by
  65   intro t v hv
  66   rw [← execution_counts_prefix I a x hzero hstep t.val (Nat.le_of_lt t.isLt) _,
  67     ← execution_counts_prefix I a x hzero hstep t.val (Nat.le_of_lt t.isLt) _]
  68   exact hchoice t v hv

  70  theorem execution_final_load (I : Instance V T) (a : Orientation T) (x : ℕ → V → ℕ)
  71     (hzero : ∀ v, x 0 v = 0)
  72     (hstep : ∀ t : Fin T, ∀ v, x (t.val+1) v = x t.val v + if destination I a t = v then 1 else 0)
  73     (v : V) : x T v = load I a v := by
  74   rw [execution_counts_prefix I a x hzero hstep T le_rfl v]
  75   unfold prefixLoad load
  76   congr 1
  77   ext t
  78   simp [t.isLt]
