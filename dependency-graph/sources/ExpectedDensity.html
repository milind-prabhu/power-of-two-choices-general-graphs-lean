   3 /-! Expected density lower bounds, maximized over all concrete vertex sets,
   4 including the rounding cost of an integer reference capacity. These are
   5 statements about the full source sampling model, not a presumed benchmark.

  12 noncomputable def BaseGraph.baseDensity (G : BaseGraph V E) (S : Finset V) : ℝ :=
  13   (G.internalEdges S).card/(S.card : ℝ)

  15 noncomputable def BaseGraph.maxBaseDensity (G : BaseGraph V E) : ℝ :=
  16   (univ.powerset).sup' (by simp) (G.baseDensity)

  18 noncomputable def BaseGraph.referenceCap (G : BaseGraph V E) : ℕ := ⌈G.maxBaseDensity⌉₊

  20 theorem expectedOpt_nonneg (G : BaseGraph V E) (T : ℕ) : 0 ≤ expectedOpt G T := by
  21   unfold expectedOpt average
  22   positivity

  29   (G.baseDensity_nonneg ∅).trans (le_sup' G.baseDensity (by simp))

  31 theorem expected_density_ratio (G : BaseGraph V E) (S : Finset V) (T : ℕ) :
  32     G.baseDensity S*((T:ℝ)/Fintype.card E) ≤ expectedOpt G T := by
  33   by_cases hz : S.card = 0
  34   · simp only [BaseGraph.baseDensity, hz, Nat.cast_zero, div_zero, zero_mul]
  35     exact expectedOpt_nonneg G T
  36   have hS : (0:ℝ) < S.card := by exact_mod_cast Nat.pos_of_ne_zero hz
  37   calc
  38     _ = ((T:ℝ)*(G.internalEdges S).card/Fintype.card E)/(S.card : ℝ) := by
  39       unfold BaseGraph.baseDensity
  40       ring
  41     _ ≤ ((S.card : ℝ)*expectedOpt G T)/(S.card : ℝ) :=
  42       div_le_div_of_nonneg_right (expected_density_lower_bound G S T) hS.le
  43     _ = _ := by field_simp
  44 
  45 theorem expected_max_density_lower_bound (G : BaseGraph V E) (T : ℕ) :
  46     G.maxBaseDensity*((T:ℝ)/Fintype.card E) ≤ expectedOpt G T := by
  47   by_cases hT : T = 0
  48   · simpa only [hT, Nat.cast_zero, zero_div, mul_zero] using expectedOpt_nonneg G 0
  49   have hz : (0:ℝ) < (T:ℝ)/Fintype.card E := by
  50     exact div_pos (by exact_mod_cast Nat.pos_of_ne_zero hT) (by exact_mod_cast Fintype.card_pos)
  51   apply (le_div_iff₀ hz).mp
  52   apply (Finset.sup'_le_iff _ _).mpr
  53   intro S _
  54   exact (le_div_iff₀ hz).mpr (expected_density_ratio G S T)

  56 /-- The expected count of a fixed source-edge label is at most twice the
  57 expected optimum. Its endpoint pair has at most two vertices, including loops. -/
  58 theorem expected_edge_rate_le_two_opt (G : BaseGraph V E) (T : ℕ) :
  59     (T:ℝ)/Fintype.card E ≤ 2*expectedOpt G T := by
  60   classical
  61   let e : E := Classical.choice (inferInstance : Nonempty E)
  62   let S : Finset V := {G.left e, G.right e}
  63   have he : e ∈ G.internalEdges S := by simp [BaseGraph.internalEdges, S]
  64   have hcount : 1 ≤ (G.internalEdges S).card := card_pos.mpr ⟨e,he⟩
  65   have hcount' : (1:ℝ) ≤ (G.internalEdges S).card := by exact_mod_cast hcount
  66   have hs : S.card ≤ 2 := (card_insert_le _ _).trans (by simp)
  67   calc
  68     _ ≤ (T:ℝ)*(G.internalEdges S).card/Fintype.card E := by
  69       apply div_le_div_of_nonneg_right _ (Nat.cast_nonneg _)
  70       simpa only [mul_one] using mul_le_mul_of_nonneg_left hcount' (Nat.cast_nonneg T)
  71     _ ≤ (S.card : ℝ)*expectedOpt G T := expected_density_lower_bound G S T
  72     _ ≤ _ := mul_le_mul_of_nonneg_right (by exact_mod_cast hs) (expectedOpt_nonneg G T)
  73 
  74 /-- The integer ceiling in the reference capacity costs at most another two
  75 expected optimum units; it is not silently dropped from the density identity. -/
  76 theorem expected_reference_cap_bound (G : BaseGraph V E) (T : ℕ) :
  77     (G.referenceCap : ℝ)*((T:ℝ)/Fintype.card E) ≤ 3*expectedOpt G T := by
  78   have hc : (G.referenceCap : ℝ) ≤ G.maxBaseDensity+1 := (Nat.ceil_lt_add_one G.maxBaseDensity_nonneg).le
  79   have hm := mul_le_mul_of_nonneg_right hc (show 0 ≤ (T:ℝ)/Fintype.card E by positivity)
  80   have hd := expected_max_density_lower_bound G T
  81   have he := expected_edge_rate_le_two_opt G T
  82   nlinarith

  84 theorem BaseGraph.referenceCap_le_edges (G : BaseGraph V E) : G.referenceCap ≤ Fintype.card E := by
  85   apply Nat.ceil_le.mpr
  86   apply (Finset.sup'_le_iff _ _).mpr
  87   intro S _
  88   by_cases hS : S.card = 0
  89   · simp [BaseGraph.baseDensity, hS]
  90   have hp : (0:ℝ) < S.card := by exact_mod_cast Nat.pos_of_ne_zero hS
  91   have hone : (1:ℝ) ≤ S.card := by exact_mod_cast Nat.pos_of_ne_zero hS
  92   have he : ((G.internalEdges S).card : ℝ) ≤ Fintype.card E := by exact_mod_cast card_le_card (subset_univ _)
  93   change ((G.internalEdges S).card : ℝ)/(S.card : ℝ) ≤ Fintype.card E
  94   apply (div_le_iff₀ hp).mpr
  95   have hE0 : (0:ℝ) ≤ Fintype.card E := Nat.cast_nonneg _

  98 theorem BaseGraph.total_edges_le_card_mul_referenceCap (G : BaseGraph V E) :
  99     (Fintype.card E : ℝ) ≤ (Fintype.card V : ℝ)*G.referenceCap := by
 100   classical
 101   have hV : Nonempty V := ⟨G.left (Classical.choice (inferInstance : Nonempty E))⟩
 102   have hp : (0:ℝ) < Fintype.card V := by exact_mod_cast Fintype.card_pos_iff.mpr hV
 103   have hd : G.baseDensity univ ≤ G.maxBaseDensity := le_sup' G.baseDensity (by simp)
 104   have hc : G.maxBaseDensity ≤ (G.referenceCap : ℝ) := Nat.le_ceil _
 105   have heq : G.baseDensity univ = (Fintype.card E : ℝ)/Fintype.card V := by
 106     simp [BaseGraph.baseDensity, BaseGraph.internalEdges]
 107   rw [heq] at hd
 108   have hh := (div_le_iff₀ hp).mp (hd.trans hc)
 109   simpa only [mul_comm] using hh
