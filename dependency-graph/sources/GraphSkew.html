   7 /-! A finite maximum log-skewness over actual bipartite subgraphs, and the
   8 bridge from this parameter to hereditary expansion and terminating peeling.
   9 Admissibility explicitly excludes zero degrees and nonpositive log denominators.
  10 -/
  11 namespace GraphChoices.Peeling
  12 open Finset
  13 variable {L R : Type*} [Fintype L] [Fintype R] [DecidableEq L] [DecidableEq R]
  14 
  15 noncomputable def minLeftDegree (F : Graph L R) (A : Finset L) : ℕ :=
  16   if h : A.Nonempty then A.inf' h (leftDegree F) else 0
  17 
  18  theorem minLeftDegree_le (F : Graph L R) {A : Finset L} {u : L} (hu : u ∈ A) :
  19     minLeftDegree F A ≤ leftDegree F u := by
  20   rw [minLeftDegree, dif_pos ((show A.Nonempty from ⟨u,hu⟩))]
  21   exact inf'_le _ hu
  22 
  23  theorem le_minLeftDegree (F : Graph L R) {A : Finset L} (hA : A.Nonempty)
  24     (k : ℕ) (hk : ∀ u ∈ A, k ≤ leftDegree F u) : k ≤ minLeftDegree F A := by
  25   rw [minLeftDegree, dif_pos hA]
  26   exact le_inf' hA (leftDegree F) hk
  27 
  28 def neighborSet (F : Graph L R) (A : Finset L) : Finset R :=
  29   GraphChoices.neighbors (fun u v => (u,v) ∈ F) A
  30 
  31 noncomputable def skewValue (F : Graph L R) (A : Finset L) (d n : ℝ) : ℝ :=
  32   GraphChoices.localLogSkewness A.card (neighborSet F A).card (minLeftDegree F A) d n
  33 
  34 noncomputable def SkewAdmissible (F : Graph L R) (A : Finset L) (d n : ℝ) : Prop :=
  35   A.Nonempty ∧ (neighborSet F A).Nonempty ∧ (neighborSet F A).card ≤ A.card ∧
  36     0 < minLeftDegree F A ∧ 1 < d * (Real.log n)^2 / minLeftDegree F A
  37 
  38 noncomputable def skewValues (G : Graph L R) (d n : ℝ) : Finset ℝ := by
  39   classical
  40   exact insert 0 (((G.powerset ×ˢ (univ : Finset L).powerset).filter
  41     fun p => SkewAdmissible p.1 p.2 d n).image fun p => skewValue p.1 p.2 d n)
  42 
  43 noncomputable def graphLogSkewness (G : Graph L R) (d n : ℝ) : ℝ :=
  44   (skewValues G d n).sup' (by
  45     classical
  46     simp [skewValues]) id
  47 
  48  theorem graphLogSkewness_nonneg (G : Graph L R) (d n : ℝ) :
  49     0 ≤ graphLogSkewness G d n := by
  50   classical
  51   unfold graphLogSkewness
  52   exact le_sup' id (by simp [skewValues])
  53 
  54  theorem skewValue_le_graphLogSkewness {G F : Graph L R} (hF : F ⊆ G)
  55     (A : Finset L) (d n : ℝ) (ha : SkewAdmissible F A d n) :
  56     skewValue F A d n ≤ graphLogSkewness G d n := by
  57   classical
  58   unfold graphLogSkewness
  59   apply le_sup' id
  60   apply mem_insert_of_mem
  61   apply mem_image.mpr
  62   refine ⟨(F,A), ?_, rfl⟩
  63   simp [hF, ha]
  64 
  65 noncomputable def scaleCapacity (G : Graph L R) (d n f : ℝ) : ℕ :=
  66   ⌈(f * Real.log n)^(2 * graphLogSkewness G d n)⌉₊
  67 
  68  theorem scaleCapacity_pos (G : Graph L R) (d n f : ℝ)
  69     (hf : 1 ≤ f) (hn : 1 < Real.log n) : 1 ≤ scaleCapacity G d n f := by
  70   have hb : 1 ≤ f * Real.log n := by nlinarith
  71   have hp : (1:ℝ) ≤ (f * Real.log n)^(2 * graphLogSkewness G d n) :=
  72     Real.one_le_rpow hb (mul_nonneg (by norm_num) (graphLogSkewness_nonneg G d n))
  73   have hc := Nat.le_ceil ((f * Real.log n)^(2 * graphLogSkewness G d n))
  74   exact_mod_cast hp.trans hc

  76 /-- Bounded log-skewness supplies the actual Hall expansion premise in every
  77 residual graph. The left-degree hypothesis is the preprocessing degree bound. -/
  78  theorem peelingExpansion_of_graphLogSkewness (G : Graph L R) (k : ℕ) (d n f : ℝ)
  79     (hd : 0 < d) (hf : 1 ≤ f) (hn : 1 < Real.log n)
  80     (hleft : ∀ u, (leftDegree G u : ℝ) ≤ d * Real.log n)
  81     (hcut : d / f^2 ≤ (k+1 : ℕ)) :
  82     PeelingExpansion G k (scaleCapacity G d n f) := by
  83   classical
  84   intro F hF A hA
  85   by_cases hAempty : A = ∅
  86   · simp [hAempty]
  87   have hAne : A.Nonempty := nonempty_iff_ne_empty.mpr hAempty
  88   have hkmin : k+1 ≤ minLeftDegree F A :=
  89     le_minLeftDegree F hAne (k+1) (fun u hu => hA u hu)
  90   have hmin : 0 < minLeftDegree F A := by omega
  91   obtain ⟨u, hu⟩ := hAne
  92   have hAne : A.Nonempty := ⟨u,hu⟩
  93   have hdmin : (0:ℝ) < minLeftDegree F A := by exact_mod_cast hmin
  94   have hminmax : (minLeftDegree F A : ℝ) ≤ d * Real.log n := by
  95     have hh := (minLeftDegree_le F hu).trans (leftDegree_mono hF u)
  96     have hreal : (minLeftDegree F A : ℝ) ≤ leftDegree G u := by exact_mod_cast hh
  97     exact hreal.trans (hleft u)
  98   have hden : 1 < d * (Real.log n)^2 / minLeftDegree F A := by
  99     apply (lt_div_iff₀ hdmin).mpr
 100     have hh : d * Real.log n < d * (Real.log n)^2 := by
 101       nlinarith [mul_pos hd (mul_pos (by linarith : 0 < Real.log n) (by linarith : 0 < Real.log n - 1))]
 102     linarith
 103   have hNne : (neighborSet F A).Nonempty := by
 104     have hdu : 0 < leftDegree F u := (Nat.zero_le k).trans_lt (hA u hu)
 105     obtain ⟨v,hv⟩ := card_pos.mp hdu
 106     refine ⟨v, ?_⟩
 107     simp only [leftNeighbors, mem_filter, mem_univ, true_and] at hv
 108     simp only [neighborSet, GraphChoices.neighbors, mem_filter, mem_univ, true_and]
 109     exact ⟨u,hu,hv⟩
 110   by_cases hbal : A.card ≤ (neighborSet F A).card
 111   · have hcap := scaleCapacity_pos G d n f hf hn
 112     change A.card ≤ scaleCapacity G d n f * (neighborSet F A).card
 113     exact hbal.trans (by nlinarith)
 114   · have hadm : SkewAdmissible F A d n :=
 115       ⟨hAne, hNne, by omega, hmin, hden⟩
 116     have hskew := skewValue_le_graphLogSkewness hF A d n hadm
 117     have hdeg : d / f^2 ≤ (minLeftDegree F A : ℝ) :=
 118       hcut.trans (by exact_mod_cast hkmin)
 119     have hexp := GraphChoices.expansion_at_scale
 120       (a := (A.card : ℝ)) (b := ((neighborSet F A).card : ℝ))
 121       (d := (minLeftDegree F A : ℝ)) (D := d) (n := n) (f := f)
 122       (by exact_mod_cast hAne.card_pos) (by exact_mod_cast hNne.card_pos) hdmin
 123       (by linarith) (by linarith) (graphLogSkewness_nonneg G d n) hdeg hden hskew
 124     have hceil := Nat.le_ceil ((f * Real.log n)^(2 * graphLogSkewness G d n))
 125     have hbnonneg : (0:ℝ) ≤ (neighborSet F A).card := Nat.cast_nonneg _
 126     have hh := hexp.trans (mul_le_mul_of_nonneg_left hceil hbnonneg)
 127     change A.card ≤ scaleCapacity G d n f * (neighborSet F A).card
 128     rw [Nat.mul_comm]
 129     exact_mod_cast hh

 131 /-- One full extraction round, including finite termination, derived from the
 132 maximum over concrete log-skewness witnesses. -/
 133  theorem exists_graph_skew_piece (G : Graph L R) (k D : ℕ) (d n f : ℝ)
 134     (hd : 0 < d) (hf : 1 ≤ f) (hn : 1 < Real.log n)
 135     (hleft : ∀ u, (leftDegree G u : ℝ) ≤ d * Real.log n)
 136     (hcut : d / f^2 ≤ (k+1 : ℕ)) (hD : ∀ u, leftDegree G u ≤ D) :
 137     ∃ P : Graph L R, P ⊆ G ∧
 138       (∀ u, leftDegree (G \ P) u ≤ k) ∧
 139       (∀ v, rightDegree P v ≤ D * scaleCapacity G d n f) := by
 140   exact exists_peeling_piece G k _ D hD
 141     (peelingExpansion_of_graphLogSkewness G k d n f hd hf hn hleft hcut)
 142 
 143 end GraphChoices.Peeling
