   4 open Finset
   5 variable {V C : Type*} [Fintype V] [DecidableEq V] [Fintype C] [DecidableEq C]
   6   {m r T : ℕ}
   7 
   8 def blockBase (left : C × Fin m → V) (right : C × Fin r → V) : BaseGraph V (C × (Fin m × Fin r)) where
   9   left e := left (e.1,e.2.1)
  10   right e := right (e.1,e.2.2)
  11 
  12 def blockAllowed (c : C) (A : Finset (Fin m)) (B : Finset (Fin r)) : Finset (C × (Fin m × Fin r)) :=
  13   {c} ×ˢ (A ×ˢ B)
  14 
  15 theorem blockAllowed_card (c : C) (A : Finset (Fin m)) (B : Finset (Fin r)) :
  16     (blockAllowed c A B).card = A.card*B.card := by simp [blockAllowed]
  17 
  18 def blockLeftSet (left : C × Fin m → V) (S : Finset V) (c : C) : Finset (Fin m) :=
  19   univ.filter fun a => left (c,a) ∈ S
  20 def blockRightSet (right : C × Fin r → V) (S : Finset V) (c : C) : Finset (Fin r) :=
  21   univ.filter fun a => right (c,a) ∈ S
  22 
  23 theorem sum_block_sets_le (f : C × Fin m → V) (hf : Function.Injective f) (S : Finset V) :
  24     (∑ c, (univ.filter fun a : Fin m => f (c,a) ∈ S).card) ≤ S.card := by
  25   have heq : (∑ c, (univ.filter fun a : Fin m => f (c,a) ∈ S).card) =
  26       (univ.filter fun p : C × Fin m => f p ∈ S).card := by
  27     simp only [card_eq_sum_ones, sum_filter, Fintype.sum_prod_type]
  28   rw [heq]
  29   apply card_le_card_of_injOn f
  30   · intro p hp
  31     exact (mem_filter.mp hp).2
  32   · exact fun _ _ _ _ h => hf h
  33 
  34 theorem internalCount_block_sum (left : C × Fin m → V) (right : C × Fin r → V)
  35     (ω : Fin T → C × (Fin m × Fin r)) (S : Finset V) :
  36     internalCount ((blockBase left right).sample ω) S =
  37       ∑ c, hitCount (blockAllowed c (blockLeftSet left S c) (blockRightSet right S c)) ω := by
  38   have heq (c : C) : hitCount (blockAllowed c (blockLeftSet left S c) (blockRightSet right S c)) ω =
  39       ((univ.filter fun j => (blockBase left right).left (ω j) ∈ S ∧
  40         (blockBase left right).right (ω j) ∈ S).filter fun j => (ω j).1 = c).card := by
  41     unfold hitCount
  42     congr 1
  43     ext j
  44     simp only [blockAllowed, blockLeftSet, blockRightSet, hitCount, mem_filter,
  45       mem_univ, true_and, mem_product, mem_singleton, blockBase]
  46     constructor
  47     · rintro ⟨h,hl,hr⟩
  48       subst c
  49       exact ⟨⟨hl,hr⟩,rfl⟩
  50     · rintro ⟨⟨hl,hr⟩,h⟩
  51       subst c
  52       exact ⟨rfl,hl,hr⟩
  53   simp_rw [heq]
  54   exact card_eq_sum_card_fiberwise (fun _ _ => mem_univ _)
  55 
  56 /-- Each vertex belongs to at most one left block and at most one right
  57 block. Density bounds within every complete bipartite block therefore add. -/
  58 theorem opt_le_of_block_density (left : C × Fin m → V) (right : C × Fin r → V)
  59     (hl : Function.Injective left) (hr : Function.Injective right)
  60     (ω : Fin T → C × (Fin m × Fin r)) (k : ℕ)
  61     (hblock : ∀ c A B, A.Nonempty → B.Nonempty →
  62       hitCount (blockAllowed c A B) ω ≤ k*(A.card+B.card)) :
  63     opt ((blockBase left right).sample ω) ≤ 2*k := by
  64   apply (opt_le_iff_density _ _).mpr
  65   intro S
  66   rw [internalCount_block_sum]
  67   have heach (c : C) :
  68       hitCount (blockAllowed c (blockLeftSet left S c) (blockRightSet right S c)) ω ≤
  69         k*((blockLeftSet left S c).card+(blockRightSet right S c).card) := by
  70     by_cases hA : (blockLeftSet left S c).Nonempty
  71     · by_cases hB : (blockRightSet right S c).Nonempty
  72       · exact hblock c _ _ hA hB
  73       · have hz := not_nonempty_iff_eq_empty.mp hB
  74         simp [blockAllowed, hz, hitCount]
  75     · have hz := not_nonempty_iff_eq_empty.mp hA
  76       simp [blockAllowed, hz, hitCount]
  77   have hh := sum_le_sum (fun c (_ : c ∈ (univ : Finset C)) => heach c)
  78   rw [← mul_sum, sum_add_distrib] at hh
  79   have hL := sum_block_sets_le left hl S
  80   have hR := sum_block_sets_le right hr S
  81   change (∑ c, (blockLeftSet left S c).card) ≤ S.card at hL
  82   change (∑ c, (blockRightSet right S c).card) ≤ S.card at hR
  83   nlinarith
  84 
  85 end GraphChoices.LowerBound
