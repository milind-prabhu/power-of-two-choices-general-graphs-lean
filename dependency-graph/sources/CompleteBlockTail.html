   7 noncomputable def sizeDensityEvent (m r T k : ℕ) (c : C) (a b : ℕ) :

  10   exact ((univ.powersetCard a) ×ˢ (univ.powersetCard b)).biUnion
  11     (fun p => hitTailEvent (blockAllowed c p.1 p.2) T (k*(a+b)))

  13 theorem sizeDensityEvent_probability (m t T k : ℕ) (c : C) (a b : ℕ)
  14     (hm : 0 < m) (ht : 1 ≤ t) (ha : 0 < a) (hb : 0 < b)
  15     (ham : a ≤ m) (hbtm : b ≤ t*m) (hk : 16 ≤ k)
  16     (hrate : (T:ℝ)*(m:ℝ) ≤ 2*(Fintype.card (C × (Fin m × Fin (t*m))):ℝ))
  17     (hscale : Real.exp 1*(t:ℝ)*(2*Real.exp 1/(k:ℝ))^k ≤ 1/256) :
  18     eventProbability (sizeDensityEvent m (t*m) T k c a b) ≤ 1/(m:ℝ)^8 := by
  19   classical
  20   letI : Nonempty (Fin m) := ⟨⟨0,hm⟩⟩
  21   letI : Nonempty (Fin (t*m)) := ⟨⟨0,by positivity⟩⟩
  22   let E := C × (Fin m × Fin (t*m))
  23   have hE : (0:ℝ) < Fintype.card E := by exact_mod_cast Fintype.card_pos
  24   have hmr : (0:ℝ) < m := by exact_mod_cast hm
  25   have hkr : (0:ℝ) < k := by exact_mod_cast (show 0 < k by omega)
  26   let tail := (2*Real.exp 1*(a:ℝ)*b/((m:ℝ)*k*(a+b)))^(k*(a+b))
  27   have heach (p : Finset (Fin m) × Finset (Fin (t*m)))
  28       (hp : p ∈ (univ.powersetCard a) ×ˢ (univ.powersetCard b)) :
  29       eventProbability (hitTailEvent (blockAllowed c p.1 p.2) T (k*(a+b))) ≤ tail := by
  30     have hcards := mem_product.mp hp
  31     have hA := (mem_powersetCard.mp hcards.1).2
  32     have hB := (mem_powersetCard.mp hcards.2).2
  33     apply (hitTail_probability_le_exp _ T (k*(a+b))).trans
  34     apply pow_le_pow_left₀ (by positivity)
  35     rw [blockAllowed_card,hA,hB]
  36     push_cast
  37     have hr : (T:ℝ)/Fintype.card E ≤ 2/(m:ℝ) :=
  38       (div_le_div_iff₀ hE hmr).mpr hrate
  39     have hh := mul_le_mul_of_nonneg_right hr
  40       (show 0 ≤ Real.exp 1*(a:ℝ)*b/((k:ℝ)*(a+b)) by positivity)
  41     convert hh using 1 <;> field_simp <;> ring
  42   apply (witness_union_bound _ _).trans
  43   apply (sum_le_sum heach).trans
  44   have heq : (∑ _p ∈ ((univ : Finset (Fin m)).powersetCard a) ×ˢ ((univ : Finset (Fin (t*m))).powersetCard b), tail) =
  45       (m.choose a:ℝ)*((t*m).choose b:ℝ)*tail := by
  46     simp only [sum_const, card_product, card_powersetCard, card_univ,
  47       Fintype.card_fin, nsmul_eq_mul, Nat.cast_mul]
  48   rw [heq]
  49   exact block_size_pair_bound m t a b k hm ht ha hb ham hbtm hk hscale

  56 theorem badBlockDensity_probability (m t T k : ℕ) (hm : 0 < m) (ht : 1 ≤ t) (hk : 16 ≤ k)
  57     (hrate : (T:ℝ)*(m:ℝ) ≤ 2*(Fintype.card (C × (Fin m × Fin (t*m))):ℝ))
  58     (hscale : Real.exp 1*(t:ℝ)*(2*Real.exp 1/(k:ℝ))^k ≤ 1/256) :
  59     eventProbability (badBlockDensity (C := C) m (t*m) T k) ≤
  60       (Fintype.card C:ℝ)*m*(t*m)/(m:ℝ)^8 := by
  61   apply (witness_union_bound _ _).trans
  62   have hh := sum_le_sum (fun p (_ : p ∈ (univ : Finset (C × (Fin m × Fin (t*m))))) =>
  63     sizeDensityEvent_probability m t T k p.1 (p.2.1.val+1) (p.2.2.val+1)
  64       hm ht (by omega) (by omega) (by omega) (by omega) hk hrate hscale)
  65   convert hh using 1 <;> simp only [sum_const, card_univ, Fintype.card_prod,
  66     Fintype.card_fin, nsmul_eq_mul, Nat.cast_mul] <;> ring

  68 theorem not_badBlockDensity (m r T k : ℕ) (ω : Fin T → C × (Fin m × Fin r))
  69     (hω : ω ∉ badBlockDensity m r T k) :
  70     ∀ c A B, A.Nonempty → B.Nonempty → hitCount (blockAllowed c A B) ω ≤ k*(A.card+B.card) := by
  71   classical
  72   intro c A B hA hB
  73   by_contra h
  74   have ha : A.card-1 < m := by have hh := card_le_univ A; have hp := hA.card_pos; simp only [Fintype.card_fin] at hh; omega
  75   have hb : B.card-1 < r := by have hh := card_le_univ B; have hp := hB.card_pos; simp only [Fintype.card_fin] at hh; omega
  76   apply hω
  77   apply mem_biUnion.mpr
  78   refine ⟨(c,⟨A.card-1,ha⟩,⟨B.card-1,hb⟩),mem_univ _,?_⟩
  79   apply mem_biUnion.mpr
  80   refine ⟨(A,B),mem_product.mpr ⟨mem_powersetCard.mpr ⟨subset_univ _,by change A.card = A.card-1+1; have := hA.card_pos; omega⟩,
  81     mem_powersetCard.mpr ⟨subset_univ _,by change B.card = B.card-1+1; have := hB.card_pos; omega⟩⟩,?_⟩
  82   apply mem_filter.mpr
  83   refine ⟨mem_univ _,?_⟩
  84   have hsum : (A.card-1+1)+(B.card-1+1) = A.card+B.card := by have := hA.card_pos; have := hB.card_pos; omega
  85   change k*((A.card-1+1)+(B.card-1+1)) ≤ hitCount (blockAllowed c A B) ω
  86   rw [hsum]
  87   omega
