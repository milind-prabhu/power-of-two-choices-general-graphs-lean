   3 /-! Exact coordinate independence and the finite witness union bound used in
   4 Section 4.2.2. These are probability primitives, not the paper's still-open
   5 pattern-counting theorem. -/
   6 namespace GraphChoices
   7 open Finset
   8 
   9 noncomputable def eventProbability {Ω : Type*} [Fintype Ω] (A : Finset Ω) : ℝ :=
  10   A.card / (Fintype.card Ω : ℝ)
  11 
  12  theorem eventProbability_nonneg {Ω : Type*} [Fintype Ω] (A : Finset Ω) :
  13     0 ≤ eventProbability A := div_nonneg (Nat.cast_nonneg _) (Nat.cast_nonneg _)
  14 
  15  theorem eventProbability_mono {Ω : Type*} [Fintype Ω] {A B : Finset Ω} (h : A ⊆ B) :
  16     eventProbability A ≤ eventProbability B := by
  17   apply div_le_div_of_nonneg_right _ (Nat.cast_nonneg _)
  18   exact_mod_cast card_le_card h
  19 
  20 /-- Probability of a union is at most the sum of probabilities, with no
  21 independence requirement on the witness events. -/
  22  theorem witness_union_bound {Ω W : Type*} [Fintype Ω] [DecidableEq Ω]
  23     (W₀ : Finset W) (events : W → Finset Ω) :
  24     eventProbability (W₀.biUnion events) ≤ ∑ w ∈ W₀, eventProbability (events w) := by
  25   unfold eventProbability
  26   have hc : ((W₀.biUnion events).card : ℝ) ≤ ∑ w ∈ W₀, ((events w).card : ℝ) := by
  27     exact_mod_cast card_biUnion_le
  28   calc
  29     _ ≤ (∑ w ∈ W₀, ((events w).card : ℝ)) / Fintype.card Ω :=
  30       div_le_div_of_nonneg_right hc (Nat.cast_nonneg _)
  31     _ = _ := by simp [div_eq_mul_inv, sum_mul]
  32 
  33 /-- Product sampling has exact independence, including empty allowed sets.
  34 A coordinate allows every edge label if it is unconstrained. -/
  35  theorem probability_coordinate_constraints {E : Type*} [Fintype E]
  36     [DecidableEq E] (T : ℕ) (allowed : Fin T → Finset E) :
  37     eventProbability (Fintype.piFinset allowed) =
  38       ∏ t : Fin T, ((allowed t).card : ℝ) / Fintype.card E := by
  39   simp only [eventProbability, Fintype.card_piFinset, Fintype.card_fun, Fintype.card_fin,
  40     Nat.cast_prod, Nat.cast_pow, prod_div_distrib, prod_const, card_univ]
  41 
  42 /-- A finite family of per-coordinate witness restrictions can be union-bounded
  43 by the corresponding products of marginal probabilities. -/
  44  theorem witness_coordinate_union_bound {E W : Type*} [Fintype E] [DecidableEq E]
  45     (T : ℕ) (W₀ : Finset W) (allowed : W → Fin T → Finset E) :
  46     eventProbability (W₀.biUnion fun w => Fintype.piFinset (allowed w)) ≤
  47       ∑ w ∈ W₀, ∏ t : Fin T, ((allowed w t).card : ℝ) / Fintype.card E := by
  48   simpa only [probability_coordinate_constraints] using
  49     witness_union_bound W₀ (fun w => Fintype.piFinset (allowed w))
  50 
  51 end GraphChoices
