   4 /-! Samplewise orientation and optimum comparisons for the vertex-copy
   5 construction in Lemma 2.4. The comparisons hold for every reference orientation,
   6 not just a density-minimizing one. -/
   7 namespace GraphChoices
   8 open Finset
   9 variable {V : Type*} [Fintype V] [DecidableEq V] {T : ℕ}
  10 
  11 def flip (a : Orientation T) : Orientation T := fun t => !(a t)
  12 
  13 def splitInstance (I : Instance V T) (b : Orientation T) : Instance (V ⊕ V) T where
  14   left t := Sum.inl (destination I b t)
  15   right t := Sum.inr (destination I (flip b) t)
  16 
  17 /-- Orientations on the copied graph and original graph are in bijection. -/
  18 def transport (b a : Orientation T) : Orientation T := fun t =>
  19   if b t then a t else !(a t)
  20 
  21  theorem transport_involutive (b a : Orientation T) : transport b (transport b a) = a := by
  22   funext t
  23   simp only [transport]
  24   cases b t <;> cases a t <;> rfl
  25 
  26 omit [Fintype V] [DecidableEq V] in
  27  theorem collapse_destination (I : Instance V T) (b a : Orientation T) (t : Fin T) :
  28     Sum.elim id id (destination (splitInstance I b) a t) = destination I (transport b a) t := by
  29   cases hb : b t <;> cases ha : a t <;>
  30     simp [destination, splitInstance, transport, flip, hb, ha]
  31 
  32 omit [Fintype V] in
  33  theorem collapse_load (I : Instance V T) (b a : Orientation T) (v : V) :
  34     load I (transport b a) v =
  35       load (splitInstance I b) a (Sum.inl v) + load (splitInstance I b) a (Sum.inr v) := by
  36   simp only [load, card_eq_sum_ones, sum_filter, ← sum_add_distrib]
  37   apply sum_congr rfl
  38   intro t _
  39   rw [← collapse_destination]
  40   cases h : destination (splitInstance I b) a t with
  41   | inl x => simp
  42   | inr x => simp
  43 
  44  theorem split_maxLoad_le (I : Instance V T) (b a : Orientation T) :
  45     maxLoad (splitInstance I b) a ≤ maxLoad I (transport b a) := by
  46   apply (maxLoad_le_iff _ _ _).2
  47   intro w
  48   cases w with
  49   | inl v =>
  50     have h := collapse_load I b a v
  51     have hmax := load_le_maxLoad I (transport b a) v
  52     omega
  53   | inr v =>
  54     have h := collapse_load I b a v
  55     have hmax := load_le_maxLoad I (transport b a) v
  56     omega
  57 
  58  theorem collapse_maxLoad_le (I : Instance V T) (b a : Orientation T) :
  59     maxLoad I (transport b a) ≤ 2 * maxLoad (splitInstance I b) a := by
  60   apply (maxLoad_le_iff _ _ _).2
  61   intro v
  62   rw [collapse_load]
  63   have hl := load_le_maxLoad (splitInstance I b) a (Sum.inl v)
  64   have hr := load_le_maxLoad (splitInstance I b) a (Sum.inr v)
  65   omega
  66 
  67  theorem split_opt_le (I : Instance V T) (b : Orientation T) :
  68     opt (splitInstance I b) ≤ opt I := by
  69   obtain ⟨a, ha⟩ := exists_exact_opt I
  70   calc
  71     _ ≤ maxLoad (splitInstance I b) (transport b a) := opt_le_maxLoad _ _
  72     _ ≤ maxLoad I (transport b (transport b a)) := split_maxLoad_le _ _ _
  73     _ = opt I := by rw [transport_involutive, ha]
  74 
  75  theorem opt_le_two_split_opt (I : Instance V T) (b : Orientation T) :
  76     opt I ≤ 2 * opt (splitInstance I b) := by
  77   obtain ⟨a, ha⟩ := exists_exact_opt (splitInstance I b)
  78   calc
  79     _ ≤ maxLoad I (transport b a) := opt_le_maxLoad _ _
  80     _ ≤ 2 * maxLoad (splitInstance I b) a := collapse_maxLoad_le _ _ _
  81     _ = _ := by rw [ha]
  82 
  83 /-- Pointwise competitive transfer; expectation preserves these inequalities. -/
  84  theorem competitive_transfer (I : Instance V T) (b a : Orientation T) (c : ℕ)
  85     (hc : maxLoad (splitInstance I b) a ≤ c * opt (splitInstance I b)) :
  86     maxLoad I (transport b a) ≤ (2*c) * opt I := by
  87   calc
  88     _ ≤ 2 * maxLoad (splitInstance I b) a := collapse_maxLoad_le _ _ _
  89     _ ≤ 2 * (c * opt (splitInstance I b)) := Nat.mul_le_mul_left _ hc
  90     _ ≤ 2 * (c * opt I) := Nat.mul_le_mul_left _ (Nat.mul_le_mul_left c (split_opt_le I b))
  91     _ = _ := by ring
  92 
  93 end GraphChoices
