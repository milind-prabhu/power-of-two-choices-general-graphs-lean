   9 theorem average_sub {Ω : Type*} [Fintype Ω] (f g : Ω → ℝ) :
  10     average (fun ω => f ω - g ω) = average f - average g := by
  11   simp [average, sum_sub_distrib, sub_div]
  12 
  13 theorem average_div {Ω : Type*} [Fintype Ω] (f : Ω → ℝ) (a : ℝ) :
  14     average (fun ω => f ω / a) = average f / a := by
  15   simp only [average, div_eq_mul_inv, ← sum_mul]
  16   ring

  18 noncomputable def successNumber {I Ω : Type*} [Fintype I] [DecidableEq Ω]
  19     (events : I → Finset Ω) (ω : Ω) : ℝ := ∑ i, if ω ∈ events i then 1 else 0

  21 theorem successNumber_nonneg {I Ω : Type*} [Fintype I] [DecidableEq Ω]
  22     (events : I → Finset Ω) (ω : Ω) : 0 ≤ successNumber events ω := by
  23   exact sum_nonneg fun _ _ => by split <;> norm_num

  25 theorem average_successNumber {I Ω : Type*} [Fintype I] [Fintype Ω] [DecidableEq Ω]
  26     (events : I → Finset Ω) :
  27     average (successNumber events) = ∑ i, eventProbability (events i) := by
  28   unfold successNumber
  29   rw [average_sum]
  30   simp_rw [average_unit_indicator]
  31 
  32 theorem average_indicator_product {Ω : Type*} [Fintype Ω] [DecidableEq Ω]
  33     (A B : Finset Ω) :
  34     average (fun ω => (if ω ∈ A then (1 : ℝ) else 0) *
  35       (if ω ∈ B then (1 : ℝ) else 0)) = eventProbability (A ∩ B) := by
  36   rw [← average_unit_indicator]
  37   congr 1
  38   funext ω
  39   by_cases hA : ω ∈ A <;> by_cases hB : ω ∈ B <;> simp [hA, hB]

  41 theorem successNumber_second_moment {I Ω : Type*} [Fintype I] [Fintype Ω]
  42     [DecidableEq Ω] (events : I → Finset Ω)
  43     (hcov : ∀ i j, i ≠ j → eventProbability (events i ∩ events j) ≤
  44       eventProbability (events i) * eventProbability (events j)) :
  45     average (fun ω => (successNumber events ω)^2) ≤
  46       (∑ i, eventProbability (events i))^2 + ∑ i, eventProbability (events i) := by
  47   classical
  48   have heq (ω : Ω) : (successNumber events ω)^2 =
  49       ∑ i, ∑ j, (if ω ∈ events i then (1 : ℝ) else 0) *
  50         (if ω ∈ events j then (1 : ℝ) else 0) := by
  51     simp only [successNumber, pow_two, sum_mul, mul_sum]
  52     rw [sum_comm]
  53   simp_rw [heq, average_sum, average_indicator_product]
  54   calc
  55     _ ≤ ∑ i, ∑ j, (eventProbability (events i) * eventProbability (events j) +
  56         if i = j then eventProbability (events i) else 0) := by
  57       apply sum_le_sum
  58       intro i _
  59       apply sum_le_sum
  60       intro j _
  61       by_cases hij : i = j
  62       · subst j
  63         simp only [inter_self, if_true]
  64         nlinarith [sq_nonneg (eventProbability (events i))]
  65       · simpa only [if_neg hij, add_zero] using hcov i j hij
  66     _ = _ := by simp [sum_add_distrib, ← mul_sum, ← sum_mul, pow_two]

  68 theorem density_quadratic_lower {z r f k a : ℝ}
  69     (hz : 0 ≤ z) (hr : 0 < r) (hk : 0 ≤ k) (ha : 0 < a)
  70     (hdensity : k*z ≤ (z+r)*f) :
  71     k*(2*z/a - (z^2+r*z)/a^2) ≤ f := by
  72   have hzr : 0 < z+r := by linarith
  73   have hf : k*z/(z+r) ≤ f := (div_le_iff₀ hzr).mpr (by simpa only [mul_comm f] using hdensity)
  74   apply le_trans _ hf
  75   apply (le_div_iff₀ hzr).mpr
  76   have ha2 : 0 < a^2 := sq_pos_of_pos ha
  77   apply (le_of_mul_le_mul_right (a := a^2) ?_ ha2)
  78   have heq : (k*(2*z/a - (z^2+r*z)/a^2)*(z+r)) * a^2 =
  79       k*(2*z*a - (z^2+r*z))*(z+r) := by field_simp <;> ring
  80   rw [heq]
  81   nlinarith [mul_nonneg (mul_nonneg hk hz) (sq_nonneg (a-(z+r)))]

  83 /-- If E[Z²] ≤ E[Z]²+E[Z] and E[Z] ≥ r ≥ 1, any sampled
  84 graph with density at least k Z/(Z+r) has expected optimum at least k/3. -/
  85 theorem expected_density_of_second_moment {Ω : Type*} [Fintype Ω] [Nonempty Ω]
  86     (Z f : Ω → ℝ) (k r : ℝ) (hz : ∀ ω, 0 ≤ Z ω) (hk : 0 ≤ k)
  87     (hr : 1 ≤ r) (hmean : r ≤ average Z)
  88     (hsecond : average (fun ω => (Z ω)^2) ≤ (average Z)^2 + average Z)
  89     (hdensity : ∀ ω, k*Z ω ≤ (Z ω+r)*f ω) :
  90     k/3 ≤ average f := by
  91   let m := average Z
  92   let a := m+r+1
  93   have hm : 1 ≤ m := hr.trans hmean
  94   have ha : 0 < a := by dsimp [a]; linarith
  95   have hminor := average_mono fun ω => density_quadratic_lower (hz ω)
  96     (by linarith : 0 < r) hk ha (hdensity ω)
  97   rw [average_mul, average_sub, average_div, average_mul, average_div,
  98     average_add, average_mul] at hminor
  99   change k*(2*m/a - (average (fun ω => (Z ω)^2)+r*m)/a^2) ≤ average f at hminor
 100   have hbound : k*m/a ≤ average f := by
 101     apply le_trans _ hminor
 102     have ha2 : 0 < a^2 := sq_pos_of_pos ha
 103     apply (le_of_mul_le_mul_right (a := a^2) ?_ ha2)
 104     have heq : k*m/a*a^2 = k*m*a := by field_simp <;> ring
 105     have heq2 : (k*(2*m/a - (average (fun ω => (Z ω)^2)+r*m)/a^2))*a^2 =
 106         k*(2*m*a - (average (fun ω => (Z ω)^2)+r*m)) := by field_simp <;> ring
 107     rw [heq, heq2]
 108     have hh : average (fun ω => (Z ω)^2) ≤ m^2+m := hsecond
 109     dsimp [a]
 110     nlinarith [mul_nonneg hk (sub_nonneg.mpr hh)]
 111   apply le_trans _ hbound
 112   apply (le_div_iff₀ ha).mpr
 113   have hma : a ≤ 3*m := by dsimp [a]; linarith
 114   nlinarith [mul_nonneg hk (sub_nonneg.mpr hma)]
