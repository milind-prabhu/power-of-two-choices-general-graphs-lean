  11 theorem graph_card_le_square (G : Graph L R) (N : ℕ)
  12     (hvertices : Fintype.card (L ⊕ R) ≤ N) : G.card ≤ N^2 := by
  13   have hsum : Fintype.card L + Fintype.card R ≤ N := by simpa only [Fintype.card_sum] using hvertices
  14   have hL : Fintype.card L ≤ N := by omega
  15   have hR : Fintype.card R ≤ N := by omega
  16   calc
  17     _ ≤ Fintype.card (L × R) := card_le_card (subset_univ _)
  18     _ = Fintype.card L*Fintype.card R := Fintype.card_prod _ _
  19     _ ≤ N*N := Nat.mul_le_mul hL hR
  20     _ = _ := by ring

  22 theorem average_le_two_referenceCap (G : Graph L R) [Nonempty G] (N : ℕ) (d : ℝ)
  23     (hN : 0 < N) (hvertices : Fintype.card (L ⊕ R) ≤ N)
  24     (haverage : 2*(G.card : ℝ) = (N:ℝ)*d) : d ≤ 2*(graphBase G).referenceCap := by
  25   have hNr : (0:ℝ) < N := by exact_mod_cast hN
  26   have hcard : (Fintype.card (L ⊕ R) : ℝ) ≤ N := by exact_mod_cast hvertices
  27   have he := (graphBase G).total_edges_le_card_mul_referenceCap
  28   rw [Fintype.card_coe] at he
  29   have hcap : (0:ℝ) ≤ (graphBase G).referenceCap := Nat.cast_nonneg _
  30   have hbound := mul_le_mul_of_nonneg_right hcard hcap
  31   apply (mul_le_mul_left hNr).mp
  32   nlinarith

  49 /-- A fully quantified preprocessed guarantee with explicit universal constant.
  50 No lower bound on log-skewness is assumed: it remains visibly additive to
  51 the expected optimum on the right-hand side. -/
  52 theorem PreparedDecomposition.expected_cost_density_skew (P : PreparedDecomposition G B N d)
  53     (hB : B = (graphBase G).referenceCap) (hvertices : Fintype.card (L ⊕ R) ≤ N)
  54     (haverage : 2*(G.card : ℝ) = (N:ℝ)*d) :
  55     expectedCost (graphBase G) (onlineOrientation (T := N) (P.policy N)) ≤
  56       1400*(iteratedLog N : ℝ)*(expectedOpt (graphBase G) N+graphLogSkewness G d (N:ℝ)) := by
  57   let q : ℝ := iteratedLog N
  58   let optE : ℝ := expectedOpt (graphBase G) N
  59   let σ : ℝ := graphLogSkewness G d (N:ℝ)
  60   have hq : 1 ≤ q := by
  61     change (1:ℝ) ≤ (iteratedLog N : ℝ)
  62     exact_mod_cast (show 1 ≤ iteratedLog N by have := iteratedLog_pos N; omega)
  63   have hopt : 1 ≤ optE := expectedOpt_ge_one _ _ P.vertex_count_pos
  64   have hs : 0 ≤ σ := graphLogSkewness_nonneg G d (N:ℝ)
  65   have hM := P.reference_density_bound hB
  66   have hbudget := P.budget_loglog (P.reference_cap_le_square hB hvertices) haverage
  67   have hbudget' : P.budget N ≤ 320*(4*optE+σ)*q := by
  68     apply hbudget.trans
  69     apply mul_le_mul_of_nonneg_right _ (by linarith)
  70     dsimp only [optE, σ]
  71     nlinarith
  72   have hcoeff : ((2*(Nat.log 2 (3*(polynomialCutoff N 5+1)^2)+1) : ℕ) : ℝ) ≤ 20*q := by
  73     change ((2*(Nat.log 2 (3*(polynomialCutoff N 5+1)^2)+1) : ℕ) : ℝ) ≤ 20*(iteratedLog N : ℝ)
  74     exact_mod_cast greedy_multiplier_le_iteratedLog N
  75   have hgreedy := mul_le_mul_of_nonneg_right hcoeff (show 0 ≤ optE by linarith)
  76   have hprod : 1 ≤ q*optE := by nlinarith
  77   have herror : 3*(N:ℝ)/(N:ℝ)^5 ≤ 3*q*optE := by
  78     apply (exceptional_error_le_three N P.vertex_count_pos).trans
  79     nlinarith
  80   have hcost := P.expected_cost_at_vertex_horizon 5 (by decide) hvertices haverage
  81   calc
  82     _ ≤ _ := hcost
  83     _ ≤ 320*(4*optE+σ)*q + (20*q)*optE + 3*q*optE :=
  84       add_le_add (add_le_add hbudget' hgreedy) herror
  85     _ ≤ _ := by nlinarith [mul_nonneg (show 0 ≤ q by linarith) hs,
  86       mul_nonneg (show 0 ≤ q by linarith) (show 0 ≤ optE by linarith)]

  88 /-- Explicitly conditional corollary: the log-skewness benchmark comparison
  89 is still an input here, rather than a postulated or claimed proved lemma. -/
  90 theorem PreparedDecomposition.competitive_of_skew_lower_bound (P : PreparedDecomposition G B N d)
  91     (hB : B = (graphBase G).referenceCap) (hvertices : Fintype.card (L ⊕ R) ≤ N)
  92     (haverage : 2*(G.card : ℝ) = (N:ℝ)*d) (K : ℝ)
  93     (hskew : graphLogSkewness G d (N:ℝ) ≤ K*expectedOpt (graphBase G) N) :
  94     expectedCost (graphBase G) (onlineOrientation (T := N) (P.policy N)) ≤
  95       1400*(1+K)*(iteratedLog N : ℝ)*expectedOpt (graphBase G) N := by
  96   apply (P.expected_cost_density_skew hB hvertices haverage).trans
  97   calc
  98     _ ≤ 1400*(iteratedLog N : ℝ)*(expectedOpt (graphBase G) N+K*expectedOpt (graphBase G) N) :=
  99       mul_le_mul_of_nonneg_left (add_le_add_left hskew _) (by positivity)
 100     _ = _ := by ring
