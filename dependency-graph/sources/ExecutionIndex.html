   5 and represents actual threshold witness times as finite subsets of Fin T. -/

  10  theorem traceFrom_mem_spec (α : C → ℕ) (s : State L R C) (k : ℕ)
  11     (es : List (Arrival L R C)) (ev : Event L R C) (hev : ev ∈ traceFrom α s k es) :
  12     ∃ i : Fin es.length, ev.time = k+i.val ∧ ev.arrival = es.get i ∧
  13       ev.threshold = isThreshold α (runFrom α s (es.take i.val)) (es.get i) := by
  14   induction es generalizing s k with
  15   | nil => simp [traceFrom] at hev
  16   | cons e es ih =>
  17     rcases List.mem_cons.mp hev with rfl | hev
  18     · exact ⟨⟨0,by simp⟩,by simp, rfl, rfl⟩
  19     · obtain ⟨i,hti,hai,hthi⟩ := ih (step α s e) (k+1) hev
  20       refine ⟨i.succ,?_,?_,?_⟩
  21       · simpa [Nat.add_assoc, Nat.add_comm, Nat.add_left_comm] using hti
  22       · simpa using hai
  23       · simpa [List.take_succ_cons, runFrom] using hthi

  25  theorem trace_mem_arrival (α : C → ℕ) (es : List (Arrival L R C))
  26     (ev : Event L R C) (hev : ev ∈ trace α es) :
  27     ∃ i : Fin es.length, ev.time = i.val ∧ ev.arrival = es.get i := by
  28   obtain ⟨i,ht,ha,_⟩ := traceFrom_mem_spec α initial 0 es ev hev
  29   exact ⟨i,by simpa using ht,ha⟩

  31  theorem step_counter_mono (α : C → ℕ) (s : State L R C) (e : Arrival L R C) (u : L) (i : C) :
  32     s.threshold u i ≤ (step α s e).threshold u i := by
  33   unfold step
  34   split_ifs <;> simp
  35 
  36  theorem runFrom_counter_mono (α : C → ℕ) (s : State L R C)
  37     (es : List (Arrival L R C)) (u : L) (i : C) :
  38     s.threshold u i ≤ (runFrom α s es).threshold u i := by
  39   induction es generalizing s with
  40   | nil => exact le_rfl
  41   | cons e es ih => exact (step_counter_mono α s e u i).trans (ih (step α s e))
  42 
  43  theorem run_counter_mono_append (α : C → ℕ) (xs ys : List (Arrival L R C)) (u : L) (i : C) :
  44     (run α xs).threshold u i ≤ (run α (xs ++ ys)).threshold u i := by
  45   unfold run
  46   rw [runFrom_append]
  47   exact runFrom_counter_mono α _ ys u i

  49 /-- A residual occurrence saturates its bucket in the complete execution. -/
  50  theorem residual_event_final_saturated (α : C → ℕ) (es : List (Arrival L R C))
  51     (ev : Event L R C) (hev : ev ∈ trace α es) (hr : ev.threshold = false) :
  52     (run α es).threshold ev.arrival.left ev.arrival.cls = α ev.arrival.cls := by
  53   obtain ⟨i,ht,ha,hth⟩ := traceFrom_mem_spec α initial 0 es ev hev
  54   have hres : isThreshold α (run α (es.take i.val)) (es.get i) = false := by
  55     simpa only [run] using hth.symm.trans hr
  56   have hsat := residual_counter_saturated α (es.take i.val) (es.get i) hres
  57   have hmono := run_counter_mono_append α (es.take i.val) (es.drop i.val) (es.get i).left (es.get i).cls
  58   rw [List.take_append_drop, hsat] at hmono
  59   have hcap := threshold_cap α es (es.get i).left (es.get i).cls
  60   simpa [ha] using Nat.le_antisymm hcap hmono

  62 noncomputable def finiteThresholdTimes (α : C → ℕ) (es : List (Arrival L R C))
  63     (T : ℕ) (hlen : es.length = T) (u : L) (i : C) : Finset (Fin T) := by
  64   classical
  65   let K := (thresholdWitnessTimes α es u i).toFinset
  66   exact K.attach.image fun t => ⟨t.val, by
  67     have ht := thresholdWitnessTimes_earlier α es u i t.val (List.mem_toFinset.mp t.property)
  68     omega⟩
  69 
  70  theorem mem_finiteThresholdTimes (α : C → ℕ) (es : List (Arrival L R C))
  71     (T : ℕ) (hlen : es.length = T) (u : L) (i : C) (t : Fin T) :
  72     t ∈ finiteThresholdTimes α es T hlen u i ↔ t.val ∈ thresholdWitnessTimes α es u i := by
  73   classical
  74   unfold finiteThresholdTimes
  75   simp only [mem_image, mem_attach, true_and]
  76   constructor
  77   · rintro ⟨s,hs⟩
  78     have hv := congrArg Fin.val hs
  79     change s.val = t.val at hv
  80     rw [← hv]
  81     exact List.mem_toFinset.mp s.property
  82   · intro ht
  83     exact ⟨⟨t.val,List.mem_toFinset.mpr ht⟩,Fin.ext rfl⟩

  85  theorem finiteThresholdTimes_card (α : C → ℕ) (es : List (Arrival L R C))
  86     (T : ℕ) (hlen : es.length = T) (u : L) (i : C) :
  87     (finiteThresholdTimes α es T hlen u i).card = (run α es).threshold u i := by
  88   classical
  89   unfold finiteThresholdTimes
  90   rw [card_image_of_injective _ (by
  91     intro a b hab
  92     apply Subtype.ext
  93     exact congrArg Fin.val hab)]
  94   rw [card_attach, List.toFinset_card_of_nodup (thresholdWitnessTimes_nodup α es u i)]
  95   exact thresholdWitnessTimes_length α es u i

  97 /-- Time labels identify a unique event, even across buckets. -/
  98  theorem trace_event_eq_of_time (α : C → ℕ) (es : List (Arrival L R C))
  99     (ev ev' : Event L R C) (h : ev ∈ trace α es) (h' : ev' ∈ trace α es)
 100     (ht : ev.time = ev'.time) : ev = ev' :=
 101   List.inj_on_of_nodup_map (traceFrom_unique_times α initial 0 es) h h' ht

 103  theorem finiteThresholdTimes_buckets_disjoint (α : C → ℕ) (es : List (Arrival L R C))
 104     (T : ℕ) (hlen : es.length = T) (u u' : L) (i i' : C) (hne : (u,i) ≠ (u',i')) :
 105     Disjoint (finiteThresholdTimes α es T hlen u i) (finiteThresholdTimes α es T hlen u' i') := by
 106   apply Finset.disjoint_left.mpr
 107   intro t ht ht'
 108   obtain ⟨ev,hev,hetime,_,hu,hi⟩ := thresholdWitnessTimes_spec α es u i t.val
 109     ((mem_finiteThresholdTimes α es T hlen u i t).mp ht)
 110   obtain ⟨ev',hev',hetime',_,hu',hi'⟩ := thresholdWitnessTimes_spec α es u' i' t.val
 111     ((mem_finiteThresholdTimes α es T hlen u' i' t).mp ht')
 112   have heq := trace_event_eq_of_time α es ev ev' hev hev' (hetime.trans hetime'.symm)
 113   apply hne
 114   exact Prod.ext (hu.symm.trans ((congrArg (fun e : Event L R C => e.arrival.left) heq).trans hu'))
 115     (hi.symm.trans ((congrArg (fun e : Event L R C => e.arrival.cls) heq).trans hi'))
