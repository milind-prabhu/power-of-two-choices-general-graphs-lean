  12  theorem finite_half_geometric (h : ℕ) :
  13     ∑ i ∈ range h, ((1:ℝ)/2)^i = 2 * (1 - ((1:ℝ)/2)^h) := by
  14   induction h with
  15   | zero => simp
  16   | succ h ih => rw [sum_range_succ, ih, pow_succ]; ring
  17 
  18  theorem inverse_powers_sum_le_two (h : ℕ) :
  19     ∑ i ∈ range h, (1:ℝ)/2^i ≤ 2 := by
  20   simp_rw [← one_div_pow]
  21   rw [finite_half_geometric]
  22   have : (0:ℝ) ≤ (1/2)^h := by positivity
  23   linarith

  25 noncomputable def roundedThreshold (K L : ℝ) (i : ℕ) : ℕ :=
  26   ⌈K * (L / 2^i + 1)⌉₊

  28 /-- An explicit version of sum alpha_i = O(K (L+h)), including ceil errors. -/
  29  theorem rounded_threshold_budget (K L : ℝ) (h : ℕ) (hK : 0 ≤ K) (hL : 0 ≤ L) :
  30     (∑ i ∈ range h, (roundedThreshold K L i : ℝ)) ≤ K * (2*L+h) + h := by
  31   have hround (i : ℕ) : (roundedThreshold K L i : ℝ) ≤ K * (L / 2^i + 1) + 1 := by
  32     exact (Nat.ceil_lt_add_one (mul_nonneg hK (by positivity))).le
  33   calc
  34     _ ≤ ∑ i ∈ range h, (K * (L / 2^i + 1) + 1) := sum_le_sum fun i _ => hround i
  35     _ = K * (L * (∑ i ∈ range h, (1:ℝ)/2^i) + h) + h := by
  36       have heq (i : ℕ) : K * (L / 2^i + 1) + 1 = K*L*((1:ℝ)/2^i) + (K+1) := by ring
  37       simp_rw [heq]
  38       rw [sum_add_distrib, ← mul_sum]
  39       simp only [sum_const, card_range, nsmul_eq_mul]
  40       ring
  41     _ ≤ K * (2*L+h) + h := by
  42       have hh := mul_le_mul_of_nonneg_left (inverse_powers_sum_le_two h) hL
  43       nlinarith
