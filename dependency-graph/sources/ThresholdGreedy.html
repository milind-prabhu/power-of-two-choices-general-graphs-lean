   3 /-! Executable Threshold-Greedy, with class counters for threshold edges and
   4 separate loads for the single residual Greedy process (Algorithm 1).
   5 Counters start at zero. A counter that reaches its threshold never changes,
   6 so using threshold counters implements the threshold phase in the paper. -/
   7 namespace GraphChoices.ThresholdGreedy
   8 open Finset
   9 variable {L R C : Type*} [DecidableEq L] [DecidableEq R] [DecidableEq C]
  10 
  11 structure Arrival (L R C : Type*) where
  12   left : L
  13   right : R
  14   cls : C
  15 
  16 structure State (L R C : Type*) where
  17   threshold : L → C → ℕ
  18   greedyLeft : L → ℕ
  19   greedyRight : R → ℕ
  20 
  21 def initial : State L R C := ⟨fun _ _ => 0, fun _ => 0, fun _ => 0⟩
  22 
  23 def isThreshold (α : C → ℕ) (s : State L R C) (e : Arrival L R C) : Bool :=
  24   decide (s.threshold e.left e.cls < α e.cls)
  25 
  26 def step (α : C → ℕ) (s : State L R C) (e : Arrival L R C) : State L R C :=
  27   if s.threshold e.left e.cls < α e.cls then
  28     { s with threshold := fun u i => s.threshold u i + if u = e.left ∧ i = e.cls then 1 else 0 }
  29   else if s.greedyLeft e.left < s.greedyRight e.right then
  30     { s with greedyLeft := fun u => s.greedyLeft u + if u = e.left then 1 else 0 }
  31   else
  32     { s with greedyRight := fun v => s.greedyRight v + if v = e.right then 1 else 0 }
  33 
  34 def runFrom (α : C → ℕ) : State L R C → List (Arrival L R C) → State L R C
  35   | s, [] => s
  36   | s, e :: es => runFrom α (step α s e) es
  37 
  38 def run (α : C → ℕ) (es : List (Arrival L R C)) : State L R C :=
  39   runFrom α initial es
  40 
  41  theorem runFrom_append (α : C → ℕ) (s : State L R C) (xs ys : List (Arrival L R C)) :
  42     runFrom α s (xs ++ ys) = runFrom α (runFrom α s xs) ys := by
  43   induction xs generalizing s with
  44   | nil => rfl
  45   | cons e es ih => exact ih (step α s e)
  46 
  47  theorem run_append_singleton (α : C → ℕ) (xs : List (Arrival L R C)) (e : Arrival L R C) :
  48     run α (xs ++ [e]) = step α (run α xs) e := by
  49   exact runFrom_append α initial xs [e]
  50 
  51  theorem step_preserves_caps (α : C → ℕ) (s : State L R C) (e : Arrival L R C)
  52     (hs : ∀ u i, s.threshold u i ≤ α i) :
  53     ∀ u i, (step α s e).threshold u i ≤ α i := by
  54   intro u i
  55   unfold step
  56   split_ifs with ht hg
  57   · dsimp
  58     by_cases h : u = e.left ∧ i = e.cls
  59     · rcases h with ⟨rfl, rfl⟩
  60       simp
  61       omega
  62     · simpa [h] using hs u i
  63   · exact hs u i
  64   · exact hs u i
  65 
  66  theorem runFrom_preserves_caps (α : C → ℕ) (s : State L R C)
  67     (es : List (Arrival L R C)) (hs : ∀ u i, s.threshold u i ≤ α i) :
  68     ∀ u i, (runFrom α s es).threshold u i ≤ α i := by
  69   induction es generalizing s with
  70   | nil => exact hs
  71   | cons e es ih => exact ih _ (step_preserves_caps α s e hs)
  72 
  73  theorem threshold_cap (α : C → ℕ) (es : List (Arrival L R C)) (u : L) (i : C) :
  74     (run α es).threshold u i ≤ α i :=
  75   runFrom_preserves_caps α initial es (fun _ _ => Nat.zero_le _) u i
  76 
  77  theorem threshold_budget [Fintype C] (α : C → ℕ) (es : List (Arrival L R C)) (u : L) :
  78     ∑ i, (run α es).threshold u i ≤ ∑ i, α i :=
  79   sum_le_sum fun i _ => threshold_cap α es u i
  80 
  81 /-- A residual edge can occur only once its class counter is saturated. -/
  82  theorem residual_counter_saturated (α : C → ℕ) (es : List (Arrival L R C))
  83     (e : Arrival L R C) (he : isThreshold α (run α es) e = false) :
  84     (run α es).threshold e.left e.cls = α e.cls := by
  85   have hlo : α e.cls ≤ (run α es).threshold e.left e.cls := by
  86     simpa [isThreshold] using he
  87   exact Nat.le_antisymm (threshold_cap α es _ _) hlo
  88 
  89 /-- The number of previous arrivals in a left-vertex/class bucket. -/
  90 def incidentCount (es : List (Arrival L R C)) (u : L) (i : C) : ℕ :=
  91   (es.filter fun e => decide (e.left = u ∧ e.cls = i)).length
  92 
  93  theorem step_counter_le (α : C → ℕ) (s : State L R C) (e : Arrival L R C) (u : L) (i : C) :
  94     (step α s e).threshold u i ≤ s.threshold u i + if e.left = u ∧ e.cls = i then 1 else 0 := by
  95   unfold step
  96   split_ifs <;> simp_all [eq_comm]
  97 
  98  theorem counter_le_incident (α : C → ℕ) (es : List (Arrival L R C)) (u : L) (i : C) :
  99     (run α es).threshold u i ≤ incidentCount es u i := by
 100   induction es using List.reverseRecOn with
 101   | nil => simp [run, runFrom, initial, incidentCount]
 102   | append_singleton es e ih =>
 103     rw [run_append_singleton]
 104     have hstep := step_counter_le α (run α es) e u i
 105     have hc : incidentCount (es ++ [e]) u i =
 106         incidentCount es u i + if e.left = u ∧ e.cls = i then 1 else 0 := by
 107       by_cases he : e.left = u ∧ e.cls = i <;> simp [incidentCount, List.filter_append, he]
 108     rw [hc]
 109     omega
 110 
 111 /-- The necessary incidence condition behind the pattern probability bound. -/
 112  theorem residual_requires_previous_arrivals (α : C → ℕ) (es : List (Arrival L R C))
 113     (e : Arrival L R C) (he : isThreshold α (run α es) e = false) :
 114     α e.cls ≤ incidentCount es e.left e.cls := by
 115   rw [← residual_counter_saturated α es e he]
 116   exact counter_le_incident α es _ _
 117 
 118 /-- On a residual arrival the counter is unchanged and the selected endpoint
 119 has minimum residual load. Ties go right, as in Algorithm 1. -/
 120  theorem residual_step (α : C → ℕ) (s : State L R C) (e : Arrival L R C)
 121     (he : isThreshold α s e = false) :
 122     (step α s e).threshold = s.threshold ∧
 123     ((s.greedyLeft e.left < s.greedyRight e.right ∧
 124         (step α s e).greedyLeft e.left = s.greedyLeft e.left + 1 ∧
 125         (step α s e).greedyRight = s.greedyRight) ∨
 126       (s.greedyRight e.right ≤ s.greedyLeft e.left ∧
 127         (step α s e).greedyRight e.right = s.greedyRight e.right + 1 ∧
 128         (step α s e).greedyLeft = s.greedyLeft)) := by
 129   have ht : ¬s.threshold e.left e.cls < α e.cls := by simpa [isThreshold] using he
 130   unfold step
 131   rw [if_neg ht]
 132   split_ifs with hg
 133   · simp [hg]
 134   · simp [hg, Nat.le_of_not_gt hg]
 135 
 136 end GraphChoices.ThresholdGreedy
