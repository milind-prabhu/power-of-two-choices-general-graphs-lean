   4 /-! Union over tree sizes as well as shapes. This avoids assuming that a
   5 large left-heavy tree contains a left-heavy subtree of an exact chosen size.
   6 Coverage by the canonical encoding is still a separate deterministic step. -/
   7 namespace GraphChoices.Patterns
   8 open Finset
   9 variable {E L R C : Type*} [Fintype E] [DecidableEq E] [Nonempty E]
  10   [Fintype L] [Fintype R] [Fintype C]
  11   [DecidableEq L] [DecidableEq R] [DecidableEq C]
  12 
  13 noncomputable def largeShapeEvent (H : C → Peeling.Graph L R)
  14     (encode : E → ThresholdGreedy.Arrival L R C) (α : C → ℕ) (T k B : ℕ) : Finset (Fin T → E) :=
  15   (range (B+1)).biUnion fun j => heavyShapeEvent H encode α T (k+j)
  16 
  17 theorem largeShape_probability_le (H : C → Peeling.Graph L R)
  18     (encode : E → ThresholdGreedy.Arrival L R C) (α : C → ℕ) (T k B : ℕ)
  19     (htail : ∀ j ≤ B, eventProbability (heavyShapeEvent H encode α T (k+j)) ≤
  20       (Fintype.card R : ℝ)/2^(k+j)) :
  21     eventProbability (largeShapeEvent H encode α T k B) ≤ 2*(Fintype.card R : ℝ)/2^k := by
  22   calc
  23     _ ≤ ∑ j ∈ range (B+1), eventProbability (heavyShapeEvent H encode α T (k+j)) :=
  24       witness_union_bound _ _
  25     _ ≤ ∑ j ∈ range (B+1), (Fintype.card R : ℝ)/2^(k+j) := by
  26       apply sum_le_sum
  27       intro j hj
  28       exact htail j (by have := mem_range.mp hj; omega)
  29     _ = ((Fintype.card R : ℝ)/2^k) * (∑ j ∈ range (B+1), (1:ℝ)/2^j) := by
  30       rw [mul_sum]
  31       apply sum_congr rfl
  32       intro j _
  33       rw [pow_add]
  34       ring
  35     _ ≤ ((Fintype.card R : ℝ)/2^k) * 2 :=
  36       mul_le_mul_of_nonneg_left (ThresholdGreedy.inverse_powers_sum_le_two _) (by positivity)
  37     _ = _ := by ring
  38 
  39 end GraphChoices.Patterns
