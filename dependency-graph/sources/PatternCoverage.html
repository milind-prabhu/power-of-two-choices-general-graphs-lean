  12 theorem menuValid_of_pointwise {V : Type*} [Fintype V] [DecidableEq V]
  13     (menu : (k : ℕ) → (Fin k → V) → Finset V) (n : ℕ) (f : Fin n → V)
  14     (h : ∀ i : Fin n, f i ∈ menu i.val (fun j => f ⟨j.val, by omega⟩)) :
  15     MenuValid menu n f := by
  16   induction n with
  17   | zero => trivial
  18   | succ n ih =>
  19     constructor
  20     · apply ih (Fin.init f)
  21       intro i
  22       exact h i.castSucc
  23     · exact h (Fin.last n)
  24 
  25 end GraphChoices

  34 theorem edgeData_labels_injective (p : ParentPattern C m) (encode : E → Arrival L R C)
  35     (f : Fin (m+1) → L ⊕ R) (g : Fin m → E)
  36     (hf : Function.Injective f) (he : EdgeData p encode f g) : Function.Injective g := by
  37   intro i j hij
  38   have hi := (he i).2
  39   have hj := (he j).2
  40   rw [hij] at hi
  41   have hpi := (p.parent i).isLt
  42   have hpj := (p.parent j).isLt
  43   cases hsi : p.childLeft i <;> cases hsj : p.childLeft j <;>
  44     simp only [hsi, hsj, Bool.false_eq_true, if_false, if_true] at hi hj
  45   · have hidx := congrArg Fin.val (hf (hi.2.trans hj.2.symm))
  46     apply Fin.ext
  47     simpa only [childIndex] using Nat.add_right_cancel hidx
  48   · have hidx1 := congrArg Fin.val (hf (hi.1.trans hj.1.symm))
  49     have hidx2 := congrArg Fin.val (hf (hi.2.trans hj.2.symm))
  50     simp only [childIndex, parentIndex] at hidx1 hidx2
  51     omega
  52   · have hidx1 := congrArg Fin.val (hf (hi.1.trans hj.1.symm))
  53     have hidx2 := congrArg Fin.val (hf (hi.2.trans hj.2.symm))
  54     simp only [childIndex, parentIndex] at hidx1 hidx2
  55     omega
  56   · have hidx := congrArg Fin.val (hf (hi.1.trans hj.1.symm))
  57     apply Fin.ext
  58     simpa only [childIndex] using Nat.add_right_cancel hidx

  60 theorem edgeData_mem_assignments (p : ParentPattern C m) (H : C → Peeling.Graph L R)
  61     (encode : E → Arrival L R C)
  62     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
  63     (f : Fin (m+1) → L ⊕ R) (g : Fin m → E)
  64     (hroot : ∃ v, f ⟨0, by omega⟩ = Sum.inr v) (he : EdgeData p encode f g) :
  65     f ∈ patternAssignments p H := by
  66   classical
  67   apply (mem_menuAssignments_iff _ _ _).mpr
  68   apply menuValid_of_pointwise
  69   intro k
  70   cases k with
  71   | mk k hk =>
  72     cases k with
  73     | zero =>
  74       obtain ⟨v, hv⟩ := hroot
  75       simp only [patternMenu, hv]
  76       exact mem_image.mpr ⟨v, mem_univ _, rfl⟩
  77     | succ k =>
  78       have hkm : k < m := by omega
  79       let i : Fin m := ⟨k, hkm⟩
  80       have hs : ((encode (g i)).left, (encode (g i)).right) ∈ H (p.cls i) := by
  81         rw [← (he i).1]
  82         exact hsource _
  83       have hed := (he i).2
  84       cases hi : p.childLeft i
  85       · simp only [hi, Bool.false_eq_true, if_false] at hed
  86         change f (childIndex i) ∈ patternMenu p H (k+1) _
  87         simp only [patternMenu, dif_pos hkm]
  88         change f (childIndex i) ∈ if p.childLeft i then _ else _
  89         rw [hi]
  90         simp only [Bool.false_eq_true, if_false]
  91         change f (childIndex i) ∈ match f (parentIndex p i) with
  92           | Sum.inl u => (Peeling.leftNeighbors (H (p.cls i)) u).image Sum.inr
  93           | Sum.inr _ => ∅
  94         rw [hed.1, hed.2]
  95         exact mem_image.mpr ⟨_, by simpa [Peeling.leftNeighbors] using hs, rfl⟩
  96       · simp only [hi, if_true] at hed
  97         change f (childIndex i) ∈ patternMenu p H (k+1) _
  98         simp only [patternMenu, dif_pos hkm]
  99         change f (childIndex i) ∈ if p.childLeft i then _ else _
 100         rw [hi]
 101         simp only [if_true]
 102         change f (childIndex i) ∈ match f (parentIndex p i) with
 103           | Sum.inl _ => ∅
 104           | Sum.inr v => (Peeling.rightNeighbors (H (p.cls i)) v).image Sum.inl
 105         rw [hed.2, hed.1]
 106         exact mem_image.mpr ⟨_, by simpa [Peeling.rightNeighbors] using hs, rfl⟩

 108 /-- Geometric vertex data and actual residual occurrences imply membership
 109 in the exact pattern event whose probability was already bounded. -/
 110 theorem mem_residualPatternEvent_of_data (p : ParentPattern C m) (H : C → Peeling.Graph L R)
 111     (encode : E → Arrival L R C) (α : C → ℕ)
 112     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
 113     (ω : Fin T → E) (f : Fin (m+1) → L ⊕ R) (g : Fin m → E)
 114     (hf : Function.Injective f) (hroot : ∃ v, f ⟨0, by omega⟩ = Sum.inr v)
 115     (he : EdgeData p encode f g)
 116     (hres : ∀ i, ∃ t : Fin T, ω t = g i ∧ residualTime α encode ω t) :
 117     ω ∈ residualPatternEvent p H encode α T := by
 118   classical
 119   apply mem_biUnion.mpr
 120   refine ⟨(f,g), mem_filter.mpr ⟨mem_univ _, ?_, hf, he⟩, ?_⟩
 121   · exact edgeData_mem_assignments p H encode hsource f g hroot he
 122   · exact (mem_fixedResidualEvent_iff α encode g (edgeData_labels_injective p encode f g hf he) ω).mpr hres
