   3 /-! Explicit logarithmic cutoffs, inverse-polynomial failure probabilities,

   8 def polynomialCutoff (N c : ℕ) : ℕ := (c+1)*(Nat.log 2 N+1)
   9 
  10 theorem polynomialCutoff_pos (N c : ℕ) : 0 < polynomialCutoff N c := by
  11   unfold polynomialCutoff
  12   positivity
  13 
  14 theorem polynomialCutoff_power (N c : ℕ) : N^(c+1) ≤ 2^polynomialCutoff N c := by
  15   have hN := (Nat.lt_pow_succ_log_self (by decide : 1 < 2) N).le
  16   calc
  17     _ ≤ (2^(Nat.log 2 N+1))^(c+1) := Nat.pow_le_pow_left hN _
  18     _ = _ := by rw [← pow_mul]; congr 1; unfold polynomialCutoff; ring

  20 theorem geometric_tail_le_polynomial (N c k A : ℕ) (hN : 0 < N) (hA : A ≤ N)
  21     (hk : polynomialCutoff N c ≤ k) (r : ℝ) (hr : 0 ≤ r) (hrsmall : r ≤ 1/2) :
  22     (A:ℝ)*r^k ≤ 1/(N:ℝ)^c := by
  23   have hNr : (0:ℝ) < N := by exact_mod_cast hN
  24   have hexp : (N:ℝ)^(c+1) ≤ (2:ℝ)^k := by
  25     exact_mod_cast (polynomialCutoff_power N c).trans (Nat.pow_le_pow_right (by decide : 0 < 2) hk)
  26   calc
  27     _ ≤ (A:ℝ)*((1:ℝ)/2)^k := mul_le_mul_of_nonneg_left (pow_le_pow_left₀ hr hrsmall k) (Nat.cast_nonneg _)
  28     _ = (A:ℝ)/(2:ℝ)^k := by rw [div_pow]; ring
  29     _ ≤ (N:ℝ)/(2:ℝ)^k := div_le_div_of_nonneg_right (by exact_mod_cast hA) (by positivity)
  30     _ ≤ (N:ℝ)/(N:ℝ)^(c+1) := div_le_div_of_nonneg_left hNr.le (by positivity) hexp
  31     _ = _ := by rw [pow_succ]; field_simp

  41 /-- Conditional component shattering with exact cutoff and failure constant.
  42 The source of the remaining coverage assumption is stated in the module header. -/
  43 theorem largeComponent_polynomial_tail_of_coverage (H : C → Peeling.Graph L R)
  44     (encode : E → Arrival L R C) (α : C → ℕ) (T D k B N c : ℕ)
  45     (hN : 0 < N) (hL : Fintype.card L ≤ N) (hR : Fintype.card R ≤ N)
  46     (hD : polynomialCutoff N c ≤ D+1) (hk : polynomialCutoff N c ≤ k)
  47     (hmean : ∀ u, Real.exp 1*T*(((leftSourceBucket encode u).card : ℝ)/Fintype.card E)/(D+1) ≤ 1/2)
  48     (hcoverage : largeHeavyTreeEvent encode α T k ⊆ largeShapeEvent H encode α T k B)
  49     (hshape : ∀ j ≤ B, eventProbability (heavyShapeEvent H encode α T (k+j)) ≤
  50       (Fintype.card R : ℝ)/2^(k+j)) :
  51     eventProbability (largeComponentEvent encode α T ((D+1)*(k+1))) ≤ 3/(N:ℝ)^c := by
  52   have h := largeComponent_tail_of_canonical_coverage H encode α T D k B (1/2) hmean hcoverage hshape
  53   have hd := geometric_tail_le_polynomial N c (D+1) (Fintype.card L) hN hL hD (1/2) (by norm_num) le_rfl
  54   have hr := geometric_tail_le_polynomial N c k (Fintype.card R) hN hR hk (1/2) (by norm_num) le_rfl
  55   have heq : (Fintype.card R : ℝ)*((1:ℝ)/2)^k = (Fintype.card R : ℝ)/2^k := by
  56     rw [div_pow]; ring
  57   rw [heq] at hr
  58   calc
  59     _ ≤ _ := h
  60     _ = (Fintype.card L : ℝ)*((1:ℝ)/2)^(D+1) + 2*((Fintype.card R : ℝ)/2^k) := by ring
  61     _ ≤ 1/(N:ℝ)^c + 2*(1/(N:ℝ)^c) := add_le_add hd (mul_le_mul_of_nonneg_left hr (by norm_num))
  62     _ = _ := by ring
