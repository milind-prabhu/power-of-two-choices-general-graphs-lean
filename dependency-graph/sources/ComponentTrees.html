   4 /-! Actual arrival components have spanning trees on exactly their vertex
   5 sets. Every selected tree edge is backed by an original arrival occurrence.
   6 Converting those trees to the counted canonical shape codes remains separate.
   7 -/
   8 namespace GraphChoices
   9 variable {V : Type*} [Fintype V] [DecidableEq V] {T : ℕ}
  10 
  11 omit [DecidableEq V] in
  12 theorem componentVertices_coe_eq_supp (I : Instance V T) (v : V) :
  13     (componentVertices I v : Set V) = ((arrivalGraph I).connectedComponentMk v).supp := by
  14   ext w
  15   simp only [Finset.mem_coe, mem_componentVertices,
  16     SimpleGraph.ConnectedComponent.mem_supp_iff, SimpleGraph.ConnectedComponent.eq]
  17   exact SimpleGraph.reachable_comm
  18 
  19 theorem component_induce_connected (I : Instance V T) (v : V) :
  20     ((arrivalGraph I).induce (componentVertices I v : Set V)).Connected := by
  21   rw [componentVertices_coe_eq_supp]
  22   exact SimpleGraph.ConnectedComponent.connected_toSimpleGraph _
  23 
  24 /-- Spanning-tree extraction from an actual component, including its exact
  25 vertex count. Edges inherit original occurrence times, with no sampling premise. -/
  26 theorem exists_component_spanning_tree (I : Instance V T) (v : V) :
  27     ∃ H : SimpleGraph {w : V // w ∈ componentVertices I v},
  28       H.IsTree ∧
  29       (∀ x y, H.Adj x y → ∃ t : Fin T,
  30         (I.left t = x.val ∧ I.right t = y.val) ∨
  31         (I.left t = y.val ∧ I.right t = x.val)) ∧
  32       Fintype.card {w : V // w ∈ componentVertices I v} = (componentVertices I v).card := by
  33   classical
  34   obtain ⟨H, hle, htree⟩ := (component_induce_connected I v).exists_isTree_le
  35   refine ⟨H, htree, ?_, Fintype.card_coe _⟩
  36   intro x y hxy
  37   exact (hle hxy).2
  38 
  39 end GraphChoices
