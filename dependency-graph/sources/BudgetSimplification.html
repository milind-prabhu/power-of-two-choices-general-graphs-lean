   9 theorem log_reference_branch_bound (B h : ℕ) (z : ℝ)
  10     (hB : 0 < B) (hh : 0 < h) (hz : 0 < z) :
  11     Real.log (16*(B:ℝ)*z/branchTarget (4*B) h z) ≤ 10+5*((B:ℝ)*z)+2*h := by
  12   have hBr : (0:ℝ) < B := by exact_mod_cast hB
  13   have hhr : (0:ℝ) < h := by exact_mod_cast hh
  14   have hne : (h:ℝ) ≠ 0 := hhr.ne'
  15   have hden : (4*(B:ℝ)*z+1) ≠ 0 := ne_of_gt (by positivity)
  16   have heq : 16*(B:ℝ)*z/branchTarget (4*B) h z =
  17       1024*((B:ℝ)*z)*(4*((B:ℝ)*z)+1)*(h:ℝ)^2 := by
  18     unfold branchTarget shapeRate
  19     push_cast
  20     field_simp
  21     ring
  22   rw [heq]
  23   have hM : 0 < (B:ℝ)*z := mul_pos hBr hz
  24   have hlog : Real.log (1024*((B:ℝ)*z)*(4*((B:ℝ)*z)+1)*(h:ℝ)^2) =
  25       Real.log 1024 + Real.log ((B:ℝ)*z) + Real.log (4*((B:ℝ)*z)+1) + 2*Real.log h := by
  26     rw [Real.log_mul (by positivity) (by positivity), Real.log_pow,
  27       Real.log_mul (by positivity) (by positivity), Real.log_mul (by norm_num) hM.ne']
  28     norm_num
  29   rw [hlog]
  30   have hconst : Real.log 1024 ≤ 10 := by
  31     have he : (1024:ℝ) = 2^10 := by norm_num
  32     rw [he, Real.log_pow]
  33     norm_num only [Nat.cast_ofNat]
  34     linarith [Real.log_two_lt_d9]
  35   have hm := Real.log_le_sub_one_of_pos hM
  36   have hm' := Real.log_le_sub_one_of_pos (by positivity : (0:ℝ) < 4*((B:ℝ)*z)+1)
  37   have hhh := Real.log_le_sub_one_of_pos hhr
  38   linarith

  40 theorem scalar_budget_le (h M σ l K : ℝ) (hh : 1 ≤ h) (hM : 0 ≤ M) (hσ : 0 ≤ σ)
  41     (hl : 0 ≤ l) (hK : 0 ≤ K) (hKbound : K ≤ 10+5*M+2*h) :
  42     h*(16*Real.exp 1*M+2*σ+2)+(K+2*σ*l)/Real.log 2 ≤
  43       64*(M+σ+1)*(h+l+1) := by
  44   have he : Real.exp 1 ≤ 3 := by linarith [Real.exp_one_lt_d9]
  45   have hlog : (1:ℝ)/2 ≤ Real.log 2 := by linarith [Real.log_two_gt_d9]
  46   have hnum : 0 ≤ K+2*σ*l := by positivity
  47   have hdiv : (K+2*σ*l)/Real.log 2 ≤ 2*(K+2*σ*l) := by
  48     calc
  49       _ ≤ (K+2*σ*l)/((1:ℝ)/2) := div_le_div_of_nonneg_left hnum (by norm_num) hlog
  50       _ = _ := by ring
  51   have heM := mul_le_mul_of_nonneg_right he hM
  52   have hhead : h*(16*Real.exp 1*M+2*σ+2) ≤ h*(48*M+2*σ+2) := by
  53     apply mul_le_mul_of_nonneg_left _ (by linarith)
  54     nlinarith
  55   have hMh := mul_nonneg hM (show 0 ≤ h by linarith)
  56   have hσh := mul_nonneg hσ (show 0 ≤ h by linarith)
  57   have hMl := mul_nonneg hM hl
  58   have hσl := mul_nonneg hσ hl
  59   nlinarith

  68 theorem PreparedDecomposition.budget_simplified (P : PreparedDecomposition G B N d)
  69     (haverage : 2*(G.card : ℝ) = (N:ℝ)*d) :
  70     P.budget N ≤ 64*((B:ℝ)*((N:ℝ)/Fintype.card G)+graphLogSkewness G d (N:ℝ)+1)*
  71       (P.pieces.length+Real.log (Real.log (N:ℝ))+1) := by
  72   have hz : 0 < (N:ℝ)/Fintype.card G := div_pos
  73     (by exact_mod_cast P.vertex_count_pos) (by exact_mod_cast Fintype.card_pos)
  74   have hh := P.class_count_pos
  75   have hhr : (1:ℝ) ≤ P.pieces.length := by exact_mod_cast hh
  76   have hb := branchTarget_pos (4*B) P.pieces.length _ hh hz.le
  77   have hb1 := (branchTarget_le_rate (4*B) P.pieces.length _ hh hz.le).trans (shapeRate_le_one _ hh)
  78   have hmass := P.mass_at_vertex_horizon haverage
  79   have hK : 1 ≤ 16*(B:ℝ)*((N:ℝ)/Fintype.card G)/branchTarget (4*B) P.pieces.length ((N:ℝ)/Fintype.card G) := by
  80     apply (le_div_iff₀ hb).mpr
  81     simpa only [one_mul] using hb1.trans hmass
  82   have hl : 0 ≤ Real.log (Real.log (N:ℝ)) := Real.log_nonneg P.large_log.le
  83   have hbound := log_reference_branch_bound B P.pieces.length ((N:ℝ)/Fintype.card G) P.reference_cap_pos hh hz
  84   have hs := scalar_budget_le (P.pieces.length : ℝ) ((B:ℝ)*((N:ℝ)/Fintype.card G))
  85     (graphLogSkewness G d (N:ℝ)) (Real.log (Real.log (N:ℝ))) _ hhr (by positivity)
  86     (graphLogSkewness_nonneg G d (N:ℝ)) hl (Real.log_nonneg hK) hbound
  87   convert hs using 1 <;> unfold PreparedDecomposition.budget <;> ring
