  80 theorem family_parameter_bounds (b : ℕ) (hb : 16 ≤ b) :
  81     4 ≤ branching b ∧ 2 ≤ width b ∧ branching b ≤ width b ∧
  82       layerSum (branching b) b ≤ width b ∧ b ≤ size b ∧ size b ≤ (width b)^2 := by
  83   let t := branching b
  84   have ht : 4 ≤ t := by
  85     change 4 ≤ (16*b)^3
  86     exact (show 4 ≤ 16*b by omega).trans (Nat.le_pow (by decide : 0 < 3))
  87   have hbt : b ≤ t := by
  88     change b ≤ (16*b)^3
  89     exact (show b ≤ 16*b by omega).trans (Nat.le_pow (by decide : 0 < 3))
  90   have hpow : 1 ≤ t^b := Nat.one_le_pow _ _ (by omega)
  91   have hpowt : t ≤ t^b := Nat.le_pow (by omega)
  92   have hm : b ≤ width b := by change b ≤ 2*t^b; omega
  93   have htm : t ≤ width b := by change t ≤ 2*t^b; omega
  94   have hQ : layerSum t b ≤ width b := by
  95     have hh := layerSum_geometric_upper t b (by omega)
  96     have hh' : layerSum t b ≤ layerSum t (b+1) := by rw [layerSum_succ]; omega
  97     exact hh'.trans hh
  98   have hS := layerSum_geometric_upper t b (by omega)
  99   have hSpos : 1 ≤ layerSum t (b+1) := layerSum_pos t (b+1) (by omega)
 100   have hN : b ≤ size b := by unfold size; change b ≤ width b*layerSum t (b+1); nlinarith
 101   have hNupper : size b ≤ (width b)^2 := by
 102     have hmdef : width b = 2*t^b := rfl
 103     unfold size
 104     change width b*layerSum t (b+1) ≤ (width b)^2
 105     nlinarith
 106   exact ⟨ht,by change 2 ≤ 2*t^b; omega,htm,hQ,hN,hNupper⟩

 117 theorem family_rate_bounds (b : ℕ) (hb : 16 ≤ b) :
 118     (Fintype.card (Edges b (branching b) (width b)):ℝ) ≤ (size b:ℝ)*width b ∧
 119       (size b:ℝ)*width b ≤ 2*(Fintype.card (Edges b (branching b) (width b)):ℝ) := by
 120   have hp := family_parameter_bounds b hb
 121   have hQ : 0 < layerSum (branching b) b := layerSum_pos _ _ (by omega)
 122   rw [size_eq_vertices,card_vertices,card_edges]
