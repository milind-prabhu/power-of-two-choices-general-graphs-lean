  11 noncomputable def extendFinZero {h : ℕ} (f : Fin h → ℝ) (i : ℕ) : ℝ :=
  12   if hi : i < h then f ⟨i,hi⟩ else 0

  14 theorem sufficientThreshold_fin_scale_budget (h : ℕ) (mean x : Fin h → ℝ) (A L σ K : ℝ)
  15     (hmean : ∀ i, 0 ≤ mean i) (hx : ∀ i, 0 ≤ x i)
  16     (hA : 0 ≤ A) (hL : 1 ≤ L) (hσ : 0 ≤ σ) (hK : 1 ≤ K)
  17     (hbase : ∀ i, (Peeling.scaleNat i.val : ℝ)*mean i ≤ A)
  18     (hbranch : ∀ i, x i ≤ K*((Peeling.scaleNat i.val : ℝ)*L)^(2*σ)) :
  19     (∑ i : Fin h, (sufficientThreshold (mean i) (Peeling.scaleNat i.val)
  20       (Real.log (x i)/Real.log (Peeling.scaleNat i.val)) : ℝ)) ≤
  21       h*(A+2*σ+2)+(Real.log K+2*σ*Real.log L)/Real.log 2 := by
  22   have ht := sufficientThreshold_scale_budget h (extendFinZero mean) (extendFinZero x) A L σ K
  23     (fun i hi => by simpa [extendFinZero, hi] using hmean ⟨i,hi⟩)
  24     (fun i hi => by simpa [extendFinZero, hi] using hx ⟨i,hi⟩) hA hL hσ hK
  25     (fun i hi => by simpa [extendFinZero, hi] using hbase ⟨i,hi⟩)
  26     (fun i hi => by simpa [extendFinZero, hi] using hbranch ⟨i,hi⟩)
  27   rw [← Fin.sum_univ_eq_sum_range] at ht
  28   simpa only [extendFinZero, Fin.isLt, dite_true] using ht

  32 /-- Budget for exactly the automatic thresholds. h is the actual finite
  33 class count, D is the actual common left cap, and z=T/|E|. -/
  34 theorem automaticThresholds_budget (h D T : ℕ) (DL DR : Fin h → ℕ) (B L σ : ℝ)
  35     (hh : 0 < h) (hB : 0 < B) (hL : 1 ≤ L) (hσ : 0 ≤ σ)
  36     (hmass : 1 ≤ 16*B*((T:ℝ)/Fintype.card E))
  37     (hDL : ∀ i, (DL i : ℝ) ≤ 16*B/(Peeling.scaleNat i.val : ℝ))
  38     (hDR : ∀ i, (DR i : ℝ) ≤ 16*B*((Peeling.scaleNat i.val : ℝ)*L)^(2*σ)) :
  39     (∑ i : Fin h, (automaticThresholds (E := E) DL DR D T (fun i => Peeling.scaleNat i.val) i : ℝ)) ≤
  40       h*(16*Real.exp 1*B*((T:ℝ)/Fintype.card E)+2*σ+2) +
  41       (Real.log (16*B*((T:ℝ)/Fintype.card E)/
  42         branchTarget D h ((T:ℝ)/Fintype.card E)) + 2*σ*Real.log L)/Real.log 2 := by
  43   have hz : 0 ≤ (T:ℝ)/Fintype.card E := by positivity
  44   have hb := branchTarget_pos D h _ hh hz
  45   have hb1 := (branchTarget_le_rate D h _ hh hz).trans (shapeRate_le_one h hh)
  46   have hK : 1 ≤ 16*B*((T:ℝ)/Fintype.card E)/branchTarget D h ((T:ℝ)/Fintype.card E) := by
  47     apply (le_div_iff₀ hb).mpr
  48     simpa only [one_mul] using hb1.trans hmass
  49   have ht := sufficientThreshold_fin_scale_budget h
  50     (fun i => Real.exp 1*T*(DL i : ℝ)/Fintype.card E)
  51     (fun i => (DR i : ℝ)*((T:ℝ)/Fintype.card E)/branchTarget D h ((T:ℝ)/Fintype.card E))
  52     (16*Real.exp 1*B*((T:ℝ)/Fintype.card E)) L σ _
  53     (fun _ => by positivity) (fun _ => by positivity) (by positivity) hL hσ hK ?_ ?_
  54   · simpa only [automaticThresholds, explicitThresholds, Fintype.card_fin] using ht
  55   · intro i
  56     have hpos : (0:ℝ) < Peeling.scaleNat i.val := by exact_mod_cast Peeling.scaleNat_pos i.val
  57     have hd := (le_div_iff₀ hpos).mp (hDL i)
  58     calc
  59       _ = (Real.exp 1*((T:ℝ)/Fintype.card E))*((DL i : ℝ)*(Peeling.scaleNat i.val : ℝ)) := by ring
  60       _ ≤ (Real.exp 1*((T:ℝ)/Fintype.card E))*(16*B) := mul_le_mul_of_nonneg_left hd (by positivity)
  61       _ = _ := by ring
  62   · intro i
  63     calc
  64       _ = (DR i : ℝ)*(((T:ℝ)/Fintype.card E)/branchTarget D h ((T:ℝ)/Fintype.card E)) := by ring
  65       _ ≤ (16*B*((Peeling.scaleNat i.val : ℝ)*L)^(2*σ))*
  66           (((T:ℝ)/Fintype.card E)/branchTarget D h ((T:ℝ)/Fintype.card E)) :=
  67         mul_le_mul_of_nonneg_right (hDR i) (by positivity)
  68       _ = _ := by ring
