  12 theorem mem_unionPieces_iff (Ps : List (Graph L R)) (e : L × R) :
  13     e ∈ unionPieces Ps ↔ ∃ P ∈ Ps, e ∈ P := by
  14   induction Ps with
  15   | nil => simp [unionPieces]
  16   | cons P Ps ih => simp [unionPieces, ih, or_and_right, exists_or]
  17 
  18 theorem exists_piece_index (G : Graph L R) (Ps : List (Graph L R))
  19     (hcover : unionPieces Ps = G) (e : G) : ∃ i : Fin Ps.length, e.val ∈ Ps.get i := by
  20   have he : e.val ∈ unionPieces Ps := by rw [hcover]; exact e.property
  21   obtain ⟨P, hP, heP⟩ := (mem_unionPieces_iff Ps e.val).mp he
  22   obtain ⟨i, hi⟩ := List.mem_iff_get.mp hP
  23   exact ⟨i, by rw [hi]; exact heP⟩

  25 noncomputable def pieceClass (G : Graph L R) (Ps : List (Graph L R))
  26     (hcover : unionPieces Ps = G) (e : G) : Fin Ps.length :=
  27   Classical.choose (exists_piece_index G Ps hcover e)
  28 
  29 theorem pieceClass_mem (G : Graph L R) (Ps : List (Graph L R))
  30     (hcover : unionPieces Ps = G) (e : G) : e.val ∈ Ps.get (pieceClass G Ps hcover e) :=
  31   Classical.choose_spec (exists_piece_index G Ps hcover e)

  33 noncomputable def pieceEncode (G : Graph L R) (Ps : List (Graph L R))
  34     (hcover : unionPieces Ps = G) (e : G) : Arrival L R (Fin Ps.length) :=
  35   ⟨e.val.1, e.val.2, pieceClass G Ps hcover e⟩
  36 
  37 theorem pieceEncode_simple (G : Graph L R) (Ps : List (Graph L R)) (hcover : unionPieces Ps = G) :
  38     Function.Injective fun e => ((pieceEncode G Ps hcover e).left, (pieceEncode G Ps hcover e).right) := by
  39   intro e f h
  40   exact Subtype.ext h

  42 theorem pieceEncode_source (G : Graph L R) (Ps : List (Graph L R)) (hcover : unionPieces Ps = G) :
  43     ∀ e, ((pieceEncode G Ps hcover e).left, (pieceEncode G Ps hcover e).right) ∈
  44       Ps.get (pieceEncode G Ps hcover e).cls := pieceClass_mem G Ps hcover

  46 theorem piece_sourceBucket_card_le (G : Graph L R) (Ps : List (Graph L R))
  47     (hcover : unionPieces Ps = G) (u : L) (i : Fin Ps.length) :
  48     (sourceBucket (pieceEncode G Ps hcover) u i).card ≤ leftDegree (Ps.get i) u := by
  49   classical
  50   apply card_le_card_of_injOn (fun e : G => e.val.2)
  51   · intro e he
  52     obtain ⟨hu, hc⟩ := (mem_filter.mp he).2
  53     have hsrc := pieceEncode_source G Ps hcover e
  54     rw [hc] at hsrc
  55     change e.val.1 = u at hu
  56     change (e.val.1, e.val.2) ∈ Ps.get i at hsrc
  57     change e.val.2 ∈ leftNeighbors (Ps.get i) u
  58     simp only [leftNeighbors, mem_filter, mem_univ, true_and]
  59     rw [← hu]
  60     exact hsrc
  61   · intro e he f hf hright
  62     have heleft := (mem_filter.mp he).2.1
  63     have hfleft := (mem_filter.mp hf).2.1
  64     apply Subtype.ext
  65     exact Prod.ext (heleft.trans hfleft.symm) hright
  66 
  67 theorem piece_leftSourceBucket_card_le (G : Graph L R) (Ps : List (Graph L R))
  68     (hcover : unionPieces Ps = G) (u : L) :
  69     (leftSourceBucket (pieceEncode G Ps hcover) u).card ≤ leftDegree G u := by
  70   classical
  71   apply card_le_card_of_injOn (fun e : G => e.val.2)
  72   · intro e he
  73     have hu := (mem_filter.mp he).2
  74     change e.val.1 = u at hu
  75     change e.val.2 ∈ leftNeighbors G u
  76     simp only [leftNeighbors, mem_filter, mem_univ, true_and]
  77     rw [← hu]
  78     exact e.property
  79   · intro e he f hf hright
  80     apply Subtype.ext
  81     exact Prod.ext ((mem_filter.mp he).2.trans (mem_filter.mp hf).2.symm) hright

  83 def maxLeftDegree (G : Graph L R) : ℕ := univ.sup (leftDegree G)
  84 
  85 def maxRightDegree (G : Graph L R) : ℕ := univ.sup (rightDegree G)

  87 theorem leftDegree_le_max (G : Graph L R) (u : L) : leftDegree G u ≤ maxLeftDegree G :=
  88   Finset.le_sup (mem_univ u)
  89 
  90 theorem rightDegree_le_max (G : Graph L R) (v : R) : rightDegree G v ≤ maxRightDegree G :=
  91   Finset.le_sup (mem_univ v)

  93 theorem maxLeftDegree_le (G : Graph L R) (D : ℕ) (h : ∀ u, leftDegree G u ≤ D) :
  94     maxLeftDegree G ≤ D := Finset.sup_le (fun u _ => h u)

  96 theorem nat_sup_cast_le {U : Type*} (s : Finset U) (f : U → ℕ) (K : ℝ)
  97     (hK : 0 ≤ K) (h : ∀ u ∈ s, (f u : ℝ) ≤ K) : ((s.sup f : ℕ) : ℝ) ≤ K := by
  98   classical
  99   induction s using Finset.induction_on with
 100   | empty => simpa using hK
 101   | @insert u s hu ih =>
 102     rw [Finset.sup_insert]
 103     change ((max (f u) (s.sup f) : ℕ) : ℝ) ≤ K
 104     rw [Nat.cast_max]
 105     exact max_le (h u (mem_insert_self _ _)) (ih (fun v hv => h v (mem_insert_of_mem hv)))
 106 
 107 theorem maxLeftDegree_real_le (G : Graph L R) (K : ℝ)
 108     (hK : 0 ≤ K) (h : ∀ u, (leftDegree G u : ℝ) ≤ K) : (maxLeftDegree G : ℝ) ≤ K :=
 109   nat_sup_cast_le _ _ _ hK (fun u _ => h u)

 111 theorem maxRightDegree_real_le (G : Graph L R) (K : ℝ)
 112     (hK : 0 ≤ K) (h : ∀ v, (rightDegree G v : ℝ) ≤ K) : (maxRightDegree G : ℝ) ≤ K :=
 113   nat_sup_cast_le _ _ _ hK (fun v _ => h v)

 115 theorem piece_maxLeftDegree_le (G : Graph L R) (Ps : List (Graph L R))
 116     (hcover : unionPieces Ps = G) (i : Fin Ps.length) :
 117     maxLeftDegree (Ps.get i) ≤ maxLeftDegree G := by
 118   apply maxLeftDegree_le
 119   intro u
 120   have hsub : Ps.get i ⊆ G := by
 121     rw [← hcover]
 122     exact mem_subset_unionPieces (List.get_mem _ _)
 123   exact (leftDegree_mono hsub u).trans (leftDegree_le_max G u)
