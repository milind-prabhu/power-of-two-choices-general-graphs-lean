   3 /-! Changing edge-label names preserves both sampling and nonanticipating
   4 policies. A once-per-base-edge instance identifies the rounded density cap. -/
   5 namespace GraphChoices
   6 open Finset
   7 variable {V E F : Type*} [Fintype V] [DecidableEq V]
   8   [Fintype E] [Fintype F] [DecidableEq E] [DecidableEq F] [Nonempty E] [Nonempty F]

  10 theorem relabel_sample (G : BaseGraph V E) (H : BaseGraph V F) (e : E ≃ F)
  11     (hl : ∀ x, H.left (e x) = G.left x) (hr : ∀ x, H.right (e x) = G.right x)
  12     {T : ℕ} (ω : Fin T → E) : H.sample (fun t => e (ω t)) = G.sample ω := by
  13   apply Instance.ext <;> funext t
  14   · exact hl (ω t)
  15   · exact hr (ω t)
  16 
  17 theorem relabel_expectedOpt (G : BaseGraph V E) (H : BaseGraph V F) (e : E ≃ F)
  18     (hl : ∀ x, H.left (e x) = G.left x) (hr : ∀ x, H.right (e x) = G.right x) (T : ℕ) :
  19     expectedOpt G T = expectedOpt H T := by
  20   unfold expectedOpt
  21   rw [← average_equiv (Equiv.arrowCongr (Equiv.refl (Fin T)) e)]
  22   congr 1
  23   funext ω
  24   congr 1
  25   change opt (G.sample ω) = opt (H.sample (fun t => e (ω t)))
  26   rw [relabel_sample G H e hl hr ω]

  28 def pullbackPolicy (e : E ≃ F) (p : OnlinePolicy F) : OnlinePolicy E :=
  29   fun history x => p (history.map e) (e x)
  30 
  31 theorem pullback_orientation (e : E ≃ F) (p : OnlinePolicy F) {T : ℕ} (ω : Fin T → E) :
  32     onlineOrientation (pullbackPolicy e p) ω = onlineOrientation p (fun t => e (ω t)) := by
  33   funext t
  34   simp only [onlineOrientation, pullbackPolicy, List.map_take, List.map_ofFn, Function.comp_def]
  35 
  36 theorem relabel_expectedCost (G : BaseGraph V E) (H : BaseGraph V F) (e : E ≃ F)
  37     (hl : ∀ x, H.left (e x) = G.left x) (hr : ∀ x, H.right (e x) = G.right x)
  38     (T : ℕ) (p : OnlinePolicy F) :
  39     expectedCost G (onlineOrientation (T := T) (pullbackPolicy e p)) =
  40       expectedCost H (onlineOrientation (T := T) p) := by
  41   unfold expectedCost
  42   rw [← average_equiv (Equiv.arrowCongr (Equiv.refl (Fin T)) e)]
  43   congr 1
  44   funext ω
  45   rw [pullback_orientation]
  46   congr 1
  47   change maxLoad (G.sample ω) (onlineOrientation p (fun t => e (ω t))) =
  48     maxLoad (H.sample (fun t => e (ω t))) (onlineOrientation p (fun t => e (ω t)))
  49   rw [relabel_sample G H e hl hr ω]

  51 theorem relabel_referenceCap (G : BaseGraph V E) (H : BaseGraph V F) (e : E ≃ F)
  52     (hl : ∀ x, H.left (e x) = G.left x) (hr : ∀ x, H.right (e x) = G.right x) :
  53     G.referenceCap = H.referenceCap := by
  54   have hcount (S : Finset V) : (G.internalEdges S).card = (H.internalEdges S).card := by
  55     apply card_equiv e
  56     intro x
  57     simp only [BaseGraph.internalEdges, mem_filter, mem_univ, true_and, hl, hr]
  58   have hd : G.maxBaseDensity = H.maxBaseDensity := by
  59     unfold BaseGraph.maxBaseDensity
  60     apply sup'_congr _ rfl
  61     intro S _
  62     simp only [BaseGraph.baseDensity, hcount]
  63   simp only [BaseGraph.referenceCap, hd]
  64 
  65 noncomputable def sourceInstance (G : BaseGraph V E) : Instance V (Fintype.card E) :=
  66   G.sample (Fintype.equivFin E).symm
  67 
  68 theorem sourceInstance_internalCount (G : BaseGraph V E) (S : Finset V) :
  69     internalCount (sourceInstance G) S = (G.internalEdges S).card := by
  70   apply card_equiv (Fintype.equivFin E).symm
  71   intro t
  72   simp only [mem_filter, mem_univ, true_and, sourceInstance, BaseGraph.sample, BaseGraph.internalEdges]
  73 
  74 theorem sourceInstance_opt (G : BaseGraph V E) : opt (sourceInstance G) = G.referenceCap := by
  75   rw [opt_eq_ceil_maxDensity]
  76   have hd : maxDensity (sourceInstance G) = G.maxBaseDensity := by
  77     unfold maxDensity BaseGraph.maxBaseDensity
  78     apply sup'_congr _ rfl
  79     intro S _
  80     simp only [density, BaseGraph.baseDensity, sourceInstance_internalCount]
  81   rw [hd]
  82   rfl
  83 
  84 theorem sourceInstance_split (G : BaseGraph V E) (b : E → Bool) :
  85     sourceInstance (G.split b) = splitInstance (sourceInstance G)
  86       (fun t => b ((Fintype.equivFin E).symm t)) := split_sample G b _
  87 
  88 theorem source_split_reference (G : BaseGraph V E) (b : E → Bool) :
  89     G.referenceCap ≤ 2*(G.split b).referenceCap := by
  90   rw [← sourceInstance_opt G, ← sourceInstance_opt (G.split b), sourceInstance_split]
  91   exact opt_le_two_split_opt _ _

  93 theorem exists_reference_source_orientation (G : BaseGraph V E) :
  94     ∃ b : E → Bool, ∀ v,
  95       (univ.filter fun x => (if b x then G.left x else G.right x) = v).card ≤ G.referenceCap := by
  96   obtain ⟨a,ha⟩ := exists_exact_opt (sourceInstance G)
  97   let b := fun x : E => a (Fintype.equivFin E x)
  98   refine ⟨b, fun v => ?_⟩
  99   have hc : (univ.filter fun x : E => (if b x then G.left x else G.right x) = v).card =
 100       load (sourceInstance G) a v := by
 101     apply card_equiv (Fintype.equivFin E)
 102     intro x
 103     simp only [mem_filter, mem_univ, true_and, destination, load, sourceInstance,
 104       BaseGraph.sample, b, Equiv.symm_apply_apply]
 105   rw [hc]
 106   have hh := load_le_maxLoad (sourceInstance G) a v
 107   rw [ha, sourceInstance_opt] at hh
 108   exact hh
