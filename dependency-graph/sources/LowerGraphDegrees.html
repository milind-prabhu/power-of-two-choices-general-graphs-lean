   7 def baseDegree {V E : Type*} [Fintype E] [DecidableEq V] (G : BaseGraph V E) (v : V) : ℕ :=
   8   (univ.filter fun e => G.left e = v ∨ G.right e = v).card

  10 theorem block_degree_upper {V C : Type*} [Fintype C] [DecidableEq C] [DecidableEq V] {m r : ℕ}
  11     (left : C × Fin m → V) (right : C × Fin r → V)
  12     (hl : Function.Injective left) (hr : Function.Injective right) (v : V) :
  13     baseDegree (blockBase left right) v ≤ m+r := by
  14   have hL : (univ.filter fun e : C × (Fin m × Fin r) => left (e.1,e.2.1) = v).card ≤ r := by
  15     let A := univ.filter fun e : C × (Fin m × Fin r) => left (e.1,e.2.1) = v
  16     let f : A → Fin r := fun e => e.val.2.2
  17     have hf : Function.Injective f := by
  18       intro e f h
  19       have hh := hl (((mem_filter.mp e.property).2).trans ((mem_filter.mp f.property).2).symm)
  20       apply Subtype.ext
  21       exact Prod.ext (congrArg (fun p : C × Fin m => p.1) hh)
  22         (Prod.ext (congrArg (fun p : C × Fin m => p.2) hh) h)
  23     simpa only [Fintype.card_coe,Fintype.card_fin] using Fintype.card_le_of_injective f hf
  24   have hR : (univ.filter fun e : C × (Fin m × Fin r) => right (e.1,e.2.2) = v).card ≤ m := by
  25     let A := univ.filter fun e : C × (Fin m × Fin r) => right (e.1,e.2.2) = v
  26     let f : A → Fin m := fun e => e.val.2.1
  27     have hf : Function.Injective f := by
  28       intro e f h
  29       have hh := hr (((mem_filter.mp e.property).2).trans ((mem_filter.mp f.property).2).symm)
  30       apply Subtype.ext
  31       exact Prod.ext (congrArg (fun p : C × Fin r => p.1) hh)
  32         (Prod.ext h (congrArg (fun p : C × Fin r => p.2) hh))
  33     simpa only [Fintype.card_coe,Fintype.card_fin] using Fintype.card_le_of_injective f hf
  34   have heq : (univ.filter fun e : C × (Fin m × Fin r) => left (e.1,e.2.1) = v ∨ right (e.1,e.2.2) = v) =
  35       (univ.filter fun e => left (e.1,e.2.1) = v) ∪ (univ.filter fun e => right (e.1,e.2.2) = v) := by ext e; simp
  36   unfold baseDegree
  37   change (univ.filter fun e : C × (Fin m × Fin r) => left (e.1,e.2.1) = v ∨ right (e.1,e.2.2) = v).card ≤ m+r
  38   rw [heq]
  39   exact (card_union_le _ _).trans (by omega)

  41 theorem rightVertex_covers_positive (b t m : ℕ) (v : Vertices b t m) (hv : v.1 ≠ 0) :
  42     ∃ p : Blocks b t × Fin (t*m), rightVertex p = v := by
  43   rcases v with ⟨i,g,a⟩
  44   obtain ⟨j,rfl⟩ := Fin.eq_succ_of_ne_zero hv
  45   let p := finProdFinEquiv.symm (Fin.cast (pow_succ t j.val) g)
  46   refine ⟨(⟨j,p.1⟩,finProdFinEquiv (p.2,a)),?_⟩
  47   apply Sigma.mk.inj_iff.mpr
  48   refine ⟨rfl,heq_of_eq ?_⟩
  49   simp only [rightVertex,childGroup,Equiv.symm_apply_apply,Prod.fst,Prod.snd]
  50   apply Prod.ext
  51   · change Fin.cast (pow_succ t j.val).symm (finProdFinEquiv p) = g
  52     change Fin.cast (pow_succ t j.val).symm
  53       (finProdFinEquiv (finProdFinEquiv.symm (Fin.cast (pow_succ t j.val) g))) = g
  54     rw [Equiv.apply_symm_apply]
  55     exact Fin.ext rfl
  56   · rfl

  58 theorem vertex_block_cover (b t m : ℕ) (hb : 0 < b) (v : Vertices b t m) :
  59     (∃ p : Blocks b t × Fin m, leftVertex p = v) ∨
  60       ∃ p : Blocks b t × Fin (t*m), rightVertex p = v := by
  61   by_cases hv : v.1.val < b
  62   · left
  63     exact ⟨(⟨⟨v.1.val,hv⟩,v.2.1⟩,v.2.2),rfl⟩
  64   · right
  65     apply rightVertex_covers_positive
  66     intro hz
  67     have hval := congrArg Fin.val hz
  68     simp only [Fin.val_zero] at hval
  69     omega

  71 theorem graph_degree_lower (b t m : ℕ) (hb : 0 < b) (ht : 1 ≤ t) (v : Vertices b t m) :
  72     m ≤ baseDegree (graph b t m) v := by
  73   rcases vertex_block_cover b t m hb v with ⟨⟨c,a⟩,hv⟩ | ⟨⟨c,r⟩,hv⟩
  74   · have hh : t*m ≤ baseDegree (graph b t m) v := by
  75       let f : Fin (t*m) → {e : Edges b t m // (graph b t m).left e = v ∨ (graph b t m).right e = v} :=
  76         fun r => ⟨(c,a,r),Or.inl hv⟩
  77       have hf : Function.Injective f := by intro r s h; exact congrArg (fun e => e.val.2.2) h
  78       simpa only [Fintype.card_fin,Fintype.card_subtype,baseDegree] using Fintype.card_le_of_injective f hf
  79     exact (by nlinarith : m ≤ t*m).trans hh
  80   · let f : Fin m → {e : Edges b t m // (graph b t m).left e = v ∨ (graph b t m).right e = v} :=
  81       fun a => ⟨(c,a,r),Or.inr hv⟩
  82     have hf : Function.Injective f := by intro a a' h; exact congrArg (fun e => e.val.2.1) h
  83     simpa only [Fintype.card_fin,Fintype.card_subtype,baseDegree] using Fintype.card_le_of_injective f hf

  87 theorem family_degree_range (b : ℕ) (hb : 16 ≤ b) (v : Vertices b (branching b) (width b)) :
  88     Real.sqrt (size b:ℝ) ≤ (baseDegree (graph b (branching b) (width b)) v:ℝ) ∧
  89       (baseDegree (graph b (branching b) (width b)) v:ℝ) ≤
  90         Real.sqrt (size b:ℝ)*(Real.log (size b:ℝ))^3 := by
  91   have hp := family_parameter_bounds b hb
  92   have hm : (0:ℝ) ≤ width b := Nat.cast_nonneg _
  93   have hN : (0:ℝ) ≤ size b := Nat.cast_nonneg _
  94   have hmin := graph_degree_lower b (branching b) (width b) (by omega) (by have := hp.1; omega) v
  95   have hmax := block_degree_upper (@leftVertex b (branching b) (width b)) (@rightVertex b (branching b) (width b))
  96     (leftVertex_injective _ _ _) (rightVertex_injective _ _ _) v
  97   have hminr : (width b:ℝ) ≤ baseDegree (graph b (branching b) (width b)) v := by exact_mod_cast hmin
  98   have hmaxr : (baseDegree (graph b (branching b) (width b)) v:ℝ) ≤ (width b:ℝ)*(branching b+1) := by
  99     have hh : baseDegree (graph b (branching b) (width b)) v ≤ width b+branching b*width b := hmax
 100     exact_mod_cast (by nlinarith : baseDegree (graph b (branching b) (width b)) v ≤ width b*(branching b+1))
 101   have hsqrt : Real.sqrt (size b:ℝ) ≤ width b := Real.sqrt_le_iff.mpr ⟨hm,by exact_mod_cast hp.2.2.2.2.2⟩
 102   have hS : (branching b)^b ≤ layerSum (branching b) (b+1) := by rw [layerSum_succ]; omega
 103   have hm2N : (width b:ℝ)^2 ≤ 2*(size b:ℝ) := by
 104     have hh := Nat.mul_le_mul_left (4*(branching b)^b) hS
 105     unfold width size at *
 106     exact_mod_cast (by nlinarith : (2*(branching b)^b)^2 ≤ 2*(2*(branching b)^b*layerSum (branching b) (b+1)))
 107   have hs0 := Real.sqrt_nonneg (size b:ℝ)
 108   have hs2 := Real.sq_sqrt hN
 109   have hmupper : (width b:ℝ) ≤ 2*Real.sqrt (size b:ℝ) := by nlinarith
 110   have hlog := family_log_large b hb
 111   have hpow := pow_le_pow_left₀ (show (0:ℝ) ≤ 30*(b:ℝ) by positivity) hlog 3
 112   have htdef : (branching b:ℝ) = 4096*(b:ℝ)^3 := by unfold branching; push_cast; ring
 113   have hb16 : (16:ℝ) ≤ b := by exact_mod_cast hb
 114   have hb3 : (1:ℝ) ≤ (b:ℝ)^3 := one_le_pow₀ (by linarith : (1:ℝ) ≤ b)
 115   have htlog : 2*((branching b:ℝ)+1) ≤ (Real.log (size b:ℝ))^3 := by rw [htdef]; nlinarith
 116   refine ⟨hsqrt.trans hminr,?_⟩
 117   have hh := mul_le_mul_of_nonneg_right hmupper (show 0 ≤ (branching b:ℝ)+1 by positivity)
 118   have hh' := mul_le_mul_of_nonneg_left htlog hs0
 119   push_cast at hmaxr
 120   nlinarith
