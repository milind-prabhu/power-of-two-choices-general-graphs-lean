   6 /-! Occurrence-preserving execution logs and exact replay of residual arrivals
   7 by a single ordinary Greedy process. Repeated edges carry distinct time labels.
   8 -/
   9 namespace GraphChoices.ThresholdGreedy
  10 variable {L R C : Type*} [DecidableEq L] [DecidableEq R] [DecidableEq C]
  11 
  12 structure Event (L R C : Type*) where
  13   time : ℕ
  14   arrival : Arrival L R C
  15   threshold : Bool
  16 
  17 def traceFrom (α : C → ℕ) : State L R C → ℕ → List (Arrival L R C) → List (Event L R C)
  18   | _, _, [] => []
  19   | s, k, e :: es =>
  20     ⟨k,e,isThreshold α s e⟩ :: traceFrom α (step α s e) (k+1) es
  21 
  22 def trace (α : C → ℕ) (es : List (Arrival L R C)) : List (Event L R C) :=
  23   traceFrom α initial 0 es
  24 
  25 def thresholdArrivals (log : List (Event L R C)) : List (Arrival L R C) :=
  26   (log.filter fun e => e.threshold).map Event.arrival
  27 
  28 def residualArrivals (log : List (Event L R C)) : List (Arrival L R C) :=
  29   (log.filter fun e => !e.threshold).map Event.arrival

  31 structure GreedyLoads (L R : Type*) where
  32   left : L → ℕ
  33   right : R → ℕ
  34 
  35 def project (s : State L R C) : GreedyLoads L R := ⟨s.greedyLeft, s.greedyRight⟩
  36 
  37 def greedyStep (s : GreedyLoads L R) (e : Arrival L R C) : GreedyLoads L R :=
  38   if s.left e.left < s.right e.right then
  39     {s with left := fun u => s.left u + if u = e.left then 1 else 0}
  40   else
  41     {s with right := fun v => s.right v + if v = e.right then 1 else 0}
  42 
  43 def greedyRun : GreedyLoads L R → List (Arrival L R C) → GreedyLoads L R
  44   | s, [] => s
  45   | s, e::es => greedyRun (greedyStep s e) es
  46 
  47  theorem project_threshold_step (α : C → ℕ) (s : State L R C) (e : Arrival L R C)
  48     (h : isThreshold α s e = true) : project (step α s e) = project s := by
  49   have ht : s.threshold e.left e.cls < α e.cls := by simpa [isThreshold] using h
  50   simp [step, ht, project]
  51 
  52  theorem project_residual_step (α : C → ℕ) (s : State L R C) (e : Arrival L R C)
  53     (h : isThreshold α s e = false) : project (step α s e) = greedyStep (project s) e := by
  54   have ht : ¬ s.threshold e.left e.cls < α e.cls := by simpa [isThreshold] using h
  55   simp only [step, if_neg ht, greedyStep, project]
  56   split_ifs <;> rfl
  57 
  58 /-- The implemented residual process is exactly ordinary Greedy replay on the
  59 extracted residual list, with all classes combined. -/
  60  theorem residual_replay_from (α : C → ℕ) (s : State L R C) (k : ℕ)
  61     (es : List (Arrival L R C)) :
  62     project (runFrom α s es) = greedyRun (project s)
  63       (residualArrivals (traceFrom α s k es)) := by
  64   induction es generalizing s k with
  65   | nil => rfl
  66   | cons e es ih =>
  67     cases h : isThreshold α s e with
  68     | false =>
  69       simp only [traceFrom, residualArrivals, h, Bool.not_false, List.filter_cons_of_pos,
  70         List.map_cons, greedyRun]
  71       rw [← project_residual_step α s e h]
  72       exact ih _ _
  73     | true =>
  74       simp only [traceFrom, residualArrivals, h, Bool.not_true, List.filter_cons_of_neg]
  75       rw [← project_threshold_step α s e h]
  76       exact ih _ _
  77 
  78  theorem residual_replay (α : C → ℕ) (es : List (Arrival L R C)) :
  79     project (run α es) = greedyRun (project (initial : State L R C))
  80       (residualArrivals (trace α es)) := residual_replay_from α initial 0 es

  82 /-- Counters count actual threshold occurrences, rather than all arrivals in
  83 the corresponding bucket. -/
  84  theorem threshold_counter_exact_from (α : C → ℕ) (s : State L R C) (k : ℕ)
  85     (es : List (Arrival L R C)) (u : L) (i : C) :
  86     (runFrom α s es).threshold u i = s.threshold u i +
  87       incidentCount (thresholdArrivals (traceFrom α s k es)) u i := by
  88   induction es generalizing s k with
  89   | nil => simp [runFrom, traceFrom, thresholdArrivals, incidentCount]
  90   | cons e es ih =>
  91     cases h : isThreshold α s e with
  92     | false =>
  93       have ht : ¬s.threshold e.left e.cls < α e.cls := by simpa [isThreshold] using h
  94       simp only [runFrom, traceFrom, thresholdArrivals, h, List.filter_cons_of_neg]
  95       rw [ih (step α s e) (k+1)]
  96       simp only [step, if_neg ht]
  97       split_ifs <;> rfl
  98     | true =>
  99       have ht : s.threshold e.left e.cls < α e.cls := by simpa [isThreshold] using h
 100       simp only [runFrom, traceFrom, thresholdArrivals, h, List.filter_cons_of_pos, List.map_cons]
 101       rw [ih (step α s e) (k+1)]
 102       simp only [thresholdArrivals, step, if_pos ht, incidentCount, List.filter_cons]
 103       by_cases he : e.left = u ∧ e.cls = i
 104       · obtain ⟨rfl,rfl⟩ := he
 105         simp
 106         omega
 107       · have he' : ¬(u = e.left ∧ i = e.cls) := by aesop
 108         simp [he, he']
 109 
 110  theorem threshold_counter_exact (α : C → ℕ) (es : List (Arrival L R C)) (u : L) (i : C) :
 111     (run α es).threshold u i = incidentCount (thresholdArrivals (trace α es)) u i := by
 112   simpa [run, trace, initial] using threshold_counter_exact_from α initial 0 es u i

 114  theorem residual_has_threshold_witnesses (α : C → ℕ) (es : List (Arrival L R C))
 115     (e : Arrival L R C) (he : isThreshold α (run α es) e = false) :

 121  theorem traceFrom_arrivals (α : C → ℕ) (s : State L R C) (k : ℕ)
 122     (es : List (Arrival L R C)) : (traceFrom α s k es).map Event.arrival = es := by
 123   induction es generalizing s k with
 124   | nil => rfl
 125   | cons e es ih => simp [traceFrom, ih]

 127  theorem traceFrom_times (α : C → ℕ) (s : State L R C) (k : ℕ)
 128     (es : List (Arrival L R C)) :
 129     (traceFrom α s k es).map Event.time = List.range' k es.length := by
 130   induction es generalizing s k with
 131   | nil => simp [traceFrom]
 132   | cons e es ih => simp [traceFrom, ih, List.range'_succ]

 134  theorem traceFrom_unique_times (α : C → ℕ) (s : State L R C) (k : ℕ)
 135     (es : List (Arrival L R C)) : ((traceFrom α s k es).map Event.time).Nodup := by
 136   rw [traceFrom_times]

 139 /-- Threshold and residual logs partition the occurrence log up to permutation.
 140 Their internal order is retained by the filter operations. -/
 141  theorem trace_partition (α : C → ℕ) (es : List (Arrival L R C)) :
 142     (thresholdArrivals (trace α es) ++ residualArrivals (trace α es)).Perm es := by
 143   have h := List.filter_append_perm (fun e : Event L R C => e.threshold) (trace α es)
 144   have hh := h.map Event.arrival
 145   simpa [thresholdArrivals, residualArrivals, List.map_append, trace, traceFrom_arrivals] using hh

 147 theorem traceFrom_append (α : C → ℕ) (s : State L R C) (k : ℕ)
 148     (xs ys : List (Arrival L R C)) :
 149     traceFrom α s k (xs ++ ys) = traceFrom α s k xs ++
 150       traceFrom α (runFrom α s xs) (k + xs.length) ys := by
 151   induction xs generalizing s k with
 152   | nil => simp [traceFrom, runFrom]
 153   | cons e es ih =>
 154     simp [traceFrom, runFrom, ih, Nat.add_assoc, Nat.add_comm, Nat.add_left_comm]

 156 /-- Distinct time labels make threshold witnesses disjoint from all residual
 157 occurrences, even when their base-edge labels are identical. -/
 158 theorem threshold_residual_times_disjoint (log : List (Event L R C))
 159     (hunique : (log.map Event.time).Nodup) :
 160     List.Disjoint ((log.filter fun e => e.threshold).map Event.time)
 161       ((log.filter fun e => !e.threshold).map Event.time) := by
 162   apply List.disjoint_left.mpr
 163   intro t ht hr
 164   obtain ⟨e,he,het⟩ := List.mem_map.mp ht
 165   obtain ⟨f,hf,hft⟩ := List.mem_map.mp hr
 166   have hmemE := (List.mem_filter.mp he).1
 167   have hmemF := (List.mem_filter.mp hf).1
 168   have hef : e = f := List.inj_on_of_nodup_map hunique hmemE hmemF (het.trans hft.symm)
 169   have hetrue := (List.mem_filter.mp he).2
 170   have hffalse := (List.mem_filter.mp hf).2
 171   simp_all

 173 theorem trace_threshold_residual_disjoint (α : C → ℕ) (es : List (Arrival L R C)) :
 174     List.Disjoint (((trace α es).filter fun e => e.threshold).map Event.time)
 175       (((trace α es).filter fun e => !e.threshold).map Event.time) :=
 176   threshold_residual_times_disjoint _ (traceFrom_unique_times α initial 0 es)
