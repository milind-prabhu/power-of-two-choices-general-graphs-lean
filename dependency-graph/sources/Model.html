  11 # Samplewise graph balancing (Definition 2.1)
  12 
  13 An instance is an indexed sequence of pairs. Parallel occurrences are kept
  14 separate. Loops are allowed in this general model, so all the lemmas also apply
  15 to the loopless simple base graphs in the paper.
  16 -/
  17 namespace GraphChoices
  18 open Finset
  19 
  20 structure Instance (V : Type*) (T : ℕ) where
  21   left : Fin T → V
  22   right : Fin T → V
  23 
  24 theorem Instance.ext {V : Type*} {T : ℕ} {I J : Instance V T}
  25     (hl : I.left = J.left) (hr : I.right = J.right) : I = J := by
  26   cases I
  27   cases J
  28   cases hl
  29   cases hr
  30   rfl
  31 
  32 variable {V : Type*} [Fintype V] [DecidableEq V] {T : ℕ}
  33 
  34 abbrev Orientation (T : ℕ) := Fin T → Bool
  35 
  36 def destination (I : Instance V T) (a : Orientation T) (t : Fin T) : V :=
  37   if a t then I.left t else I.right t
  38 
  39 def load (I : Instance V T) (a : Orientation T) (v : V) : ℕ :=
  40   (univ.filter fun t => destination I a t = v).card
  41 
  42 def maxLoad (I : Instance V T) (a : Orientation T) : ℕ :=
  43   univ.sup (load I a)
  44 
  45 def internalCount (I : Instance V T) (S : Finset V) : ℕ :=
  46   (univ.filter fun t => I.left t ∈ S ∧ I.right t ∈ S).card
  47 
  48 def assignedCount (I : Instance V T) (a : Orientation T) (S : Finset V) : ℕ :=
  49   (univ.filter fun t => destination I a t ∈ S).card
  50 
  51 omit [Fintype V] [DecidableEq V] in
  52  theorem destination_endpoint (I : Instance V T) (a : Orientation T) (t : Fin T) :
  53     destination I a t = I.left t ∨ destination I a t = I.right t := by
  54   simp only [destination]; split <;> simp_all
  55 
  56  theorem load_le_maxLoad (I : Instance V T) (a : Orientation T) (v : V) :
  57     load I a v ≤ maxLoad I a := by
  58   exact le_sup (f := load I a) (mem_univ v)
  59 
  60  theorem maxLoad_le_iff (I : Instance V T) (a : Orientation T) (c : ℕ) :
  61     maxLoad I a ≤ c ↔ ∀ v, load I a v ≤ c := by
  62   simp [maxLoad]
  63 
  64 omit [Fintype V] in
  65  theorem load_le_length (I : Instance V T) (a : Orientation T) (v : V) :
  66     load I a v ≤ T := by
  67   simpa [load] using card_filter_le (univ : Finset (Fin T))
  68     (fun t => destination I a t = v)
  69 
  70  theorem maxLoad_le_length (I : Instance V T) (a : Orientation T) :
  71     maxLoad I a ≤ T :=
  72   (maxLoad_le_iff I a T).2 (load_le_length I a)
  73 
  74 omit [Fintype V] in
  75  theorem assignedCount_eq_sum (I : Instance V T) (a : Orientation T) (S : Finset V) :
  76     assignedCount I a S = ∑ v ∈ S, load I a v := by
  77   classical
  78   simp only [assignedCount, load, card_eq_sum_ones, sum_filter]
  79   rw [sum_comm]
  80   apply sum_congr rfl
  81   intro t ht
  82   simp
  83 
  84  theorem total_load (I : Instance V T) (a : Orientation T) :
  85     ∑ v, load I a v = T := by
  86   rw [← assignedCount_eq_sum I a univ]
  87   simp [assignedCount]
  88 
  89 omit [Fintype V] in
  90  theorem internalCount_le_assignedCount (I : Instance V T) (a : Orientation T)
  91     (S : Finset V) : internalCount I S ≤ assignedCount I a S := by
  92   apply card_le_card
  93   intro t ht
  94   simp only [mem_filter, mem_univ, true_and] at ht ⊢
  95   rcases destination_endpoint I a t with h | h <;> simp [h, ht.1, ht.2]
  96 
  97 /-- The exact, ceiling-sensitive density obstruction used in Section 2.2. -/
  98  theorem density_obstruction (I : Instance V T) (a : Orientation T) (S : Finset V) :
  99     internalCount I S ≤ S.card * maxLoad I a := by
 100   calc
 101     internalCount I S ≤ assignedCount I a S := internalCount_le_assignedCount I a S
 102     _ = ∑ v ∈ S, load I a v := assignedCount_eq_sum I a S
 103     _ ≤ ∑ _v ∈ S, maxLoad I a := sum_le_sum fun v _ => load_le_maxLoad I a v
 104     _ = S.card * maxLoad I a := by simp
 105 
 106 /-- Existence is constructive at the level of a finite search space. -/
 107 def opt (I : Instance V T) : ℕ :=
 108   Nat.find (show ∃ c, ∃ a : Orientation T, maxLoad I a ≤ c from
 109     ⟨T, (fun _ => false), maxLoad_le_length I (fun _ => false)⟩)
 110 
 111  theorem exists_opt_orientation (I : Instance V T) :
 112     ∃ a : Orientation T, maxLoad I a ≤ opt I := by
 113   unfold opt
 114   exact Nat.find_spec (p := fun c => ∃ a : Orientation T, maxLoad I a ≤ c) _
 115 
 116  theorem opt_le_maxLoad (I : Instance V T) (a : Orientation T) :
 117     opt I ≤ maxLoad I a := Nat.find_min' _ ⟨a, le_rfl⟩
 118 
 119  theorem exists_exact_opt (I : Instance V T) :
 120     ∃ a : Orientation T, maxLoad I a = opt I := by
 121   obtain ⟨a, ha⟩ := exists_opt_orientation I
 122   exact ⟨a, Nat.le_antisymm ha (opt_le_maxLoad I a)⟩
 123 
 124  theorem opt_le_length (I : Instance V T) : opt I ≤ T :=
 125   (opt_le_maxLoad I (fun _ => false)).trans (maxLoad_le_length I _)
 126 
 127  theorem density_le_opt (I : Instance V T) (S : Finset V) :
 128     internalCount I S ≤ S.card * opt I := by
 129   obtain ⟨a, ha⟩ := exists_exact_opt I
 130   simpa [ha] using density_obstruction I a S
 131 
 132  theorem total_le_card_mul_opt (I : Instance V T) : T ≤ Fintype.card V * opt I := by
 133   simpa [internalCount] using density_le_opt I univ
 134 
 135  theorem opt_pos (I : Instance V T) (hT : 0 < T) : 0 < opt I := by
 136   have h := total_le_card_mul_opt I
 137   by_contra hn
 138   have : opt I = 0 := Nat.eq_zero_of_not_pos hn
 139   simp [this] at h
 140   omega
 141 
 142 /-- Every occurrence of a fixed pair must be assigned to one of two vertices. -/
 143  theorem multiplicity_le_two_opt (I : Instance V T) (u v : V) :
 144     (univ.filter fun t => I.left t = u ∧ I.right t = v).card ≤ 2 * opt I := by
 145   calc
 146     _ ≤ internalCount I {u, v} := by
 147       apply card_le_card
 148       intro t ht
 149       simp only [mem_filter, mem_univ, true_and] at ht ⊢
 150       simp [ht.1, ht.2]
 151     _ ≤ ({u, v} : Finset V).card * opt I := density_le_opt I _
 152     _ ≤ 2 * opt I := Nat.mul_le_mul_right _ (card_insert_le _ _ |>.trans (by simp))
 153
