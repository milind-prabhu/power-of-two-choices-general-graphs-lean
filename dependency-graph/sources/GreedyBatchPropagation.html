   7 /-- A deterministic forcing lemma. Every marked arrival favors u when its
   8 neighbor already has the preceding load level; unrelated arrivals help. -/
   9 theorem batch_forces_load (G : BaseGraph V E) (k s : ℕ)
  10     (ω : Fin k → Fin s → E × Bool) (i : Fin k) (h : ℕ) (u : V)
  11     (A : Finset (Fin s)) (hcount : h+1 ≤ A.card)
  12     (hsource : ∀ j ∈ A, G.left (ω i j).1 = u ∧ (ω i j).2 = true ∧
  13       h ≤ state G (blockFlatten k s ω) (i.val*s) (G.right (ω i j).1)) :
  14     h+1 ≤ state G (blockFlatten k s ω) ((i.val+1)*s) u := by
  15   let τ := blockFlatten k s ω
  16   have hend : (i.val+1)*s ≤ k*s := Nat.mul_le_mul_right s (by omega)
  17   by_contra hbad
  18   have hu : state G τ ((i.val+1)*s) u ≤ h := by
  19     change ¬ h+1 ≤ state G τ ((i.val+1)*s) u at hbad
  20     omega
  21   have htime (j : Fin s) : (finProdFinEquiv (i,j)).val = i.val*s+j.val := by
  22     change j.val+s*i.val = i.val*s+j.val
  23     ring
  24   have hdest (j : Fin s) (hj : j ∈ A) :
  25       destination ((coinGraph G).sample τ) (onlineOrientation (policy G) τ) (finProdFinEquiv (i,j)) = u := by
  26     obtain ⟨hl,hcoin,hr⟩ := hsource j hj
  27     let t : Fin (k*s) := finProdFinEquiv (i,j)
  28     have ht : t.val < (i.val+1)*s := by dsimp only [t]; rw [htime]; nlinarith [j.isLt]
  29     have hstart : i.val*s ≤ t.val := by dsimp only [t]; rw [htime]; omega
  30     have hlcur : state G τ t.val u ≤ h :=
  31       (state_mono G τ ht.le hend u).trans hu
  32     have hrcur : h ≤ state G τ t.val (G.right (ω i j).1) :=
  33       hr.trans (state_mono G τ hstart (Nat.le_of_lt t.isLt) _)
  34     have hmark : τ t = ω i j := blockFlatten_apply k s ω i j
  35     change (if choice G (state G τ t.val) (τ t) then G.left (τ t).1 else G.right (τ t).1) = u
  36     rw [hmark]
  37     rw [choice_favours_left G _ _ hcoin (by rw [hl]; omega), if_pos rfl, hl]
  38   have hc : A.card ≤ prefixLoad ((coinGraph G).sample τ) (onlineOrientation (policy G) τ)
  39       ((i.val+1)*s) u := by
  40     apply card_le_card_of_injOn (fun j => finProdFinEquiv (i,j))
  41     · intro j hj
  42       change finProdFinEquiv (i,j) ∈ univ.filter (fun t => t.val < (i.val+1)*s ∧
  43         destination ((coinGraph G).sample τ) (onlineOrientation (policy G) τ) t = u)
  44       exact mem_filter.mpr ⟨mem_univ _,by rw [htime]; nlinarith [j.isLt],hdest j hj⟩
  45     · intro j _ j' _ he
  46       exact congrArg Prod.snd (finProdFinEquiv.injective he)
  47   rw [← state_eq_prefixLoad G τ _ hend] at hc
  48   omega
  49 
  50 end GraphChoices.RandomGreedy

  55 def markedBucket {b t m : ℕ} (c : Blocks b t) (a : Fin m) : Finset (Edges b t m × Bool) :=
  56   (({c} : Finset (Blocks b t)) ×ˢ (({a} : Finset (Fin m)) ×ˢ univ)) ×ˢ {true}
  57 
  58 theorem markedBucket_card {b t m : ℕ} (c : Blocks b t) (a : Fin m) :
  59     (markedBucket c a).card = t*m := by simp [markedBucket]
  60 
  61 theorem markedBucket_source {b t m : ℕ} (c : Blocks b t) (a : Fin m)
  62     (x : Edges b t m × Bool) (hx : x ∈ markedBucket c a) :
  63     (graph b t m).left x.1 = leftVertex (c,a) ∧ x.2 = true ∧
  64       ((graph b t m).right x.1).1.val = c.1.val+1 := by
  65   simp only [markedBucket, mem_product, mem_singleton, mem_univ, and_true] at hx
  66   rcases x with ⟨⟨c',a',r⟩,coin⟩
  67   rcases hx with ⟨⟨rfl,rfl⟩,rfl⟩
  68   exact ⟨rfl,rfl,rfl⟩
  69 
  70 def allBatchesGood (b t m s : ℕ) (ω : Fin b → Fin s → Edges b t m × Bool) : Prop :=
  71   ∀ i c a, b ≤ hitCount (markedBucket c a) (ω i)

  73 theorem propagation (b t m s : ℕ) (ω : Fin b → Fin s → Edges b t m × Bool)
  74     (hgood : allBatchesGood b t m s ω) :
  75     ∀ i, i ≤ b → ∀ v : Vertices b t m, v.1.val = b-i →
  76       i ≤ RandomGreedy.state (graph b t m) (blockFlatten b s ω) (i*s) v := by
  77   intro i
  78   induction i with
  79   | zero => intro _ _ _; omega
  80   | succ i ih =>
  81     intro hib v hv
  82     have hib' : i < b := by omega
  83     let ci : Fin b := ⟨v.1.val,by omega⟩
  84     let c : Blocks b t := ⟨ci,v.2.1⟩
  85     have hleft : leftVertex (c,v.2.2) = v := by
  86       rcases v with ⟨j,g,a⟩
  87       rfl
  88     let A := univ.filter fun j : Fin s => ω ⟨i,hib'⟩ j ∈ markedBucket c v.2.2
  89     have hcount : i+1 ≤ A.card := (by omega : i+1 ≤ b).trans (hgood ⟨i,hib'⟩ c v.2.2)
  90     have hh := RandomGreedy.batch_forces_load (graph b t m) b s ω ⟨i,hib'⟩ i v A hcount
  91       (fun j hj => by
  92         have hs := markedBucket_source c v.2.2 _ ((mem_filter.mp hj).2)
  93         refine ⟨hs.1.trans hleft,hs.2.1,?_⟩
  94         exact ih (by omega) _ (by rw [hs.2.2]; dsimp only [c,ci]; omega))
  95     exact hh
  96 
  97 end GraphChoices.LowerBound
