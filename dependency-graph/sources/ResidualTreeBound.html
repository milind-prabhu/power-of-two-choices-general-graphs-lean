   4 /-! The fixed-residual-tree event is contained in the counted joint witness
   5 event. The proof uses actual execution counters and time labels, not an
   6 assumed witness-encoding hypothesis. Tree geometry is only needed later for
   7 counting embeddings and ensuring that left-vertex buckets are distinct. -/
   8 namespace GraphChoices.ThresholdGreedy
   9 open Finset
  10 variable {E L R C J : Type*}
  11   [Fintype E] [DecidableEq E] [Nonempty E]
  12   [DecidableEq L] [DecidableEq R] [DecidableEq C]
  13   [Fintype J] [DecidableEq J] {T m : ℕ}

  15 def sampleArrivals (encode : E → Arrival L R C) (ω : Fin T → E) : List (Arrival L R C) :=
  16   List.ofFn fun t => encode (ω t)
  17 
  18  theorem sampleArrivals_length (encode : E → Arrival L R C) (ω : Fin T → E) :
  19     (sampleArrivals encode ω).length = T := by simp [sampleArrivals]
  20 
  21 def residualTime (α : C → ℕ) (encode : E → Arrival L R C) (ω : Fin T → E) (t : Fin T) : Prop :=
  22   ∃ ev ∈ trace α (sampleArrivals encode ω), ev.time = t.val ∧ ev.threshold = false
  23 
  24  theorem sample_event_arrival (α : C → ℕ) (encode : E → Arrival L R C) (ω : Fin T → E)
  25     (ev : Event L R C) (hev : ev ∈ trace α (sampleArrivals encode ω))
  26     (t : Fin T) (ht : ev.time = t.val) : ev.arrival = encode (ω t) := by
  27   obtain ⟨i,hi,ha⟩ := trace_mem_arrival α (sampleArrivals encode ω) ev hev
  28   have hval : i.val = t.val := hi.symm.trans ht
  29   simpa only [sampleArrivals, List.get_eq_getElem, List.getElem_ofFn, hval] using ha

  31 noncomputable def fixedResidualEvent (α : C → ℕ) (encode : E → Arrival L R C)
  32     (T m : ℕ) (tree : Fin m → E) : Finset (Fin T → E) := by
  33   classical
  34   exact univ.filter fun ω => ∃ τ : Fin m → Fin T, Function.Injective τ ∧
  35     ∀ i, ω (τ i) = tree i ∧ residualTime α encode ω (τ i)
  36 
  37 def sourceBucket (encode : E → Arrival L R C) (u : L) (c : C) : Finset E :=
  38   univ.filter fun e => (encode e).left = u ∧ (encode e).cls = c
  39 
  40 /-- For distinct fixed edge labels, occurrence of every label in the residual
  41 process is equivalent to the injectively time-labelled event definition. -/
  42  theorem mem_fixedResidualEvent_iff (α : C → ℕ) (encode : E → Arrival L R C)
  43     (tree : Fin m → E) (htree : Function.Injective tree) (ω : Fin T → E) :
  44     ω ∈ fixedResidualEvent α encode T m tree ↔
  45       ∀ i, ∃ t : Fin T, ω t = tree i ∧ residualTime α encode ω t := by
  46   classical
  47   simp only [fixedResidualEvent, mem_filter, mem_univ, true_and]
  48   constructor
  49   · rintro ⟨τ,hτ,hh⟩ i
  50     exact ⟨τ i,hh i⟩
  51   · intro hh
  52     choose τ hτ using hh
  53     refine ⟨τ,?_,hτ⟩
  54     intro i j hij
  55     apply htree
  56     exact (hτ i).1.symm.trans ((congrArg ω hij).trans (hτ j).1)

  58 /-- Every fixed residual-edge event forces the exact witnesses counted in the
  59 joint probability bound. There is one representative residual edge per bucket. -/
  60  theorem fixedResidualEvent_subset_witness (α : C → ℕ) (encode : E → Arrival L R C)
  61     (tree : Fin m → E) (vertex : J → L) (cls : J → C) (representative : J → Fin m)
  62     (hrepresentative : ∀ j, (encode (tree (representative j))).left = vertex j ∧
  63       (encode (tree (representative j))).cls = cls j)
  64     (hdistinct : Function.Injective (fun j => (vertex j,cls j))) :
  65     fixedResidualEvent α encode T m tree ⊆
  66       GraphChoices.patternWitnessEvent T m (fun j => α (cls j)) tree
  67         (fun j => sourceBucket encode (vertex j) (cls j)) := by
  68   classical
  69   intro ω hω
  70   obtain ⟨τ,hτ,hocc⟩ := (mem_filter.mp hω).2
  71   let es := sampleArrivals encode ω
  72   have hlen : es.length = T := sampleArrivals_length encode ω
  73   let K (j : J) := finiteThresholdTimes α es T hlen (vertex j) (cls j)
  74   apply GraphChoices.mem_patternWitnessEvent_of_times ω tree
  75     (fun j => sourceBucket encode (vertex j) (cls j)) τ K
  76   · intro j
  77     rw [finiteThresholdTimes_card]
  78     obtain ⟨ev,hev,htime,hr⟩ := (hocc (representative j)).2
  79     have ha : ev.arrival = encode (tree (representative j)) :=
  80       (sample_event_arrival α encode ω ev hev _ htime).trans (congrArg encode (hocc _).1)
  81     have hs := residual_event_final_saturated α es ev hev hr
  82     simpa [ha, (hrepresentative j).1, (hrepresentative j).2] using hs
  83   · exact hτ
  84   · exact fun i => (hocc i).1
  85   · intro i j ht
  86     obtain ⟨ev,hev,htime,hth,_,_⟩ := thresholdWitnessTimes_spec α es (vertex j) (cls j) (τ i).val
  87       ((mem_finiteThresholdTimes α es T hlen _ _ _).mp ht)
  88     obtain ⟨ev',hev',htime',hr⟩ := (hocc i).2
  89     have heq := trace_event_eq_of_time α es ev ev' hev hev' (htime.trans htime'.symm)
  90     have : true = false := hth.symm.trans ((congrArg Event.threshold heq).trans hr)
  91     cases this
  92   · intro j j' hj
  93     exact finiteThresholdTimes_buckets_disjoint α es T hlen _ _ _ _ (fun h => hj (hdistinct h))
  94   · intro j t ht
  95     obtain ⟨ev,hev,htime,_,hu,hc⟩ := thresholdWitnessTimes_spec α es (vertex j) (cls j) t.val
  96       ((mem_finiteThresholdTimes α es T hlen _ _ _).mp ht)
  97     have ha := sample_event_arrival α encode ω ev hev t htime
  98     simp only [sourceBucket, mem_filter, mem_univ, true_and]
  99     exact ⟨(congrArg Arrival.left ha).symm.trans hu,(congrArg Arrival.cls ha).symm.trans hc⟩

 101 /-- The raw exponential fixed-tree probability bound (Equation (12)), now
 102 proved for the actual residual execution. -/
 103  theorem fixedResidual_probability_le (α : C → ℕ) (encode : E → Arrival L R C)
 104     (tree : Fin m → E) (vertex : J → L) (cls : J → C) (representative : J → Fin m)
 105     (hrepresentative : ∀ j, (encode (tree (representative j))).left = vertex j ∧
 106       (encode (tree (representative j))).cls = cls j)
 107     (hdistinct : Function.Injective (fun j => (vertex j,cls j))) :
 108     GraphChoices.eventProbability (fixedResidualEvent α encode T m tree) ≤
 109       ((T:ℝ) / Fintype.card E)^m *
 110         ∏ j, (Real.exp 1 * T * ((sourceBucket encode (vertex j) (cls j)).card : ℝ) /
 111           Fintype.card E / α (cls j))^(α (cls j)) := by
 112   have hs := GraphChoices.eventProbability_mono
 113     (fixedResidualEvent_subset_witness (T := T) α encode tree vertex cls representative hrepresentative hdistinct)
 114   have hp := GraphChoices.patternWitness_probability_le_exp T m (fun j => α (cls j)) tree
 115     (fun j => sourceBucket encode (vertex j) (cls j))
 116   apply hs.trans
 117   simpa only [div_eq_mul_inv, mul_assoc, mul_left_comm, mul_comm] using hp
