  15 theorem PreparedDecomposition.component_tail (P : PreparedDecomposition G B N d) (T c : ℕ)
  16     (hT : 0 < T) (hTN : T ≤ N) (hc : 5 ≤ c)
  17     (hvertices : Fintype.card (L ⊕ R) ≤ N)
  18     (haverage : 2*(G.card : ℝ) = (N:ℝ)*d) :
  19     eventProbability (largeComponentEvent P.encode (P.thresholds T) T
  20       (3*(polynomialCutoff N c+1)^2)) ≤ 3/(N:ℝ)^c := by
  21   have hv : Fintype.card L + Fintype.card R ≤ N := by simpa only [Fintype.card_sum] using hvertices
  22   have hLN : Fintype.card L ≤ N := by omega
  23   have hRN : Fintype.card R ≤ N := by omega
  24   exact preprocessed_log_squared_component_tail P.pieces.get P.encode P.DL P.DR
  25     (4*B) (maxLeftDegree G) T N c (fun i => scaleNat i.val) d P.simple P.source
  26     P.bucket_bound P.piece_left_cap P.piece_right_cap P.global_bucket_bound
  27     hT hTN P.scale_gt_one P.vertex_count_pos hLN hRN hc P.max_degree_preprocessing
  28     (by simpa only [Fintype.card_coe] using haverage)

  30 /-- Concrete finite form of the preprocessed T=N algorithmic guarantee.
  31 The policy is nonanticipating by construction, and its cost and benchmark
  32 are the original indexed sampling definitions. The threshold budget is the
  33 explicit formula derived from this exact decomposition's real degree bounds. -/
  34 theorem PreparedDecomposition.expected_cost_at_vertex_horizon (P : PreparedDecomposition G B N d)
  35     (c : ℕ) (hc : 5 ≤ c) (hvertices : Fintype.card (L ⊕ R) ≤ N)
  36     (haverage : 2*(G.card : ℝ) = (N:ℝ)*d) :
  37     expectedCost (graphBase G) (onlineOrientation (T := N) (P.policy N)) ≤
  38       P.budget N + (2*(Nat.log 2 (3*(polynomialCutoff N c+1)^2)+1) : ℕ)*
  39         expectedOpt (graphBase G) N + 3*N/(N:ℝ)^c := by
  40   have ht := P.component_tail N c P.vertex_count_pos le_rfl hc hvertices haverage
  41   have h := expected_load_of_component_tail P.encode (P.thresholds N) N _ _ ht
  42   rw [expectedTotalMaxLoad_eq_expectedCost, expectedReplayOpt_eq_expectedOpt] at h
  43   change expectedCost (graphBase G) (onlineOrientation (T := N) (P.policy N)) ≤
  44     (∑ i, (P.thresholds N i : ℝ)) + (2*(Nat.log 2 (3*(polynomialCutoff N c+1)^2)+1) : ℕ)*
  45       expectedOpt (graphBase G) N + N*(3/(N:ℝ)^c) at h
  46   have hb := P.threshold_budget N (P.mass_at_vertex_horizon haverage)
  47   calc
  48     _ ≤ _ := h
  49     _ ≤ P.budget N + (2*(Nat.log 2 (3*(polynomialCutoff N c+1)^2)+1) : ℕ)*
  50         expectedOpt (graphBase G) N + N*(3/(N:ℝ)^c) := add_le_add_right (add_le_add_right hb _) _
  51     _ = _ := by ring

  53 /-- Existence of an actual source-edge online policy from the graph-level
  54 preprocessing assumptions. The decomposition is constructed by the proved
  55 peeling theorem; it is not an input existence hypothesis. -/
  56 theorem exists_decomposed_online_policy (G : Graph L R) [Nonempty G] (B N : ℕ) (d : ℝ)
  57     (hd : 0 < d) (hlog : 1 < Real.log (N:ℝ))
  58     (hleft : ∀ u, (leftDegree G u : ℝ) ≤ d*Real.log (N:ℝ))
  59     (hB : d ≤ 16*B) (hD : ∀ u, leftDegree G u ≤ 4*B)
  60     (hvertices : Fintype.card (L ⊕ R) ≤ N)
  61     (haverage : 2*(G.card : ℝ) = (N:ℝ)*d) (c : ℕ) (hc : 5 ≤ c) :
  62     ∃ P : PreparedDecomposition G B N d, ∃ p : OnlinePolicy G,
  63       p = P.policy N ∧
  64       expectedCost (graphBase G) (onlineOrientation (T := N) p) ≤ P.budget N +
  65         (2*(Nat.log 2 (3*(polynomialCutoff N c+1)^2)+1) : ℕ)*expectedOpt (graphBase G) N +
  66           3*N/(N:ℝ)^c := by
  67   obtain ⟨P⟩ := exists_preparedDecomposition G B N d hd hlog hleft hB hD
  68   exact ⟨P, P.policy N, rfl, P.expected_cost_at_vertex_horizon c hc hvertices haverage⟩
