  10 theorem badBatches_probability (b t m s : ℕ) [Nonempty (Edges b t m)] :
  11     eventProbability (badBatches b t m s) ≤
  12       (b:ℝ)*(Fintype.card (Blocks b t):ℝ)*m*
  13         Real.exp ((b:ℝ)*Real.log 2-(s:ℝ)*(t*m)/(4*(Fintype.card (Edges b t m):ℝ))) := by
  14   classical
  15   let events := fun p : Fin b × (Blocks b t × Fin m) =>
  16     univ.filter fun ω : Fin b → Fin s → Edges b t m × Bool =>
  17       hitCount (markedBucket p.2.1 p.2.2) (ω p.1) < b
  18   have hsub : badBatches b t m s ⊆ univ.biUnion events := by
  19     intro ω hω
  20     have hh := (mem_filter.mp hω).2
  21     simp only [allBatchesGood, not_forall, not_le] at hh
  22     obtain ⟨i,c,a,h⟩ := hh
  23     exact mem_biUnion.mpr ⟨(i,c,a),mem_univ _,mem_filter.mpr ⟨mem_univ _,h⟩⟩
  24   have heach (p : Fin b × (Blocks b t × Fin m)) : eventProbability (events p) ≤
  25       Real.exp ((b:ℝ)*Real.log 2-(s:ℝ)*(t*m)/(4*(Fintype.card (Edges b t m):ℝ))) := by
  26     have hc := coordinate_event_probability
  27       (univ.filter fun η : Fin s → Edges b t m × Bool => hitCount (markedBucket p.2.1 p.2.2) η < b) p.1
  28     have heq : events p = univ.filter (fun ω : Fin b → Fin s → Edges b t m × Bool =>
  29       ω p.1 ∈ univ.filter fun η : Fin s → Edges b t m × Bool => hitCount (markedBucket p.2.1 p.2.2) η < b) := by
  30       ext ω
  31       simp [events]
  32     have hceq : eventProbability (events p) = eventProbability
  33         (univ.filter fun η : Fin s → Edges b t m × Bool => hitCount (markedBucket p.2.1 p.2.2) η < b) := by
  34       convert hc using 1
  35       congr 1
  36       ext ω
  37       simp [events]
  38     rw [hceq]
  39     apply (hitLowerTail_probability (markedBucket p.2.1 p.2.2) s b).trans_eq
  40     congr 1
  41     rw [markedBucket_card, Fintype.card_prod, Fintype.card_bool]
  42     push_cast
  43     ring
  44   apply (eventProbability_mono hsub).trans
  45   apply (witness_union_bound univ events).trans
  46   have hh := sum_le_sum (fun p (_ : p ∈ (univ : Finset (Fin b × (Blocks b t × Fin m)))) => heach p)
  47   simpa only [sum_const, card_univ, Fintype.card_prod, Fintype.card_fin, nsmul_eq_mul, Nat.cast_mul, mul_assoc] using hh

  49 theorem goodBatches_maxLoad (b t m s : ℕ) (hm : 0 < m)
  50     (ω : Fin b → Fin s → Edges b t m × Bool) (hgood : allBatchesGood b t m s ω) :
  51     b ≤ maxLoad ((RandomGreedy.coinGraph (graph b t m)).sample (blockFlatten b s ω))
  52       (onlineOrientation (RandomGreedy.policy (graph b t m)) (blockFlatten b s ω)) := by
  53   let v : Vertices b t m := ⟨0,(⟨0,by simp⟩,⟨0,hm⟩)⟩
  54   have hh := propagation b t m s ω hgood b le_rfl v (by simp [v])
  55   have heq := execution_final_load _ _ _
  56     (RandomGreedy.state_zero (graph b t m) (blockFlatten b s ω))
  57     (RandomGreedy.state_step (graph b t m) (blockFlatten b s ω)) v
  58   rw [heq] at hh
  59   exact hh.trans (load_le_maxLoad _ _ _)

  61 theorem expectedGreedy_lower (b t m s : ℕ) [Nonempty (Edges b t m)] (hm : 0 < m) :
  62     (b:ℝ)*(1-(b:ℝ)*(Fintype.card (Blocks b t):ℝ)*m*
  63       Real.exp ((b:ℝ)*Real.log 2-(s:ℝ)*(t*m)/(4*(Fintype.card (Edges b t m):ℝ)))) ≤
  64       RandomGreedy.expectedLoad (graph b t m) (b*s) := by
  65   classical
  66   let good := univ \ badBatches b t m s
  67   have hpoint (ω : Fin b → Fin s → Edges b t m × Bool) :
  68       (b:ℝ)*(if ω ∈ good then (1:ℝ) else 0) ≤
  69         (maxLoad ((RandomGreedy.coinGraph (graph b t m)).sample (blockFlatten b s ω))
  70           (onlineOrientation (RandomGreedy.policy (graph b t m)) (blockFlatten b s ω)):ℝ) := by
  71     split_ifs with hω
  72     · rw [mul_one]
  73       have hh : allBatchesGood b t m s ω := by
  74         have h := (mem_sdiff.mp hω).2
  75         simpa only [badBatches, mem_filter, mem_univ, true_and, not_not] using h
  76       exact_mod_cast goodBatches_maxLoad b t m s hm ω hh
  77     · rw [mul_zero]
  78       positivity
  79   have hh := average_mono hpoint
  80   rw [average_mul, average_unit_indicator, eventProbability_compl,
  81     average_equiv (blockFlatten b s) (fun ω => (maxLoad ((RandomGreedy.coinGraph (graph b t m)).sample ω)
  82       (onlineOrientation (RandomGreedy.policy (graph b t m)) ω):ℝ))] at hh
  83   apply le_trans _ hh
  84   have hb := badBatches_probability b t m s
  85   have hbr : (0:ℝ) ≤ b := Nat.cast_nonneg _
  86   nlinarith
