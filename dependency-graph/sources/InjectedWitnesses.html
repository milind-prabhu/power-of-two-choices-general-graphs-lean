   3 /-! Exact probabilities for injectively assigned witness time steps. This is
   4 the independence step in the pattern bound: every required occurrence must
   5 have a distinct time, including repeated labels of the same base edge. -/
   6 namespace GraphChoices
   7 open Finset
   8 variable {E J : Type*} [Fintype E] [DecidableEq E] [Nonempty E]
   9   [Fintype J] [DecidableEq J] {T : ℕ}
  10 
  11 def requirementsEvent (τ : J → Fin T) (A : J → Finset E) : Finset (Fin T → E) :=
  12   univ.filter fun ω => ∀ j, ω (τ j) ∈ A j
  13 
  14 noncomputable def constraintsAt (τ : J → Fin T) (A : J → Finset E) (t : Fin T) : Finset E := by
  15   classical
  16   exact if h : ∃ j, τ j = t then A (Classical.choose h) else univ
  17 
  18  theorem constraintsAt_selected (τ : J → Fin T) (A : J → Finset E)
  19     (hτ : Function.Injective τ) (j : J) : constraintsAt τ A (τ j) = A j := by
  20   classical
  21   unfold constraintsAt
  22   rw [dif_pos (show ∃ k, τ k = τ j from ⟨j,rfl⟩)]
  23   congr 1
  24   exact hτ (Classical.choose_spec (show ∃ k, τ k = τ j from ⟨j,rfl⟩))
  25 
  26  theorem requirementsEvent_eq_pi (τ : J → Fin T) (A : J → Finset E)
  27     (hτ : Function.Injective τ) :
  28     requirementsEvent τ A = Fintype.piFinset (constraintsAt τ A) := by
  29   classical
  30   ext ω
  31   simp only [requirementsEvent, mem_filter, mem_univ, true_and, Fintype.mem_piFinset]
  32   constructor
  33   · intro h t
  34     unfold constraintsAt
  35     split_ifs with ht
  36     · have hchoose := Classical.choose_spec ht
  37       simpa [hchoose] using h (Classical.choose ht)
  38     · exact mem_univ _
  39   · intro h j
  40     have ht := h (τ j)
  41     simpa [constraintsAt_selected τ A hτ j] using ht
  42 
  43 /-- Injective assignment of all witness slots makes the probability a product
  44 of the permitted fractions of base-edge labels. -/
  45  theorem requirements_probability (τ : J → Fin T) (A : J → Finset E)
  46     (hτ : Function.Injective τ) :
  47     eventProbability (requirementsEvent τ A) =
  48       ∏ j, ((A j).card : ℝ) / Fintype.card E := by
  49   classical
  50   rw [requirementsEvent_eq_pi τ A hτ, probability_coordinate_constraints]
  51   let S : Finset (Fin T) := univ.image τ
  52   have hm : (Fintype.card E : ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  53   calc
  54     _ = ∏ t ∈ S, ((constraintsAt τ A t).card : ℝ) / Fintype.card E := by
  55       symm
  56       apply prod_subset (subset_univ _)
  57       intro t ht hnot
  58       have hnone : ¬∃ j, τ j = t := by
  59         rintro ⟨j,hj⟩
  60         exact hnot (mem_image.mpr ⟨j,mem_univ _,hj⟩)
  61       simp [constraintsAt, hnone, hm]
  62     _ = ∏ j, ((constraintsAt τ A (τ j)).card : ℝ) / Fintype.card E := by
  63       exact prod_image hτ.injOn
  64     _ = _ := by simp only [constraintsAt_selected τ A hτ]

  66 /-- A finite family of injective time witnesses can be union-bounded without
  67 assuming independence between the different witness events. -/
  68  theorem injected_witness_union_bound {W : Type*} [Fintype W] [DecidableEq W]
  69     (τ : W → J → Fin T) (A : J → Finset E) :
  70     eventProbability (univ.biUnion fun w : W =>
  71       if Function.Injective (τ w) then requirementsEvent (τ w) A else ∅) ≤
  72       Fintype.card W * (∏ j, ((A j).card : ℝ) / Fintype.card E) := by
  73   classical
  74   calc
  75     _ ≤ ∑ w : W, eventProbability
  76         (if Function.Injective (τ w) then requirementsEvent (τ w) A else ∅) := witness_union_bound _ _
  77     _ ≤ ∑ _w : W, ∏ j, ((A j).card : ℝ) / Fintype.card E := by
  78       apply sum_le_sum
  79       intro w _
  80       split_ifs with h
  81       · rw [requirements_probability _ _ h]
  82       · have hp : (0:ℝ) ≤ ∏ j : J, ((A j).card : ℝ) / Fintype.card E :=
  83           prod_nonneg (fun _ _ => div_nonneg (Nat.cast_nonneg _) (Nat.cast_nonneg _))
  84         simpa [eventProbability] using hp
  85     _ = _ := by simp
