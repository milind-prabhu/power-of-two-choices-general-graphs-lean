   5 theorem expected_total_arrivals_bound {V E : Type*} [Fintype V] [DecidableEq V]
   6     [Fintype E] [DecidableEq E] [Nonempty E] (G : BaseGraph V E) (T N : ℕ)
   7     (hvertices : Fintype.card V ≤ N) : (T:ℝ) ≤ (N:ℝ)*expectedOpt G T := by
   8   have hh := expected_density_lower_bound G Finset.univ T
   9   have hm : (Fintype.card E:ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  10   simp only [BaseGraph.internalEdges, Finset.mem_univ, and_self, Finset.filter_true,
  11     Finset.card_univ, mul_div_cancel_right₀ _ hm] at hh
  12   exact hh.trans (mul_le_mul_of_nonneg_right (by exact_mod_cast hvertices) (expectedOpt_nonneg G T))

  14 theorem loglogFactor_square_bound (N M : ℕ) (hN : 2 ≤ N) (hM : M ≤ N^2) :
  15     loglogFactor M ≤ 2*loglogFactor N := by
  16   have hNr : (2:ℝ) ≤ N := by exact_mod_cast hN
  17   have hMr : (M:ℝ) ≤ (N:ℝ)^2 := by exact_mod_cast hM
  18   have hmax : max 2 (M:ℝ) ≤ (N:ℝ)^2 := max_le (by nlinarith) hMr
  19   have hlog := Real.log_le_log (by have := le_max_left (2:ℝ) (M:ℝ); linarith) hmax
  20   rw [Real.log_pow] at hlog
  21   norm_num only [Nat.cast_ofNat] at hlog
  22   have hlogN : 0 ≤ Real.log (N:ℝ) := Real.log_nonneg (by linarith)
  23   have hlogM : 0 ≤ Real.log (max 2 (M:ℝ)) := Real.log_nonneg (by have := le_max_left (2:ℝ) (M:ℝ); linarith)
  24   have hh := Real.log_le_log (by linarith : 0 < 1+Real.log (max 2 (M:ℝ)))
  25     (show 1+Real.log (max 2 (M:ℝ)) ≤ 2*(1+Real.log (N:ℝ)) by linarith)
  26   rw [Real.log_mul (by norm_num) (by positivity)] at hh
  27   have hh' : 0 ≤ Real.log (1+Real.log (N:ℝ)) := Real.log_nonneg (by linarith)
  28   have hlog2 : Real.log 2 ≤ 1 := by linarith [Real.log_two_lt_d9]
  29   unfold loglogFactor
  30   rw [max_eq_right hNr]
  31   norm_num only [Nat.cast_ofNat] at hlog
  32   linarith

  40 /-- All arrival horizons on every left-degree-bounded bipartite graph.
  41 The rate-adapted proof needs no auxiliary random clique augmentation. -/
  42 theorem exists_bipartite_policy_all_horizons (G : Graph L R) [Nonempty G] (N T : ℕ)
  43     (hvertices : Fintype.card (L ⊕ R) ≤ N)
  44     (hD : ∀ u, leftDegree G u ≤ 4*(graphBase G).referenceCap) :
  45     ∃ p : OnlinePolicy G, expectedCost (graphBase G) (onlineOrientation (T := T) p) ≤
  46       30000*loglogFactor N*expectedOpt (graphBase G) T := by
  47   have hfactor := loglogFactor_ge_one N
  48   have ho := expectedOpt_nonneg (graphBase G) T
  49   have htotal := expected_total_arrivals_bound (graphBase G) T N hvertices
  50   by_cases hT0 : T = 0
  51   · refine ⟨fun _ _ => true, (expectedCost_le_horizon (graphBase G) T _).trans ?_⟩
  52     rw [hT0, Nat.cast_zero]
  53     have hh := expectedOpt_nonneg (graphBase G) 0
  54     have hf : 0 ≤ loglogFactor N := by linarith
  55     positivity
  56   have hT : 0 < T := Nat.pos_of_ne_zero hT0
  57   have hopt := expectedOpt_ge_one (graphBase G) T hT
  58   by_cases hlog : 2 ≤ Real.log (N:ℝ)
  59   · have hN : 2 ≤ N := by
  60       by_contra hn
  61       have hz : N = 0 ∨ N = 1 := by omega
  62       rcases hz with rfl | rfl <;> norm_num at hlog
  63     have hNr : (0:ℝ) < N := by exact_mod_cast (show 0 < N by omega)
  64     by_cases hshort : (T:ℝ) ≤ Real.log (N:ℝ)
  65     · refine ⟨simpleGreedyPolicy G, (short_horizon_policy G N T hT hlog hshort).trans ?_⟩
  66       have hh : 0 ≤ loglogFactor N*expectedOpt (graphBase G) T := by positivity
  67       nlinarith
  68     by_cases hlong : (N:ℝ)*Real.log (N:ℝ) < T
  69     · refine ⟨fun _ _ => true, (allLeft_reference_bound G T N hvertices hD).trans ?_⟩
  70       have hlower : Real.log (N:ℝ) ≤ expectedOpt (graphBase G) T := by nlinarith
  71       have hm : expectedOpt (graphBase G) T ≤ loglogFactor N*expectedOpt (graphBase G) T := by nlinarith
  72       nlinarith
  73     let M := max N T
  74     have hTM : T ≤ M := le_max_right _ _
  75     have hNM : N ≤ M := le_max_left _ _
  76     have hverticesM : Fintype.card (L ⊕ R) ≤ M := hvertices.trans hNM
  77     have hlogM : 2 ≤ Real.log (M:ℝ) := hlog.trans (Real.log_le_log hNr (by exact_mod_cast hNM))
  78     have hlogMT : Real.log (M:ℝ) ≤ T := by
  79       by_cases hTN : T ≤ N
  80       · rw [show M = N by exact max_eq_left hTN]
  81         exact le_of_not_ge hshort
  82       · rw [show M = T by exact max_eq_right (le_of_not_ge hTN)]
  83         exact (Real.log_le_sub_one_of_pos (by exact_mod_cast hT)).trans (by linarith)
  84     have hMsquare : M ≤ N^2 := by
  85       have hNN : N ≤ N^2 := by nlinarith
  86       apply max_le hNN
  87       have hlogN := Real.log_le_sub_one_of_pos hNr
  88       have hbound : (T:ℝ) ≤ (N:ℝ)^2 := by
  89         have hh : (T:ℝ) ≤ (N:ℝ)*Real.log (N:ℝ) := le_of_not_gt hlong
  90         nlinarith
  91       exact_mod_cast hbound
  92     let d : ℝ := 2*(G.card:ℝ)/T
  93     have hMpos : (0:ℝ) < G.card := by
  94       exact_mod_cast (show 0 < G.card by simpa only [← Fintype.card_coe] using Fintype.card_pos (α := G))
  95     have hTr : (0:ℝ) < T := by exact_mod_cast hT
  96     have hd : 0 < d := by dsimp [d]; positivity
  97     have havg : 2*(G.card:ℝ) = (T:ℝ)*d := by dsimp [d]; field_simp
  98     by_cases hleft : ∀ u, (leftDegree G u:ℝ) ≤ d*Real.log (M:ℝ)
  99     · obtain ⟨p,hp⟩ := exists_rate_competitive_policy G M T d hT hd hlogM hlogMT hTM hleft hD hverticesM havg
 100       refine ⟨p, hp.trans ?_⟩
 101       have hh := mul_le_mul_of_nonneg_right (loglogFactor_square_bound N M hN hMsquare)
