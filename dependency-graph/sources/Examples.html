  12 /-- Three parallel arrivals need optimum load two, rather than density 3/2. -/
  13 theorem repeated_opt : opt repeated = 2 := by
  14   let a : Orientation 3 := fun t => decide (t.val < 2)
  15   have hm : maxLoad repeated a = 2 := by decide
  16   have hu := opt_le_maxLoad repeated a
  17   rw [hm] at hu
  18   have hl := total_le_card_mul_opt repeated
  19   norm_num at hl
  20   omega
