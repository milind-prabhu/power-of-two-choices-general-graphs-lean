  12 noncomputable def shapeRate (h : ℕ) : ℝ := 1/(8*h)
  13 
  14 noncomputable def branchTarget (D h : ℕ) (z : ℝ) : ℝ := (shapeRate h)^2/((D:ℝ)*z+1)
  15 
  16 theorem shapeRate_pos (h : ℕ) (hh : 0 < h) : 0 < shapeRate h := by
  17   unfold shapeRate
  18   positivity
  19 
  20 theorem shapeRate_le_one (h : ℕ) (hh : 0 < h) : shapeRate h ≤ 1 := by
  21   have hhr : (1:ℝ) ≤ h := by exact_mod_cast hh
  22   unfold shapeRate
  23   apply (div_le_iff₀ (by positivity : (0:ℝ) < 8*h)).mpr
  24   linarith
  25 
  26 theorem shapeRate_class_factor (h : ℕ) (hh : 0 < h) :
  27     4*(h:ℝ)*shapeRate h = 1/2 := by
  28   have hne : (h:ℝ) ≠ 0 := by exact_mod_cast Nat.ne_of_gt hh
  29   unfold shapeRate
  30   field_simp
  31   ring
  32 
  33 theorem branchTarget_pos (D h : ℕ) (z : ℝ) (hh : 0 < h) (hz : 0 ≤ z) :
  34     0 < branchTarget D h z := by
  35   unfold branchTarget
  36   exact div_pos (sq_pos_of_pos (shapeRate_pos h hh)) (by positivity)
  37 
  38 theorem branchTarget_le_rate (D h : ℕ) (z : ℝ) (hh : 0 < h) (hz : 0 ≤ z) :
  39     branchTarget D h z ≤ shapeRate h := by
  40   have hr := shapeRate_pos h hh
  41   have hr1 := shapeRate_le_one h hh
  42   unfold branchTarget
  43   apply (div_le_iff₀ (by positivity : (0:ℝ) < (D:ℝ)*z+1)).mpr
  44   have hDz : 0 ≤ (D:ℝ)*z := mul_nonneg (Nat.cast_nonneg _) hz
  45   have hmul := mul_nonneg hr.le hDz
  46   nlinarith
  47 
  48 theorem branchTarget_pair (D h : ℕ) (z : ℝ) (hh : 0 < h) (hz : 0 ≤ z) :
  49     (D:ℝ)*z*branchTarget D h z ≤ (shapeRate h)^2 := by
  50   have hb := (branchTarget_pos D h z hh hz).le
  51   have heq : branchTarget D h z * ((D:ℝ)*z+1) = (shapeRate h)^2 := by
  52     unfold branchTarget
  53     exact div_mul_cancel₀ _ (ne_of_gt (by positivity : (0:ℝ) < (D:ℝ)*z+1))
  54   nlinarith

  60 theorem class_card_pos (encode : E → Arrival L R C) : 0 < Fintype.card C := by
  61   classical
  62   exact Fintype.card_pos_iff.mpr ⟨(encode (Classical.choice (inferInstance : Nonempty E))).cls⟩

  64 noncomputable def automaticThresholds (DL DR : C → ℕ) (D T : ℕ) (f : C → ℝ) : C → ℕ :=
  65   explicitThresholds (E := E) DL DR T f
  66     (branchTarget D (Fintype.card C) ((T:ℝ)/Fintype.card E))

  68 /-- Canonical-shape geometric decay for an explicit integer threshold vector.
  69 Every local numerical inequality is proved from the parameter definitions. -/
  70 theorem heavyShape_geometric_automatic (H : C → Peeling.Graph L R)
  71     (encode : E → Arrival L R C) (DL DR : C → ℕ) (D T m : ℕ) (f : C → ℝ)
  72     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
  73     (hDL : ∀ u c, (sourceBucket encode u c).card ≤ DL c)
  74     (hL : ∀ c u, Peeling.leftDegree (H c) u ≤ D)
  75     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c)
  76     (hT : 0 < T) (hf : ∀ c, 1 < f c) :
  77     eventProbability (heavyShapeEvent H encode (automaticThresholds (E := E) DL DR D T f) T m) ≤
  78       (Fintype.card R : ℝ)/2^m := by
  79   have hh := class_card_pos encode
  80   have hz : 0 ≤ (T:ℝ)/Fintype.card E := by positivity
  81   have hdec := heavyShape_decay_explicit H encode DL DR D T m f hsimple hDL hL hR
  82     (branchTarget D (Fintype.card C) ((T:ℝ)/Fintype.card E)) (shapeRate (Fintype.card C)) hT
  83     (branchTarget_pos _ _ _ hh hz) (shapeRate_pos _ hh).le
  84     (branchTarget_le_rate _ _ _ hh hz) (branchTarget_pair _ _ _ hh hz) hf
  85   exact heavyShape_geometric_tail H encode _ T m _ (shapeRate_pos _ hh).le hdec
  86     (shapeRate_class_factor _ hh).le

  90     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))

  93     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c)

  98   intro j _

 101 /-- Component tail for the concrete threshold vector. The only combinatorial
 102 coverage premise is explicitly visible; no canonical probability bound or
 103 local decay inequality is assumed. D bounds base-piece left degrees, while
 104 d bounds sampled residual left degrees outside the degree exception. -/

 106     (encode : E → Arrival L R C) (DL DR : C → ℕ) (D T d k B N c : ℕ) (f : C → ℝ)
 107     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
 108     (hDL : ∀ u c, (sourceBucket encode u c).card ≤ DL c)
 109     (hL : ∀ c u, Peeling.leftDegree (H c) u ≤ D)
 110     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c)
 111     (hT : 0 < T) (hf : ∀ c, 1 < f c)
 112     (hN : 0 < N) (hLN : Fintype.card L ≤ N) (hRN : Fintype.card R ≤ N)
 113     (hd : polynomialCutoff N c ≤ d+1) (hk : polynomialCutoff N c ≤ k)
 114     (hmean : ∀ u, Real.exp 1*T*(((leftSourceBucket encode u).card : ℝ)/Fintype.card E)/(d+1) ≤ 1/2)
