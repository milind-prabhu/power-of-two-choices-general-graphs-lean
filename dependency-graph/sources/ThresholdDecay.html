   8  theorem inverse_power_le_target (f target : ℝ) (k : ℕ)
   9     (hf : 1 < f) (ht : 0 < target)
  10     (hk : Real.log (1/target) / Real.log f ≤ (k:ℝ)) :
  11     ((1:ℝ)/f)^k ≤ target := by
  12   have hfpos : 0 < f := by linarith
  13   apply (Real.log_le_log_iff (pow_pos (div_pos zero_lt_one hfpos) _) ht).mp
  14   rw [Real.log_pow, Real.log_div (by norm_num) (ne_of_gt hfpos), Real.log_one]
  15   have hh := (div_le_iff₀ (Real.log_pos hf)).mp hk
  16   rw [Real.log_div (by norm_num) (ne_of_gt ht), Real.log_one] at hh
  17   linarith
  18 
  19  theorem power_le_target_of_threshold (x f target : ℝ) (k : ℕ)
  20     (hx : 0 ≤ x) (hf : 1 < f) (ht : 0 < target)
  21     (hbase : x ≤ 1/f) (hk : Real.log (1/target) / Real.log f ≤ (k:ℝ)) :
  22     x^k ≤ target :=
  23   (pow_le_pow_left₀ hx hbase k).trans (inverse_power_le_target f target k hf ht hk)
  24 
  25 end GraphChoices

  34  theorem bucketDecay_small_of_threshold (α DL DR : C → ℕ) (f : C → ℝ) (T : ℕ)
  35     (b : ℝ) (hT : 0 < T) (hb : 0 < b) (hf : ∀ c, 1 < f c)
  36     (hbase : ∀ c, Real.exp 1*T*(DL c : ℝ)/Fintype.card E/α c ≤ 1/f c)
  37     (hthreshold : ∀ c, 0 < DR c →
  38       Real.log (((DR c : ℝ)*((T:ℝ)/Fintype.card E))/b) / Real.log (f c) ≤ (α c : ℝ)) :
  39     ∀ c, (DR c : ℝ)*((T:ℝ)/Fintype.card E)*bucketDecay (E := E) α DL T c ≤ b := by
  40   intro c
  41   by_cases hDR : DR c = 0
  42   · simp [hDR, le_of_lt hb]
  43   have hp : (0:ℝ) < (DR c : ℝ)*((T:ℝ)/Fintype.card E) := by
  44     have hd : (0:ℝ) < DR c := by exact_mod_cast Nat.pos_of_ne_zero hDR
  45     have ht : (0:ℝ) < T := by exact_mod_cast hT
  46     exact mul_pos hd (div_pos ht (by exact_mod_cast Fintype.card_pos))
  47   have htarget : 0 < b / ((DR c : ℝ)*((T:ℝ)/Fintype.card E)) := div_pos hb hp
  48   have hlog : Real.log (1 / (b / ((DR c : ℝ)*((T:ℝ)/Fintype.card E)))) /
  49       Real.log (f c) ≤ (α c : ℝ) := by
  50     simpa [one_div_div] using hthreshold c (Nat.pos_of_ne_zero hDR)
  51   have hdecay : bucketDecay (E := E) α DL T c ≤ b / ((DR c : ℝ)*((T:ℝ)/Fintype.card E)) :=
  52     GraphChoices.power_le_target_of_threshold _ _ _ _ (by positivity) (hf c) htarget (hbase c) hlog
  53   calc
  54     _ ≤ ((DR c : ℝ)*((T:ℝ)/Fintype.card E)) * (b / ((DR c : ℝ)*((T:ℝ)/Fintype.card E))) :=
  55       mul_le_mul_of_nonneg_left hdecay hp.le
  56     _ = b := by field_simp

  58 /-- Conditional fixed-pattern decay with explicit base and logarithmic
  59 threshold inequalities. Instantiation by the paper's alpha_i remains separate. -/
  60  theorem residualPattern_decay_of_thresholds (p : ParentPattern C m) (H : C → Peeling.Graph L R)
  61     (encode : E → Arrival L R C) (α : C → ℕ) (DL DR : C → ℕ) (D : ℕ) (f : C → ℝ)
  62     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
  63     (hDL : ∀ u c, (sourceBucket encode u c).card ≤ DL c)
  64     (hL : ∀ c u, Peeling.leftDegree (H c) u ≤ D)
  65     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c)
  66     (b r : ℝ) (hT : 0 < T) (hb : 0 < b) (hr : 0 ≤ r) (hbr : b ≤ r)
  67     (hpair : (D : ℝ)*((T:ℝ)/Fintype.card E)*b ≤ r^2)
  68     (hf : ∀ c, 1 < f c)
  69     (hbase : ∀ c, Real.exp 1*T*(DL c : ℝ)/Fintype.card E/α c ≤ 1/f c)
  70     (hthreshold : ∀ c, 0 < DR c →
  71       Real.log (((DR c : ℝ)*((T:ℝ)/Fintype.card E))/b) / Real.log (f c) ≤ (α c : ℝ))
  72     (hheavy : m ≤ 2*(univ.filter fun i : Fin m => p.childLeft i = true).card) :
  73     eventProbability (residualPatternEvent p H encode α T) ≤ (Fintype.card R : ℝ)*r^m :=
  74   residualPattern_probability_decay p H encode α DL DR D hsimple hDL hL hR b r hb.le hr hbr hpair
  75     (bucketDecay_small_of_threshold α DL DR f T b hT hb hf hbase hthreshold) hheavy
