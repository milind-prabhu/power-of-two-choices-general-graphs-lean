   4 is chosen from an actual source-bucket bound and the logarithmic polynomial

  14 noncomputable def sampledDegreeCutoff (M T N c : ℕ) : ℕ :=
  15   ⌈2*Real.exp 1*T*(M:ℝ)/Fintype.card E⌉₊ + polynomialCutoff N c

  17 theorem sampledDegreeCutoff_ge (M T N c : ℕ) :
  18     polynomialCutoff N c ≤ sampledDegreeCutoff (E := E) M T N c + 1 := by
  19   unfold sampledDegreeCutoff
  20   omega

  22 theorem sampledDegreeCutoff_mean (encode : E → Arrival L R C) (M T N c : ℕ)
  23     (hM : ∀ u, (leftSourceBucket encode u).card ≤ M) :
  24     ∀ u, Real.exp 1*T*(((leftSourceBucket encode u).card : ℝ)/Fintype.card E)/
  25       ((sampledDegreeCutoff (E := E) M T N c : ℕ)+1 : ℝ) ≤ 1/2 := by
  26   intro u
  27   have hbucket : ((leftSourceBucket encode u).card : ℝ) ≤ M := by exact_mod_cast hM u
  28   have hmean := mul_le_mul_of_nonneg_left
  29     (div_le_div_of_nonneg_right hbucket (Nat.cast_nonneg (Fintype.card E)))
  30     (show 0 ≤ Real.exp 1*(T:ℝ) by positivity)
  31   have hc := Nat.le_ceil (2*Real.exp 1*T*(M:ℝ)/Fintype.card E)
  32   have hcut : ⌈2*Real.exp 1*T*(M:ℝ)/Fintype.card E⌉₊ ≤ sampledDegreeCutoff (E := E) M T N c := by
  33     unfold sampledDegreeCutoff
  34     omega
  35   have hcut' : (⌈2*Real.exp 1*T*(M:ℝ)/Fintype.card E⌉₊ : ℝ) ≤ sampledDegreeCutoff (E := E) M T N c := by
  36     exact_mod_cast hcut
  37   apply (div_le_iff₀ (by positivity : (0:ℝ) < (sampledDegreeCutoff (E := E) M T N c : ℝ)+1)).mpr
  38   have heq : Real.exp 1*T*((M:ℝ)/Fintype.card E) = Real.exp 1*T*(M:ℝ)/Fintype.card E := by ring
  39   rw [heq] at hmean
  40   have htwo : 2*(Real.exp 1*T*(M:ℝ)/Fintype.card E) ≤
  41       (⌈2*Real.exp 1*T*(M:ℝ)/Fintype.card E⌉₊ : ℝ) := by
  42     convert hc using 1 <;> ring
  43   linarith

  45 /-- Actual component failure at most 3/N^c with a concrete cutoff. The graph
  46 and sampling degree bounds are explicit input data, not probability premises. -/
  47 theorem concrete_component_tail (H : C → Peeling.Graph L R)
  48     (encode : E → Arrival L R C) (DL DR : C → ℕ) (D M T N c : ℕ) (f : C → ℝ)
  49     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
  50     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
  51     (hDL : ∀ u a, (sourceBucket encode u a).card ≤ DL a)
  52     (hL : ∀ a u, Peeling.leftDegree (H a) u ≤ D)
  53     (hR : ∀ a v, Peeling.rightDegree (H a) v ≤ DR a)
  54     (hM : ∀ u, (leftSourceBucket encode u).card ≤ M)
  55     (hT : 0 < T) (hf : ∀ a, 1 < f a)
  56     (hN : 0 < N) (hLN : Fintype.card L ≤ N) (hRN : Fintype.card R ≤ N) :
  57     eventProbability (largeComponentEvent encode (automaticThresholds (E := E) DL DR D T f)
  58       T ((sampledDegreeCutoff (E := E) M T N c+1)*(polynomialCutoff N c+1))) ≤ 3/(N:ℝ)^c :=
  59   largeComponent_polynomial_automatic H encode DL DR D T _ _ N c f hsimple hsource hDL hL hR hT hf
  60     hN hLN hRN (sampledDegreeCutoff_ge M T N c) le_rfl (sampledDegreeCutoff_mean encode M T N c hM)
  61 
  62 theorem concrete_expected_load (H : C → Peeling.Graph L R)
  63     (encode : E → Arrival L R C) (DL DR : C → ℕ) (D M T N c : ℕ) (f : C → ℝ)
  64     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
  65     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
  66     (hDL : ∀ u a, (sourceBucket encode u a).card ≤ DL a)
  67     (hL : ∀ a u, Peeling.leftDegree (H a) u ≤ D)
  68     (hR : ∀ a v, Peeling.rightDegree (H a) v ≤ DR a)
  69     (hM : ∀ u, (leftSourceBucket encode u).card ≤ M)
  70     (hT : 0 < T) (hf : ∀ a, 1 < f a)
  71     (hN : 0 < N) (hLN : Fintype.card L ≤ N) (hRN : Fintype.card R ≤ N) :
  72     expectedTotalMaxLoad encode (automaticThresholds (E := E) DL DR D T f) T ≤
  73       (∑ a, (automaticThresholds (E := E) DL DR D T f a : ℝ)) +
  74       (2*(Nat.log 2 ((sampledDegreeCutoff (E := E) M T N c+1)*(polynomialCutoff N c+1))+1) : ℕ)*
  75         expectedReplayOpt encode T + 3*T/(N:ℝ)^c :=
  76   expected_load_polynomial_automatic H encode DL DR D T _ _ N c f hsimple hsource hDL hL hR hT hf
  77     hN hLN hRN (sampledDegreeCutoff_ge M T N c) le_rfl (sampledDegreeCutoff_mean encode M T N c hM)

  79 theorem sampledDegreeCutoff_le_three_cutoffs (M T N c : ℕ)
  80     (hmean : Real.exp 1*T*(M:ℝ)/Fintype.card E ≤ (polynomialCutoff N c : ℝ)) :
  81     sampledDegreeCutoff (E := E) M T N c ≤ 3*polynomialCutoff N c := by
  82   have hceil : ⌈2*Real.exp 1*T*(M:ℝ)/Fintype.card E⌉₊ ≤ 2*polynomialCutoff N c := by
  83     apply Nat.ceil_le.mpr
  84     push_cast
  85     calc
  86       _ = 2*(Real.exp 1*T*(M:ℝ)/Fintype.card E) := by ring
  87       _ ≤ _ := mul_le_mul_of_nonneg_left hmean (by norm_num)
  88   unfold sampledDegreeCutoff
  89   omega

  91 theorem largeComponentEvent_antitone (encode : E → Arrival L R C) (α : C → ℕ)
  92     (T a b : ℕ) (hab : a ≤ b) :
  93     largeComponentEvent encode α T b ⊆ largeComponentEvent encode α T a := by
  94   classical
  95   intro ω hω
  96   obtain ⟨v, hv⟩ := (mem_filter.mp hω).2
  97   exact mem_filter.mpr ⟨mem_univ _, v, hab.trans_lt hv⟩

  99 /-- The log-squared component bound, with all tree-coverage and numerical
 100 tail conditions discharged. The sole additional mean condition is an explicit
 101 upper bound on the expected sampled left degree (including its factor e). -/
 102 theorem log_squared_component_tail (H : C → Peeling.Graph L R)
 103     (encode : E → Arrival L R C) (DL DR : C → ℕ) (D M T N c : ℕ) (f : C → ℝ)
 104     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
 105     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
 106     (hDL : ∀ u a, (sourceBucket encode u a).card ≤ DL a)
 107     (hL : ∀ a u, Peeling.leftDegree (H a) u ≤ D)
 108     (hR : ∀ a v, Peeling.rightDegree (H a) v ≤ DR a)
 109     (hM : ∀ u, (leftSourceBucket encode u).card ≤ M)
 110     (hT : 0 < T) (hf : ∀ a, 1 < f a)
 111     (hN : 0 < N) (hLN : Fintype.card L ≤ N) (hRN : Fintype.card R ≤ N)
 112     (hmean : Real.exp 1*T*(M:ℝ)/Fintype.card E ≤ (polynomialCutoff N c : ℝ)) :
 113     eventProbability (largeComponentEvent encode (automaticThresholds (E := E) DL DR D T f)
 114       T (3*(polynomialCutoff N c+1)^2)) ≤ 3/(N:ℝ)^c := by
 115   have hd := sampledDegreeCutoff_le_three_cutoffs M T N c hmean
 116   have hsize : (sampledDegreeCutoff (E := E) M T N c+1)*(polynomialCutoff N c+1) ≤
 117       3*(polynomialCutoff N c+1)^2 := by
 118     have hm := Nat.mul_le_mul_right (polynomialCutoff N c+1) (Nat.add_le_add_right hd 1)
 119     nlinarith
 120   apply (eventProbability_mono (largeComponentEvent_antitone encode _ T _ _ hsize)).trans
 121   exact concrete_component_tail H encode DL DR D M T N c f hsimple hsource hDL hL hR hM hT hf hN hLN hRN

 123 theorem log_squared_expected_load (H : C → Peeling.Graph L R)
 124     (encode : E → Arrival L R C) (DL DR : C → ℕ) (D M T N c : ℕ) (f : C → ℝ)
 125     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
 126     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
 127     (hDL : ∀ u a, (sourceBucket encode u a).card ≤ DL a)
 128     (hL : ∀ a u, Peeling.leftDegree (H a) u ≤ D)
 129     (hR : ∀ a v, Peeling.rightDegree (H a) v ≤ DR a)
 130     (hM : ∀ u, (leftSourceBucket encode u).card ≤ M)
 131     (hT : 0 < T) (hf : ∀ a, 1 < f a)
 132     (hN : 0 < N) (hLN : Fintype.card L ≤ N) (hRN : Fintype.card R ≤ N)
 133     (hmean : Real.exp 1*T*(M:ℝ)/Fintype.card E ≤ (polynomialCutoff N c : ℝ)) :

 136       (2*(Nat.log 2 (3*(polynomialCutoff N c+1)^2)+1) : ℕ)*expectedReplayOpt encode T + 3*T/(N:ℝ)^c := by
 137   have ht := log_squared_component_tail H encode DL DR D M T N c f hsimple hsource hDL hL hR hM hT hf
