   3 /-! The oriented vertex-copy construction as a concrete simple bipartite
   4 edge set, together with its proved degree certificate and label bijection. -/
   5 namespace GraphChoices
   6 open Finset
   7 variable {V E : Type*} [Fintype V] [DecidableEq V]
   8   [Fintype E] [DecidableEq E] [Nonempty E]
   9 
  10 def orientedPair (G : BaseGraph V E) (b : E → Bool) (x : E) : V × V :=
  11   (if b x then G.left x else G.right x, if b x then G.right x else G.left x)
  12 
  13 def splitGraph (G : BaseGraph V E) (b : E → Bool) : Peeling.Graph V V :=
  14   univ.image (orientedPair G b)
  15 
  16 theorem orientedPair_injective (G : BaseGraph V E) (hsimple : G.IsSimple) (b : E → Bool) :
  17     Function.Injective (orientedPair G b) := by
  18   intro x y hxy
  19   have h1 := congrArg Prod.fst hxy
  20   have h2 := congrArg Prod.snd hxy
  21   cases hx : b x <;> cases hy : b y <;>
  22     simp only [orientedPair, hx, hy, Bool.false_eq_true, ↓reduceIte] at h1 h2
  23   · exact hsimple.2 x y (Or.inl ⟨h2,h1⟩)
  24   · exact hsimple.2 x y (Or.inr ⟨h2,h1⟩)
  25   · exact hsimple.2 x y (Or.inr ⟨h1,h2⟩)
  26   · exact hsimple.2 x y (Or.inl ⟨h1,h2⟩)
  27 
  28 noncomputable def splitEdgeEquiv (G : BaseGraph V E) (hsimple : G.IsSimple) (b : E → Bool) :
  29     E ≃ splitGraph G b := Equiv.ofBijective
  30       (fun x => ⟨orientedPair G b x, mem_image.mpr ⟨x,mem_univ _,rfl⟩⟩)
  31       ⟨fun _ _ h => orientedPair_injective G hsimple b (congrArg Subtype.val h), by
  32         intro y
  33         obtain ⟨x,_,hx⟩ := mem_image.mp y.property
  34         exact ⟨x,Subtype.ext hx⟩⟩
  35 
  36 theorem splitEdgeEquiv_left (G : BaseGraph V E) (hsimple : G.IsSimple) (b : E → Bool) (x : E) :
  37     (Peeling.graphBase (splitGraph G b)).left (splitEdgeEquiv G hsimple b x) = (G.split b).left x := rfl
  38 
  39 theorem splitEdgeEquiv_right (G : BaseGraph V E) (hsimple : G.IsSimple) (b : E → Bool) (x : E) :
  40     (Peeling.graphBase (splitGraph G b)).right (splitEdgeEquiv G hsimple b x) = (G.split b).right x := rfl
  41 
  42 theorem exists_preprocessed_split (G : BaseGraph V E) (hsimple : G.IsSimple) :
  43     ∃ b : E → Bool, ∀ u,
  44       Peeling.leftDegree (splitGraph G b) u ≤
  45         4*(Peeling.graphBase (splitGraph G b)).referenceCap := by
  46   classical
  47   obtain ⟨b,hb⟩ := exists_reference_source_orientation G
  48   refine ⟨b, fun u => ?_⟩
  49   let H := splitGraph G b
  50   let e := splitEdgeEquiv G hsimple b
  51   letI : Nonempty H := ⟨e (Classical.choice (inferInstance : Nonempty E))⟩
  52   have hc : (univ.filter fun x : E => (if b x then G.left x else G.right x) = u).card =
  53       (Peeling.leftBucket H u).card := by
  54     apply card_equiv e
  55     intro x
  56     simp only [Peeling.leftBucket, mem_filter, mem_univ, true_and]
  57     rfl
  58   have hleft : Peeling.leftDegree H u ≤ G.referenceCap := by
  59     rw [← Peeling.leftBucket_card, ← hc]
  60     exact hb u
  61   have hcap : (G.split b).referenceCap = (Peeling.graphBase H).referenceCap :=
  62     relabel_referenceCap (G.split b) (Peeling.graphBase H) e
  63       (splitEdgeEquiv_left G hsimple b) (splitEdgeEquiv_right G hsimple b)
  64   have hs := source_split_reference G b
  65   rw [hcap] at hs
  66   change Peeling.leftDegree H u ≤ 4*(Peeling.graphBase H).referenceCap
  67   omega
  68 
  69 end GraphChoices
