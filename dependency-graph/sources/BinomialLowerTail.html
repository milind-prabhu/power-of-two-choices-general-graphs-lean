   7 theorem average_pow_hits (A : Finset E) (T : ℕ) (r : ℝ) :
   8     average (fun ω : Fin T → E => r^(hitCount A ω)) =
   9       (1+((A.card:ℝ)/Fintype.card E)*(r-1))^T := by
  10   have hp (ω : Fin T → E) : r^(hitCount A ω) = ∏ t, if ω t ∈ A then r else 1 := by
  11     simp only [prod_ite, prod_const_one, mul_one, prod_const, hitCount]
  12   have hfilter : (univ.filter fun e : E => e ∈ A) = A := by ext e; simp
  13   have hs : (∑ e : E, if e ∈ A then r else 1) = Fintype.card E+(r-1)*A.card := by
  14     have hh (e : E) : (if e ∈ A then r else 1) = 1+(r-1)*(if e ∈ A then (1:ℝ) else 0) := by
  15       split <;> ring
  16     calc
  17       _ = ∑ e : E, (1+(r-1)*(if e ∈ A then (1:ℝ) else 0)) := sum_congr rfl (fun e _ => hh e)
  18       _ = _ := by simp only [sum_add_distrib, sum_const, card_univ, nsmul_eq_mul, mul_one, ← mul_sum, sum_boole, hfilter]
  19   simp only [average, hp]
  20   rw [← Fintype.prod_sum (fun (_ : Fin T) (e : E) => if e ∈ A then r else 1)]
  21   simp only [hs, prod_const, card_univ, Fintype.card_fin, Fintype.card_fun, Nat.cast_pow, ← div_pow]
  22   have hm : (Fintype.card E:ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  23   congr 1
  24   field_simp

  26 theorem hitLowerTail_probability (A : Finset E) (T k : ℕ) :
  27     eventProbability (univ.filter fun ω : Fin T → E => hitCount A ω < k) ≤
  28       Real.exp ((k:ℝ)*Real.log 2-(T:ℝ)*((A.card:ℝ)/Fintype.card E)/2) := by
  29   let bad := univ.filter fun ω : Fin T → E => hitCount A ω < k
  30   have hm : average (fun ω : Fin T → E => (if ω ∈ bad then (1:ℝ) else 0)*(1/2)^k) ≤
  31       average (fun ω : Fin T → E => (1/2:ℝ)^(hitCount A ω)) := by
  32     apply average_mono
  33     intro ω
  34     split_ifs with hω
  35     · simp only [one_mul]
  36       exact pow_le_pow_of_le_one (by norm_num) (by norm_num) ((mem_filter.mp hω).2.le)
  37     · simp only [zero_mul]
  38       positivity
  39   have heq : average (fun ω : Fin T → E => (if ω ∈ bad then (1:ℝ) else 0)*(1/2)^k) =
  40       (1/2:ℝ)^k*eventProbability bad := by
  41     conv_lhs => arg 1; ext ω; rw [mul_comm]
  42     rw [average_mul, average_unit_indicator]
  43   rw [heq, average_pow_hits] at hm
  44   have hexp : (1+((A.card:ℝ)/Fintype.card E)*((1/2:ℝ)-1))^T ≤
  45       Real.exp (-(T:ℝ)*((A.card:ℝ)/Fintype.card E)/2) := by
  46     have hp : 0 ≤ 1+((A.card:ℝ)/Fintype.card E)*((1/2:ℝ)-1) := by
  47       have hh : (A.card:ℝ)/Fintype.card E ≤ 1 := eventProbability_le_one A
  48       linarith
  49     calc
  50       _ ≤ (Real.exp (-((A.card:ℝ)/Fintype.card E)/2))^T := by
  51         apply pow_le_pow_left₀ hp
  52         have hh := Real.add_one_le_exp (-((A.card:ℝ)/Fintype.card E)/2)
  53         linarith
  54       _ = _ := by rw [← Real.exp_nat_mul]; congr 1; ring
  55   have htwo : (1/2:ℝ)^k*Real.exp ((k:ℝ)*Real.log 2) = 1 := by
  56     rw [Real.exp_nat_mul, Real.exp_log (by norm_num), ← mul_pow]
  57     norm_num
  58   have hh := mul_le_mul_of_nonneg_left (hm.trans hexp) (le_of_lt (Real.exp_pos ((k:ℝ)*Real.log 2)))
  59   have hright : Real.exp ((k:ℝ)*Real.log 2)*Real.exp (-(T:ℝ)*((A.card:ℝ)/Fintype.card E)/2) =
  60       Real.exp ((k:ℝ)*Real.log 2-(T:ℝ)*((A.card:ℝ)/Fintype.card E)/2) := by rw [← Real.exp_add]; congr 1; ring
  61   rw [← mul_assoc, mul_comm (Real.exp _), htwo, one_mul, hright] at hh
  62   exact hh

  64 theorem coordinate_event_probability {Ω : Type*} [Fintype Ω] [DecidableEq Ω] [Nonempty Ω]
  65     (B : Finset Ω) {k : ℕ} (i : Fin k) :
