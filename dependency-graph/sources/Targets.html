  11 namespace GraphChoices

  13 variable {V E : Type*} [Fintype V] [DecidableEq V]
  14   [Fintype E] [DecidableEq E] [Nonempty E] {T : ℕ}
  15 
  16 /-- No loops and no repeated unordered base edges. -/
  17 def BaseGraph.IsSimple (G : BaseGraph V E) : Prop :=
  18   (∀ e, G.left e ≠ G.right e) ∧
  19     ∀ e f, ((G.left e = G.left f ∧ G.right e = G.right f) ∨
  20       (G.left e = G.right f ∧ G.right e = G.left f)) → e = f
  21 
  22 abbrev OnlinePolicy (E : Type*) := List E → E → Bool
  23 
  24 def onlineOrientation (p : OnlinePolicy E) (ω : Fin T → E) : Orientation T :=
  25   fun t => p ((List.ofFn ω).take t.val) (ω t)
  26 
  27 def collapsePolicy (b : E → Bool) (p : OnlinePolicy E) : OnlinePolicy E :=
  28   fun history e => if b e then p history e else !(p history e)
  29 
  30 omit [Fintype E] [DecidableEq E] [Nonempty E] in
  31  theorem collapsePolicy_orientation (b : E → Bool) (p : OnlinePolicy E) (ω : Fin T → E) :
  32     onlineOrientation (collapsePolicy b p) ω =
  33       transport (fun t => b (ω t)) (onlineOrientation p ω) := rfl
  34 
  35 /-- The expected transfer also preserves the explicit online-policy type. -/
  36  theorem online_expected_transfer (G : BaseGraph V E) (b : E → Bool)
  37     (p : OnlinePolicy E) (c : ℝ) (hc : 0 ≤ c)
  38     (hcost : expectedCost (T := T) (G.split b) (onlineOrientation p) ≤
  39       c * expectedOpt (G.split b) T) :
  40     expectedCost (T := T) G (onlineOrientation (collapsePolicy b p)) ≤
  41       (2*c) * expectedOpt G T := by
  42   exact expected_competitive_transfer G b (onlineOrientation p) c hc hcost
  43 
  44 /-- Positive for small n and asymptotically Theta(log log n). -/
  45 noncomputable def loglogFactor (n : ℕ) : ℝ :=
  46   1 + Real.log (1 + Real.log (max 2 (n : ℝ)))
  47 
  48 /-- Theorem 1.1 specification, proved by GraphChoices.mainTheorem, with a
  49 horizon-aware deterministic online policy and
  50 an explicit bounded-at-small-n interpretation of O(log log n).
  51 Edge labels use Fin(m+1) to ensure that the sampling space is nonempty. -/
  52 def MainTheorem : Prop :=
  53   ∃ C : ℝ, 0 < C ∧
  54     ∀ (n m : ℕ) (G : BaseGraph (Fin n) (Fin (m+1))), G.IsSimple →
  55       ∀ T : ℕ, ∃ p : OnlinePolicy (Fin (m+1)),
  56         expectedCost G (onlineOrientation (T := T) p) ≤
  57           C * loglogFactor n * expectedOpt G T
  58 
  59 end GraphChoices
