   4 /-! The degree condition used in the right-leaf pruning reduction: deleting
   5 threshold occurrences cannot increase any sampled left neighborhood. Its
   6 probability tail follows in the same i.i.d. space as the pattern bounds. -/
   7 namespace GraphChoices.ThresholdGreedy
   8 open Finset
   9 variable {E L R C : Type*} [Fintype E] [DecidableEq E] [Nonempty E]
  10   [Fintype L] [Fintype R] [DecidableEq L] [DecidableEq R] [DecidableEq C]
  11 
  12 def listLeftNeighbors (es : List (Arrival L R C)) (u : L) : Finset R :=
  13   ((es.filter fun e => decide (e.left = u)).map Arrival.right).toFinset
  14 
  15 omit [Fintype L] [Fintype R] [DecidableEq C] in
  16 theorem listLeftNeighbors_card_le (es : List (Arrival L R C)) (u : L) :
  17     (listLeftNeighbors es u).card ≤ es.countP (fun e => decide (e.left = u)) := by
  18   exact (List.toFinset_card_le _).trans_eq (by simp [List.countP_eq_length_filter])
  19 
  20 omit [Fintype L] [Fintype R] [DecidableEq C] in
  21 theorem listLeftNeighbors_mono {xs ys : List (Arrival L R C)} (h : xs.Sublist ys) (u : L) :
  22     listLeftNeighbors xs u ⊆ listLeftNeighbors ys u := by
  23   intro v hv
  24   exact List.mem_toFinset.mpr (((h.filter _).map Arrival.right).subset (List.mem_toFinset.mp hv))
  25 
  26 theorem residual_left_neighbors_subset (α : C → ℕ) (es : List (Arrival L R C)) (u : L) :
  27     listLeftNeighbors (residualArrivals (trace α es)) u ⊆ listLeftNeighbors es u :=
  28   listLeftNeighbors_mono (residual_sublist α es) u
  29 
  30 def leftSourceBucket (encode : E → Arrival L R C) (u : L) : Finset E :=
  31   univ.filter fun e => (encode e).left = u
  32 
  33 omit [Nonempty E] [Fintype L] [Fintype R] [DecidableEq R] [DecidableEq C] in
  34 theorem sample_left_count_eq_hits (encode : E → Arrival L R C) {T : ℕ}
  35     (ω : Fin T → E) (u : L) :
  36     (sampleArrivals encode ω).countP (fun e => decide (e.left = u)) =
  37       hitCount (leftSourceBucket encode u) ω := by
  38   unfold sampleArrivals
  39   rw [countP_ofFn_eq_sum]
  40   simp only [hitCount, leftSourceBucket, card_eq_sum_ones, sum_filter,
  41     mem_filter, mem_univ, true_and, decide_eq_true_eq]
  42 
  43 theorem residual_left_degree_le_hits (encode : E → Arrival L R C) (α : C → ℕ) {T : ℕ}
  44     (ω : Fin T → E) (u : L) :
  45     (listLeftNeighbors (residualArrivals (trace α (sampleArrivals encode ω))) u).card ≤
  46       hitCount (leftSourceBucket encode u) ω := by
  47   calc
  48     _ ≤ (listLeftNeighbors (sampleArrivals encode ω) u).card :=
  49       card_le_card (residual_left_neighbors_subset α _ u)
  50     _ ≤ _ := (listLeftNeighbors_card_le _ u).trans_eq (sample_left_count_eq_hits encode ω u)
  51 
  52 /-- Left-degree tail for the actual residual graph, counting distinct neighbors. -/
  53 theorem residual_left_degree_tail (encode : E → Arrival L R C) (α : C → ℕ)
  54     (T k : ℕ) (r : ℝ)
  55     (hmean : ∀ u, Real.exp 1*T*(((leftSourceBucket encode u).card : ℝ)/Fintype.card E)/k ≤ r) :
  56     eventProbability (univ.filter fun ω : Fin T → E => ∃ u,
  57       k ≤ (listLeftNeighbors (residualArrivals (trace α (sampleArrivals encode ω))) u).card) ≤
  58       Fintype.card L * r^k := by
  59   apply (eventProbability_mono (B := univ.filter fun ω : Fin T → E =>
  60     ∃ u, k ≤ hitCount (leftSourceBucket encode u) ω) ?_).trans
  61   · exact some_bucket_tail_bound (leftSourceBucket encode) T k r hmean
  62   · intro ω hω
  63     obtain ⟨u, hu⟩ := (mem_filter.mp hω).2
  64     exact mem_filter.mpr ⟨mem_univ _, u, hu.trans (residual_left_degree_le_hits encode α ω u)⟩
  65 
  66 end GraphChoices.ThresholdGreedy
