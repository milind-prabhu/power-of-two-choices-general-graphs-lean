   4 /-! The block occupancy argument applied to actual sampled bipartite graphs.

  10 theorem average_equiv {Ω Ω' : Type*} [Fintype Ω] [Fintype Ω']
  11     (e : Ω ≃ Ω') (f : Ω' → ℝ) : average (fun ω => f (e ω)) = average f := by
  12   simp only [average, e.sum_comp, Fintype.card_congr e]

  14 def blockFlatten {E : Type*} (k s : ℕ) : (Fin k → Fin s → E) ≃ (Fin (k*s) → E) :=
  15   (Equiv.curry (Fin k) (Fin s) E).symm.trans
  16     (Equiv.arrowCongr finProdFinEquiv (Equiv.refl E))
  17 
  18 theorem blockFlatten_apply {E : Type*} (k s : ℕ) (ω : Fin k → Fin s → E)
  19     (i : Fin k) (j : Fin s) : blockFlatten k s ω (finProdFinEquiv (i,j)) = ω i j := by
  20   simp [blockFlatten, Equiv.arrowCongr]

  22 variable {L R E : Type*} [Fintype L] [Fintype R] [Fintype E]
  23   [DecidableEq L] [DecidableEq R] [DecidableEq E] [Nonempty E]

  25 def bipartiteBase (left : E → L) (right : E → R) : BaseGraph (L ⊕ R) E where
  26   left e := Sum.inl (left e)
  27   right e := Sum.inr (right e)

  29 /-- No independence is used here: selected left vertices hit in every block
  30 force at least k edges per selected vertex in the same sampled induced set. -/
  31 theorem block_success_density (left : E → L) (right : E → R)
  32     (A : Finset L) (B : Finset R) (bucket : L → Finset E)
  33     (hsource : ∀ u ∈ A, ∀ e ∈ bucket u, left e = u ∧ right e ∈ B)
  34     (k s : ℕ) (ω : Fin k → Fin s → E) :
  35     let Z := successNumber (fun u : A => allBlocksEvent (bucket u.val) k s) ω
  36     (k:ℝ)*Z ≤ (Z+B.card)*(opt ((bipartiteBase left right).sample (blockFlatten k s ω)) : ℝ) := by
  37   classical
  38   let H : Finset L := A.filter fun u => ω ∈ allBlocksEvent (bucket u) k s
  39   let S : Finset (L ⊕ R) := H.map Function.Embedding.inl ∪ B.map Function.Embedding.inr
  40   let I := (bipartiteBase left right).sample (blockFlatten k s ω)
  41   have hsuccess (u : H) (i : Fin k) : ∃ j : Fin s, ω i j ∈ bucket u.val := by
  42     have hu := (mem_filter.mp u.property).2
  43     have hh := (Fintype.mem_piFinset.mp hu) i
  44     exact (mem_filter.mp hh).2
  45   let time (u : H) (i : Fin k) : Fin (k*s) :=
  46     finProdFinEquiv (i, Classical.choose (hsuccess u i))
  47   have htime (u : H) (i : Fin k) :
  48       I.left (time u i) ∈ S ∧ I.right (time u i) ∈ S := by
  49     have hb := Classical.choose_spec (hsuccess u i)
  50     have hs := hsource u.val (mem_filter.mp u.property).1 _ hb
  51     dsimp only [I, BaseGraph.sample, bipartiteBase, time]
  52     rw [blockFlatten_apply]
  53     constructor
  54     · apply mem_union_left
  55       exact mem_map.mpr ⟨u.val, u.property, by simp [hs.1]⟩
  56     · apply mem_union_right
  57       exact mem_map.mpr ⟨right (ω i (Classical.choose (hsuccess u i))), hs.2, rfl⟩
  58   let f : H × Fin k → {t : Fin (k*s) // I.left t ∈ S ∧ I.right t ∈ S} :=
  59     fun p => ⟨time p.1 p.2, htime p.1 p.2⟩
  60   have hinj : Function.Injective f := by
  61     rintro ⟨u,i⟩ ⟨v,j⟩ h
  62     have he : time u i = time v j := congrArg Subtype.val h
  63     have hij := congrArg Prod.fst (finProdFinEquiv.injective he)
  64     have hs1 := hsource u.val (mem_filter.mp u.property).1 _ (Classical.choose_spec (hsuccess u i))
  65     have hs2 := hsource v.val (mem_filter.mp v.property).1 _ (Classical.choose_spec (hsuccess v j))
  66     have ht : ω i (Classical.choose (hsuccess u i)) = ω j (Classical.choose (hsuccess v j)) :=
  67       by simpa only [time, blockFlatten_apply] using congrArg (fun t => blockFlatten k s ω t) he
  68     have huv : u = v := Subtype.ext (by rw [← hs1.1, ht, hs2.1])
  69     exact Prod.ext huv hij
  70   have hedge : H.card*k ≤ internalCount I S := by
  71     have hh := Fintype.card_le_of_injective f hinj
  72     rw [Fintype.card_prod, Fintype.card_coe, Fintype.card_fin] at hh
  73     simpa only [Fintype.card_subtype, internalCount] using hh
  74   have hdisj : Disjoint (H.map Function.Embedding.inl) (B.map Function.Embedding.inr) := by
  75     apply disjoint_left.mpr
  76     intro x hx hy
  77     obtain ⟨u,_,hu⟩ := mem_map.mp hx
  78     obtain ⟨v,_,hv⟩ := mem_map.mp hy
  79     exact Sum.noConfusion (hu.trans hv.symm)
  80   have hScard : S.card = H.card+B.card := by simp [S, card_union_of_disjoint hdisj]
  81   have hZ : successNumber (fun u : A => allBlocksEvent (bucket u.val) k s) ω = H.card := by
  82     simp only [successNumber, H, card_filter]
  83     push_cast
  84     exact Finset.sum_coe_sort A (fun u => if ω ∈ allBlocksEvent (bucket u) k s then (1:ℝ) else 0)
  85   dsimp only
  86   rw [hZ]
  87   have hd := hedge.trans (density_le_opt I S)
  88   rw [hScard] at hd
  89   exact_mod_cast (by simpa only [Nat.mul_comm] using hd)

  91 theorem expectedOpt_block_lower (left : E → L) (right : E → R)
  92     (A : Finset L) (B : Finset R) (bucket : L → Finset E)
  93     (hsource : ∀ u ∈ A, ∀ e ∈ bucket u, left e = u ∧ right e ∈ B)
  94     (k s : ℕ) (hB : B.Nonempty)
  95     (hmean : (B.card:ℝ) ≤ ∑ u : A, eventProbability (allBlocksEvent (bucket u.val) k s)) :
  96     (k:ℝ)/3 ≤ expectedOpt (bipartiteBase left right) (k*s) := by
  97   let events := fun u : A => allBlocksEvent (bucket u.val) k s
  98   have hcov : ∀ u v : A, u ≠ v → eventProbability (events u ∩ events v) ≤
  99       eventProbability (events u)*eventProbability (events v) := by
 100     intro u v huv
 101     apply allBlocks_disjoint_covariance
 102     apply disjoint_left.mpr
 103     intro e he hf
 104     apply huv
 105     exact Subtype.ext (((hsource u.val u.property e he).1).symm.trans (hsource v.val v.property e hf).1)
 106   have hh := expected_density_of_second_moment (successNumber events)
 107     (fun ω => (opt ((bipartiteBase left right).sample (blockFlatten k s ω)):ℝ))
 108     k B.card (successNumber_nonneg events) (Nat.cast_nonneg k)
 109     (by exact_mod_cast hB.card_pos) (by rwa [average_successNumber])
 110     (by rw [average_successNumber]; exact successNumber_second_moment events hcov)
 111     (block_success_density left right A B bucket hsource k s)
 112   rw [average_equiv (blockFlatten k s) (fun ω => (opt ((bipartiteBase left right).sample ω):ℝ))] at hh
 113   exact hh
