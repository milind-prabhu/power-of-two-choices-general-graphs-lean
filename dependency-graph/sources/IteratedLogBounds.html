   7 def iteratedLog (N : ℕ) : ℕ := Nat.log 2 (Nat.log 2 N+1)+1
   8 
   9 theorem iteratedLog_pos (N : ℕ) : 0 < iteratedLog N := by unfold iteratedLog; omega

  11 theorem nat_log_mul_two_pow (a k : ℕ) (ha : 0 < a) : Nat.log 2 (a*2^k) = Nat.log 2 a+k := by
  12   induction k with
  13   | zero => simp
  14   | succ k ih =>
  15     rw [pow_succ, ← mul_assoc, Nat.log_mul_base (by decide) (by positivity), ih]
  16     omega
  17 
  18 theorem nat_log_square_bound (a : ℕ) : Nat.log 2 (a^2)+1 ≤ 2*(Nat.log 2 a+1) := by
  19   by_cases ha : a = 0
  20   · simp [ha]
  21   have hp := Nat.lt_pow_succ_log_self (by decide : 1 < 2) a
  22   have hs : a^2 < 2^(2*(Nat.log 2 a+1)) := by
  23     calc
  24       _ < (2^(Nat.log 2 a+1))^2 := Nat.pow_lt_pow_left hp (by decide)
  25       _ = _ := by rw [← pow_mul]; congr 1; omega
  26   have hh := Nat.log_lt_of_lt_pow (pow_ne_zero 2 ha) hs
  27   omega

  29 theorem numberRounds_le_iteratedLog (B N : ℕ) (hN : 0 < N) (hB : B ≤ N^2) :
  30     Peeling.numberRounds B ≤ iteratedLog N+2 := by
  31   by_cases hB0 : B = 0
  32   · simp [hB0, Peeling.numberRounds]
  33   let l := Nat.log 2 N+1
  34   have hl : 1 ≤ l := by dsimp [l]; omega
  35   have hpow := Nat.lt_pow_succ_log_self (by decide : 1 < 2) N
  36   have hbound : 16*B < 2^(2*l+4) := by
  37     calc
  38       _ ≤ 16*N^2 := Nat.mul_le_mul_left _ hB
  39       _ < 16*(2^l)^2 := Nat.mul_lt_mul_of_pos_left (Nat.pow_lt_pow_left hpow (by decide)) (by decide)
  40       _ = _ := by
  41         have he : 2^(2*l+4) = (2^l)^2*16 := by
  42           rw [pow_add, show 2*l = l*2 by omega, pow_mul]
  43           norm_num
  44         rw [he]
  45         ring
  46   have hin := Nat.log_lt_of_lt_pow (by positivity : 16*B ≠ 0) hbound
  47   have hinner : Nat.log 2 (16*B)+1 ≤ 8*l := by omega
  48   have hmono := Nat.log_mono_right hinner (b := 2)
  49   have heq : Nat.log 2 (8*l) = Nat.log 2 l+3 := by
  50     simpa only [show 2^3 = 8 by norm_num, mul_comm] using nat_log_mul_two_pow l 3 (by omega)
  51   unfold Peeling.numberRounds
  52   rw [heq] at hmono
  53   change Nat.log 2 (Nat.log 2 (16*B)+1) ≤ Nat.log 2 l+1+2
  54   omega
  55 
  56 theorem real_loglog_le_iteratedLog (N : ℕ) (hlog : 1 < Real.log (N:ℝ)) :
  57     Real.log (Real.log (N:ℝ)) ≤ (iteratedLog N : ℝ) := by
  58   have hN : 0 < N := by
  59     by_contra hn
  60     have hz : N = 0 := by omega
  61     norm_num [hz] at hlog
  62   have hl := Patterns.real_log_le_nat_log_shift N hN
  63   have hpos : 0 < Real.log (N:ℝ) := by linarith
  64   have hh := Real.log_le_log hpos hl
  65   exact hh.trans (Patterns.real_log_le_nat_log_shift (Nat.log 2 N+1) (by omega))

  67 theorem greedy_multiplier_le_iteratedLog (N : ℕ) :
  68     2*(Nat.log 2 (3*(polynomialCutoff N 5+1)^2)+1) ≤ 20*iteratedLog N := by
  69   let l := Nat.log 2 N+1
  70   have hl : 1 ≤ l := by dsimp [l]; omega
  71   have hl2 : l ≤ l*l := Nat.le_mul_self l
  72   have hsize : 3*(6*l+1)^2 ≤ (16*l)^2 := by nlinarith
  73   have hmono := Nat.log_mono_right hsize (b := 2)
  74   have hsquare := nat_log_square_bound (16*l)
  75   have hlog : Nat.log 2 (16*l) = Nat.log 2 l+4 := by
  76     simpa only [show 2^4 = 16 by norm_num, mul_comm] using nat_log_mul_two_pow l 4 (by omega)
  77   rw [hlog] at hsquare
  78   change 2*(Nat.log 2 (3*(6*l+1)^2)+1) ≤ 20*(Nat.log 2 l+1)
  79   omega

  81 theorem exceptional_error_le_three (N : ℕ) (hN : 0 < N) : 3*(N:ℝ)/(N:ℝ)^5 ≤ 3 := by
  82   have hNr : (0:ℝ) < N := by exact_mod_cast hN
  83   have hN1 : (1:ℝ) ≤ N := by exact_mod_cast hN
  84   have hp : (1:ℝ) ≤ (N:ℝ)^4 := one_le_pow₀ hN1
  85   calc
  86     _ = 3/(N:ℝ)^4 := by rw [pow_succ]; field_simp
  87     _ ≤ 3 := (div_le_iff₀ (by positivity)).mpr (by linarith)

  96 theorem PreparedDecomposition.budget_loglog (P : PreparedDecomposition G B N d)
  97     (hB : B ≤ N^2) (haverage : 2*(G.card : ℝ) = (N:ℝ)*d) :
  98     P.budget N ≤ 320*((B:ℝ)*((N:ℝ)/Fintype.card G)+graphLogSkewness G d (N:ℝ)+1)*(iteratedLog N : ℝ) := by
  99   have hn : P.pieces.length ≤ iteratedLog N+2 := by
 100     rw [P.count]
 101     exact numberRounds_le_iteratedLog B N P.vertex_count_pos hB
 102   have hnr : (P.pieces.length : ℝ) ≤ (iteratedLog N : ℝ)+2 := by exact_mod_cast hn
 103   have hQ : (1:ℝ) ≤ iteratedLog N := by exact_mod_cast iteratedLog_pos N
 104   have hl := real_loglog_le_iteratedLog N P.large_log
 105   have hs := graphLogSkewness_nonneg G d (N:ℝ)
 106   have hfactor : (P.pieces.length : ℝ)+Real.log (Real.log (N:ℝ))+1 ≤ 5*(iteratedLog N : ℝ) := by linarith
 107   calc
 108     _ ≤ _ := P.budget_simplified haverage
 109     _ ≤ 64*((B:ℝ)*((N:ℝ)/Fintype.card G)+graphLogSkewness G d (N:ℝ)+1)*(5*(iteratedLog N : ℝ)) :=
 110       mul_le_mul_of_nonneg_left hfactor (by positivity)
 111     _ = _ := by ring
