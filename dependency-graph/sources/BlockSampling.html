   9 theorem average_unit_indicator {Ω : Type*} [Fintype Ω] [DecidableEq Ω] (A : Finset Ω) :
  10     average (fun ω => if ω ∈ A then (1 : ℝ) else 0) = eventProbability A := by
  11   simp [average, eventProbability]

  13 theorem eventProbability_le_one {Ω : Type*} [Fintype Ω] (A : Finset Ω) :
  14     eventProbability A ≤ 1 := by
  15   unfold eventProbability
  16   by_cases h : Fintype.card Ω = 0
  17   · simp [h]
  18   · apply (div_le_one (by exact_mod_cast Nat.pos_of_ne_zero h)).mpr
  19     exact_mod_cast card_le_univ A

  21 theorem eventProbability_compl {Ω : Type*} [Fintype Ω] [Nonempty Ω]
  22     [DecidableEq Ω] (A : Finset Ω) :
  23     eventProbability (univ \ A) = 1 - eventProbability A := by
  24   simp only [eventProbability, card_sdiff, inter_univ, card_univ]
  25   rw [Nat.cast_sub (card_le_univ A)]
  26   have h : (Fintype.card Ω : ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  27   field_simp
  28 
  29 theorem eventProbability_union_inter {Ω : Type*} [Fintype Ω] [DecidableEq Ω]
  30     (A B : Finset Ω) :
  31     eventProbability (A ∪ B) + eventProbability (A ∩ B) =
  32       eventProbability A + eventProbability B := by
  33   unfold eventProbability
  34   rw [← add_div, ← add_div]
  35   congr 1
  36   exact_mod_cast card_union_add_card_inter A B

  40 def blockHitEvent (A : Finset E) (s : ℕ) : Finset (Fin s → E) :=
  41   univ.filter fun ω => ∃ t, ω t ∈ A
  42 
  43 theorem blockHitEvent_compl (A : Finset E) (s : ℕ) :
  44     univ \ blockHitEvent A s = Fintype.piFinset (fun _ : Fin s => univ \ A) := by
  45   ext ω
  46   simp [blockHitEvent, Fintype.mem_piFinset]
  47 
  48 theorem blockHit_probability (A : Finset E) (s : ℕ) :
  49     eventProbability (blockHitEvent A s) =
  50       1 - (1 - (A.card : ℝ) / Fintype.card E)^s := by
  51   have h := eventProbability_compl (blockHitEvent A s)
  52   rw [blockHitEvent_compl, probability_coordinate_constraints] at h
  53   have hm : (Fintype.card E : ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  54   have hc : ((univ \ A).card : ℝ) / Fintype.card E =
  55       1 - (A.card : ℝ) / Fintype.card E := by
  56     rw [card_sdiff, inter_univ, card_univ, Nat.cast_sub (card_le_univ A)]
  57     field_simp
  58   simp only [hc, prod_const, card_univ, Fintype.card_fin] at h
  59   linarith
  60 
  61 theorem blockHit_union (A B : Finset E) (s : ℕ) :
  62     blockHitEvent A s ∪ blockHitEvent B s = blockHitEvent (A ∪ B) s := by
  63   ext ω
  64   simp only [mem_union, blockHitEvent, mem_filter, mem_univ, true_and, mem_union]
  65   exact exists_or.symm
  66 
  67 theorem blockHit_disjoint_covariance (A B : Finset E) (s : ℕ) (hAB : Disjoint A B) :
  68     eventProbability (blockHitEvent A s ∩ blockHitEvent B s) ≤
  69       eventProbability (blockHitEvent A s) * eventProbability (blockHitEvent B s) := by
  70   have hi := eventProbability_union_inter (blockHitEvent A s) (blockHitEvent B s)
  71   rw [blockHit_union] at hi
  72   simp only [blockHit_probability] at hi ⊢
  73   have ha : (0 : ℝ) ≤ 1 - (A.card : ℝ) / Fintype.card E := by
  74     have h := eventProbability_le_one A
  75     unfold eventProbability at h
  76     linarith
  77   have hb : (0 : ℝ) ≤ 1 - (B.card : ℝ) / Fintype.card E := by
  78     have h := eventProbability_le_one B
  79     unfold eventProbability at h
  80     linarith
  81   have hab : (0 : ℝ) ≤ 1 - ((A.card : ℝ) + B.card) / Fintype.card E := by
  82     have h := eventProbability_le_one (A ∪ B)
  83     simp only [eventProbability, card_union_of_disjoint hAB, Nat.cast_add] at h
  84     linarith
  85   have hh : 1 - ((A.card : ℝ) + B.card) / Fintype.card E ≤
  86       (1 - (A.card : ℝ) / Fintype.card E) * (1 - (B.card : ℝ) / Fintype.card E) := by
  87     have hprod : (0 : ℝ) ≤ ((A.card : ℝ) / Fintype.card E) *
  88         ((B.card : ℝ) / Fintype.card E) := by positivity
  89     rw [add_div]
  90     nlinarith
  91   have hp := pow_le_pow_left₀ hab hh s
  92   rw [mul_pow] at hp
  93   simp only [card_union_of_disjoint hAB, Nat.cast_add] at hi
  94   nlinarith

  96 noncomputable def allBlocksEvent (A : Finset E) (k s : ℕ) :
  97     Finset (Fin k → Fin s → E) := Fintype.piFinset fun _ => blockHitEvent A s

  99 theorem allBlocks_probability (A : Finset E) (k s : ℕ) :
 100     eventProbability (allBlocksEvent A k s) = (eventProbability (blockHitEvent A s))^k := by
 101   rw [allBlocksEvent, probability_coordinate_constraints]
 102   simp [eventProbability, div_pow]

 104 theorem allBlocks_inter (A B : Finset E) (k s : ℕ) :
 105     allBlocksEvent A k s ∩ allBlocksEvent B k s =
 106       Fintype.piFinset (fun _ : Fin k => blockHitEvent A s ∩ blockHitEvent B s) := by
 107   ext ω
 108   simp only [allBlocksEvent, mem_inter, Fintype.mem_piFinset]
 109   exact forall_and.symm

 111 theorem allBlocks_disjoint_covariance (A B : Finset E) (k s : ℕ) (hAB : Disjoint A B) :
 112     eventProbability (allBlocksEvent A k s ∩ allBlocksEvent B k s) ≤
 113       eventProbability (allBlocksEvent A k s) * eventProbability (allBlocksEvent B k s) := by
 114   rw [allBlocks_inter, probability_coordinate_constraints]
 115   simp only [prod_const, card_univ, Fintype.card_fin]
 116   change (eventProbability (blockHitEvent A s ∩ blockHitEvent B s))^k ≤ _
 117   rw [allBlocks_probability, allBlocks_probability, ← mul_pow]
 118   exact pow_le_pow_left₀ (eventProbability_nonneg _) (blockHit_disjoint_covariance A B s hAB) k

 120 theorem bernoulli_miss_bound (p : ℝ) (hp : 0 ≤ p) (hp1 : p ≤ 1) (s : ℕ) :
 121     (1-p)^s * (1+s*p) ≤ 1 := by
 122   induction s with
 123   | zero => simp
 124   | succ s ih =>
 125     have hh : (1-p)*(1+((s+1 : ℕ) : ℝ)*p) ≤ 1+s*p := by
 126       push_cast
 127       nlinarith [mul_nonneg (by positivity : (0 : ℝ) ≤ (s:ℝ)+1) (sq_nonneg p)]
 128     calc
 129       _ = (1-p)^s * ((1-p)*(1+((s+1 : ℕ) : ℝ)*p)) := by rw [pow_succ]; ring
 130       _ ≤ (1-p)^s * (1+s*p) := mul_le_mul_of_nonneg_left hh (pow_nonneg (by linarith) _)
 131       _ ≤ 1 := ih

 133 /-- A direct binomial lower bound, valid without an asymptotic quantile lemma. -/
 134 theorem blockHit_probability_lower (A : Finset E) (s : ℕ) :
 135     ((s:ℝ)*((A.card : ℝ)/Fintype.card E)) /
 136       (1+(s:ℝ)*((A.card : ℝ)/Fintype.card E)) ≤ eventProbability (blockHitEvent A s) := by
 137   have hp : (0 : ℝ) ≤ (A.card : ℝ)/Fintype.card E := by positivity
 138   have hp1 : (A.card : ℝ)/Fintype.card E ≤ 1 := eventProbability_le_one A
 139   rw [blockHit_probability]
 140   apply (div_le_iff₀ (by positivity : (0 : ℝ) < 1+(s:ℝ)*((A.card : ℝ)/Fintype.card E))).mpr
 141   have h := bernoulli_miss_bound _ hp hp1 s
 142   nlinarith

 144 theorem blockHit_probability_lower_of_mean (A : Finset E) (s : ℕ) (x : ℝ)
 145     (hx : 0 ≤ x) (hx1 : x ≤ 1)
 146     (hmean : x ≤ (s:ℝ)*((A.card : ℝ)/Fintype.card E)) :
 147     x/2 ≤ eventProbability (blockHitEvent A s) := by
 148   have hm : (0 : ℝ) ≤ (s:ℝ)*((A.card : ℝ)/Fintype.card E) := by positivity
 149   apply le_trans _ (blockHit_probability_lower A s)
 150   apply (le_div_iff₀ (by positivity : (0 : ℝ) < 1+(s:ℝ)*((A.card : ℝ)/Fintype.card E))).mpr
 151   nlinarith [mul_nonneg (sub_nonneg.mpr hmean) (sub_nonneg.mpr hx1)]
