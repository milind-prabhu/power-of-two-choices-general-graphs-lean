   3 /-! A canonical geometric shape encoding for every finite connected graph,
   4 with a prescribed right root and an exact bijection of vertex sets. -/

   6 open Finset

   9 theorem exists_canonical_embedding (G : SimpleGraph V) (hc : G.Connected)
  10     (color : V → Bool) (hcolor : ∀ u v, G.Adj u v → color u ≠ color v)
  11     (root : V) (hroot : color root = false) :
  12     ∃ m : ℕ, ∃ s : UnlabelledShape m, ∃ e : Fin (m+1) ≃ V,
  13       m+1 = Fintype.card V ∧ e ⟨0, by omega⟩ = root ∧
  14       ∀ i : Fin m, G.Adj (e (childIndex i)) (e (parentIndex (skeletonPattern s) i)) ∧
  15         color (e (childIndex i)) = (skeletonPattern s).childLeft i := by
  16   classical
  17   obtain ⟨t, ht, hn, hcover, hsize⟩ := exists_spanning_labelled_forest G hc root
  18   let xs := root :: forestLabels t
  19   have hlen : xs.length = t.numNodes+1 := by simp [xs, forestLabels_length]
  20   let F : ℕ → V := fun i => if hi : i < xs.length then xs[i] else root
  21   have hF (i : ℕ) (hi : i < xs.length) : F i = xs[i] := dif_pos hi
  22   have hF0 : F 0 = root := by simp [F, xs]
  23   have hbij : Function.Bijective (fun i : Fin (t.numNodes+1) => F i.val) := by
  24     constructor
  25     · intro i j hij
  26       change F i.val = F j.val at hij
  27       have hi : i.val < xs.length := by rw [hlen]; exact i.isLt
  28       have hj : j.val < xs.length := by rw [hlen]; exact j.isLt
  29       rw [hF i.val hi, hF j.val hj] at hij
  30       exact Fin.ext ((hn.getElem_inj_iff).mp hij)
  31     · intro v
  32       obtain ⟨i, hi⟩ := List.mem_iff_get.mp (hcover v)
  33       let j : Fin (t.numNodes+1) := ⟨i.val, by simpa only [List.length_cons, forestLabels_length] using i.isLt⟩
  34       refine ⟨j, ?_⟩
  35       change F j.val = v
  36       rw [hF j.val (by exact i.isLt)]
  37       exact hi
  38   let e : Fin (t.numNodes+1) ≃ V := Equiv.ofBijective _ hbij
  39   let s : UnlabelledShape t.numNodes :=
  40     ⟨eraseForest t, Tree.mem_treesOfNumNodesEq.mpr (eraseForest_numNodes t)⟩
  41   have hvalid : ForestValid G (F 0) t := by simpa only [hF0] using ht
  42   have hparent : color (F 0) = !true := by simpa only [hF0] using hroot
  43   have hvalues : ∀ i < t.numNodes, F (0+i+1) = forestValue t root i := by
  44     intro i hi
  45     rw [hF (0+i+1) (by rw [hlen]; omega)]
  46     simpa only [xs, Nat.zero_add, List.getElem_cons_succ] using (forestValue_eq_get t root i hi).symm
  47   have hspec := forest_index_spec G color hcolor t F root 0 0 true hvalid hparent hvalues
  48   refine ⟨t.numNodes, s, e, hsize, hF0, ?_⟩
  49   intro i
  50   have h := hspec i.val i.isLt
  51   have hpar : (parentIndex (skeletonPattern s) i).val = parentPos t 0 0 i.val := by
  52     change (forestParent (eraseForest t) 0 0 le_rfl _).val = _
  53     exact forestParent_eq_parentPos t 0 0 le_rfl _
  54   have hside : (skeletonPattern s).childLeft i = sidePos t true i.val := by
  55     change forestSide (eraseForest t) true _ = _
  56     exact forestSide_eq_sidePos t true _
  57   constructor
  58   · change G.Adj (F (i.val+1)) (F (parentIndex (skeletonPattern s) i).val)
  59     rw [hpar]
  60     simpa only [Nat.zero_add] using h.1
  61   · change color (F (i.val+1)) = _
  62     rw [hside]
  63     simpa only [Nat.zero_add] using h.2

  65 /-- The right root contributes no left vertex, so the non-root left-index
  66 count equals the number of left vertices of the encoded graph exactly. -/
  67 theorem left_index_card (m : ℕ) (e : Fin (m+1) ≃ V) (color : V → Bool)
  68     (hroot : color (e ⟨0, by omega⟩) = false) :
  69     (univ.filter fun i : Fin m => color (e (childIndex i)) = true).card =
  70       Nat.card {v : V // color v = true} := by
  71   classical
  72   let A := {i : Fin m // i ∈ univ.filter (fun i => color (e (childIndex i)) = true)}
  73   let f : A → {v : V // color v = true} := fun i =>
  74     ⟨e (childIndex i.val), (mem_filter.mp i.property).2⟩
  75   have hf : Function.Bijective f := by
  76     constructor
  77     · intro i j hij
  78       have hidx := e.injective (congrArg Subtype.val hij)
  79       apply Subtype.ext
  80       apply Fin.ext
  81       have hval := congrArg Fin.val hidx
  82       simp only [childIndex] at hval
  83       omega
  84     · intro v
  85       let j := e.symm v.val
  86       have hj0 : j.val ≠ 0 := by
  87         intro hj
  88         have hj' : j = ⟨0, by omega⟩ := Fin.ext hj
  89         have he : e ⟨0, by omega⟩ = v.val := by rw [← hj']; exact e.apply_symm_apply _
  90         rw [he, v.property] at hroot
  91         contradiction
  92       let i : Fin m := ⟨j.val-1, by have := j.isLt; omega⟩
  93       have hij : childIndex i = j := by apply Fin.ext; dsimp [childIndex, i]; omega
  94       have hcol : color (e (childIndex i)) = true := by
  95         rw [hij]
  96         simpa only [j, e.apply_symm_apply] using v.property
  97       refine ⟨⟨i, mem_filter.mpr ⟨mem_univ _, hcol⟩⟩, ?_⟩
  98       apply Subtype.ext
  99       change e (childIndex i) = v.val
 100       rw [hij]
 101       exact e.apply_symm_apply _
 102   have hcard := Fintype.card_congr (Equiv.ofBijective f hf)
 103   simpa only [A, Fintype.card_coe, Nat.card_eq_fintype_card] using hcard
