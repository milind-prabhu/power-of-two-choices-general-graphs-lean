  11 theorem PreparedDecomposition.mass_at_horizon (P : PreparedDecomposition G B N d) (T : ℕ)
  12     (haverage : 2*(G.card : ℝ) = (T:ℝ)*d) :
  13     1 ≤ 16*(B:ℝ)*((T:ℝ)/Fintype.card G) := by
  14   have hE : (0:ℝ) < Fintype.card G := by exact_mod_cast Fintype.card_pos
  15   rw [← mul_div_assoc]
  16   apply (le_div_iff₀ hE).mpr
  17   have hp := mul_le_mul_of_nonneg_left P.average_cap (Nat.cast_nonneg T : (0:ℝ) ≤ T)
  18   have heq : (Fintype.card G : ℝ) = G.card := by rw [Fintype.card_coe]
  19   rw [heq]

  23 theorem PreparedDecomposition.budget_simplified_horizon (P : PreparedDecomposition G B N d) (T : ℕ) (hT : 0 < T)
  24     (haverage : 2*(G.card : ℝ) = (T:ℝ)*d) :
  25     P.budget T ≤ 64*((B:ℝ)*((T:ℝ)/Fintype.card G)+graphLogSkewness G d (N:ℝ)+1)*
  26       (P.pieces.length+Real.log (Real.log (N:ℝ))+1) := by
  27   have hz : 0 < (T:ℝ)/Fintype.card G := div_pos
  28     (by exact_mod_cast hT) (by exact_mod_cast Fintype.card_pos)
  29   have hh := P.class_count_pos
  30   have hhr : (1:ℝ) ≤ P.pieces.length := by exact_mod_cast hh
  31   have hb := branchTarget_pos (4*B) P.pieces.length _ hh hz.le
  32   have hb1 := (branchTarget_le_rate (4*B) P.pieces.length _ hh hz.le).trans (shapeRate_le_one _ hh)
  33   have hmass := P.mass_at_horizon T haverage
  34   have hK : 1 ≤ 16*(B:ℝ)*((T:ℝ)/Fintype.card G)/branchTarget (4*B) P.pieces.length ((T:ℝ)/Fintype.card G) := by
  35     apply (le_div_iff₀ hb).mpr
  36     simpa only [one_mul] using hb1.trans hmass
  37   have hl : 0 ≤ Real.log (Real.log (N:ℝ)) := Real.log_nonneg P.large_log.le
  38   have hbound := log_reference_branch_bound B P.pieces.length ((T:ℝ)/Fintype.card G) P.reference_cap_pos hh hz
  39   have hs := scalar_budget_le (P.pieces.length : ℝ) ((B:ℝ)*((T:ℝ)/Fintype.card G))
  40     (graphLogSkewness G d (N:ℝ)) (Real.log (Real.log (N:ℝ))) _ hhr (by positivity)
  41     (graphLogSkewness_nonneg G d (N:ℝ)) hl (Real.log_nonneg hK) hbound
  42   convert hs using 1 <;> unfold PreparedDecomposition.budget <;> ring

  45 theorem PreparedDecomposition.budget_loglog_horizon (P : PreparedDecomposition G B N d) (T : ℕ) (hT : 0 < T)
  46     (hB : B ≤ N^2) (haverage : 2*(G.card : ℝ) = (T:ℝ)*d) :
  47     P.budget T ≤ 320*((B:ℝ)*((T:ℝ)/Fintype.card G)+graphLogSkewness G d (N:ℝ)+1)*(iteratedLog N : ℝ) := by
  48   have hn : P.pieces.length ≤ iteratedLog N+2 := by
  49     rw [P.count]
  50     exact numberRounds_le_iteratedLog B N P.vertex_count_pos hB
  51   have hnr : (P.pieces.length : ℝ) ≤ (iteratedLog N : ℝ)+2 := by exact_mod_cast hn
  52   have hQ : (1:ℝ) ≤ iteratedLog N := by exact_mod_cast iteratedLog_pos N
  53   have hl := real_loglog_le_iteratedLog N P.large_log
  54   have hs := graphLogSkewness_nonneg G d (N:ℝ)
  55   have hfactor : (P.pieces.length : ℝ)+Real.log (Real.log (N:ℝ))+1 ≤ 5*(iteratedLog N : ℝ) := by linarith
  56   calc
  57     _ ≤ _ := P.budget_simplified_horizon T hT haverage
  58     _ ≤ 64*((B:ℝ)*((T:ℝ)/Fintype.card G)+graphLogSkewness G d (N:ℝ)+1)*(5*(iteratedLog N : ℝ)) :=
  59       mul_le_mul_of_nonneg_left hfactor (by positivity)

  63 theorem PreparedDecomposition.component_tail_horizon (P : PreparedDecomposition G B N d)
  64     (T c : ℕ) (hT : 0 < T) (hc : 5 ≤ c)
  65     (hvertices : Fintype.card (L ⊕ R) ≤ N)
  66     (haverage : 2*(G.card:ℝ) = (T:ℝ)*d) :
  67     eventProbability (largeComponentEvent P.encode (P.thresholds T) T
  68       (3*(polynomialCutoff N c+1)^2)) ≤ 3/(N:ℝ)^c := by
  69   have hv : Fintype.card L+Fintype.card R ≤ N := by simpa only [Fintype.card_sum] using hvertices
  70   apply log_squared_component_tail P.pieces.get P.encode P.DL P.DR (4*B)
  71     (maxLeftDegree G) T N c (fun i => scaleNat i.val) P.simple P.source P.bucket_bound
  72     P.piece_left_cap P.piece_right_cap P.global_bucket_bound hT P.scale_gt_one
  73     P.vertex_count_pos (by omega) (by omega)
  74   have hm : (T:ℝ)*(maxLeftDegree G:ℝ)/G.card ≤ 2*Real.log (N:ℝ) := by
  75     have he : (0:ℝ) < G.card := by exact_mod_cast (show 0 < G.card by simpa only [← Fintype.card_coe] using Fintype.card_pos (α := G))
  76     apply (div_le_iff₀ he).mpr
  77     have hh := mul_le_mul_of_nonneg_left P.max_degree_preprocessing (Nat.cast_nonneg T : (0:ℝ) ≤ T)
  78     rw [← mul_assoc, ← haverage] at hh
  79     nlinarith
  80   have hh := preprocessed_mean_below_cutoff N c ((T:ℝ)*(maxLeftDegree G:ℝ)/G.card)
  81     P.vertex_count_pos hc (by positivity) hm
  82   rw [Fintype.card_coe]
  83   convert hh using 1 <;> ring

  85 theorem PreparedDecomposition.expected_cost_horizon (P : PreparedDecomposition G B N d)
  86     (T : ℕ) (hT : 0 < T) (hTN : T ≤ N)
  87     (hvertices : Fintype.card (L ⊕ R) ≤ N)
  88     (haverage : 2*(G.card:ℝ) = (T:ℝ)*d) (hB : B ≤ N^2) :
  89     expectedCost (graphBase G) (onlineOrientation (T := T) (P.policy T)) ≤
  90       320*((B:ℝ)*((T:ℝ)/G.card)+graphLogSkewness G d (N:ℝ)+1)*(iteratedLog N:ℝ)+
  91         20*(iteratedLog N:ℝ)*expectedOpt (graphBase G) T+3 := by
  92   have ht := P.component_tail_horizon T 5 hT (by decide) hvertices haverage
  93   have hh := expected_load_of_component_tail P.encode (P.thresholds T) T _ _ ht
  94   rw [expectedTotalMaxLoad_eq_expectedCost, expectedReplayOpt_eq_expectedOpt] at hh
  95   change expectedCost (graphBase G) (onlineOrientation (T := T) (P.policy T)) ≤
  96     (∑ i, (P.thresholds T i:ℝ))+
  97       (2*(Nat.log 2 (3*(polynomialCutoff N 5+1)^2)+1):ℕ)*expectedOpt (graphBase G) T+
  98         T*(3/(N:ℝ)^5) at hh
  99   have hb := (P.threshold_budget T (P.mass_at_horizon T haverage)).trans
 100     (P.budget_loglog_horizon T hT hB haverage)
 101   rw [Fintype.card_coe] at hb
 102   have hg : (((2*(Nat.log 2 (3*(polynomialCutoff N 5+1)^2)+1):ℕ):ℝ)) ≤ 20*(iteratedLog N:ℝ) := by
 103     exact_mod_cast greedy_multiplier_le_iteratedLog N
 104   have hgm := mul_le_mul_of_nonneg_right hg (expectedOpt_nonneg (graphBase G) T)
 105   have he : (T:ℝ)*(3/(N:ℝ)^5) ≤ 3 := by
 106     apply le_trans _ (exceptional_error_le_three N P.vertex_count_pos)
 107     have htr : (T:ℝ) ≤ N := by exact_mod_cast hTN
 108     have hmul := mul_le_mul_of_nonneg_right htr (show 0 ≤ 3/(N:ℝ)^5 by positivity)
 109     convert hmul using 1 <;> ring
 110   linarith
