   6 theorem greedy_bound_by_arrivals {V : Type*} [Fintype V] [DecidableEq V] {T : ℕ}
   7     (I : Instance V T) (a : Orientation T) (hg : IsGreedy I a) (hT : 0 < T) :
   8     maxLoad I a ≤ 2*opt I*(Nat.log 2 (2*T)+1) := by
   9   let S := univ.image I.left ∪ univ.image I.right
  10   have hl (t : Fin T) : I.left t ∈ S := mem_union_left _ (mem_image.mpr ⟨t,mem_univ _,rfl⟩)
  11   have hr (t : Fin T) : I.right t ∈ S := mem_union_right _ (mem_image.mpr ⟨t,mem_univ _,rfl⟩)
  12   have hc : S.card ≤ 2*T := by
  13     have h1 := card_image_le (f := I.left) (s := univ)
  14     have h2 := card_image_le (f := I.right) (s := univ)
  15     have h3 := card_union_le (univ.image I.left) (univ.image I.right)
  16     simp only [card_univ, Fintype.card_fin] at h1 h2
  17     dsimp only [S]
  18     omega
  19   apply (maxLoad_le_iff _ _ _).mpr
  20   intro v
  21   by_cases hv : v ∈ S
  22   · exact (greedy_load_lt_of_closed I a hg S (fun t => ⟨fun _ => hr t,fun _ => hl t⟩)
  23       hT (Nat.log 2 (2*T)+1) (hc.trans_lt (Nat.lt_pow_succ_log_self (by decide) _)) v hv).le
  24   · have hz : load I a v = 0 := by
  25       apply card_eq_zero.mpr
  26       apply Finset.eq_empty_iff_forall_not_mem.mpr
  27       intro t ht
  28       have hd := (mem_filter.mp ht).2
  29       have hm : destination I a t ∈ S := by
  30         rcases destination_endpoint I a t with he | he
  31         · rw [he]; exact hl t
  32         · rw [he]; exact hr t
  33       exact hv (hd ▸ hm)
  34     rw [hz]
  35     omega
  36 
  37 end GraphChoices

  72 theorem short_horizon_policy (G : Graph L R) [Nonempty G] (N T : ℕ)
  73     (hT : 0 < T) (hlog : 2 ≤ Real.log (N:ℝ)) (hshort : (T:ℝ) ≤ Real.log (N:ℝ)) :
  74     expectedCost (graphBase G) (onlineOrientation (T := T) (simpleGreedyPolicy G)) ≤
  75       12*loglogFactor N*expectedOpt (graphBase G) T := by
  76   have hgreedy : expectedCost (graphBase G) (onlineOrientation (T := T) (simpleGreedyPolicy G)) ≤
  77       (2*(Nat.log 2 (2*T)+1):ℕ)*expectedOpt (graphBase G) T := by
  78     unfold expectedCost expectedOpt
  79     rw [← average_mul]
  80     apply average_mono
  81     intro ω
  82     have hh := greedy_bound_by_arrivals ((graphBase G).sample ω) _ (simpleGreedyPolicy_isGreedy G ω) hT
  83     exact_mod_cast (by nlinarith : maxLoad ((graphBase G).sample ω) (onlineOrientation (simpleGreedyPolicy G) ω) ≤
  84       (2*(Nat.log 2 (2*T)+1))*opt ((graphBase G).sample ω))
  85   have hln := nat_log_le_twice_real_log (2*T) (by omega)
  86   have hTp : (0:ℝ) < T := by exact_mod_cast hT
  87   have hlogT := Real.log_le_log hTp hshort
  88   have hloglog := Real.log_le_log (by linarith : 0 < Real.log (N:ℝ)) (by linarith : Real.log (N:ℝ) ≤ 1+Real.log (N:ℝ))
  89   have hN2 : (2:ℝ) ≤ N := by
  90     by_contra hn
  91     have hnat : N < 2 := by exact_mod_cast lt_of_not_ge hn
  92     have hz : N = 0 ∨ N = 1 := by omega
  93     rcases hz with rfl | rfl <;> norm_num at hlog
  94   have hfactor : loglogFactor N = 1+Real.log (1+Real.log (N:ℝ)) := by unfold loglogFactor; rw [max_eq_right hN2]
  95   have hlog2 : Real.log 2 ≤ 1 := by linarith [Real.log_two_lt_d9]
  96   rw [Nat.cast_mul, Real.log_mul (by norm_num) hTp.ne'] at hln
  97   norm_num only [Nat.cast_ofNat] at hln
  98   have hcoeff : (((2*(Nat.log 2 (2*T)+1):ℕ):ℝ)) ≤ 12*loglogFactor N := by
  99     push_cast
 100     rw [hfactor]
 101     have hh := Real.log_nonneg (show 1 ≤ 1+Real.log (N:ℝ) by linarith)
 102     linarith
