  11  theorem countP_ofFn_eq_sum {A : Type*} {n : ℕ} (f : Fin n → A) (P : A → Bool) :
  12     (List.ofFn f).countP P = ∑ i, if P (f i) then 1 else 0 := by
  13   induction n with
  14   | zero => simp
  15   | succ n ih =>
  16     rw [List.ofFn_succ, List.countP_cons, ih, Fin.sum_univ_succ]
  17     simp only [Nat.add_comm]
  18 
  19  theorem card_filter_list_indices {A : Type*} (es : List A) (P : A → Bool) :
  20     (univ.filter fun i : Fin es.length => P (es.get i) = true).card = es.countP P := by
  21   have hh := countP_ofFn_eq_sum es.get P
  22   rw [List.ofFn_get] at hh
  23   simp only [card_eq_sum_ones, sum_filter]
  24   exact hh.symm

  26  theorem opt_le_of_internalCount_le {V : Type*} [Fintype V] [DecidableEq V]
  27     {S T : ℕ} (J : Instance V S) (I : Instance V T)
  28     (h : ∀ A, internalCount J A ≤ internalCount I A) : opt J ≤ opt I := by
  29   apply (opt_le_iff_density J _).2
  30   intro A
  31   exact (h A).trans (density_le_opt I A)

  40  theorem replay_internalCount (es : List (Arrival L R C)) (A : Finset (L ⊕ R)) :
  41     internalCount (replayInstance es) A =
  42       es.countP (fun e => decide (Sum.inl e.left ∈ A ∧ Sum.inr e.right ∈ A)) := by
  43   simpa only [internalCount, replayInstance, decide_eq_true_eq] using
  44     GraphChoices.card_filter_list_indices es
  45       (fun e => decide (Sum.inl e.left ∈ A ∧ Sum.inr e.right ∈ A))

  47  theorem replay_opt_mono_sublist (xs ys : List (Arrival L R C)) (h : xs.Sublist ys) :
  48     opt (replayInstance xs) ≤ opt (replayInstance ys) := by
  49   apply GraphChoices.opt_le_of_internalCount_le
  50   intro A
  51   rw [replay_internalCount, replay_internalCount]
  52   have hh := (h.filter (fun e => decide (Sum.inl e.left ∈ A ∧ Sum.inr e.right ∈ A))).length_le
  53   simpa only [List.countP_eq_length_filter] using hh

  55  theorem residual_sublist (α : C → ℕ) (es : List (Arrival L R C)) :
  56     (residualArrivals (trace α es)).Sublist es := by
  57   have hh := List.Sublist.map Event.arrival (List.filter_sublist
  58     (l := trace α es) (p := fun ev => !ev.threshold))
  59   simpa only [residualArrivals, trace, traceFrom_arrivals] using hh

  61  theorem residual_opt_le_full (α : C → ℕ) (es : List (Arrival L R C)) :
  62     opt (replayInstance (residualArrivals (trace α es))) ≤ opt (replayInstance es) :=
  63   replay_opt_mono_sublist _ _ (residual_sublist α es)
