  15 def scaleNat (i : ℕ) : ℕ := 2^(2^(i+1))

  17 def roundBudget (B i : ℕ) : ℕ := 16*B / scaleNat i

  19  theorem scaleNat_pos (i : ℕ) : 0 < scaleNat i := by unfold scaleNat; positivity

  21  theorem scaleNat_succ (i : ℕ) : scaleNat (i+1) = (scaleNat i)^2 := by
  22   unfold scaleNat
  23   rw [← pow_mul, pow_succ]

  25  theorem roundBudget_zero (B : ℕ) : roundBudget B 0 = 4*B := by
  26   simp only [roundBudget, scaleNat]

  30 /-- Floor rounding leaves enough slack for the minimum active integer degree. -/
  31  theorem active_cutoff_at_scale (B i : ℕ) (d : ℝ) (hd : d ≤ 16*B) :
  32     d / (scaleNat i : ℝ)^2 ≤ (roundBudget B (i+1) + 1 : ℕ) := by
  33   have hpos := scaleNat_pos (i+1)
  34   have hdiv : 16*B < (16*B / scaleNat (i+1) + 1) * scaleNat (i+1) := by
  35     have hm := Nat.mod_lt (16*B) hpos
  36     have he := Nat.mod_add_div (16*B) (scaleNat (i+1))
  37     nlinarith
  38   have hcast : (16*B : ℝ) ≤ ((roundBudget B (i+1) + 1 : ℕ) : ℝ) * (scaleNat i : ℝ)^2 := by
  39     have he : scaleNat (i+1) = scaleNat i ^ 2 := scaleNat_succ i
  40     have hr : ((16*B : ℕ) : ℝ) < ((16*B / scaleNat (i+1) + 1 : ℕ) : ℝ) * (scaleNat (i+1) : ℝ) := by
  41       exact_mod_cast hdiv
  42     simpa [roundBudget, he] using hr.le
  43   have hscale : (0:ℝ) < (scaleNat i : ℝ) := by exact_mod_cast scaleNat_pos i
  44   apply (div_le_iff₀ (sq_pos_of_pos hscale)).mpr
  45   exact hd.trans hcast

  47 noncomputable def skewSchedule (G : Graph L R) (B h : ℕ) (d n : ℝ) : List (ℕ × ℕ) :=
  48   List.ofFn fun i : Fin h =>
  49     (roundBudget B (i.val+1), scaleCapacity G d n (scaleNat i.val))

  51  theorem skewSchedule_length (G : Graph L R) (B h : ℕ) (d n : ℝ) :
  52     (skewSchedule G B h d n).length = h := by simp [skewSchedule]

  54  theorem skewSchedule_expands (G : Graph L R) (B h : ℕ) (d n : ℝ)
  55     (hd : 0 < d) (hn : 1 < Real.log n)
  56     (hleft : ∀ u, (leftDegree G u : ℝ) ≤ d * Real.log n) (hB : d ≤ 16*B) :
  57     ∀ kb ∈ skewSchedule G B h d n, PeelingExpansion G kb.1 kb.2 := by
  58   intro kb hkb
  59   obtain ⟨i,rfl⟩ := List.mem_ofFn.mp hkb
  60   apply peelingExpansion_of_graphLogSkewness G _ d n (scaleNat i.val) hd _ hn hleft
  61     (active_cutoff_at_scale B i.val d hB)
  62   exact_mod_cast (scaleNat_pos i.val)

  64 /-- A zero cutoff in the last round makes the schedule end at zero. -/
  65  theorem finalBudget_append (D : ℕ) (ss : List (ℕ × ℕ)) (kb : ℕ × ℕ) :
  66     finalBudget D (ss ++ [kb]) = kb.1 := by
  67   induction ss generalizing D with
  68   | nil => rfl
  69   | cons a ss ih => exact ih a.1

  71  theorem skewSchedule_finalBudget (G : Graph L R) (B h : ℕ) (d n : ℝ) :
  72     finalBudget (roundBudget B 0) (skewSchedule G B h d n) = roundBudget B h := by
  73   induction h with
  74   | zero => simp [skewSchedule, finalBudget]
  75   | succ h ih =>
  76     unfold skewSchedule
  77     rw [List.ofFn_succ']
  78     simp only [List.concat_eq_append]
  79     rw [finalBudget_append]
  80     rfl

  82 /-- Concrete doubly exponential decomposition. There is one piece for each
  83 scale 2^(2^(i+1)); their exact integer degree caps are recorded by pieceBudgets.
  84 The condition 16 B < scaleNat h certifies that all edges have been extracted. -/
  85  theorem exists_doubly_exponential_decomposition (G : Graph L R) (B h : ℕ) (d n : ℝ)
  86     (hd : 0 < d) (hn : 1 < Real.log n)
  87     (hleft : ∀ u, (leftDegree G u : ℝ) ≤ d * Real.log n)
  88     (hB : d ≤ 16*B) (hD : ∀ u, leftDegree G u ≤ 4*B)
  89     (hstop : 16*B < scaleNat h) :
  90     ∃ Ps : List (Graph L R), Ps.length = h ∧ Ps.Pairwise Disjoint ∧
  91       unionPieces Ps = G ∧
  92       List.Forall₂ (fun P kb => (∀ u, leftDegree P u ≤ kb.1) ∧
  93         (∀ v, rightDegree P v ≤ kb.2)) Ps
  94         (pieceBudgets (4*B) (skewSchedule G B h d n)) := by
  95   have hfinal : finalBudget (4*B) (skewSchedule G B h d n) = 0 := by
  96     rw [← roundBudget_zero B, skewSchedule_finalBudget]
  97     exact Nat.div_eq_of_lt hstop
  98   obtain ⟨Ps,hlen,hdisj,hcover,hbounds⟩ := exists_edge_partition G (4*B) hD
  99     (skewSchedule G B h d n) (skewSchedule_expands G B h d n hd hn hleft hB) hfinal
 100   exact ⟨Ps,hlen.trans (skewSchedule_length G B h d n),hdisj,hcover,hbounds⟩

 102 theorem pieceBudgets_append (D : ℕ) (ss : List (ℕ × ℕ)) (kb : ℕ × ℕ) :
 103     pieceBudgets D (ss ++ [kb]) = pieceBudgets D ss ++
 104       [(finalBudget D ss, finalBudget D ss * kb.2)] := by
 105   induction ss generalizing D with
 106   | nil => rfl
 107   | cons a ss ih => simpa [pieceBudgets, finalBudget] using congrArg (List.cons (D,D*a.2)) (ih a.1)

 109 /-- The generic iteration certificate has exactly the intended scale-by-scale
 110 integer degree caps. -/
 111 theorem pieceBudgets_skewSchedule (G : Graph L R) (B h : ℕ) (d n : ℝ) :
 112     pieceBudgets (4*B) (skewSchedule G B h d n) =
 113       List.ofFn (fun i : Fin h => (roundBudget B i.val,
 114         roundBudget B i.val * scaleCapacity G d n (scaleNat i.val))) := by
 115   induction h with
 116   | zero => simp [skewSchedule, pieceBudgets]
 117   | succ h ih =>
 118     have heq : skewSchedule G B (h+1) d n = skewSchedule G B h d n ++
 119         [(roundBudget B (h+1), scaleCapacity G d n (scaleNat h))] := by
 120       unfold skewSchedule
 121       rw [List.ofFn_succ']
 122       simp only [List.concat_eq_append]
 123       rfl
 124     rw [heq, pieceBudgets_append, ih]
 125     have hfinal : finalBudget (4*B) (skewSchedule G B h d n) = roundBudget B h := by
 126       rw [← roundBudget_zero B, skewSchedule_finalBudget]
 127     rw [hfinal, List.ofFn_succ']
 128     simp only [List.concat_eq_append]
 129     rfl

 131 theorem scaleNat_ge_two (i : ℕ) : 2 ≤ scaleNat i := by
 132   have hp : 1 ≤ 2^(i+1) := Nat.one_le_iff_ne_zero.mpr (pow_ne_zero _ (by decide))
 133   simpa [scaleNat] using Nat.pow_le_pow_right (by decide : 0 < 2) hp

 135 theorem roundBudget_mul_scale (B i : ℕ) : roundBudget B i * scaleNat i ≤ 16*B :=
 136   Nat.div_mul_le_self _ _

 138 theorem roundBudget_real_le (B i : ℕ) :
 139     (roundBudget B i : ℝ) ≤ (16*B : ℝ) / scaleNat i := by
 140   have hp : (0:ℝ) < scaleNat i := by exact_mod_cast scaleNat_pos i
 141   apply (le_div_iff₀ hp).mpr
 142   exact_mod_cast roundBudget_mul_scale B i

 144 /-- Integer rounding of the right capacity costs no extra constant, since each
 145 scale is at least two and the unrounded right-capacity factor is at least one. -/
 146 theorem rounded_capacity_product (B i : ℕ) (Q : ℝ) (hQ : 1 ≤ Q) :
 147     ((roundBudget B i * ⌈Q⌉₊ : ℕ) : ℝ) ≤ 16*B*Q := by
 148   have hscale : (2:ℝ) ≤ scaleNat i := by exact_mod_cast scaleNat_ge_two i
 149   have hmul : (roundBudget B i : ℝ) * scaleNat i ≤ 16*B := by
 150     exact_mod_cast roundBudget_mul_scale B i
 151   have hnonneg : (0:ℝ) ≤ roundBudget B i := Nat.cast_nonneg _
 152   have hsmall : (roundBudget B i : ℝ)*2 ≤ 16*B := by nlinarith
 153   have hceil : (⌈Q⌉₊ : ℝ) ≤ 2*Q := by
 154     have hh := Nat.ceil_lt_add_one (by linarith : 0 ≤ Q)
 155     linarith
 156   calc
 157     _ = (roundBudget B i : ℝ) * (⌈Q⌉₊ : ℝ) := by norm_cast
 158     _ ≤ (roundBudget B i : ℝ) * (2*Q) := mul_le_mul_of_nonneg_left hceil hnonneg
 159     _ = ((roundBudget B i : ℝ)*2)*Q := by ring
 160     _ ≤ 16*B*Q := mul_le_mul_of_nonneg_right hsmall (by linarith)

 162 /-- An explicit log-log number of matching-extraction rounds. -/
 163 def numberRounds (B : ℕ) : ℕ := Nat.log 2 (Nat.log 2 (16*B) + 1)
 164 
 165  theorem numberRounds_suffices (B : ℕ) : 16*B < scaleNat (numberRounds B) := by
 166   have houter := Nat.lt_pow_succ_log_self (by decide : 1 < 2) (16*B)
 167   have hinner := Nat.lt_pow_succ_log_self (by decide : 1 < 2) (Nat.log 2 (16*B) + 1)
 168   exact houter.trans_le (Nat.pow_le_pow_right (by decide : 0 < 2) (Nat.le_of_lt hinner))

 170 /-- A bounded-left-degree graph admits an exact partition with the explicit
 171 log-log piece count; no termination or expansion conclusion is assumed. -/
 172 theorem exists_loglog_skew_decomposition (G : Graph L R) (B : ℕ) (d n : ℝ)
 173     (hd : 0 < d) (hn : 1 < Real.log n)
 174     (hleft : ∀ u, (leftDegree G u : ℝ) ≤ d * Real.log n)
 175     (hB : d ≤ 16*B) (hD : ∀ u, leftDegree G u ≤ 4*B) :
 176     ∃ Ps : List (Graph L R), Ps.length = numberRounds B ∧ Ps.Pairwise Disjoint ∧
 177       unionPieces Ps = G ∧
 178       List.Forall₂ (fun P kb => (∀ u, leftDegree P u ≤ kb.1) ∧
 179         (∀ v, rightDegree P v ≤ kb.2)) Ps
 180         (pieceBudgets (4*B) (skewSchedule G B (numberRounds B) d n)) :=
 181   exists_doubly_exponential_decomposition G B _ d n hd hn hleft hB hD (numberRounds_suffices B)

 183 /-- Structural existence part of the skew-biregular decomposition, stated with
 184 explicit real degree bounds and an exact log-log piece count. -/
 185 theorem exists_skew_biregular_partition (G : Graph L R) (B : ℕ) (d n : ℝ)
 186     (hd : 0 < d) (hn : 1 < Real.log n)
 187     (hleft : ∀ u, (leftDegree G u : ℝ) ≤ d * Real.log n)
 188     (hB : d ≤ 16*B) (hD : ∀ u, leftDegree G u ≤ 4*B) :
 189     ∃ Ps : List (Graph L R), Ps.length = numberRounds B ∧ Ps.Pairwise Disjoint ∧
 190       unionPieces Ps = G ∧ ∀ i : Fin Ps.length,
 191         (∀ u, (leftDegree (Ps.get i) u : ℝ) ≤ 16*B/(scaleNat i.val)) ∧
 192         (∀ v, (rightDegree (Ps.get i) v : ℝ) ≤
 193           16*B*((scaleNat i.val : ℝ)*Real.log n)^(2*graphLogSkewness G d n)) := by
 194   obtain ⟨Ps,hlen,hdisj,hcover,hbounds⟩ := exists_loglog_skew_decomposition G B d n hd hn hleft hB hD
 195   rw [pieceBudgets_skewSchedule] at hbounds
 196   refine ⟨Ps,hlen,hdisj,hcover,?_⟩
 197   intro i
 198   have hi : i.val < numberRounds B := by rw [← hlen]; exact i.isLt
 199   have hh := hbounds.get i.isLt (by simp [hi])
 200   simp only [List.get_eq_getElem, List.getElem_ofFn] at hh
 201   constructor
 202   · intro u
 203     exact (by exact_mod_cast hh.1 u : (leftDegree (Ps.get i) u : ℝ) ≤ roundBudget B i.val).trans
 204       (roundBudget_real_le B i.val)
 205   · intro v
 206     have hQ : (1:ℝ) ≤ ((scaleNat i.val : ℝ)*Real.log n)^(2*graphLogSkewness G d n) := by
 207       have hs : (2:ℝ) ≤ scaleNat i.val := by exact_mod_cast scaleNat_ge_two i.val
 208       apply Real.one_le_rpow
 209       · nlinarith
 210       · exact mul_nonneg (by norm_num) (graphLogSkewness_nonneg G d n)
 211     have hright : (rightDegree (Ps.get i) v : ℝ) ≤
 212         ((roundBudget B i.val * scaleCapacity G d n (scaleNat i.val) : ℕ) : ℝ) := by
 213       exact_mod_cast hh.2 v
 214     exact hright.trans (rounded_capacity_product B i.val _ hQ)
