  12 theorem mem_residual_sample_iff (encode : E → Arrival L R C) (α : C → ℕ)
  13     (ω : Fin T → E) (e : Arrival L R C) :
  14     e ∈ residualArrivals (trace α (sampleArrivals encode ω)) ↔
  15       ∃ t : Fin T, encode (ω t) = e ∧ residualTime α encode ω t := by
  16   constructor
  17   · intro he
  18     obtain ⟨ev, hev, ha⟩ := List.mem_map.mp he
  19     obtain ⟨htrace, hres⟩ := List.mem_filter.mp hev
  20     have hfalse : ev.threshold = false := by simpa using hres
  21     obtain ⟨i, htime, _⟩ := trace_mem_arrival α (sampleArrivals encode ω) ev htrace
  22     let t : Fin T := ⟨ev.time, by
  23       have hi := i.isLt
  24       have hlen := sampleArrivals_length encode ω
  25       omega⟩
  26     have hs := sample_event_arrival α encode ω ev htrace t rfl
  27     exact ⟨t, hs.symm.trans ha, ev, htrace, rfl, hfalse⟩
  28   · rintro ⟨t, he, ev, hev, ht, hres⟩
  29     apply List.mem_map.mpr
  30     refine ⟨ev, List.mem_filter.mpr ⟨hev, by simpa [hres]⟩, ?_⟩
  31     exact (sample_event_arrival α encode ω ev hev t ht).trans he

  33 theorem residual_adj_iff_time (encode : E → Arrival L R C) (α : C → ℕ)
  34     (ω : Fin T → E) (x y : L ⊕ R) :
  35     (arrivalGraph (residualInstance α (sampleArrivals encode ω))).Adj x y ↔
  36       x ≠ y ∧ ∃ t : Fin T, residualTime α encode ω t ∧
  37         ((Sum.inl (encode (ω t)).left = x ∧ Sum.inr (encode (ω t)).right = y) ∨
  38          (Sum.inl (encode (ω t)).left = y ∧ Sum.inr (encode (ω t)).right = x)) := by
  39   rw [residualInstance, replay_adj_iff]
  40   constructor
  41   · rintro ⟨hne, e, he, hxy⟩
  42     obtain ⟨t, ht, hres⟩ := (mem_residual_sample_iff encode α ω e).mp he
  43     exact ⟨hne, t, hres, by simpa only [ht] using hxy⟩
  44   · rintro ⟨hne, t, hres, hxy⟩
  45     exact ⟨hne, encode (ω t), (mem_residual_sample_iff encode α ω _).mpr ⟨t, rfl, hres⟩, hxy⟩
  46 
  47 theorem residual_adj_inl_inr_iff_time (encode : E → Arrival L R C) (α : C → ℕ)
  48     (ω : Fin T → E) (u : L) (v : R) :
  49     (arrivalGraph (residualInstance α (sampleArrivals encode ω))).Adj (Sum.inl u) (Sum.inr v) ↔
  50       ∃ t : Fin T, residualTime α encode ω t ∧
  51         (encode (ω t)).left = u ∧ (encode (ω t)).right = v := by
  52   rw [residual_adj_iff_time]
  53   simp
