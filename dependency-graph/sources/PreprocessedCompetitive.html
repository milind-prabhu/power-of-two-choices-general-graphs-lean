   9 theorem loglogFactor_ge_one (N : ℕ) : 1 ≤ loglogFactor N := by
  10   unfold loglogFactor
  11   have hh : 0 ≤ Real.log (max 2 (N:ℝ)) := Real.log_nonneg (by have := le_max_left (2:ℝ) (N:ℝ); linarith)
  12   have hh' : 0 ≤ Real.log (1+Real.log (max 2 (N:ℝ))) := Real.log_nonneg (by linarith)
  13   linarith

  15 theorem small_log_vertex_bound (N : ℕ) (hlog : Real.log (N:ℝ) < 2) : N ≤ 8 := by
  16   by_cases hN : N = 0
  17   · omega
  18   have hNr : (0:ℝ) < N := by exact_mod_cast Nat.pos_of_ne_zero hN
  19   have hh := Real.exp_lt_exp.mpr hlog
  20   rw [Real.exp_log hNr] at hh
  21   have he : Real.exp 2 = (Real.exp 1)^2 := by simpa using Real.exp_nat_mul 1 2
  22   have he3 : Real.exp 1 < 3 := by linarith [Real.exp_one_lt_d9]
  23   have hepos := Real.exp_pos 1
  24   have he9 : (Real.exp 1)^2 < 9 := by nlinarith
  25   have hn9 : (N:ℝ) < 9 := by rw [he] at hh; linarith
  26   have hn : N < 9 := by exact_mod_cast hn9
  27   omega

  43 /-- The constructed Threshold-Greedy policy is O(log log N)-competitive
  44 under the graph preprocessing conditions, with no unproved skewness premise. -/
  45 theorem PreparedDecomposition.expected_cost_competitive (P : PreparedDecomposition G B N d)
  46     (hB : B = (graphBase G).referenceCap) (hvertices : Fintype.card (L ⊕ R) ≤ N)
  47     (haverage : 2*(G.card:ℝ) = (N:ℝ)*d) :
  48     expectedCost (graphBase G) (onlineOrientation (T := N) (P.policy N)) ≤
  49       37800*loglogFactor N*expectedOpt (graphBase G) N := by
  50   have hfactor := loglogFactor_ge_one N
  51   by_cases hlog : 2 ≤ Real.log (N:ℝ)
  52   · have hs := graphLogSkewness_le_eight_expectedOpt G N d P.positive_average hlog
  53       P.preprocessing hvertices haverage
  54     have hh := P.real_competitive_of_skew_lower_bound hB hvertices haverage 8 hs
  55     norm_num at hh
  56     exact hh
  57   · have hN8 := small_log_vertex_bound N (lt_of_not_ge hlog)
  58     have hopt := expectedOpt_ge_one (graphBase G) N P.vertex_count_pos
  59     apply (expectedCost_le_horizon (graphBase G) N _).trans
  60     have hNr : (N:ℝ) ≤ 8 := by exact_mod_cast hN8
  61     have hh : 1 ≤ loglogFactor N*expectedOpt (graphBase G) N := by nlinarith
  62     nlinarith

  64 /-- A graph-level competitive theorem at T=N. The remaining assumptions
  65 are solely concrete preprocessing, vertex-count and average-degree facts. -/
  66 theorem exists_preprocessed_competitive_policy (G : Graph L R) [Nonempty G]
  67     (N : ℕ) (d : ℝ) (hd : 0 < d)
  68     (hleft : ∀ u, (leftDegree G u:ℝ) ≤ d*Real.log (N:ℝ))
  69     (hD : ∀ u, leftDegree G u ≤ 4*(graphBase G).referenceCap)
  70     (hvertices : Fintype.card (L ⊕ R) ≤ N)
  71     (haverage : 2*(G.card:ℝ) = (N:ℝ)*d) :
  72     ∃ p : OnlinePolicy G, expectedCost (graphBase G) (onlineOrientation (T := N) p) ≤
  73       37800*loglogFactor N*expectedOpt (graphBase G) N := by
  74   have hN : 0 < N := by
  75     have hm : (0:ℝ) < G.card := by
  76       have hh : 0 < G.card := by simpa only [← Fintype.card_coe] using Fintype.card_pos (α := G)
  77       exact_mod_cast hh
  78     by_contra h
  79     have hz : N = 0 := by omega
  80     simp only [hz, Nat.cast_zero, zero_mul] at haverage
  81     linarith
  82   by_cases hlog : 2 ≤ Real.log (N:ℝ)
  83   · have hlog1 : 1 < Real.log (N:ℝ) := by linarith
  84     have hcap := average_le_two_referenceCap G N d hN hvertices haverage
  85     have hB : d ≤ 16*(graphBase G).referenceCap := by
  86       have hh : (0:ℝ) ≤ (graphBase G).referenceCap := Nat.cast_nonneg _
  87       linarith
  88     obtain ⟨P⟩ := exists_preparedDecomposition G (graphBase G).referenceCap N d hd hlog1 hleft hB hD
  89     exact ⟨P.policy N, P.expected_cost_competitive rfl hvertices haverage⟩
  90   · refine ⟨fun _ _ => false, (expectedCost_le_horizon (graphBase G) N _).trans ?_⟩
  91     have hN8 : (N:ℝ) ≤ 8 := by exact_mod_cast small_log_vertex_bound N (lt_of_not_ge hlog)
  92     have hopt := expectedOpt_ge_one (graphBase G) N hN
  93     have hfactor := loglogFactor_ge_one N
  94     have hh : 1 ≤ loglogFactor N*expectedOpt (graphBase G) N := by nlinarith
  95     nlinarith
