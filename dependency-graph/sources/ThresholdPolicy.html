   4 /-! The implemented full Threshold-Greedy state is an actual online policy.
   5 Its expected load and full-sample benchmark agree with the original indexed
   6 sample model, rather than a separate list-only performance notion. -/
   7 namespace GraphChoices.ThresholdGreedy
   8 open Finset
   9 variable {E L R C : Type*} [Fintype E] [DecidableEq E] [Nonempty E]
  10   [Fintype L] [Fintype R] [Fintype C]
  11   [DecidableEq L] [DecidableEq R] [DecidableEq C]
  12 
  13 def encodedBaseGraph (encode : E → Arrival L R C) : BaseGraph (L ⊕ R) E where
  14   left e := Sum.inl (encode e).left
  15   right e := Sum.inr (encode e).right
  16 
  17 theorem encoded_internalCount (encode : E → Arrival L R C) {T : ℕ} (ω : Fin T → E)
  18     (A : Finset (L ⊕ R)) :
  19     internalCount (replayInstance (sampleArrivals encode ω)) A =
  20       internalCount ((encodedBaseGraph encode).sample ω) A := by
  21   rw [replay_internalCount]
  22   unfold sampleArrivals
  23   rw [countP_ofFn_eq_sum]
  24   simp only [internalCount, BaseGraph.sample, encodedBaseGraph, card_eq_sum_ones, sum_filter,
  25     decide_eq_true_eq]
  26 
  27 theorem encoded_opt (encode : E → Arrival L R C) {T : ℕ} (ω : Fin T → E) :
  28     opt (replayInstance (sampleArrivals encode ω)) = opt ((encodedBaseGraph encode).sample ω) := by
  29   apply Nat.le_antisymm
  30   · apply opt_le_of_internalCount_le
  31     intro A
  32     exact (encoded_internalCount encode ω A).le
  33   · apply opt_le_of_internalCount_le
  34     intro A
  35     exact (encoded_internalCount encode ω A).symm.le
  36 
  37 theorem expectedReplayOpt_eq_expectedOpt (encode : E → Arrival L R C) (T : ℕ) :
  38     expectedReplayOpt encode T = expectedOpt (encodedBaseGraph encode) T := by
  39   unfold expectedReplayOpt expectedOpt
  40   congr 1
  41   funext ω
  42   rw [encoded_opt]
  43 
  44 def stateLoad (s : State L R C) : L ⊕ R → ℕ :=
  45   Sum.elim (fun u => (∑ c, s.threshold u c) + s.greedyLeft u) s.greedyRight
  46 
  47 def stepChoice (α : C → ℕ) (s : State L R C) (e : Arrival L R C) : Bool :=
  48   if s.threshold e.left e.cls < α e.cls then true
  49   else decide (s.greedyLeft e.left < s.greedyRight e.right)
  50 
  51 theorem step_stateLoad (α : C → ℕ) (s : State L R C) (e : Arrival L R C) (v : L ⊕ R) :
  52     stateLoad (step α s e) v = stateLoad s v +
  53       if (if stepChoice α s e then Sum.inl e.left else Sum.inr e.right) = v then 1 else 0 := by
  54   by_cases ht : s.threshold e.left e.cls < α e.cls
  55   · cases v with
  56     | inl u =>
  57       by_cases hu : u = e.left <;>
  58         simp [stateLoad, step, stepChoice, ht, hu, sum_add_distrib, eq_comm, Nat.add_assoc,
  59           Nat.add_comm, Finset.filter_eq']
  60     | inr w => simp [stateLoad, step, stepChoice, ht]
  61   · by_cases hg : s.greedyLeft e.left < s.greedyRight e.right <;> cases v <;>
  62       simp [stateLoad, step, stepChoice, ht, hg, eq_comm, Nat.add_assoc]
  63 
  64 /-- Only the preceding source labels and current label are inputs. -/
  65 def thresholdPolicy (encode : E → Arrival L R C) (α : C → ℕ) : OnlinePolicy E :=
  66   fun history e => stepChoice α (run α (history.map encode)) (encode e)
  67 
  68 def policyPrefixLoad (encode : E → Arrival L R C) (α : C → ℕ) {T : ℕ}
  69     (ω : Fin T → E) (k : ℕ) : L ⊕ R → ℕ :=
  70   stateLoad (run α (((List.ofFn ω).take k).map encode))
  71 
  72 theorem policyPrefixLoad_zero (encode : E → Arrival L R C) (α : C → ℕ) {T : ℕ}
  73     (ω : Fin T → E) (v : L ⊕ R) : policyPrefixLoad encode α ω 0 v = 0 := by
  74   cases v <;> simp [policyPrefixLoad, run, runFrom, initial, stateLoad]
  75 
  76 theorem policyPrefixLoad_step (encode : E → Arrival L R C) (α : C → ℕ) {T : ℕ}
  77     (ω : Fin T → E) (t : Fin T) (v : L ⊕ R) :
  78     policyPrefixLoad encode α ω (t.val+1) v = policyPrefixLoad encode α ω t.val v +
  79       if destination ((encodedBaseGraph encode).sample ω) (onlineOrientation (thresholdPolicy encode α) ω) t = v
  80       then 1 else 0 := by
  81   have hi : t.val < (List.ofFn ω).length := by simpa using t.isLt
  82   unfold policyPrefixLoad
  83   rw [List.take_succ_eq_append_getElem hi, List.map_append, List.map_singleton, run_append_singleton]
  84   rw [step_stateLoad]
  85   simp [List.getElem_ofFn, destination, encodedBaseGraph, BaseGraph.sample,
  86     onlineOrientation, thresholdPolicy]
  87 
  88 theorem policy_final_load (encode : E → Arrival L R C) (α : C → ℕ) {T : ℕ}
  89     (ω : Fin T → E) (v : L ⊕ R) :
  90     totalVertexLoad α (sampleArrivals encode ω) v =
  91       load ((encodedBaseGraph encode).sample ω) (onlineOrientation (thresholdPolicy encode α) ω) v := by
  92   have h := execution_final_load ((encodedBaseGraph encode).sample ω)
  93     (onlineOrientation (thresholdPolicy encode α) ω) (policyPrefixLoad encode α ω)
  94     (policyPrefixLoad_zero encode α ω) (policyPrefixLoad_step encode α ω) v
  95   have htake : (List.ofFn ω).take T = List.ofFn ω := List.take_of_length_le (by simp)
  96   simpa only [policyPrefixLoad, htake, sampleArrivals, List.map_ofFn, stateLoad, totalVertexLoad] using h
  97 
  98 theorem policy_maxLoad (encode : E → Arrival L R C) (α : C → ℕ) {T : ℕ} (ω : Fin T → E) :
  99     totalMaxLoad α (sampleArrivals encode ω) =
 100       maxLoad ((encodedBaseGraph encode).sample ω) (onlineOrientation (thresholdPolicy encode α) ω) := by
 101   unfold totalMaxLoad maxLoad
 102   apply Finset.sup_congr rfl
 103   intro v _
 104   exact policy_final_load encode α ω v
 105 
 106 theorem expectedTotalMaxLoad_eq_expectedCost (encode : E → Arrival L R C) (α : C → ℕ) (T : ℕ) :
 107     expectedTotalMaxLoad encode α T =
 108       expectedCost (encodedBaseGraph encode) (onlineOrientation (T := T) (thresholdPolicy encode α)) := by
 109   unfold expectedTotalMaxLoad expectedCost
 110   congr 1
 111   funext ω
 112   rw [policy_maxLoad]

 114 end GraphChoices.ThresholdGreedy
