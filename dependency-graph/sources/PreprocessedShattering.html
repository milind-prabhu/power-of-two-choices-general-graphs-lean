   9 theorem real_log_le_nat_log_shift (N : ℕ) (hN : 0 < N) :
  10     Real.log (N:ℝ) ≤ (Nat.log 2 N+1 : ℕ) := by
  11   have hNr : (0:ℝ) < N := by exact_mod_cast hN
  12   have hp : (N:ℝ) ≤ (2:ℝ)^(Nat.log 2 N+1) := by
  13     exact_mod_cast (Nat.lt_pow_succ_log_self (by decide : 1 < 2) N).le
  14   have hlog := Real.log_le_log hNr hp
  15   rw [Real.log_pow] at hlog
  16   have htwo : Real.log 2 ≤ 1 := by
  17     have hh := Real.log_le_sub_one_of_pos (by norm_num : (0:ℝ) < 2)
  18     linarith
  19   have hmul := mul_le_mul_of_nonneg_left htwo (Nat.cast_nonneg (Nat.log 2 N+1) : (0:ℝ) ≤ (Nat.log 2 N+1 : ℕ))
  20   exact hlog.trans (by simpa using hmul)

  22 theorem preprocessing_mean_bound (M T N E : ℕ) (d : ℝ)
  23     (hE : 0 < E) (hT : T ≤ N)
  24     (hdegree : (M:ℝ) ≤ d*Real.log (N:ℝ))
  25     (haverage : 2*(E:ℝ) = (N:ℝ)*d) :
  26     (T:ℝ)*M/E ≤ 2*Real.log (N:ℝ) := by
  27   have hEr : (0:ℝ) < E := by exact_mod_cast hE
  28   apply (div_le_iff₀ hEr).mpr
  29   calc
  30     (T:ℝ)*M ≤ (N:ℝ)*M := mul_le_mul_of_nonneg_right (by exact_mod_cast hT) (Nat.cast_nonneg M)
  31     _ ≤ (N:ℝ)*(d*Real.log (N:ℝ)) := mul_le_mul_of_nonneg_left hdegree (Nat.cast_nonneg N)
  32     _ = (2*(E:ℝ))*Real.log (N:ℝ) := by rw [haverage]; ring
  33     _ = _ := by ring

  35 theorem preprocessed_mean_below_cutoff (N c : ℕ) (μ : ℝ) (hN : 0 < N)
  36     (hc : 5 ≤ c) (hμ : 0 ≤ μ) (hmean : μ ≤ 2*Real.log (N:ℝ)) :
  37     Real.exp 1*μ ≤ (polynomialCutoff N c : ℝ) := by
  38   have he : Real.exp 1 ≤ 3 := by linarith [Real.exp_one_lt_d9]
  39   have hlog := real_log_le_nat_log_shift N hN
  40   have hc' : (6:ℝ) ≤ (c:ℝ)+1 := by exact_mod_cast (by omega : 6 ≤ c+1)
  41   calc
  42     _ ≤ 3*μ := mul_le_mul_of_nonneg_right he hμ
  43     _ ≤ 3*(2*Real.log (N:ℝ)) := mul_le_mul_of_nonneg_left hmean (by norm_num)
  44     _ ≤ 6*(Nat.log 2 N+1 : ℕ) := by linarith
  45     _ ≤ ((c:ℝ)+1)*(Nat.log 2 N+1 : ℕ) :=
  46       mul_le_mul_of_nonneg_right hc' (Nat.cast_nonneg _)
  47     _ = _ := by simp only [polynomialCutoff, Nat.cast_mul, Nat.cast_add, Nat.cast_one]

  54 /-- Finite form of the component-shattering lemma in the preprocessed regime.
  55 All coverage and sampling-tail premises are proved, including the source mean. -/
  56 theorem preprocessed_log_squared_component_tail (H : C → Peeling.Graph L R)
  57     (encode : E → Arrival L R C) (DL DR : C → ℕ) (D M T N c : ℕ) (f : C → ℝ) (d : ℝ)
  58     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
  59     (hsource : ∀ e, ((encode e).left, (encode e).right) ∈ H (encode e).cls)
  60     (hDL : ∀ u a, (sourceBucket encode u a).card ≤ DL a)
  61     (hL : ∀ a u, Peeling.leftDegree (H a) u ≤ D)
  62     (hR : ∀ a v, Peeling.rightDegree (H a) v ≤ DR a)
  63     (hM : ∀ u, (leftSourceBucket encode u).card ≤ M)
  64     (hT : 0 < T) (hTN : T ≤ N) (hf : ∀ a, 1 < f a)
  65     (hN : 0 < N) (hLN : Fintype.card L ≤ N) (hRN : Fintype.card R ≤ N)
  66     (hc : 5 ≤ c) (hdegree : (M:ℝ) ≤ d*Real.log (N:ℝ))
  67     (haverage : 2*(Fintype.card E : ℝ) = (N:ℝ)*d) :
  68     eventProbability (largeComponentEvent encode (automaticThresholds (E := E) DL DR D T f)
  69       T (3*(polynomialCutoff N c+1)^2)) ≤ 3/(N:ℝ)^c := by
  70   apply log_squared_component_tail H encode DL DR D M T N c f hsimple hsource hDL hL hR hM hT hf hN hLN hRN
  71   have hmean := preprocessing_mean_bound M T N (Fintype.card E) d Fintype.card_pos hTN hdegree haverage
  72   have hbound := preprocessed_mean_below_cutoff N c ((T:ℝ)*M/Fintype.card E) hN hc (by positivity) hmean
  73   convert hbound using 1 <;> ring
