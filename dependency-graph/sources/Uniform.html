   7 /-! An exact finite probability space: every function Fin T → E is equally
   8 likely. Thus edge samples are independent and uniform, with replacement.
   9 Expectations are finite averages, so no measure-theoretic assumptions are hidden. -/
  10 namespace GraphChoices
  11 open Finset
  12 
  13 noncomputable def average {Ω : Type*} [Fintype Ω] (f : Ω → ℝ) : ℝ :=
  14   (∑ ω, f ω) / Fintype.card Ω
  15 
  16  theorem average_mono {Ω : Type*} [Fintype Ω] {f g : Ω → ℝ} (h : ∀ ω, f ω ≤ g ω) :
  17     average f ≤ average g := by
  18   unfold average
  19   exact div_le_div_of_nonneg_right (sum_le_sum fun ω _ => h ω) (Nat.cast_nonneg _)
  20 
  21  theorem average_sum {Ω K : Type*} [Fintype Ω] [Fintype K] (f : K → Ω → ℝ) :
  22     average (fun ω => ∑ k, f k ω) = ∑ k, average (f k) := by
  23   unfold average
  24   simp only [div_eq_mul_inv, sum_mul]
  25   rw [sum_comm]
  26 
  27  theorem average_mul {Ω : Type*} [Fintype Ω] (c : ℝ) (f : Ω → ℝ) :
  28     average (fun ω => c * f ω) = c * average f := by
  29   simp only [average, ← mul_sum]
  30   ring
  31 
  32  theorem average_const {Ω : Type*} [Fintype Ω] [Nonempty Ω] (c : ℝ) :
  33     average (fun _ : Ω => c) = c := by
  34   simp [average, Fintype.card_ne_zero]

  39  theorem sample_fiber_card (t : Fin T) (e : E) :
  40     (univ.filter fun ω : Fin T → E => ω t = e).card = Fintype.card E ^ (T-1) := by
  41   simpa using Fintype.card_filter_piFinset_const_eq_of_mem
  42     (ι := Fin T) (univ : Finset E) t (mem_univ e)

  44 omit [Nonempty E] in
  45  theorem sum_coordinate (t : Fin T) (f : E → ℝ) :
  46     (∑ ω : Fin T → E, f (ω t)) =
  47       (Fintype.card E : ℝ)^(T-1) * ∑ e, f e := by
  48   calc
  49     _ = ∑ e : E, ∑ ω ∈ (univ.filter fun ω : Fin T → E => ω t = e), f (ω t) :=
  50       (sum_fiberwise univ (fun ω : Fin T → E => ω t) (fun ω => f (ω t))).symm
  51     _ = ∑ e : E, (Fintype.card E : ℝ)^(T-1) * f e := by
  52       apply sum_congr rfl
  53       intro e _
  54       calc
  55         _ = ∑ _ω ∈ (univ.filter fun ω : Fin T → E => ω t = e), f e := by
  56           apply sum_congr rfl
  57           intro ω hω
  58           rw [(mem_filter.mp hω).2]
  59         _ = _ := by simp [sample_fiber_card, nsmul_eq_mul]
  60     _ = _ := by rw [mul_sum]
  61 
  62 /-- Each coordinate of the product space is exactly uniform. -/
  63  theorem average_coordinate (t : Fin T) (f : E → ℝ) :
  64     average (fun ω : Fin T → E => f (ω t)) = average f := by
  65   unfold average
  66   rw [sum_coordinate]
  67   have hT : T = (T-1)+1 := by have := t.isLt; omega
  68   have hm : (Fintype.card E : ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  69   simp only [Fintype.card_fun, Fintype.card_fin, Nat.cast_pow]
  70   have heq : (Fintype.card E : ℝ)^T = (Fintype.card E : ℝ)^(T-1) * Fintype.card E := by
  71     nth_rw 1 [hT]
  72     rw [pow_succ]
  73   rw [heq]
  74   field_simp

  76 /-- A base graph is represented by a finite, nonempty set of edge labels.
  77 Taking E to be the edges of a simple graph gives exactly the paper's model. -/
  78 structure BaseGraph (V E : Type*) where
  79   left : E → V
  80   right : E → V
  81 
  82 variable {V : Type*} [Fintype V] [DecidableEq V]
  83 
  84 def BaseGraph.sample (G : BaseGraph V E) (ω : Fin T → E) : Instance V T where
  85   left t := G.left (ω t)
  86   right t := G.right (ω t)
  87 
  88 def BaseGraph.internalEdges (G : BaseGraph V E) (S : Finset V) : Finset E :=
  89   univ.filter fun e => G.left e ∈ S ∧ G.right e ∈ S
  90 
  91 noncomputable def expectedOpt (G : BaseGraph V E) (T : ℕ) : ℝ :=
  92   average (fun ω : Fin T → E => (opt (G.sample ω) : ℝ))
  93 
  94 omit [Fintype V] in
  95  theorem average_internalCount (G : BaseGraph V E) (S : Finset V) (T : ℕ) :
  96     average (fun ω : Fin T → E => (internalCount (G.sample ω) S : ℝ)) =
  97       T * ((G.internalEdges S).card : ℝ) / Fintype.card E := by
  98   have hsum (ω : Fin T → E) : (internalCount (G.sample ω) S : ℝ) =
  99       ∑ t : Fin T, if G.left (ω t) ∈ S ∧ G.right (ω t) ∈ S then (1:ℝ) else 0 := by
 100     simp only [internalCount, BaseGraph.sample, card_eq_sum_ones, sum_filter, Nat.cast_sum, Nat.cast_ite, Nat.cast_one, Nat.cast_zero]
 101   simp_rw [hsum]
 102   rw [average_sum]
 103   have hk (k : Fin T) :
 104       average (fun ω : Fin T → E => if G.left (ω k) ∈ S ∧ G.right (ω k) ∈ S then (1:ℝ) else 0) =
 105       average (fun e : E => if G.left e ∈ S ∧ G.right e ∈ S then (1:ℝ) else 0) :=
 106     average_coordinate k (fun e : E => if G.left e ∈ S ∧ G.right e ∈ S then (1:ℝ) else 0)
 107   simp_rw [hk]
 108   simp [average, BaseGraph.internalEdges, sum_boole, mul_div_assoc]
 109 
 110 /-- Density lower bound for every subset and every number of arrivals.
 111 Equivalent to (16) after substituting d_av = 2|E|/n and maximizing over S. -/
 112  theorem expected_density_lower_bound (G : BaseGraph V E) (S : Finset V) (T : ℕ) :
 113     T * ((G.internalEdges S).card : ℝ) / Fintype.card E ≤
 114       S.card * expectedOpt G T := by
 115   rw [← average_internalCount]
 116   unfold expectedOpt
 117   rw [← average_mul]
 118   apply average_mono
 119   intro ω
 120   exact_mod_cast density_le_opt (G.sample ω) S

 123  theorem expectedOpt_ge_one (G : BaseGraph V E) (T : ℕ) (hT : 0 < T) :
 124     1 ≤ expectedOpt G T := by
 125   rw [← average_const (Ω := Fin T → E) (1:ℝ)]
