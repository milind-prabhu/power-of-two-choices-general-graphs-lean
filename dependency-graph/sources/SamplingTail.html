   6 /-! An explicit binomial upper tail proved directly in the finite i.i.d.
   7 sampling model, by selecting distinct witness time steps and union-bounding.

   9 namespace GraphChoices
  10 open Finset
  11 variable {E : Type*} [Fintype E] [DecidableEq E] [Nonempty E] {T : ℕ}

  13 def hitCount (A : Finset E) (ω : Fin T → E) : ℕ :=
  14   (univ.filter fun t => ω t ∈ A).card
  15 
  16 def hitTailEvent (A : Finset E) (T k : ℕ) : Finset (Fin T → E) :=
  17   univ.filter fun ω => k ≤ hitCount A ω

  19 noncomputable def requiredHitsEvent (A : Finset E) (K : Finset (Fin T)) : Finset (Fin T → E) :=
  20   Fintype.piFinset fun t => if t ∈ K then A else univ

  22  theorem requiredHits_probability (A : Finset E) (K : Finset (Fin T)) :
  23     eventProbability (requiredHitsEvent A K) = ((A.card : ℝ) / Fintype.card E)^K.card := by
  24   classical
  25   rw [requiredHitsEvent, probability_coordinate_constraints]
  26   have hm : (Fintype.card E : ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  27   have hfactor (t : Fin T) :
  28       (((if t ∈ K then A else univ : Finset E).card : ℝ) / Fintype.card E) =
  29       if t ∈ K then (A.card : ℝ) / Fintype.card E else 1 := by
  30     by_cases ht : t ∈ K <;> simp [ht,hm]
  31   simp_rw [hfactor]
  32   simp [div_pow]
  33 
  34  theorem hitTail_subset_witness_union (A : Finset E) (T k : ℕ) :
  35     hitTailEvent A T k ⊆
  36       (univ.powersetCard k).biUnion (fun K : Finset (Fin T) => requiredHitsEvent A K) := by
  37   classical
  38   intro ω hω
  39   have hk := (mem_filter.mp hω).2
  40   obtain ⟨K,hK,hcard⟩ := exists_subset_card_eq hk
  41   apply mem_biUnion.mpr
  42   refine ⟨K, mem_powersetCard.mpr ⟨subset_univ _,hcard⟩, ?_⟩
  43   simp only [requiredHitsEvent, Fintype.mem_piFinset]
  44   intro t
  45   by_cases ht : t ∈ K
  46   · simp only [if_pos ht]
  47     exact (mem_filter.mp (hK ht)).2
  48   · simp [ht]

  50 /-- P[Bin(T,p) >= k] <= choose(T,k) p^k, with p = |A|/|E|.
  51 This is an upper bound for every k, including k=0 and k>T. -/
  52  theorem hitTail_probability_le (A : Finset E) (T k : ℕ) :
  53     eventProbability (hitTailEvent A T k) ≤
  54       (T.choose k : ℝ) * ((A.card : ℝ) / Fintype.card E)^k := by
  55   calc
  56     _ ≤ eventProbability ((univ.powersetCard k).biUnion
  57         (fun K : Finset (Fin T) => requiredHitsEvent A K)) :=
  58       eventProbability_mono (hitTail_subset_witness_union A T k)
  59     _ ≤ ∑ K ∈ univ.powersetCard k, eventProbability (requiredHitsEvent A K) :=
  60       witness_union_bound _ _
  61     _ = ∑ _K ∈ (univ : Finset (Fin T)).powersetCard k,
  62         ((A.card : ℝ) / Fintype.card E)^k := by
  63       apply sum_congr rfl
  64       intro K hK
  65       rw [requiredHits_probability, (mem_powersetCard.mp hK).2]
  66     _ = _ := by simp [card_powersetCard]

  68 /-- The standard exponential estimate for binomial coefficients, including
  69 k=0. It is derived from the exponential series and the factorial bound. -/
  70 theorem choose_le_exp_div_pow (T k : ℕ) :
  71     (T.choose k : ℝ) ≤ (Real.exp 1 * T / k)^k := by
  72   by_cases hk : k = 0
  73   · simp [hk]
  74   have hkpos : (0:ℝ) < k := by exact_mod_cast Nat.pos_of_ne_zero hk
  75   have hkne : (k:ℝ) ≠ 0 := ne_of_gt hkpos
  76   have he : Real.exp (k : ℝ) = (Real.exp 1)^k := by simpa using Real.exp_nat_mul 1 k
  77   have hp := Real.pow_div_factorial_le_exp (k : ℝ) (Nat.cast_nonneg k : (0:ℝ) ≤ k) k
  78   rw [he] at hp
  79   have hscale := mul_le_mul_of_nonneg_left hp (by positivity : (0:ℝ) ≤ ((T:ℝ)/k)^k)
  80   calc
  81     (T.choose k : ℝ) ≤ (T:ℝ)^k / k.factorial := Nat.choose_le_pow_div k T
  82     _ = ((T:ℝ)/k)^k * ((k:ℝ)^k / k.factorial) := by
  83       rw [div_pow]
  84       field_simp
  85     _ ≤ ((T:ℝ)/k)^k * (Real.exp 1)^k := hscale
  86     _ = _ := by rw [← mul_pow]; congr 1; ring
  87 
  88 /-- A Chernoff-style upper tail in the paper's exact sampling space. -/
  89 theorem hitTail_probability_le_exp (A : Finset E) (T k : ℕ) :
  90     eventProbability (hitTailEvent A T k) ≤
  91       (Real.exp 1 * T * ((A.card : ℝ) / Fintype.card E) / k)^k := by
  92   calc
  93     _ ≤ (T.choose k : ℝ) * ((A.card : ℝ) / Fintype.card E)^k := hitTail_probability_le A T k
  94     _ ≤ (Real.exp 1 * T / k)^k * ((A.card : ℝ) / Fintype.card E)^k :=
  95       mul_le_mul_of_nonneg_right (choose_le_exp_div_pow T k) (by positivity)
  96     _ = _ := by rw [← mul_pow]; congr 1; ring
  97 
  98 /-- A useful geometric form once the threshold dominates the bucket's mean. -/
  99 theorem hitTail_probability_le_geometric (A : Finset E) (T k : ℕ) (r : ℝ)
 100     (h : Real.exp 1 * T * ((A.card : ℝ) / Fintype.card E) / k ≤ r) :
 101     eventProbability (hitTailEvent A T k) ≤ r^k := by
 102   apply (hitTail_probability_le_exp A T k).trans
 103   exact pow_le_pow_left₀ (by positivity) h k

 105 /-- Union bound for large sampled degrees across any finite family of buckets. -/
 106 theorem some_bucket_tail_bound {V : Type*} [Fintype V] [DecidableEq V]
 107     (A : V → Finset E) (T k : ℕ) (r : ℝ)
 108     (h : ∀ v, Real.exp 1 * T * (((A v).card : ℝ) / Fintype.card E) / k ≤ r) :
 109     eventProbability (univ.filter fun ω : Fin T → E => ∃ v, k ≤ hitCount (A v) ω) ≤
 110       Fintype.card V * r^k := by
 111   have hsub : (univ.filter fun ω : Fin T → E => ∃ v, k ≤ hitCount (A v) ω) ⊆
 112       univ.biUnion (fun v : V => hitTailEvent (A v) T k) := by
 113     intro ω hω
 114     obtain ⟨v,hv⟩ := (mem_filter.mp hω).2
 115     exact mem_biUnion.mpr ⟨v,mem_univ _,mem_filter.mpr ⟨mem_univ _,hv⟩⟩
 116   calc
 117     _ ≤ eventProbability (univ.biUnion fun v : V => hitTailEvent (A v) T k) := eventProbability_mono hsub
 118     _ ≤ ∑ v, eventProbability (hitTailEvent (A v) T k) := witness_union_bound _ _
 119     _ ≤ ∑ _v : V, r^k := sum_le_sum fun v _ => hitTail_probability_le_geometric (A v) T k r (h v)
 120     _ = _ := by simp
