   4 /-! Component-local bounds for the actual Threshold-Greedy state. A bound on
   5 all residual component sizes gives a deterministic bound on the full load,
   6 including thresholds, against the full sample's offline optimum. -/
   7 namespace GraphChoices.ThresholdGreedy
   8 open Finset
   9 variable {L R C : Type*} [Fintype L] [Fintype R]
  10   [DecidableEq L] [DecidableEq R] [DecidableEq C]

  12 def residualVertexLoad (α : C → ℕ) (es : List (Arrival L R C)) : L ⊕ R → ℕ :=
  13   Sum.elim (run α es).greedyLeft (run α es).greedyRight
  14 
  15 def residualInstance (α : C → ℕ) (es : List (Arrival L R C)) :=
  16   replayInstance (residualArrivals (trace α es))
  17 
  18 theorem residual_vertex_load_eq (α : C → ℕ) (es : List (Arrival L R C)) (v : L ⊕ R) :
  19     residualVertexLoad α es v =
  20       load (residualInstance α es) (replayOrientation (residualArrivals (trace α es))) v := by
  21   have hr := residual_replay α es
  22   have hh := congrArg (fun s : GreedyLoads L R => greedyVertexLoad s v) hr
  23   exact hh.trans (replay_final_load _ v)
  24 
  25 theorem residual_vertex_load_le_length (α : C → ℕ) (es : List (Arrival L R C)) (v : L ⊕ R) :
  26     residualVertexLoad α es v ≤ es.length := by
  27   rw [residual_vertex_load_eq]
  28   exact (load_le_length _ _ _).trans (residual_sublist α es).length_le
  29 
  30 /-- This bound also covers an empty residual sequence, without a positivity premise. -/
  31 theorem residual_vertex_component_bound (α : C → ℕ) (es : List (Arrival L R C)) (v : L ⊕ R) :
  32     residualVertexLoad α es v ≤ 2 * opt (replayInstance es) *
  33       (Nat.log 2 (componentVertices (residualInstance α es) v).card + 1) := by
  34   rw [residual_vertex_load_eq]
  35   by_cases hres : 0 < (residualArrivals (trace α es)).length
  36   · apply (greedy_component_competitive _ _ (replay_isGreedy _) hres v).le.trans
  37     exact Nat.mul_le_mul_right _ (Nat.mul_le_mul_left 2 (residual_opt_le_full α es))
  38   · have hz : (residualArrivals (trace α es)).length = 0 := by omega
  39     have hh := load_le_length (residualInstance α es)
  40       (replayOrientation (residualArrivals (trace α es))) v
  41     change _ ≤ (residualArrivals (trace α es)).length at hh
  42     exact (hh.trans hz.le).trans (Nat.zero_le _)
  43 
  44 theorem residual_vertex_bound_of_small_components (α : C → ℕ) (es : List (Arrival L R C))
  45     (s : ℕ) (hsmall : ∀ v, (componentVertices (residualInstance α es) v).card ≤ s)
  46     (v : L ⊕ R) :
  47     residualVertexLoad α es v ≤ 2 * opt (replayInstance es) * (Nat.log 2 s + 1) := by
  48   apply (residual_vertex_component_bound α es v).trans
  49   exact Nat.mul_le_mul_left _ (Nat.add_le_add_right (Nat.log_mono_right (hsmall v)) 1)
  50 
  51 def totalVertexLoad [Fintype C] (α : C → ℕ) (es : List (Arrival L R C)) : L ⊕ R → ℕ :=
  52   Sum.elim (fun u => (∑ c, (run α es).threshold u c) + (run α es).greedyLeft u)
  53     (run α es).greedyRight
  54 
  55 def totalMaxLoad [Fintype C] (α : C → ℕ) (es : List (Arrival L R C)) : ℕ :=
  56   univ.sup (totalVertexLoad α es)
  57 
  58 omit [Fintype L] [Fintype R] in
  59 theorem total_vertex_le_threshold_add_residual [Fintype C] (α : C → ℕ)
  60     (es : List (Arrival L R C)) (v : L ⊕ R) :
  61     totalVertexLoad α es v ≤ (∑ c, α c) + residualVertexLoad α es v := by
  62   cases v with
  63   | inl u => exact Nat.add_le_add_right (threshold_budget α es u) _
  64   | inr v => exact Nat.le_add_left _ _
  65 
  66 theorem totalMaxLoad_le_budget_add_length [Fintype C] (α : C → ℕ)
  67     (es : List (Arrival L R C)) : totalMaxLoad α es ≤ (∑ c, α c) + es.length := by
  68   apply Finset.sup_le
  69   intro v _
  70   exact (total_vertex_le_threshold_add_residual α es v).trans
  71     (Nat.add_le_add_left (residual_vertex_load_le_length α es v) _)

  73 /-- The deterministic consequence of component shattering needed by Theorem 4.1. -/
  74 theorem totalMaxLoad_of_small_components [Fintype C] (α : C → ℕ)
  75     (es : List (Arrival L R C)) (s : ℕ)
  76     (hsmall : ∀ v, (componentVertices (residualInstance α es) v).card ≤ s) :
  77     totalMaxLoad α es ≤ (∑ c, α c) + 2 * opt (replayInstance es) * (Nat.log 2 s + 1) := by
  78   apply Finset.sup_le
  79   intro v _
  80   exact (total_vertex_le_threshold_add_residual α es v).trans
  81     (Nat.add_le_add_left (residual_vertex_bound_of_small_components α es s hsmall v) _)

  83 /-- For components of at most ell squared vertices, the Greedy multiplier is
  84 at most 4(log_2 ell + 1), with exact integer constants. -/
  85 theorem totalMaxLoad_of_square_components [Fintype C] (α : C → ℕ)
  86     (es : List (Arrival L R C)) (ell : ℕ)
  87     (hsmall : ∀ v, (componentVertices (residualInstance α es) v).card ≤ ell^2) :
  88     totalMaxLoad α es ≤ (∑ c, α c) + 4 * opt (replayInstance es) * (Nat.log 2 ell + 1) := by
  89   have hlog : Nat.log 2 (ell^2) + 1 ≤ 2*(Nat.log 2 ell + 1) := by
  90     by_cases hz : ell = 0
  91     · simp [hz]
  92     have hpow := Nat.lt_pow_succ_log_self (by decide : 1 < 2) ell
  93     have hsquare : ell^2 < 2^(2*(Nat.log 2 ell + 1)) := by
  94       calc
  95         ell^2 < (2^(Nat.log 2 ell+1))^2 := Nat.pow_lt_pow_left hpow (by decide)
  96         _ = _ := by rw [← pow_mul]; congr 1; omega
  97     have hh := Nat.log_lt_of_lt_pow (pow_ne_zero 2 hz) hsquare
  98     omega
  99   apply (totalMaxLoad_of_small_components α es (ell^2) hsmall).trans
 100   apply Nat.add_le_add_left
 101   calc
 102     _ ≤ 2 * opt (replayInstance es) * (2*(Nat.log 2 ell + 1)) := Nat.mul_le_mul_left _ hlog
 103     _ = _ := by ring
