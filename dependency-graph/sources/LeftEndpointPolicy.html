   9 theorem leftBucket_card (G : Graph L R) (u : L) : (leftBucket G u).card = leftDegree G u := by
  10   classical
  11   apply card_bij (fun e _ => e.val.2)
  12   · intro e he
  13     have hh := (mem_filter.mp he).2
  14     simp only [leftNeighbors, mem_filter, mem_univ, true_and]
  15     simpa only [← hh] using e.property
  16   · intro e he f hf h
  17     apply Subtype.ext
  18     exact Prod.ext (((mem_filter.mp he).2).trans ((mem_filter.mp hf).2).symm) h
  19   · intro v hv
  20     have hh : (u,v) ∈ G := (mem_filter.mp hv).2
  21     exact ⟨⟨(u,v),hh⟩, mem_filter.mpr ⟨mem_univ _,rfl⟩, rfl⟩

  23 theorem allLeft_maxLoad (G : Graph L R) {T : ℕ} (ω : Fin T → G) :
  24     maxLoad ((graphBase G).sample ω) (fun _ => true) =
  25       univ.sup (fun u : L => hitCount (leftBucket G u) ω) := by
  26   have hl (u : L) : load ((graphBase G).sample ω) (fun _ => true) (Sum.inl u) =
  27       hitCount (leftBucket G u) ω := by simp [load, destination, graphBase, BaseGraph.sample, hitCount, leftBucket]
  28   have hr (v : R) : load ((graphBase G).sample ω) (fun _ => true) (Sum.inr v) = 0 := by
  29     simp [load, destination, graphBase, BaseGraph.sample]
  30   apply Nat.le_antisymm
  31   · apply (maxLoad_le_iff _ _ _).mpr
  32     intro v
  33     cases v with
  34     | inl u => rw [hl]; exact Finset.le_sup (f := fun u : L => hitCount (leftBucket G u) ω) (mem_univ u)
  35     | inr v => rw [hr]; omega
  36   · apply Finset.sup_le
  37     intro u _
  38     rw [← hl]
  39     exact load_le_maxLoad _ _ _

  41 theorem allLeft_expectedCost (G : Graph L R) [Nonempty G] (T N : ℕ)
  42     (hvertices : Fintype.card (L ⊕ R) ≤ N) :
  43     expectedCost (graphBase G) (onlineOrientation (T := T) (fun _ _ => true)) ≤
  44       2*Real.log (N:ℝ)+2*(maxLeftDegree G:ℝ)*((T:ℝ)/G.card) := by

  46   let e : G := Classical.choice (inferInstance : Nonempty G)

  49   have hLN : Fintype.card L ≤ N := by simp only [Fintype.card_sum] at hvertices; omega
  50   have hh := expected_max_hits (leftBucket G) T ((maxLeftDegree G:ℝ)*((T:ℝ)/G.card)) (fun u => by

  53     have he := mul_le_mul_of_nonneg_right hd (show 0 ≤ (T:ℝ)/G.card by positivity)
