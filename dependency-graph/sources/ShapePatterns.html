   5 /-! Canonical plane rooted patterns via first-child/next-sibling binary forests.
   6 A forest with m nodes encodes the m non-root vertices of a rooted tree. There
   7 are at most 4^m such shapes and |C|^m ways to label their edges. -/
   8 namespace GraphChoices.Patterns
   9 open Finset

  11 /-- Parent-before-child indices from a first-child/next-sibling forest. The
  12 extra parent p is outside the forest, at or before its starting position. -/
  13 def forestParent : (t : Tree Unit) → (p start : ℕ) → p ≤ start →
  14     (i : Fin t.numNodes) → Fin (start+i.val+1)
  15   | .nil, _, _, _, i => Fin.elim0 i
  16   | .node _ l r, p, start, hp, i =>
  17     if hz : i.val = 0 then ⟨p,by omega⟩
  18     else if hl : i.val ≤ l.numNodes then
  19       let j : Fin l.numNodes := ⟨i.val-1,by omega⟩
  20       let q := forestParent l (start+1) (start+1) le_rfl j
  21       ⟨q.val,by have := q.isLt; dsimp [j] at this; omega⟩
  22     else
  23       let j : Fin r.numNodes := ⟨i.val-1-l.numNodes,by
  24         have hi := i.isLt
  25         simp only [Tree.numNodes] at hi
  26         omega⟩
  27       let q := forestParent r p (start+1+l.numNodes) (by omega) j
  28       ⟨q.val,by have := q.isLt; dsimp [j] at this; omega⟩
  29 
  30 def forestSide : (t : Tree Unit) → Bool → Fin t.numNodes → Bool
  31   | .nil, _, i => Fin.elim0 i
  32   | .node _ l r, side, i =>
  33     if hz : i.val = 0 then side
  34     else if hl : i.val ≤ l.numNodes then
  35       forestSide l (!side) ⟨i.val-1,by omega⟩
  36     else
  37       forestSide r side ⟨i.val-1-l.numNodes,by
  38         have hi := i.isLt
  39         simp only [Tree.numNodes] at hi
  40         omega⟩

  42 abbrev ShapeCode (C : Type*) (m : ℕ) :=
  43   {t : Tree Unit // t ∈ Tree.treesOfNumNodesEq m} × (Fin m → C)
  44 
  45 def shapePattern {C : Type*} {m : ℕ} (s : ShapeCode C m) : ParentPattern C m where
  46   parent i :=
  47     let j : Fin s.1.val.numNodes := Fin.cast (Tree.mem_treesOfNumNodesEq.mp s.1.property).symm i
  48     let q := forestParent s.1.val 0 0 le_rfl j
  49     ⟨q.val,by simpa [j] using q.isLt⟩
  50   cls := s.2
  51   childLeft i := forestSide s.1.val true (Fin.cast (Tree.mem_treesOfNumNodesEq.mp s.1.property).symm i)

  53  theorem shapes_card_le (m : ℕ) : (Tree.treesOfNumNodesEq m).card ≤ 4^m := by
  54   rw [Tree.treesOfNumNodesEq_card_eq_catalan, catalan_eq_centralBinom_div]
  55   calc
  56     _ ≤ Nat.centralBinom m := Nat.div_le_self _ _
  57     _ ≤ 2^(2*m) := Nat.choose_le_two_pow _ _
  58     _ = _ := by rw [pow_mul]; norm_num
  59 
  60  theorem shapeCode_card_le {C : Type*} [Fintype C] [DecidableEq C] (m : ℕ) :
  61     Fintype.card (ShapeCode C m) ≤ 4^m * (Fintype.card C)^m := by
  62   simp only [ShapeCode, Fintype.card_prod, Fintype.card_coe, Fintype.card_fun, Fintype.card_fin]
  63   exact Nat.mul_le_mul_right _ (shapes_card_le m)

  68 /-- Union-bound over canonical plane-tree shapes and class labels. Only the
  69 left-heavy patterns are retained; the per-pattern bound is a stated premise. -/
  70 noncomputable def heavyShapeEvent (H : C → Peeling.Graph L R) (encode : E → ThresholdGreedy.Arrival L R C)
  71     (α : C → ℕ) (T m : ℕ) : Finset (Fin T → E) := by
  72   classical
  73   exact univ.biUnion fun s : ShapeCode C m =>
  74     if m ≤ 2*(univ.filter fun i : Fin m => (shapePattern s).childLeft i = true).card
  75     then residualPatternEvent (shapePattern s) H encode α T else ∅
  76 
  77  theorem heavyShape_probability_le (H : C → Peeling.Graph L R) (encode : E → ThresholdGreedy.Arrival L R C)
  78     (α : C → ℕ) (T m : ℕ) (N r : ℝ) (hN : 0 ≤ N) (hr : 0 ≤ r)
  79     (hpattern : ∀ s : ShapeCode C m,
  80       m ≤ 2*(univ.filter fun i : Fin m => (shapePattern s).childLeft i = true).card →
  81       eventProbability (residualPatternEvent (shapePattern s) H encode α T) ≤ N*r^m) :
  82     eventProbability (heavyShapeEvent H encode α T m) ≤ N*(4*(Fintype.card C : ℝ)*r)^m := by
  83   classical
  84   calc
  85     _ ≤ ∑ s : ShapeCode C m, eventProbability
  86       (if m ≤ 2*(univ.filter fun i : Fin m => (shapePattern s).childLeft i = true).card
  87       then residualPatternEvent (shapePattern s) H encode α T else ∅) := witness_union_bound _ _
  88     _ ≤ ∑ _s : ShapeCode C m, N*r^m := by
  89       apply sum_le_sum
  90       intro s _
  91       split_ifs with h
  92       · exact hpattern s h
  93       · simpa [eventProbability] using mul_nonneg hN (pow_nonneg hr m)
  94     _ = (Fintype.card (ShapeCode C m) : ℝ)*(N*r^m) := by simp
  95     _ ≤ (4^m*(Fintype.card C)^m : ℕ)*(N*r^m) := by
  96       apply mul_le_mul_of_nonneg_right _ (mul_nonneg hN (pow_nonneg hr m))
  97       exact_mod_cast shapeCode_card_le (C := C) m
  98     _ = _ := by push_cast; rw [mul_pow, mul_pow]; ring
  99
