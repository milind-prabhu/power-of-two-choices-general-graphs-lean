   4 /-! Cancellation of branching factors against threshold-witness probabilities
   5 for a left-heavy fixed pattern. Local numerical hypotheses are explicit; the
   6 paper's chosen thresholds must still be shown to satisfy them. -/
   7 namespace GraphChoices.Patterns
   8 open Finset
   9 open ThresholdGreedy
  10 variable {E L R C : Type*} [Fintype E] [DecidableEq E] [Nonempty E]
  11   [Fintype L] [Fintype R] [DecidableEq L] [DecidableEq R] [DecidableEq C]
  12   {T m : ℕ}

  14  theorem weighted_branch_identity (p : ParentPattern C m) (D z : ℝ) (DR γ : C → ℝ) :
  15     (∏ i : Fin m, if p.childLeft i then DR (p.cls i) else D) *
  16       (z^m * ∏ j : LeftChildren p, γ (p.cls j.val)) =
  17     ∏ i : Fin m, if p.childLeft i then DR (p.cls i)*z*γ (p.cls i) else D*z := by
  18   classical
  19   rw [GraphChoices.subtype_product_as_ite (fun i : Fin m => p.childLeft i = true) (fun i => γ (p.cls i))]
  20   have hz : z^m = ∏ _i : Fin m, z := by simp
  21   rw [hz, ← prod_mul_distrib, ← prod_mul_distrib]
  22   apply prod_congr rfl
  23   intro i _
  24   cases h : p.childLeft i <;> simp [h, mul_assoc]

  26 noncomputable def bucketDecay (α : C → ℕ) (DL : C → ℕ) (T : ℕ) (c : C) : ℝ :=
  27   (Real.exp 1 * T * (DL c : ℝ) / Fintype.card E / α c)^(α c)

  29 /-- A fixed left-heavy pattern has geometric probability decay when the local
  30 threshold tails beat their right branching factors. This combines actual
  31 execution witnesses and the concrete graph-embedding count. -/
  32  theorem residualPattern_probability_decay (p : ParentPattern C m) (H : C → Peeling.Graph L R)
  33     (encode : E → Arrival L R C) (α : C → ℕ) (DL DR : C → ℕ) (D : ℕ)
  34     (hsimple : Function.Injective fun e => ((encode e).left,(encode e).right))
  35     (hDL : ∀ u c, (sourceBucket encode u c).card ≤ DL c)
  36     (hL : ∀ c u, Peeling.leftDegree (H c) u ≤ D)
  37     (hR : ∀ c v, Peeling.rightDegree (H c) v ≤ DR c)
  38     (b r : ℝ) (hb : 0 ≤ b) (hr : 0 ≤ r) (hbr : b ≤ r)
  39     (hpair : (D : ℝ)*((T:ℝ)/Fintype.card E)*b ≤ r^2)
  40     (hlocal : ∀ c, (DR c : ℝ)*((T:ℝ)/Fintype.card E)*bucketDecay (E := E) α DL T c ≤ b)
  41     (hheavy : m ≤ 2*(univ.filter fun i : Fin m => p.childLeft i = true).card) :
  42     eventProbability (residualPatternEvent p H encode α T) ≤ (Fintype.card R : ℝ)*r^m := by
  43   classical
  44   have hp := residualPattern_probability_le (T := T) p H encode α DL DR D hsimple hDL hL hR
  45   have heq :
  46       ((Fintype.card R * (∏ i : Fin m, if p.childLeft i then DR (p.cls i) else D) : ℕ) : ℝ) *
  47         patternProbabilityFactor (E := E) p α DL T =
  48       (Fintype.card R : ℝ) * (∏ i : Fin m, if p.childLeft i then
  49         (DR (p.cls i) : ℝ)*((T:ℝ)/Fintype.card E)*bucketDecay (E := E) α DL T (p.cls i)
  50         else (D:ℝ)*((T:ℝ)/Fintype.card E)) := by
  51     simp only [Nat.cast_mul, Nat.cast_prod, Nat.cast_ite, patternProbabilityFactor]
  52     rw [mul_assoc]
  53     exact congrArg (fun x : ℝ => (Fintype.card R : ℝ)*x)
  54       (weighted_branch_identity p (D:ℝ) ((T:ℝ)/Fintype.card E)
  55         (fun c => (DR c : ℝ)) (bucketDecay (E := E) α DL T))
  56   rw [heq] at hp
  57   calc
  58     _ ≤ _ := hp
  59     _ ≤ (Fintype.card R : ℝ) * (∏ i : Fin m, if p.childLeft i then b else (D:ℝ)*((T:ℝ)/Fintype.card E)) := by
  60       apply mul_le_mul_of_nonneg_left _ (Nat.cast_nonneg _)
  61       apply prod_le_prod
  62       · intro i _
  63         split_ifs <;> (try dsimp only [bucketDecay]) <;> positivity
  64       · intro i _
  65         split_ifs
  66         · exact hlocal _
  67         · exact le_rfl
  68     _ ≤ (Fintype.card R : ℝ) * r^m := by
  69       apply mul_le_mul_of_nonneg_left _ (Nat.cast_nonneg _)
  70       simpa only [Fintype.card_fin] using GraphChoices.two_class_product_bound (fun i : Fin m => p.childLeft i = true)
  71         _ b r (by positivity) hb hr hpair hbr (by simpa using hheavy)
  72 
  73 end GraphChoices.Patterns
