  11 /-- A union of components: no arrival crosses the boundary. -/
  12 def EdgeClosed (I : Instance V T) (S : Finset V) : Prop :=
  13   ∀ t, I.left t ∈ S ↔ I.right t ∈ S

  16 theorem endpoints_mem_of_destination_mem (I : Instance V T) (a : Orientation T)
  17     (S : Finset V) (hS : EdgeClosed I S) (t : Fin T)
  18     (ht : destination I a t ∈ S) : I.left t ∈ S ∧ I.right t ∈ S := by
  19   rcases destination_endpoint I a t with h | h
  20   · rw [h] at ht
  21     exact ⟨ht, (hS t).mp ht⟩
  22   · rw [h] at ht
  23     exact ⟨(hS t).mpr ht, ht⟩

  25 def localHighVertices (I : Instance V T) (a : Orientation T) (S : Finset V) (k : ℕ) :=
  26   S.filter fun v => k ≤ load I a v
  27 
  28 def localHighArrivals (I : Instance V T) (a : Orientation T) (S : Finset V) (k : ℕ) :=
  29   (highArrivals I a k).filter fun t => destination I a t ∈ S

  31 theorem localHighArrivals_internal (I : Instance V T) (a : Orientation T)
  32     (hg : IsGreedy I a) (S : Finset V) (hS : EdgeClosed I S) (k : ℕ) :
  33     localHighArrivals I a S k ⊆ univ.filter (fun t =>
  34       I.left t ∈ localHighVertices I a S k ∧ I.right t ∈ localHighVertices I a S k) := by
  35   intro t ht
  36   obtain ⟨hhigh, hmem⟩ := mem_filter.mp ht
  37   have he := endpoints_mem_of_destination_mem I a S hS t hmem
  38   have hh := (mem_filter.mp (highArrivals_internal I a hg k hhigh)).2
  39   simp only [mem_filter, mem_univ, true_and, localHighVertices, highVertices] at hh ⊢
  40   exact ⟨⟨he.1, hh.1⟩, ⟨he.2, hh.2⟩⟩

  42 theorem greedy_local_layer_bound (I : Instance V T) (a : Orientation T)
  43     (hg : IsGreedy I a) (S : Finset V) (hS : EdgeClosed I S) (k d : ℕ) :
  44     d * (localHighVertices I a S (k+d)).card ≤
  45       opt I * (localHighVertices I a S k).card := by
  46   calc
  47     _ = ∑ v ∈ localHighVertices I a S (k+d), d := by simp [mul_comm]
  48     _ ≤ ∑ v ∈ localHighVertices I a S (k+d),
  49         ((localHighArrivals I a S k).filter fun t => destination I a t = v).card := by
  50       apply sum_le_sum
  51       intro v hv
  52       obtain ⟨hvS, hvload⟩ := mem_filter.mp hv
  53       have heq : ((localHighArrivals I a S k).filter fun t => destination I a t = v) =
  54           ((highArrivals I a k).filter fun t => destination I a t = v) := by
  55         ext t
  56         simp only [localHighArrivals, mem_filter]
  57         constructor
  58         · exact fun h => ⟨h.1.1, h.2⟩
  59         · exact fun h => ⟨⟨h.1, h.2 ▸ hvS⟩, h.2⟩
  60       rw [heq]
  61       have hb := load_le_low_add_high I a v k
  62       omega
  63     _ ≤ ∑ v, ((localHighArrivals I a S k).filter fun t => destination I a t = v).card :=
  64       sum_le_sum_of_subset_of_nonneg (subset_univ _) (fun _ _ _ => Nat.zero_le _)
  65     _ = (localHighArrivals I a S k).card := by
  66       symm
  67       exact card_eq_sum_card_fiberwise (fun _ _ => mem_univ _)
  68     _ ≤ internalCount I (localHighVertices I a S k) :=
  69       card_le_card (localHighArrivals_internal I a hg S hS k)
  70     _ ≤ opt I * (localHighVertices I a S k).card := by
  71       simpa [mul_comm] using density_le_opt I (localHighVertices I a S k)

  73 theorem greedy_local_layers_decay (I : Instance V T) (a : Orientation T)
  74     (hg : IsGreedy I a) (S : Finset V) (hS : EdgeClosed I S)
  75     (hc : 0 < opt I) (j : ℕ) :
  76     2^j * (localHighVertices I a S (2*opt I*j)).card ≤ S.card := by
  77   induction j with
  78   | zero => simp [localHighVertices]
  79   | succ j ih =>
  80     have h := greedy_local_layer_bound I a hg S hS (2*opt I*j) (2*opt I)
  81     have hstep : 2 * (localHighVertices I a S (2*opt I*(j+1))).card ≤
  82         (localHighVertices I a S (2*opt I*j)).card := by
  83       have heq : 2*opt I*j + 2*opt I = 2*opt I*(j+1) := by ring
  84       rw [heq] at h
  85       nlinarith
  86     calc
  87       _ = 2^j * (2 * (localHighVertices I a S (2*opt I*(j+1))).card) := by
  88         rw [pow_succ]; ring
  89       _ ≤ 2^j * (localHighVertices I a S (2*opt I*j)).card := Nat.mul_le_mul_left _ hstep
  90       _ ≤ _ := ih

  92 theorem greedy_load_lt_of_closed (I : Instance V T) (a : Orientation T)
  93     (hg : IsGreedy I a) (S : Finset V) (hS : EdgeClosed I S)
  94     (hT : 0 < T) (j : ℕ) (hj : S.card < 2^j) (v : V) (hv : v ∈ S) :
  95     load I a v < 2*opt I*j := by
  96   have hd := greedy_local_layers_decay I a hg S hS (opt_pos I hT) j
  97   by_contra h
  98   have hm : v ∈ localHighVertices I a S (2*opt I*j) := mem_filter.mpr ⟨hv, by omega⟩
  99   have hp := card_pos.mpr ⟨v, hm⟩
 100   have hh := Nat.mul_le_mul_left (2^j) (show 1 ≤ (localHighVertices I a S (2*opt I*j)).card by omega)
 101   omega

 103 /-- Forget multiplicities only for connectivity. Loads still count all times. -/
 104 def arrivalGraph (I : Instance V T) : SimpleGraph V where
 105   Adj u v := u ≠ v ∧ ∃ t, (I.left t = u ∧ I.right t = v) ∨
 106     (I.left t = v ∧ I.right t = u)
 107   symm := by
 108     intro u v h
 109     obtain ⟨hne, t, ht⟩ := h
 110     exact ⟨hne.symm, t, ht.symm⟩
 111   loopless := by intro u h; exact h.1 rfl
 112 
 113 noncomputable def componentVertices (I : Instance V T) (v : V) : Finset V := by
 114   classical
 115   exact univ.filter fun w => (arrivalGraph I).Reachable v w

 117 omit [DecidableEq V] in
 118 theorem mem_componentVertices (I : Instance V T) (v w : V) :
 119     w ∈ componentVertices I v ↔ (arrivalGraph I).Reachable v w := by
 120   classical
 121   simp [componentVertices]

 123 omit [DecidableEq V] in
 124 theorem self_mem_componentVertices (I : Instance V T) (v : V) :
 125     v ∈ componentVertices I v :=
 126   (mem_componentVertices I v v).mpr (SimpleGraph.Reachable.refl v)
 127 
 128 omit [Fintype V] in
 129 theorem arrival_endpoints_reachable (I : Instance V T) (t : Fin T) :
 130     (arrivalGraph I).Reachable (I.left t) (I.right t) := by
 131   by_cases h : I.left t = I.right t
 132   · rw [h]
 133   · exact SimpleGraph.Adj.reachable ⟨h, t, Or.inl ⟨rfl, rfl⟩⟩
 134 
 135 theorem componentVertices_closed (I : Instance V T) (v : V) :
 136     EdgeClosed I (componentVertices I v) := by
 137   intro t
 138   rw [mem_componentVertices, mem_componentVertices]
 139   have h := arrival_endpoints_reachable I t
 140   exact ⟨fun hv => hv.trans h, fun hv => hv.trans h.symm⟩

 142 /-- Component-local Fact 3.6, benchmarked against the full arrival instance. -/
 143 theorem greedy_component_competitive (I : Instance V T) (a : Orientation T)
 144     (hg : IsGreedy I a) (hT : 0 < T) (v : V) :
 145     load I a v < 2*opt I*(Nat.log 2 (componentVertices I v).card + 1) :=
 146   greedy_load_lt_of_closed I a hg _ (componentVertices_closed I v) hT _
 147     (Nat.lt_pow_succ_log_self (by decide) _) v (self_mem_componentVertices I v)
